//+------------------------------------------------------------------+
double GetPipSize()
{
   return (_Digits == 3 || _Digits == 5) ? _Point * 10 : _Point;
}

//+------------------------------------------------------------------+
//| Insert value into sorted double array                             |
//+------------------------------------------------------------------+
void InsertSorted(double &arr[], int &count, double value)
{
   ArrayResize(arr, count + 1);
   int i = count;
   while(i > 0 && arr[i-1] > value)
   {
      arr[i] = arr[i-1];
      i--;
   }
   arr[i] = value;
   count++;
}

//+------------------------------------------------------------------+
//| Insert value into sorted int array                                |
//+------------------------------------------------------------------+
void InsertSortedInt(int &arr[], int &count, int value)
{
   ArrayResize(arr, count + 1);
   int i = count;
   while(i > 0 && arr[i-1] > value)
   {
      arr[i] = arr[i-1];
      i--;
   }
   arr[i] = value;
   count++;
}

//+------------------------------------------------------------------+
//| Update pivot arrays from current ZigZag state                     |
//+------------------------------------------------------------------+
void UpdatePivots(int rates_total)
{
   double zz_buffer[];
   ArraySetAsSeries(zz_buffer, false);
   int copied = CopyBuffer(zz_handle, 0, 0, rates_total, zz_buffer);
   if(copied <= 0) return;
   
   ArrayResize(zz_pivots, 0);
   ArrayResize(zz_pivot_bars, 0);
   zz_pivot_count = 0;
   
   for(int i = 0; i < rates_total; i++)
   {
      if(zz_buffer[i] != 0)
      {
         ArrayResize(zz_pivots, zz_pivot_count + 1);
         ArrayResize(zz_pivot_bars, zz_pivot_count + 1);
         zz_pivots[zz_pivot_count] = zz_buffer[i];
         zz_pivot_bars[zz_pivot_count] = i;
         zz_pivot_count++;
      }
   }
}

//+------------------------------------------------------------------+
//| Add new completed legs to distributions                           |
//+------------------------------------------------------------------+
void UpdateDistributions()
{
   if(zz_pivot_count < 2) return;
   
   double pip_size = GetPipSize();
   
   // Process any new confirmed pivots (exclude the last one as it may repaint)
   int confirmed_pivots = zz_pivot_count - 1;
   
   while(pivots_in_distributions < confirmed_pivots)
   {
      int idx = pivots_in_distributions;
      
      // New leg from idx to idx+1
      double leg_size = MathAbs(zz_pivots[idx + 1] - zz_pivots[idx]) / pip_size;
      int duration = MathAbs(zz_pivot_bars[idx + 1] - zz_pivot_bars[idx]);
      
      if(leg_size > 0)
         InsertSorted(Leg_Dist, Leg_Dist_Count, leg_size);
      
      if(duration > 0)
         InsertSortedInt(Time_Dist, Time_Dist_Count, duration);
      
      if(duration > 0 && leg_size > 0)
         InsertSorted(Velocity_Dist, Velocity_Dist_Count, leg_size / duration);
      
      // Retracement needs prior leg
      if(idx > 0)
      {
         double prior_leg = MathAbs(zz_pivots[idx] - zz_pivots[idx - 1]);
         if(prior_leg > 0)
         {
            double retrace_pct = (leg_size * pip_size / prior_leg) * 100;
            InsertSorted(Retrace_Dist, Retrace_Dist_Count, retrace_pct);
         }
      }
      
      pivots_in_distributions++;
   }
}

//+------------------------------------------------------------------+
//| Add volume from new bars to profile                               |
//+------------------------------------------------------------------+
void UpdateVolumeProfile(int bar, double bar_high, double bar_low, long bar_volume)
{
   if(Vol_Level_Size <= 0) return;
   
   // Extend bounds if price moved beyond
   bool bounds_changed = false;
   if(bar_low < Vol_Price_Min)
   {
      Vol_Price_Min = bar_low;
      bounds_changed = true;
   }
   if(bar_high > Vol_Price_Max)
   {
      Vol_Price_Max = bar_high;
      bounds_changed = true;
   }
   
   if(bounds_changed)
   {
      // Recalculate level size with new bounds
      Vol_Level_Size = (Vol_Price_Max - Vol_Price_Min) / Volume_Levels_Count;
      if(Vol_Level_Size <= 0) Vol_Level_Size = GetPipSize();
   }
   
   int low_level = (int)MathFloor((bar_low - Vol_Price_Min) / Vol_Level_Size);
   int high_level = (int)MathFloor((bar_high - Vol_Price_Min) / Vol_Level_Size);
   
   low_level = MathMax(0, MathMin(Volume_Levels_Count - 1, low_level));
   high_level = MathMax(0, MathMin(Volume_Levels_Count - 1, high_level));
   
   int levels_touched = high_level - low_level + 1;
   double vol_per_level = (double)bar_volume / levels_touched;
   
   for(int lv = low_level; lv <= high_level; lv++)
   {
      Vol_At_Level[lv] += vol_per_level;
      if(Vol_At_Level[lv] > Vol_Max)
         Vol_Max = Vol_At_Level[lv];
   }
}

//+------------------------------------------------------------------+
//| Build initial distributions (called once)                         |
//+------------------------------------------------------------------+
void BuildDistributions(int rates_total, const double &high[], const double &low[], 
                        const double &close[], const long &tick_volume[])
{
   int bars_to_use = MathMin(History_Bars, rates_total);
   int start_bar = rates_total - bars_to_use;
   double pip_size = GetPipSize();
   
   UpdatePivots(rates_total);
   
   if(zz_pivot_count < 3)
   {
      Print("Insufficient ZigZag pivots: ", zz_pivot_count);
      return;
   }
   
   // Build leg size distribution (exclude last pivot - may repaint)
   ArrayResize(Leg_Dist, 0);
   Leg_Dist_Count = 0;
   
   for(int i = 0; i < zz_pivot_count - 2; i++)
   {
      double leg_size = MathAbs(zz_pivots[i + 1] - zz_pivots[i]) / pip_size;
      if(leg_size > 0)
         InsertSorted(Leg_Dist, Leg_Dist_Count, leg_size);
   }
   
   // Build retracement distribution
   ArrayResize(Retrace_Dist, 0);
   Retrace_Dist_Count = 0;
   
   for(int i = 1; i < zz_pivot_count - 2; i++)
   {
      double prior_leg = MathAbs(zz_pivots[i] - zz_pivots[i - 1]);
      double current_leg = MathAbs(zz_pivots[i + 1] - zz_pivots[i]);
      if(prior_leg > 0)
      {
         double retrace_pct = (current_leg / prior_leg) * 100;
         InsertSorted(Retrace_Dist, Retrace_Dist_Count, retrace_pct);
      }
   }
   
   // Build time distribution
   ArrayResize(Time_Dist, 0);
   Time_Dist_Count = 0;
   
   for(int i = 0; i < zz_pivot_count - 2; i++)
   {
      int duration = MathAbs(zz_pivot_bars[i + 1] - zz_pivot_bars[i]);
      if(duration > 0)
         InsertSortedInt(Time_Dist, Time_Dist_Count, duration);
   }
   
   // Build velocity distribution
   ArrayResize(Velocity_Dist, 0);
   Velocity_Dist_Count = 0;
   
   for(int i = 0; i < zz_pivot_count - 2; i++)
   {
      double leg_size = MathAbs(zz_pivots[i + 1] - zz_pivots[i]) / pip_size;
      int duration = MathAbs(zz_pivot_bars[i + 1] - zz_pivot_bars[i]);
      if(duration > 0 && leg_size > 0)
         InsertSorted(Velocity_Dist, Velocity_Dist_Count, leg_size / duration);
   }
   
   // Track how many pivots we've processed
   pivots_in_distributions = zz_pivot_count - 2;
   
   // Build volume profile
   Vol_Price_Min = low[ArrayMinimum(low, start_bar, bars_to_use)];
   Vol_Price_Max = high[ArrayMaximum(high, start_bar, bars_to_use)];
   Vol_Level_Size = (Vol_Price_Max - Vol_Price_Min) / Volume_Levels_Count;
   if(Vol_Level_Size <= 0) Vol_Level_Size = pip_size;
   
   ArrayResize(Vol_At_Level, Volume_Levels_Count);
   ArrayInitialize(Vol_At_Level, 0);
   
   for(int i = start_bar; i < rates_total; i++)
   {
      int low_level = (int)MathFloor((low[i] - Vol_Price_Min) / Vol_Level_Size);
      int high_level = (int)MathFloor((high[i] - Vol_Price_Min) / Vol_Level_Size);
      
      low_level = MathMax(0, MathMin(Volume_Levels_Count - 1, low_level));
      high_level = MathMax(0, MathMin(Volume_Levels_Count - 1, high_level));
      
      int levels_touched = high_level - low_level + 1;
      double vol_per_level = (double)tick_volume[i] / levels_touched;
      
      for(int lv = low_level; lv <= high_level; lv++)
         Vol_At_Level[lv] += vol_per_level;
   }
   
   Vol_Max = Vol_At_Level[ArrayMaximum(Vol_At_Level)];
   
   distributions_built = true;
   last_bar_processed = rates_total;
   
   Print("Distributions built - Legs: ", Leg_Dist_Count, ", Retraces: ", Retrace_Dist_Count, 
         ", Durations: ", Time_Dist_Count, ", Velocities: ", Velocity_Dist_Count);
}

//+------------------------------------------------------------------+
double GetPercentileRank(const double &arr[], int count, double value)
{
   if(count == 0) return 50.0;
   
   int below = 0;
   for(int i = 0; i < count; i++)
   {
      if(arr[i] < value)
         below++;
      else
         break;
   }
   return (below * 100.0) / count;
}

//+------------------------------------------------------------------+
double GetPercentileRankInt(const int &arr[], int count, int value)
{
   if(count == 0) return 50.0;
   
   int below = 0;
   for(int i = 0; i < count; i++)
   {
      if(arr[i] < value)
         below++;
      else
         break;
   }
   return (below * 100.0) / count;
}

//+------------------------------------------------------------------+
double GetVolumeSupport(double leg_start, double current_price)
{
   if(Vol_Level_Size <= 0 || Vol_Max <= 0) return 50.0;
   
   int current_level = (int)MathFloor((current_price - Vol_Price_Min) / Vol_Level_Size);
   int start_level = (int)MathFloor((leg_start - Vol_Price_Min) / Vol_Level_Size);
   
   current_level = MathMax(0, MathMin(Volume_Levels_Count - 1, current_level));
   start_level = MathMax(0, MathMin(Volume_Levels_Count - 1, start_level));
   
   double total_vol = 0;
   int start_lv = MathMin(current_level, start_level);
   int end_lv = MathMax(current_level, start_level);
   int levels = end_lv - start_lv + 1;
   
   for(int i = start_lv; i <= end_lv; i++)
      total_vol += Vol_At_Level[i];
   
   double avg_vol = total_vol / levels;
   return (avg_vol / Vol_Max) * 100.0;
}

//+------------------------------------------------------------------+
int FindRecentPivotIndex(int bar)
{
   int result = -1;
   for(int i = 0; i < zz_pivot_count; i++)
   {
      if(zz_pivot_bars[i] < bar)
         result = i;
      else
         break;
   }
   return result;
}

//+------------------------------------------------------------------+
int FindNextPivotIndex(int bar)
{
   for(int i = 0; i < zz_pivot_count; i++)
   {
      if(zz_pivot_bars[i] > bar)
         return i;
   }
   return -1;
}

void ExportCSV(int rates_total, const datetime &time[], const double &close[])
{
   if(csv_exported || !Export_CSV) return;
   
   string filename = _Symbol + "_" + CSV_Filename;
   int handle = FileOpen(filename, FILE_WRITE|FILE_CSV|FILE_COMMON, ',');
   if(handle == INVALID_HANDLE) return;
   
   double pip_size = GetPipSize();
   
   FileWrite(handle, "Timestamp","Close","LegStart","LegDirection","LegSizePips",
      "BarsSinceLegStart","PivotHistoric","PivotPrevious","TimeExt","VolSupport",
      "Velocity","ReversalProb","NextPivotPrice","NextPivotBars","LegEndPips","Profitable");
   
   for(int i = 0; i < rates_total; i++)
   {
      int pivot_idx = FindRecentPivotIndex(i);
      if(pivot_idx < 0) continue;
      
      double leg_start = zz_pivots[pivot_idx];
      bool leg_up = close[i] > leg_start;
      double leg_size_pips = MathAbs(close[i] - leg_start) / pip_size;
      int bars_since = i - zz_pivot_bars[pivot_idx];
      
      int next_idx = FindNextPivotIndex(i);
      double next_price = 0;
      int next_bars = 0;
      double leg_end_pips = 0;
      int profitable = -1;
      
      if(next_idx >= 0)
      {
         next_price = zz_pivots[next_idx];
         next_bars = zz_pivot_bars[next_idx] - i;
         leg_end_pips = MathAbs(next_price - leg_start) / pip_size;
         profitable = leg_up ? (next_price < close[i] ? 1 : 0) : (next_price > close[i] ? 1 : 0);
      }
      
      FileWrite(handle,
         TimeToString(time[i], TIME_DATE|TIME_MINUTES),
         DoubleToString(close[i], _Digits),
         DoubleToString(leg_start, _Digits),
         leg_up ? "SELL" : "BUY",
         DoubleToString(leg_size_pips, 1),
         IntegerToString(bars_since),
         DoubleToString(Historic_Buffer[i], 1),
         DoubleToString(Previous_Buffer[i], 1),
         DoubleToString(Time_Buffer[i], 1),
         DoubleToString(Volume_Buffer[i], 1),
         DoubleToString(Velocity_Buffer[i], 1),
         DoubleToString(Combined_Buffer[i], 1),
         next_price > 0 ? DoubleToString(next_price, _Digits) : "",
         next_bars > 0 ? IntegerToString(next_bars) : "",
         leg_end_pips > 0 ? DoubleToString(leg_end_pips, 1) : "",
         profitable >= 0 ? IntegerToString(profitable) : "");
   }
   
   FileClose(handle);
   csv_exported = true;
   Print("CSV exported: ", filename);
}

//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   // Initial build
   if(prev_calculated == 0 || !distributions_built)
   {
      BuildDistributions(rates_total, high, low, close, tick_volume);
      if(!distributions_built)
         return(0);
   }
   else
   {
      // Incremental updates for new bars
      UpdatePivots(rates_total);
      UpdateDistributions();
      
      // Add volume from new bars
      for(int i = last_bar_processed; i < rates_total; i++)
         UpdateVolumeProfile(i, high[i], low[i], tick_volume[i]);
      
      last_bar_processed = rates_total;
   }
   
   double pip_size = GetPipSize();
   int start = (prev_calculated == 0) ? 0 : prev_calculated - 1;
   
   for(int i = start; i < rates_total; i++)
   {
      int pivot_idx = FindRecentPivotIndex(i);
      
      if(pivot_idx < 0)
      {
         Historic_Buffer[i] = 50;
         Previous_Buffer[i] = 50;
         Time_Buffer[i] = 50;
         Volume_Buffer[i] = 50;
         Velocity_Buffer[i] = 50;
         Combined_Buffer[i] = 50;
         Combined_Color_Buffer[i] = 0;
         continue;
      }
      
      double leg_start = zz_pivots[pivot_idx];
      int leg_start_bar = zz_pivot_bars[pivot_idx];
      bool leg_up = close[i] > leg_start;
      
      double current_leg_size = MathAbs(close[i] - leg_start) / pip_size;
      int current_duration = i - leg_start_bar;
      if(current_duration < 1) current_duration = 1;
      double current_velocity = current_leg_size / current_duration;
      
      Historic_Buffer[i] = GetPercentileRank(Leg_Dist, Leg_Dist_Count, current_leg_size);
      
      double prior_leg_size = 0;
      if(pivot_idx > 0)
         prior_leg_size = MathAbs(zz_pivots[pivot_idx] - zz_pivots[pivot_idx - 1]);
      
      double current_retrace = 0;
      if(prior_leg_size > 0)
         current_retrace = (current_leg_size * pip_size / prior_leg_size) * 100;
      
      Previous_Buffer[i] = GetPercentileRank(Retrace_Dist, Retrace_Dist_Count, current_retrace);
      Time_Buffer[i] = GetPercentileRankInt(Time_Dist, Time_Dist_Count, current_duration);
      Volume_Buffer[i] = GetVolumeSupport(leg_start, close[i]);
      Velocity_Buffer[i] = GetPercentileRank(Velocity_Dist, Velocity_Dist_Count, current_velocity);
      
      double pivot_base = (Historic_Buffer[i] + Previous_Buffer[i]) / 2.0;
      
      double volume_factor = 1.0;
      if(Volume_Include)
         volume_factor = 0.8 + (Volume_Buffer[i] / 100.0) * 0.4;
      
      double velocity_factor = 1.0;
      if(Velocity_Include)
         velocity_factor = 0.8 + (Velocity_Buffer[i] / 100.0) * 0.4;
      
      Combined_Buffer[i] = MathMin(100.0, MathMax(0.0, pivot_base * volume_factor * velocity_factor));
      Combined_Color_Buffer[i] = leg_up ? 1 : 0;
   }
   
   if(Export_CSV && !csv_exported && prev_calculated == 0)
      ExportCSV(rates_total, time, close);
   
   return(rates_total);
}

The color logic is comparing current price to the leg start pivot, not measuring current price movement direction.

Fix:

// Replace this line (around line 525):
Combined_Color_Buffer[i] = leg_up ? 1 : 0;

// With:
// Determine current price direction using lookback
bool price_moving_up = false;
int lookback = MathMin(Lookback_Bars, i);
if(lookback > 0)
   price_moving_up = close[i] > close[i - lookback];
else
   price_moving_up = (i > 0) ? close[i] > close[i - 1] : true;

Combined_Color_Buffer[i] = price_moving_up ? 1 : 0;

This uses the existing Lookback_Bars parameter (default 5) to determine if price is moving up or down, independent of which leg type we're in.
