bool Detect3Bar(const MqlRates &rates[], int i, int iDir)
{
   if(i < 2) return false;
   if(BarRange(rates[i-1]) < InpMinSpread) return false;

   double ext0 = Extreme(rates[i-2], iDir);
   double ext1 = Extreme(rates[i-1], iDir);
   double ext2 = Extreme(rates[i],   iDir);

   // Middle bar must exceed both neighbors
   if(ext1 * iDir <= ext0 * iDir) return false;
   if(ext1 * iDir <= ext2 * iDir) return false;

   // Body direction: entry bar toward extreme, exit bar away
   if(BodyDir(rates[i-2]) * iDir <= 0) return false;
   if(BodyDir(rates[i])   * iDir >= 0) return false;

   // Close tent: middle bar's close is local extreme among closes
   if(rates[i-1].close * iDir <= rates[i-2].close * iDir) return false;
   if(rates[i-1].close * iDir <= rates[i].close   * iDir) return false;

   // Wick on pivot bar (always required)
   double range1 = BarRange(rates[i-1]);
   if(range1 <= 0) return false;

   double wick1 = WickIntoExtreme(rates[i-1], iDir);
   if(wick1 / range1 < InpWickRatio) return false;

   // Wick on flanking bars (optional)
   if(InpWickAllBars)
   {
      double range0 = BarRange(rates[i-2]);
      double range2 = BarRange(rates[i]);
      if(range0 <= 0 || range2 <= 0) return false;

      double wick0 = WickIntoExtreme(rates[i-2], iDir);
      double wick2 = WickIntoExtreme(rates[i],   iDir);
      if(wick0 / range0 < InpWickRatio) return false;
      if(wick2 / range2 < InpWickRatio) return false;
   }

   return true;
}
