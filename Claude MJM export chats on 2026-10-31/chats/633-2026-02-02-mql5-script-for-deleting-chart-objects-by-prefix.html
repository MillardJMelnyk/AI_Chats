void OnStart()
{
   int chartsProcessed = 0;
   int totalDeleted = 0;
   
   long chartID = ChartFirst();
   
   while(chartID >= 0)
   {
      int deleted = ObjectsDeleteAll(chartID, InpPrefix);
      totalDeleted += deleted;
      chartsProcessed++;
      
      chartID = ChartNext(chartID);
   }
   
   Print("Processed ", chartsProcessed, " charts. Deleted ", totalDeleted, " objects with prefix '", InpPrefix, "'");
}

what is the temporary template for?
Claude2026-02-02 12:43 UTC

The template is the only mechanism to capture and restore the indicators, colors, objects, and visual settings. Without it, you'd have to:

  1. Enumerate indicators with ChartIndicatorsTotal() / ChartIndicatorName() — but you can only get the short name, not the parameters
  2. Enumerate all objects with ObjectsTotal() / ObjectName() / ObjectGet*() — tedious and error-prone
  3. Read every color and display property individually

The template captures all of that in one call.

I called it "temporary" because it's created solely for this recreation process, not as a permanent user template. But it's essential — it's doing the heavy lifting.

If you want it cleaned up afterward, you could delete the .tpl file manually or via a separate script, but MQL5 file operations are sandboxed to MQL5/Files/, while templates live in MQL5/Profiles/Templates/, so you can't delete it programmatically from MQL5.

struct PercentileStats
{
   int    longs;
   int    shorts;
   int    dojis;
   double minVal;
   double maxVal;

//+------------------------------------------------------------------+
//| Get percentile bucket index for a value                           |
//+------------------------------------------------------------------+
int GetBucket(double value, const double &sortedVals[], int total, int numBuckets)
{
   for(int i = 1; i <= numBuckets; i++)
   {
      int idx = (int)MathFloor((double)i / numBuckets * (total - 1));
      if(value <= sortedVals[idx])
         return i - 1;
   }
   return numBuckets - 1;
}

//+------------------------------------------------------------------+
//| Print statistics table                                            |
//+------------------------------------------------------------------+
void PrintStats(string title, PercentileStats &stats[], int numBuckets, int digits)
{
   Print("================================================================================");
   Print(title);
   Print("--------------------------------------------------------------------------------");
   Print("Percentile |     Range            | Longs  |  Shorts |  Dojis | Total | Long %");
   Print("--------------------------------------------------------------------------------");
   
   int totalLongs = 0, totalShorts = 0, totalDojis = 0;
   
   for(int i = 0; i < numBuckets; i++)
   {
      int pctLow = i * (100 / numBuckets);
      int pctHigh = (i + 1) * (100 / numBuckets);
      int bucketTotal = stats[i].longs + stats[i].shorts + stats[i].dojis;
      int directional = stats[i].longs + stats[i].shorts;
      double longPct = directional > 0 ? 100.0 * stats[i].longs / directional : 0;
      
      Print(StringFormat(" %2d - %3d%% | %s - %s | %5d  |  %5d  |  %4d  | %5d | %5.1f%%",
            pctLow, pctHigh,
            DoubleToString(stats[i].minVal, digits),
            DoubleToString(stats[i].maxVal, digits),
            stats[i].longs,
            stats[i].shorts,
            stats[i].dojis,
            bucketTotal,
            longPct));
      
      totalLongs += stats[i].longs;
      totalShorts += stats[i].shorts;
      totalDojis += stats[i].dojis;
   }
   
   Print("--------------------------------------------------------------------------------");
   int grandTotal = totalLongs + totalShorts + totalDojis;
   int directionalTotal = totalLongs + totalShorts;
   double overallLongPct = directionalTotal > 0 ? 100.0 * totalLongs / directionalTotal : 0;
   Print(StringFormat(" TOTAL     |                      | %5d  |  %5d  |  %4d  | %5d | %5.1f%%",

   ArrayResize(body, total);

   double bodyMin = DBL_MAX, bodyMax = 0;
   
   for(int i = 0; i < total; i++)
   {
      hiLo[i] = rates[i].high - rates[i].low;
      body[i] = MathAbs(rates[i].close - rates[i].open);
      
      if(hiLo[i] < hiLoMin) hiLoMin = hiLo[i];
      if(hiLo[i] > hiLoMax) hiLoMax = hiLo[i];
      if(body[i] < bodyMin) bodyMin = body[i];
      if(body[i] > bodyMax) bodyMax = body[i];
   }
   
   // Bucket arrays
   int hiLoLongs[], hiLoShorts[], hiLoDojis[];
   int bodyLongs[], bodyShorts[], bodyDojis[];
   
   ArrayResize(hiLoLongs, InpBuckets);  ArrayInitialize(hiLoLongs, 0);

   ArrayResize(hiLoDojis, InpBuckets);  ArrayInitialize(hiLoDojis, 0);

   ArrayResize(bodyShorts, InpBuckets); ArrayInitialize(bodyShorts, 0);
   ArrayResize(bodyDojis, InpBuckets);  ArrayInitialize(bodyDojis, 0);
   
   // Assign bars to buckets
   for(int i = 0; i < total; i++)
   {
      int hBucket = GetBucket(hiLo[i], hiLoMin, hiLoMax, InpBuckets);
      int bBucket = GetBucket(body[i], bodyMin, bodyMax, InpBuckets);
      
      if(rates[i].close > rates[i].open)
      {
         hiLoLongs[hBucket]++;
         bodyLongs[bBucket]++;
      }
      else if(rates[i].close < rates[i].open)
      {
         hiLoShorts[hBucket]++;
         bodyShorts[bBucket]++;
      }
      else
      {
         hiLoDojis[hBucket]++;
         bodyDojis[bBucket]++;
      }
   }
   
   // Print results
   PrintStats("HI-LO (High - Low)", hiLoLongs, hiLoShorts, hiLoDojis, hiLoMin, hiLoMax, InpBuckets, _Digits);
   PrintStats("BODY SIZE (|Close - Open|)", bodyLongs, bodyShorts, bodyDojis, bodyMin, bodyMax, InpBuckets, _Digits);
   
   Print("Analysis complete.");
}
