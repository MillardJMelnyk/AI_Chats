//+------------------------------------------------------------------+
void UpdateOpenTradeLiveState(const int bar, const double liveExitPrice)
{
   if(g_currentTrade.direction == 0 || g_currentTrade.entryBar < 0)
      return;

   if(bar >= ArraySize(BufferTradePL))
      TS_GrowStatsArrays(bar + 1, BufferTradePL, BufferRunningPL, BufferTradeCount, BufferWinPct,
                         BufferShortCount, BufferShortWinPct, BufferLongCount, BufferLongWinPct);

   //--- Compute in raw price
   double livePL = (liveExitPrice - g_currentTrade.entryPrice) * g_currentTrade.direction;
   double liveRunning = g_tradeStats.runningPL + livePL;

   //--- Store display values in Data Window buffers
   BufferTradePL[bar]    = PriceDeltaToReadout(livePL);
   BufferRunningPL[bar]  = PriceDeltaToReadout(liveRunning);
   BufferTradeCount[bar] = g_tradeStats.TotalTrades();
   BufferWinPct[bar]     = g_tradeStats.WinPct();
   BufferShortCount[bar] = g_tradeStats.shortCount;
   BufferShortWinPct[bar]= g_tradeStats.ShortWinPct();
   BufferLongCount[bar]  = g_tradeStats.longCount;
   BufferLongWinPct[bar] = g_tradeStats.LongWinPct();

   if(g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0)
   {
      int displayDigits   = ReadoutDigits();
      string readoutUnit  = ReadoutUnitText();
      string tip = StringFormat("Entry: %s | Exit: %s | P/L: %s %s | Running: %s %s",
                                DoubleToString(g_currentTrade.entryPrice, _Digits),
                                DoubleToString(liveExitPrice, _Digits),
                                DoubleToString(BufferTradePL[bar], displayDigits), readoutUnit,
                                DoubleToString(BufferRunningPL[bar], displayDigits), readoutUnit);
      ObjectSetString(0, g_currentTrade.arrowName, OBJPROP_TOOLTIP, tip);
   }
}

//+------------------------------------------------------------------+
int T2ESign(double v, double eps)
{
   if(v > eps)  return 1;
   if(v < -eps) return -1;
   return 0;
}

//+------------------------------------------------------------------+
int KCOuterState(const double lineValue,
                 const int lineToMidState,
                 const double kcUpper,
                 const double kcLower,
                 const double eps)
{
   if(lineToMidState == 0)
      return 0;
   if(lineValue == EMPTY_VALUE || kcUpper == EMPTY_VALUE || kcLower == EMPTY_VALUE)
      return 0;

   double selectedOuter = (lineToMidState > 0) ? kcUpper : kcLower;
   return T2ESign((lineValue - selectedOuter) * lineToMidState, eps);
}

int EffectiveAggressiveBlackout(const int bar)
{
   double blackout = InpSignalBlackout;
   if(bar >= 0 && bar < ArraySize(BufferDynBlackout) && BufferDynBlackout[bar] != EMPTY_VALUE)
      blackout = BufferDynBlackout[bar];

   blackout = blackout - 1.0;
   if(blackout < 0.0)
      blackout = 0.0;
   return (int)MathRound(blackout);
}

//+------------------------------------------------------------------+
bool IsWrongSideAggressiveTarget(const int bar,
                                 const double &close[],
                                 const double triggerSensPrice,
                                 const int finalizedDir,
                                 const ENUM_MOCK_PRICE_SOURCE finalizedMode)
{
   if(finalizedDir == 0 || bar < 0)
      return false;

   double base = EMPTY_VALUE;
   if(finalizedMode == MOCK_PRICE_AGG_TEMA)
      base = BufferTEMA[bar];
   else if(finalizedMode == MOCK_PRICE_AGG_EMA)
      base = BufferEMAofTEMA[bar];

   if(base == EMPTY_VALUE)
      return false;

   double target = base + finalizedDir * triggerSensPrice;
   if(finalizedDir > 0)
      return (close[bar] < target);
   return (close[bar] > target);
}

//+------------------------------------------------------------------+
//| Spike Detection: Check for KC breach + sufficient extension       |
//+------------------------------------------------------------------+
bool CheckSpikeCondition(const int bar,
                         const double barHigh,
                         const double barLow,
                         const double barClose,
                         const double kcUpper,
                         const double kcLower)
{
   if(!InpEnableSpikeDetection)
      return false;
   if(kcUpper == EMPTY_VALUE || kcLower == EMPTY_VALUE)
      return false;
   if(g_spikeActive)
      return false;  // Already tracking a spike
      
   //--- Calculate threshold as percentage of KC width
   double kcWidth = kcUpper - kcLower;
   if(kcWidth <= 0)
      return false;
   double dSpikeExtentMIN = InpSpikeKCWidthPct * kcWidth;
   
   //--- Calculate extension beyond KC (not bar range)
   double dOutsideKC_UP = (barHigh > kcUpper) ? (barHigh - kcUpper) : 0.0;
   double dOutsideKC_DN = (barLow < kcLower) ? (kcLower - barLow) : 0.0;
   
   bool breachUp = (dOutsideKC_UP > 0);
   bool breachDn = (dOutsideKC_DN > 0);
   
   if(!breachUp && !breachDn)
      return false;
      
   bool qualifiedUp = (dOutsideKC_UP >= dSpikeExtentMIN);
   bool qualifiedDn = (dOutsideKC_DN >= dSpikeExtentMIN);
   
   if(!qualifiedUp && !qualifiedDn)
      return false;
      
   //--- Determine spike direction (if both qualified, use larger extension)
   int spikeDir = 0;
   double kcLevel = 0.0;
   
   if(qualifiedUp && qualifiedDn)
   {
      if(dOutsideKC_UP >= dOutsideKC_DN)
      {
         spikeDir = 1;
         kcLevel = kcUpper;
      }
      else
      {
         spikeDir = -1;
         kcLevel = kcLower;
      }
   }
   else if(qualifiedUp)
   {
      spikeDir = 1;
      kcLevel = kcUpper;
   }
   else
   {
      spikeDir = -1;
      kcLevel = kcLower;
   }
   
   //--- Spike must align with current hunting direction (or no direction yet)
   if(iDir != 0 && spikeDir != iDir)
   {
      if(InpEnableDiagPrints)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " SPIKE_IGNORED_OPPOSITE_DIR spikeDir<",spikeDir,"> iDir<",iDir,">");
      return false;
   }
   
   //--- Snapshot pre-spike state before modifying anything
   SnapshotPreSpikeState();
   
   //--- Initialize spike tracking
   g_spikeActive = true;
   g_spikeDir = spikeDir;
   g_spikeBar = bar;
   g_spikeKCLevel = kcLevel;
   g_spikeExtremum = (spikeDir > 0) ? barHigh : barLow;
   g_spikeFirstRetraceExtent = 0.0;
   g_spikeFlipFlopCount = 0;
   g_spikeFalterConfirmed = false;
   g_spikeSecondSpurtDone = false;
   g_spikeLastMoveTimeMs = GetTickCount64();
   g_spikeLastPrice = barClose;
   g_spikeMovementDir = 0;
   g_spikePreSpurtExtremum = g_spikeExtremum;
   g_spikeFromPriorBar = false;
   
   if(InpEnableDiagPrints)
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " SPIKE_DETECTED bar<",bar,"> dir<",spikeDir,"> kcLevel<",kcLevel,
            "> extremum<",g_spikeExtremum,"> extension<",(spikeDir > 0 ? dOutsideKC_UP : dOutsideKC_DN),
            "> threshold<",dSpikeExtentMIN,"> kcWidth<",kcWidth,">");
   
   return true;
}

//+------------------------------------------------------------------+
//| Spike Intra-bar Processing: Track faltering, second spurt, pause  |
//+------------------------------------------------------------------+
ENUM_SIGNAL_TYPE ProcessSpikeIntraBar(const int bar,
                                       const double currentPrice,
                                       const double barHigh,
                                       const double barLow,
                                       const datetime barTime)
{
   if(!g_spikeActive)
      return SIGNAL_NONE;
      
   ulong currentTimeMs = GetTickCount64();
   ENUM_SIGNAL_TYPE spikeSignal = SIGNAL_NONE;
   
   //--- Update extremum if price extends further in spike direction
   bool newExtremum = false;
   if(g_spikeDir > 0 && barHigh > g_spikeExtremum)
   {
      g_spikeExtremum = barHigh;
      newExtremum = true;
   }
   else if(g_spikeDir < 0 && barLow < g_spikeExtremum)
   {
      g_spikeExtremum = barLow;
      newExtremum = true;
   }
   
   //--- Track price movement direction for flip-flop counting
   int priceDir = 0;
   if(currentPrice > g_spikeLastPrice)
      priceDir = 1;
   else if(currentPrice < g_spikeLastPrice)
      priceDir = -1;
      
   //--- Detect flip-flops (direction reversal then reversal back)
   if(priceDir != 0 && g_spikeMovementDir != 0 && priceDir != g_spikeMovementDir)
   {
      //--- Direction changed - this is half a flip-flop
      //--- Count full flip-flop when returning to original direction
      if((g_spikeDir > 0 && priceDir > 0) || (g_spikeDir < 0 && priceDir < 0))
      {
         //--- Moving back toward spike direction after moving away = flip-flop complete
         g_spikeFlipFlopCount++;
         
         if(InpEnableDiagPrints)
            Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                  " SPIKE_FLIPFLOP count<",g_spikeFlipFlopCount,"> bar<",bar,">");
      }
      
      //--- Track first retrace extent (for second spurt threshold)
      if(g_spikeFirstRetraceExtent == 0.0 && !g_spikeFalterConfirmed)
      {
         if(g_spikeDir > 0)
            g_spikeFirstRetraceExtent = g_spikeExtremum - barLow;
         else
            g_spikeFirstRetraceExtent = barHigh - g_spikeExtremum;
      }
   }
   
   //--- Update movement tracking
   if(priceDir != 0)
   {
      g_spikeMovementDir = priceDir;
      g_spikeLastMoveTimeMs = currentTimeMs;
   }
   g_spikeLastPrice = currentPrice;
   
   //--- Check for faltering confirmation
   if(!g_spikeFalterConfirmed && g_spikeFlipFlopCount >= InpSpikeFalterFlipFlops)
   {
      g_spikeFalterConfirmed = true;
      g_spikePreSpurtExtremum = g_spikeExtremum;
      
      if(InpEnableDiagPrints)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " SPIKE_FALTER_CONFIRMED bar<",bar,"> preSpurtExtremum<",g_spikePreSpurtExtremum,">");
   }
   
   //--- Check for second spurt (after faltering confirmed)
   if(g_spikeFalterConfirmed && !g_spikeSecondSpurtDone)
   {
      double spurtThreshold = g_spikeFirstRetraceExtent * InpSpikeSecondSpurtMultiple;
      double spurtDistance = 0.0;
      
      if(g_spikeDir > 0)
         spurtDistance = g_spikeExtremum - g_spikePreSpurtExtremum;
      else
         spurtDistance = g_spikePreSpurtExtremum - g_spikeExtremum;
         
      if(spurtDistance >= spurtThreshold && spurtThreshold > 0)
      {
         g_spikeSecondSpurtDone = true;
         g_spikeLastMoveTimeMs = currentTimeMs;  // Reset pause timer
         
         if(InpEnableDiagPrints)
            Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                  " SPIKE_SECOND_SPURT bar<",bar,"> spurtDistance<",spurtDistance,
                  "> threshold<",spurtThreshold,">");
      }
   }
   
   //--- Check for PAUSE trigger (after second spurt)
   if(g_spikeSecondSpurtDone)
   {
      ulong elapsedMs = currentTimeMs - g_spikeLastMoveTimeMs;
      
      if(elapsedMs >= (ulong)InpSpikePauseMs)
      {
         //--- PAUSE DETECTED - TRIGGER SIGNAL OPPOSITE TO SPIKE
         spikeSignal = (g_spikeDir > 0) ? SIGNAL_SHORT : SIGNAL_LONG;
         
         if(InpEnableDiagPrints)
            Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                  " SPIKE_PAUSE_TRIGGER bar<",bar,"> elapsedMs<",elapsedMs,
                  "> signal<",SignalToText(spikeSignal),">");
                  
         ResetSpikeState();
         return spikeSignal;
      }
   }
   
   //--- Check for JUST-JUMP trigger (immediate retrace without pause)
   double kcToExtremum = MathAbs(g_spikeExtremum - g_spikeKCLevel);
   double justJumpThreshold = kcToExtremum * InpSpikeJustJumpPct;
   double currentRetrace = 0.0;
   
   if(g_spikeDir > 0)
      currentRetrace = g_spikeExtremum - currentPrice;
   else
      currentRetrace = currentPrice - g_spikeExtremum;
      
   if(currentRetrace >= justJumpThreshold && justJumpThreshold > 0)
   {
      //--- JUST-JUMP - Price retraced enough, trigger immediately
      spikeSignal = (g_spikeDir > 0) ? SIGNAL_SHORT : SIGNAL_LONG;
      
      if(InpEnableDiagPrints)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " SPIKE_JUSTJUMP_TRIGGER bar<",bar,"> retrace<",currentRetrace,
               "> threshold<",justJumpThreshold,"> signal<",SignalToText(spikeSignal),">");
               
      ResetSpikeState();
      return spikeSignal;
   }
   
   return SIGNAL_NONE;
}

//+------------------------------------------------------------------+
//| Check for spike reset condition (KC cross-back)                   |
//+------------------------------------------------------------------+
bool CheckSpikeReset(const int bar,
                     const double currentPrice,
                     const double kcUpper,
                     const double kcLower)
{
   if(!g_spikeActive)
      return false;
      
   //--- Check if price crossed back through the same KC line in opposite direction
   bool resetCondition = false;
   
   if(g_spikeDir > 0)
   {
      //--- Up spike: reset if price drops below KC upper (crosses back down)
      if(currentPrice < g_spikeKCLevel)
         resetCondition = true;
   }
   else
   {
      //--- Down spike: reset if price rises above KC lower (crosses back up)
      if(currentPrice > g_spikeKCLevel)
         resetCondition = true;
   }
   
   if(resetCondition)
   {
      if(InpEnableDiagPrints)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " SPIKE_RESET bar<",bar,"> restoring pre-spike state");
               
      RestorePreSpikeState();
      return true;
   }
   
   return false;
}

//| Historical Spike Detection for closed bars                        |
//+------------------------------------------------------------------+
ENUM_SIGNAL_TYPE CheckSpikeHistorical(const int bar,
                                       const double &high[],
                                       const double &low[],
                                       const double kcUpper,
                                       const double kcLower,
//                                       const double highNext,
//                                       const double lowNext,
                                       double &signalPrice)
{
   signalPrice = 0.0;
   
   if(!InpEnableSpikeDetection)
      return SIGNAL_NONE;
   if(kcUpper == EMPTY_VALUE || kcLower == EMPTY_VALUE)
      return SIGNAL_NONE;
      
   //--- Calculate threshold as percentage of KC width
   double kcWidth = kcUpper - kcLower;
   if(kcWidth <= 0)
      return SIGNAL_NONE;
   double dSpikeExtentMIN = InpSpikeKCWidthPct * kcWidth;
   
   //--- Calculate extension beyond KC for this bar
   double dOutsideKC_UP = (high[bar] > kcUpper) ? (high[bar] - kcUpper) : 0.0;
   double dOutsideKC_DN = (low[bar] < kcLower) ? (kcLower - low[bar]) : 0.0;
//   double dInsideKC_UP = (kcUpper > low[bar]) ? (kcUpper - low[bar]) : 0.0;
//   double dInsideKC_DN = (kcLower < high[bar]) ? (high[bar] - kcLower) : 0.0;
//
   bool qualifiedUp = (dOutsideKC_UP >= dSpikeExtentMIN && low[bar] < kcUpper);
   bool qualifiedDn = (dOutsideKC_DN >= dSpikeExtentMIN && high[bar] > kcLower);
   
   if(!qualifiedUp && !qualifiedDn)
      return SIGNAL_NONE;
      
   //--- Determine spike direction (if both qualified, use larger extension)
   int spikeDir = 0;
   double kcLevel = 0.0;
   double barExtremum = 0.0;
   
   if(qualifiedUp && qualifiedDn)
   {
      if(dOutsideKC_UP >= dOutsideKC_DN)
      {
         spikeDir = 1;
         kcLevel = kcUpper;
         barExtremum = high[bar];
      }
      else
      {
         spikeDir = -1;
         kcLevel = kcLower;
         barExtremum = low[bar];
      }
   }
   else if(qualifiedUp)
   {
      spikeDir = 1;
      kcLevel = kcUpper;
      barExtremum = high[bar];
   }
   else
   {
      spikeDir = -1;
      kcLevel = kcLower;
      barExtremum = low[bar];
   }
   
   //--- Spike must align with current hunting direction (or no direction yet)
   if(iDir != 0 && spikeDir != iDir)
   {
      if(InpEnableDiagPrints)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " SPIKE_HISTORICAL_IGNORED_OPPOSITE_DIR bar<",bar,"> spikeDir<",spikeDir,"> iDir<",iDir,">");
      return SIGNAL_NONE;
   }
   
   //--- Find maximum extremum between this bar and next bar
   double maxExtremum = barExtremum;
//   if(highNext != EMPTY_VALUE && lowNext != EMPTY_VALUE)
//   {
//      if(spikeDir > 0 && highNext > maxExtremum)
//         maxExtremum = highNext;
//      else if(spikeDir < 0 && lowNext < maxExtremum)
//         maxExtremum = lowNext;
//   }
//   
   //--- Calculate signal price using historical multiplier and just-jump percentage
   double spikeRange = MathAbs(maxExtremum - kcLevel);
   double retraceAmount = spikeRange * InpSpikeJustJumpPct * InpSpikeHistSignalMultiplier;
//   double retraceAmount = MathAbs(maxExtremum - kcLevel) - (InpSpikeHistSignalMultiplier * InpSpikeJustJumpPct);
   
   if(spikeDir > 0)
      signalPrice = maxExtremum - retraceAmount;
   else
      signalPrice = maxExtremum + retraceAmount;
   
   //--- Signal is opposite to spike direction
   ENUM_SIGNAL_TYPE signal = (spikeDir > 0) ? SIGNAL_SHORT : SIGNAL_LONG;
   
   if(InpEnableDiagPrints)
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " SPIKE_HISTORICAL_DETECTED bar<",bar,"> spikeDir<",spikeDir,
            "> maxExtremum<",maxExtremum,"> signalPrice<",signalPrice,
            "> dSpikeExtentMIN<",dSpikeExtentMIN,"> dOutsideKC_UP<",dOutsideKC_UP,"> dOutsideKC_DN<",dOutsideKC_DN,
            "> high[bar]<",high[bar],"> low[bar]<",low[bar],
            "> signal<",SignalToText(signal),">");

   return signal;
}

            " Failed to create KC handle. err=", GetLastError());
      return(INIT_FAILED);
   }
   
   g_arrowCfg.SetDefaults();
   g_labelCfg.SetDefaults();
   g_arrowCfg.longColor  = clrLime;
   g_arrowCfg.shortColor = clrRed;
   g_labelCfg.prefix = "T/E Trigger";
   g_maxPeriodsBack = MathMax(InpTEMAPeriod, InpEMAPeriod);
   g_maxPeriodsBack = MathMax(g_maxPeriodsBack, InpKCPeriod);
   g_maxPeriodsBack = MathMax(g_maxPeriodsBack, InpKCATRPeriod);
   g_maxPeriodsBack += (3 + InpOpenRefOffset);
   
   if(g_effectiveTradeStartDate == 0)
   {
      g_effectiveTradeStartDate = iTime(_Symbol, PERIOD_CURRENT, 0);
      Print("");
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " InpTradeStartDate=0; using current bar time<",
            TimeToString(g_effectiveTradeStartDate, TIME_DATE|TIME_MINUTES), ">");
   }
   
   if(InpTradeStopDate <= 0)
      g_tradeStopDT = D'3000.12.31 23:59';
   else
      g_tradeStopDT = InpTradeStopDate;
      
   if(InpEnableDiagPrints)
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__," InpTradeStartDate<",InpTradeStartDate,
            "> InpTradeStopDate<",InpTradeStopDate,"> g_tradeStopDT<",g_tradeStopDT,">");
            
   return(INIT_SUCCEEDED);
}

void OnDeinit(const int reason)
{
   ObjectsDeleteAll(0, SIGNAL_PREFIX+"_");
   if(iHndlKC != INVALID_HANDLE)
      IndicatorRelease(iHndlKC);
   if(iHndlEMAofTEMA != INVALID_HANDLE)
      IndicatorRelease(iHndlEMAofTEMA);
   if(iHndlTEMA != INVALID_HANDLE)
      IndicatorRelease(iHndlTEMA);

   //--- Clean up trade simulation objects
   if(InpEnableTradeSim)
   {
      TS_Cleanup();
   }
}

//+------------------------------------------------------------------+
//| Custom indicator iteration function                              |
//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   if(rates_total <= g_maxPeriodsBack)
      return(0);

   int start = (prev_calculated == 0) ? g_maxPeriodsBack : prev_calculated - 1;
   if(start < 1)
      start = 1;

   if(prev_calculated == 0)
   {
      ResetAllStateAndBuffers();
      g_lastExportedBar = -1;
      InitCsvDebugFile();
   }

   int copyStart = MathMax(0, start - 3);
   int copyCount = rates_total - copyStart;

   double temaTail[];
   double emaTail[];
   double kcUpperTail[];
   double kcMiddleTail[];
   double kcLowerTail[];
   ArrayResize(temaTail, copyCount);
   ArrayResize(emaTail, copyCount);
   ArrayResize(kcUpperTail, copyCount);
   ArrayResize(kcMiddleTail, copyCount);
   ArrayResize(kcLowerTail, copyCount);
   ArraySetAsSeries(temaTail, false);
   ArraySetAsSeries(emaTail, false);
   ArraySetAsSeries(kcUpperTail, false);
   ArraySetAsSeries(kcMiddleTail, false);
   ArraySetAsSeries(kcLowerTail, false);

   int temaCopied = CopyBuffer(iHndlTEMA, 0, 0, copyCount, temaTail);
   int emaCopied  = CopyBuffer(iHndlEMAofTEMA, 0, 0, copyCount, emaTail);
   int kcUpperCopied  = CopyBuffer(iHndlKC, 0, 0, copyCount, kcUpperTail);
   int kcMiddleCopied = CopyBuffer(iHndlKC, 1, 0, copyCount, kcMiddleTail);
   int kcLowerCopied  = CopyBuffer(iHndlKC, 2, 0, copyCount, kcLowerTail);
   
   if(temaCopied != copyCount || emaCopied != copyCount ||
      kcUpperCopied != copyCount || kcMiddleCopied != copyCount || kcLowerCopied != copyCount)
   {
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " CopyBuffer failed. temaCopied=",temaCopied,
            " emaCopied=",emaCopied,
            " kcUpperCopied=",kcUpperCopied,
            " kcMiddleCopied=",kcMiddleCopied,
            " kcLowerCopied=",kcLowerCopied,
            " expected=",copyCount,
            " err=",GetLastError());
      return(prev_calculated);
   }

   ArrayResize(BufferAggExecPrice, rates_total);

   double pointSize = MathPow(10.0, -_Digits);
   double eps = pointSize * 0.1;
   double pipSize = pointSize * 10.0;
   //--- Convert user inputs from readout units to raw price (once)
   double triggerSensInputPrice = ReadoutToPrice(InpTriggerSensitivity, pipSize);
   double flatSlopePrice   = ReadoutToPrice(InpFlatSlopePerBar, pipSize);
   double arrowOffsetPrice = ReadoutToPrice(InpObjectArrowOffset, pipSize);
   ENUM_TS_READOUT_MODE tsReadoutMode = TSReadoutMode();

   //--- Copy indicator values to buffers and calculate dynamic blackout
   for(int iKX = 0; iKX < copyCount; iKX++)
   {
      int iIX = copyStart + iKX;
      BufferTEMA[iIX] = temaTail[iKX];
      BufferEMAofTEMA[iIX] = emaTail[iKX];
      BufferKCUpper[iIX] = kcUpperTail[iKX];
      BufferKCMiddle[iIX] = kcMiddleTail[iKX];
      BufferKCLower[iIX] = kcLowerTail[iKX];
      BufferTemaEmaSpread[iIX] = PriceDeltaToReadout(BufferTEMA[iIX] - BufferEMAofTEMA[iIX]);
   
      int activeBlackout = InpSignalBlackout;
      if(InpUseDynamicBlackout && 
         BufferKCUpper[iIX] != EMPTY_VALUE && 
         BufferKCLower[iIX] != EMPTY_VALUE && 
         BufferKCMiddle[iIX] != EMPTY_VALUE)
      {
         double currWidth = MathAbs(BufferKCUpper[iIX] - BufferKCLower[iIX]);
         int lb = MathMin(InpKCWidthLookback, iIX + 1);
         double widthSum = 0.0;
         int widthCnt = 0;
         int from = iIX - lb + 1;
         if(from < 0) from = 0;
         for(int j = from; j <= iIX; j++)
         {
            if(BufferKCUpper[j] == EMPTY_VALUE || BufferKCLower[j] == EMPTY_VALUE)
               continue;
            widthSum += MathAbs(BufferKCUpper[j] - BufferKCLower[j]);
            widthCnt++;
         }
         double baselineWidth = (widthCnt > 0) ? (widthSum / widthCnt) : currWidth;
         double widthRatio = (baselineWidth > 0.0) ? (currWidth / baselineWidth) : 1.0;
         double slopePerBar = 0.0;
         if(iIX >= InpKCMidSlopePeriod && BufferKCMiddle[iIX - InpKCMidSlopePeriod] != EMPTY_VALUE)
            slopePerBar = (BufferKCMiddle[iIX] - BufferKCMiddle[iIX - InpKCMidSlopePeriod]) / InpKCMidSlopePeriod;
         
         bool isNarrow = (widthRatio < InpNarrowWidthRatio);
         bool isFlat = (MathAbs(slopePerBar) <= flatSlopePrice);
         if(isNarrow && isFlat)
         {
            double extra = (InpNarrowWidthRatio - widthRatio) * InpBlackoutScaleK;
            if(extra < 0.0) extra = 0.0;
            activeBlackout = InpSignalBlackout + (int)MathRound(extra);
            if(activeBlackout > InpBlackoutMax)
               activeBlackout = InpBlackoutMax;
         }
         BufferKCWidthRatio[iIX] = widthRatio;
      }
      BufferDynBlackout[iIX] = activeBlackout;
   }
   
   bool isNewBar = (rates_total != prev_calculated);
   bool isAggressive = IsAggressiveMode();
   int liveBar = rates_total - 1;
   
   if(isNewBar && g_provisionalSignal)
   {
      g_lastSignalBar = liveBar - 1;
      if(isAggressive && !g_aggrRecoveryActive && g_currentTrade.direction != 0)
         ArmAggressiveFlipCheck(g_lastSignalBar, g_currentTrade.direction, InpMockTradePriceSource);
      g_provisionalSignal = false;
      g_isCocked = false;
      g_cockedDir = 0;
      iDir = g_currentTrade.direction;
      g_crossBar = g_lastSignalBar;
   }

   //--- Handle spike that carried over from prior bar
   if(isNewBar && g_spikeActive && g_spikeBar < liveBar)
   {
      g_spikeFromPriorBar = true;
   }

   if(BufferTEMA[liveBar] != EMPTY_VALUE && BufferEMAofTEMA[liveBar] != EMPTY_VALUE)
      BufferTemaEmaSpread[liveBar] = PriceDeltaToReadout(BufferTEMA[liveBar] - BufferEMAofTEMA[liveBar]);

   if(InpEnableTradeSim)
   {
      double liveExitPrice = GetMockTradePrice(liveBar, close);
      UpdateOpenTradeLiveState(liveBar, liveExitPrice);
   }

   //--- Signals processing loop
   int loopEnd = rates_total;
   
   //--- Determine signal loop start: warmup bars before start date
   int signalStart = start;
   if(time[start] < g_effectiveTradeStartDate)
   {
      int warmup = 2 + InpOpenRefOffset;
      int startDateBar = -1;
      for(int i = start; i < rates_total; i++)
      {
         if(time[i] >= g_effectiveTradeStartDate)
         {
            startDateBar = i;
            break;
         }
      }
      signalStart = MathMax(start, startDateBar - warmup);
   }

   //+------------------------------------------------------------------+
   //| Signals process main loop                                        |
   //+------------------------------------------------------------------+
   for(int iIX = signalStart; iIX < loopEnd; iIX++)
   {
      //--- Stop processing if past stop date
      if(time[iIX] > g_tradeStopDT)
      {
         if(InpEnableDiagPrints)
            Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                  " DIAG STOP_DATE_BREAK iIX<",iIX,"> time[iIX]<",TimeToString(time[iIX],TIME_DATE|TIME_MINUTES),
                  "> g_tradeStopDT<",TimeToString(g_tradeStopDT,TIME_DATE|TIME_MINUTES),
                  "> signalStart<",signalStart,">");
         break;
      }
      
      if(prev_calculated == 0 || iIX >= prev_calculated)
      {
         BufferArrowBuy[iIX] = EMPTY_VALUE;
         BufferArrowSell[iIX] = EMPTY_VALUE;
      }
            
      int prevT2E = T2ESign(BufferTEMA[iIX - 1] - BufferEMAofTEMA[iIX - 1], eps);
      int currT2E = T2ESign(BufferTEMA[iIX] - BufferEMAofTEMA[iIX], eps);

      if(prevT2E == 0) prevT2E = iDir;
      if(currT2E == 0) currT2E = prevT2E;

      int prevT2KCM = 0;
      int currT2KCM = 0;
      int prevE2KCM = 0;
      int currE2KCM = 0;
      if(BufferTEMA[iIX - 1] != EMPTY_VALUE && BufferKCMiddle[iIX - 1] != EMPTY_VALUE)
         prevT2KCM = T2ESign(BufferTEMA[iIX - 1] - BufferKCMiddle[iIX - 1], eps);
      if(BufferTEMA[iIX] != EMPTY_VALUE && BufferKCMiddle[iIX] != EMPTY_VALUE)
         currT2KCM = T2ESign(BufferTEMA[iIX] - BufferKCMiddle[iIX], eps);
      if(BufferEMAofTEMA[iIX - 1] != EMPTY_VALUE && BufferKCMiddle[iIX - 1] != EMPTY_VALUE)
         prevE2KCM = T2ESign(BufferEMAofTEMA[iIX - 1] - BufferKCMiddle[iIX - 1], eps);
      if(BufferEMAofTEMA[iIX] != EMPTY_VALUE && BufferKCMiddle[iIX] != EMPTY_VALUE)
         currE2KCM = T2ESign(BufferEMAofTEMA[iIX] - BufferKCMiddle[iIX], eps);

      int prevT2KC = KCOuterState(BufferTEMA[iIX - 1], prevT2KCM, BufferKCUpper[iIX - 1], BufferKCLower[iIX - 1], eps);
      int currT2KC = KCOuterState(BufferTEMA[iIX], currT2KCM, BufferKCUpper[iIX], BufferKCLower[iIX], eps);
      int prevE2KC = KCOuterState(BufferEMAofTEMA[iIX - 1], prevE2KCM, BufferKCUpper[iIX - 1], BufferKCLower[iIX - 1], eps);
      int currE2KC = KCOuterState(BufferEMAofTEMA[iIX], currE2KCM, BufferKCUpper[iIX], BufferKCLower[iIX], eps);

      //=================================================================
      // SPIKE DETECTION AND PROCESSING (aggressive mode only)
      //=================================================================
      ENUM_SIGNAL_TYPE spikeSignal = SIGNAL_NONE;
      double spikeSignalPrice = 0.0;
      
      if(InpEnableSpikeDetection && isAggressive)
      {
         bool isLiveBar = (iIX == liveBar);
         
         if(isLiveBar)
         {
            //--- Live bar: intra-bar spike tracking
            
            //--- Check for spike reset condition first
            if(g_spikeActive)
            {
               CheckSpikeReset(iIX, close[iIX], BufferKCUpper[iIX], BufferKCLower[iIX]);
            }
            
            //--- Check for new spike condition (if not already tracking one)
            if(!g_spikeActive)
            {
               CheckSpikeCondition(iIX, high[iIX], low[iIX], close[iIX],
                                  BufferKCUpper[iIX], BufferKCLower[iIX]);
            }
            
            //--- Process active spike (intra-bar tracking)
            if(g_spikeActive)
            {
               spikeSignal = ProcessSpikeIntraBar(iIX, close[iIX], high[iIX], low[iIX], time[iIX]);
               if(spikeSignal != SIGNAL_NONE)
                  spikeSignalPrice = close[iIX];  // Use current price for live signals
            }
         }
         else
         {
            //--- Historical bar: use hindsight approximation
            double highNext = (iIX + 1 < rates_total) ? high[iIX + 1] : EMPTY_VALUE;
            double lowNext = (iIX + 1 < rates_total) ? low[iIX + 1] : EMPTY_VALUE;
            
            spikeSignal = CheckSpikeHistorical(iIX, high, low,
                                               BufferKCUpper[iIX], BufferKCLower[iIX],
//                                               highNext, lowNext,
                                               spikeSignalPrice);
         }
      }

      //=================================================================
      // EXISTING MA-BASED COCKING AND TRIGGER LOGIC
      //=================================================================
      
      // Cocking logic: only run on closed bars (not the nascent live bar)
      if(isNewBar && iIX < rates_total - 1)
      {
         if(iDir != 0)
         {
            bool candidateFound = false;
            bool cockedThisBar = false;
            for(int d = 1; d <= 2; d++)
            {
               int cand = iIX - d;
               if(cand <= g_crossBar || cand < 1 || cand + 1 >= rates_total)
                  continue;
               candidateFound = true;
               int refBar = MathMax(0, cand - InpOpenRefOffset);
               double refOpen = open[refBar];

               g_extremeBar = cand;
               g_extremeOpenRefBar = refBar;
               g_extremeOpen = refOpen;
               g_extremePrice = (iDir > 0) ? high[cand] : low[cand];
               if((iDir > 0 && close[iIX] < refOpen) || (iDir < 0 && close[iIX] > refOpen))
               {
                  g_isCocked = true;
                  g_cockedDir = (iDir > 0) ? -1 : 1;
                  g_cockLevel = refOpen;
                  cockedThisBar = true;

                  if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT)
                     Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                           " DIAG#3 COCKED iIX<",iIX,"> time<",TimeToString(time[iIX],TIME_DATE|TIME_MINUTES),"> iDir<",iDir,"> g_cockedDir<",g_cockedDir,
                           "> g_cockLevel<",g_cockLevel,"> close[iIX]<",close[iIX],
                           "> refOpen<",refOpen,"> cand<",cand,">");

                  break;
               }
            }
            g_pendingConfirm = (candidateFound && !cockedThisBar);
         }
      }

      bool rawCrossUp = (prevT2E < 0 && currT2E > 0);
      bool rawCrossDn = (prevT2E > 0 && currT2E < 0);
      bool rawCross = (rawCrossUp || rawCrossDn);

      bool revertedWas = g_revertedMode;
      bool revertedSetThisBar = false;
      if(isAggressive && rawCross)
      {
         int rawCrossDir = rawCrossUp ? 1 : -1;
         if(g_lastAggRawCrossDir != 0 && rawCrossDir != g_lastAggRawCrossDir)
         {
            g_revertedMode = true;
            revertedSetThisBar = true;
         }
         g_lastAggRawCrossDir = rawCrossDir;
      }

      double effectiveTriggerSensReadout = g_revertedMode ? 0.0 : InpTriggerSensitivity;
      double effectiveTriggerSensPrice = ReadoutToPrice(effectiveTriggerSensReadout, pipSize);
      double instantSpread = 0.0;
      if(BufferTEMA[iIX] != EMPTY_VALUE && BufferEMAofTEMA[iIX] != EMPTY_VALUE)
         instantSpread = MathAbs(BufferTEMA[iIX] - BufferEMAofTEMA[iIX]);
      bool qualifiedCross = (rawCross && (instantSpread >= effectiveTriggerSensPrice));

      if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT && rawCross)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " DIAG#4 CROSS iIX<",iIX,"> rawCrossUp<",rawCrossUp,"> rawCrossDn<",rawCrossDn,
               "> qualifiedCross<",qualifiedCross,"> revertedMode<",g_revertedMode,
               "> prevT2E<",prevT2E,"> currT2E<",currT2E,
               "> instantSpread<",instantSpread,"> effectiveSens<",effectiveTriggerSensPrice,
               "> g_isCocked<",g_isCocked,"> g_cockedDir<",g_cockedDir,">");
               
      if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT && revertedSetThisBar)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " DIAG#4A REVERTED_MODE_ON iIX<",iIX,"> revertedWas<",revertedWas,
               "> revertedNow<",g_revertedMode,"> rawCrossUp<",rawCrossUp,"> rawCrossDn<",rawCrossDn,
               "> lastAggRawCrossDir<",g_lastAggRawCrossDir,">");
               
      if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT && rawCross && !qualifiedCross)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " DIAG#4B CROSS_REJECTED iIX<",iIX,"> instantSpread<",instantSpread,
               "> requiredSens<",effectiveTriggerSensPrice,"> revertedMode<",g_revertedMode,">");

      BufferTriggerThreshold[iIX] = effectiveTriggerSensReadout;

      if(iDir == 0 && currT2E != 0 && isNewBar && iIX < rates_total - 1)
         ResetCycleForDirection(currT2E, iIX, open, high, low);

      ENUM_SIGNAL_TYPE signal = SIGNAL_NONE;
      ENUM_SIGNAL_TYPE signalCandidate = SIGNAL_NONE;
      bool forceCommittedSignal = false;
      bool isBackstopSignal = false;
      bool isSpikeTriggered = false;
      bool useAggressive = (isAggressive && !g_aggrRecoveryActive);
      bool conditionMet = false;
      
      if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT && rawCross)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " DIAG#4C MODE_SELECT iIX<",iIX,"> isAggressive<",isAggressive,
               "> g_aggrRecoveryActive<",g_aggrRecoveryActive,"> useAggressive<",useAggressive,
               "> g_triggerPending<",g_triggerPending,">");

      //=================================================================
      // SPIKE SIGNAL TAKES PRIORITY - merges into existing signal flow
      //=================================================================
      if(spikeSignal != SIGNAL_NONE)
      {
         signalCandidate = spikeSignal;
         isSpikeTriggered = true;
         forceCommittedSignal = true;  // Spike signals bypass provisional logic
         
         if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT)
            Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                  " SPIKE_SIGNAL_CANDIDATE iIX<",iIX,"> signal<",SignalToText(spikeSignal),
                  "> signalPrice<",spikeSignalPrice,">");
      }
      
      //=================================================================
      // EXISTING MA-BASED TRIGGERING LOGIC (only if no spike signal)
      //=================================================================
      if(signalCandidate == SIGNAL_NONE)
      {
         if(useAggressive)
         {
            if(g_isCocked)
            {
               double liveClose = close[iIX];
               double temaVal = BufferTEMA[iIX];
               double emaVal = BufferEMAofTEMA[iIX];

               if(InpMockTradePriceSource == MOCK_PRICE_AGG_TEMA)
               {
                  if(g_cockedDir == -1)
                     conditionMet = (liveClose < temaVal - effectiveTriggerSensPrice);
                  else
                     conditionMet = (liveClose > temaVal + effectiveTriggerSensPrice);
               }
               else if(InpMockTradePriceSource == MOCK_PRICE_AGG_EMA)
               {
                  if(g_cockedDir == -1)
                     conditionMet = (liveClose < emaVal - effectiveTriggerSensPrice);
                  else
                     conditionMet = (liveClose > emaVal + effectiveTriggerSensPrice);
               }
               else
               {
                  if(g_cockedDir == -1)
                     conditionMet = (liveClose < temaVal && liveClose < emaVal - effectiveTriggerSensPrice);
                  else
                     conditionMet = (liveClose > temaVal && liveClose > emaVal + effectiveTriggerSensPrice);
               }

               if(conditionMet)
                  signalCandidate = (g_cockedDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
            }
         }
         else
         {
            if(rawCross)
            {
               if(g_isCocked)
               {
                  if(rawCrossDn && g_cockedDir == -1)
                  {
                     g_triggerPending = true;
                     g_triggerDir = -1;
                     g_triggerCrossBar = iIX;

                     if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT)
                        Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                              " DIAG#5 TRIGGER_PENDING iIX<",iIX,"> dir<SHORT> g_triggerCrossBar<",iIX,
                              "> g_isCocked<",g_isCocked,"> g_cockedDir<",g_cockedDir,">");
                  }
                  else if(rawCrossUp && g_cockedDir == 1)
                  {
                     g_triggerPending = true;
                     g_triggerDir = 1;
                     g_triggerCrossBar = iIX;

                     if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT)
                        Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                              " DIAG#5 TRIGGER_PENDING iIX<",iIX,"> dir<LONG> g_triggerCrossBar<",iIX,
                              "> g_isCocked<",g_isCocked,"> g_cockedDir<",g_cockedDir,">");
                  }
                  else
                  {
                     g_triggerPending = false;
                     g_triggerDir = 0;
                     g_triggerCrossBar = -1;
                  }
               }
               else
               {
                  g_triggerPending = false;
                  g_triggerDir = 0;
                  g_triggerCrossBar = -1;
               }
            }

            if(g_triggerPending)
            {
               bool wrongSide = ((g_triggerDir == 1 && currT2E < 0) || (g_triggerDir == -1 && currT2E > 0));
               if(wrongSide)
               {
                  if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT)
                     Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                           " DIAG#6A PENDING_CLEARED_WRONG_SIDE iIX<",iIX,"> g_triggerDir<",g_triggerDir,
                           "> currT2E<",currT2E,"> g_triggerCrossBar<",g_triggerCrossBar,">");
                  g_triggerPending = false;
                  g_triggerDir = 0;
                  g_triggerCrossBar = -1;
               }
               else
               {
                  double cumulativeSpread = 0.0;
                  int sumFrom = MathMax(0, g_triggerCrossBar);
                  if(sumFrom <= iIX)
                  {
                     for(int s = sumFrom; s <= iIX; s++)
                     {
                        if(BufferTEMA[s] == EMPTY_VALUE || BufferEMAofTEMA[s] == EMPTY_VALUE)
                           continue;
                        cumulativeSpread += MathAbs(BufferTEMA[s] - BufferEMAofTEMA[s]);
                     }
                  }

                  if(InpEnableDiagPrints && time[iIX] >= g_effectiveTradeStartDate && time[iIX] <= g_tradeStopDT)
                     Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
