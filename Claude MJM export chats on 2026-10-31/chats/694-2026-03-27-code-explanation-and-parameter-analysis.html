"""Detect composite-body SCR signals and analyze them with opposite-signal boundaries.

SCR definition used here:
- Prior context is the two-bar composite made from bars t-2 and t-1.
- Current bar t must move in the opposite direction of the composite.
- Current body must cancel enough of the prior composite body.
- Required cancel percent starts at a base threshold and increases as the
  prior composite becomes more waffly (wick-heavy).
"""
from __future__ import annotations

import argparse
from dataclasses import dataclass

import numpy as np
import pandas as pd


@dataclass(frozen=True)
class CompositeBar:
    start_ix: int
    end_ix: int
    direction: int
    body_top: float
    body_bot: float
    body_size: float
    high: float
    low: float
    total_range: float
    wick_total: float
    body_percent: float
    wick_percent: float


def load(path: str) -> pd.DataFrame:
    df = pd.read_csv(path)
    df["Datetime"] = pd.to_datetime(df["Datetime"], format="%Y.%m.%d %H:%M:%S", errors="coerce")
    for col in ["Open", "High", "Low", "Close", "TEMA9", "EMA4_TEMA9", "EMA2_HLCC4"]:
        df[col] = pd.to_numeric(df[col], errors="coerce")
    return df.dropna(subset=["Datetime", "Open", "High", "Low", "Close"]).reset_index(drop=True)


def bar_direction(open_val: float, close_val: float) -> int:
    if close_val > open_val:
        return 1
    if close_val < open_val:
        return -1
    return 0


def build_composite(open_: np.ndarray, high: np.ndarray, low: np.ndarray, close: np.ndarray, end_ix: int) -> CompositeBar | None:
    start_ix = end_ix - 1
    if start_ix < 0:
        return None

    composite_dir = bar_direction(open_[start_ix], close[end_ix])
    if composite_dir == 0:
        return None

    body_top = max(open_[start_ix], close[end_ix])
    body_bot = min(open_[start_ix], close[end_ix])
    body_size = body_top - body_bot
    composite_high = max(high[start_ix], high[end_ix])
    composite_low = min(low[start_ix], low[end_ix])
    total_range = composite_high - composite_low
    if total_range <= 0.0:
        return None

    wick_total = max(0.0, total_range - body_size)
    body_percent = body_size / total_range
    wick_percent = wick_total / total_range

    return CompositeBar(
        start_ix=start_ix,
        end_ix=end_ix,
        direction=composite_dir,
        body_top=body_top,
        body_bot=body_bot,
        body_size=body_size,
        high=composite_high,
        low=composite_low,
        total_range=total_range,
        wick_total=wick_total,
        body_percent=body_percent,
        wick_percent=wick_percent,
    )


def calc_cancel_amount(composite: CompositeBar, curr_body_top: float, curr_body_bot: float) -> float:
    if composite.direction > 0:
        cancel_amount = composite.body_top - curr_body_bot
    else:
        cancel_amount = curr_body_top - composite.body_bot
    return max(0.0, min(cancel_amount, composite.body_size))


def detect_scrs(
    df: pd.DataFrame,
    base_cancel_pct: float = 0.50,
    waffle_penalty_scale: float = 0.25,
    min_composite_body: float = 0.01,
) -> pd.DataFrame:
    open_ = df["Open"].to_numpy(dtype=float)
    high = df["High"].to_numpy(dtype=float)
    low = df["Low"].to_numpy(dtype=float)
    close = df["Close"].to_numpy(dtype=float)
    times = df["Datetime"].to_numpy()
    n = len(df)

    rows: list[dict[str, object]] = []
    for i in range(2, n):
        composite = build_composite(open_, high, low, close, i - 1)
        if composite is None or composite.body_size < min_composite_body:
            continue

        curr_dir = bar_direction(open_[i], close[i])
        if curr_dir == 0 or curr_dir == composite.direction:
            continue

        curr_body_top = max(open_[i], close[i])
        curr_body_bot = min(open_[i], close[i])
        curr_body_size = curr_body_top - curr_body_bot
        if curr_body_size <= 0.0:
            continue

        cancel_amount = calc_cancel_amount(composite, curr_body_top, curr_body_bot)
        cancel_pct = cancel_amount / composite.body_size
        required_cancel_pct = base_cancel_pct + composite.wick_percent * waffle_penalty_scale
        if cancel_pct < required_cancel_pct:
            continue

        signal_dir = "long" if curr_dir > 0 else "short"
        rows.append(
            {
                "signal_ix": i,
                "time": pd.Timestamp(times[i]).strftime("%Y.%m.%d %H:%M:%S"),
                "direction": signal_dir,
                "entry_price": float(close[i]),
                "prior_comp_start_ix": composite.start_ix,
                "prior_comp_end_ix": composite.end_ix,
                "prior_comp_body_top": composite.body_top,
                "prior_comp_body_bot": composite.body_bot,
                "prior_comp_body_size": composite.body_size,
                "prior_comp_high": composite.high,
                "prior_comp_low": composite.low,
                "prior_comp_total_range": composite.total_range,
                "prior_comp_wick_total": composite.wick_total,
                "prior_comp_body_percent": composite.body_percent,
                "prior_comp_wick_percent": composite.wick_percent,
                "current_body_size": curr_body_size,
                "cancel_amount": cancel_amount,
                "cancel_pct": cancel_pct,
                "required_cancel_pct": required_cancel_pct,
                "rule": "SCR_COMPOSITE_2BAR",
            }
        )

    return pd.DataFrame(rows)


def dedupe_same_direction(signals: pd.DataFrame) -> pd.DataFrame:
    if signals.empty:
        return signals.copy()

    keep_rows: list[int] = []
    last_dir = ""
    for row in signals.itertuples():
        if row.direction != last_dir:
            keep_rows.append(int(row.Index))
            last_dir = row.direction
    return signals.loc[keep_rows].reset_index(drop=True)


def analyze_to_next_opposite(
    df: pd.DataFrame,
    signals: pd.DataFrame,
    min_gain: float,
) -> tuple[pd.DataFrame, pd.DataFrame]:
    if signals.empty:
        return signals.copy(), pd.DataFrame(columns=["body_q", "count", "q_lo", "q_hi"])

    highs = df["High"].to_numpy(dtype=float)
    lows = df["Low"].to_numpy(dtype=float)
    closes = df["Close"].to_numpy(dtype=float)
    signal_ix = signals["signal_ix"].to_numpy(dtype=int)
    dirs = signals["direction"].to_numpy()
    count = len(signals)

    next_opp = np.full(count, -1, dtype=int)
    for i in range(count):
        for j in range(i + 1, count):
            if dirs[j] != dirs[i]:
                next_opp[i] = j
                break

    mfe_vals: list[float] = []
    qualifies: list[bool] = []
    valid: list[bool] = []
    for i in range(count):
        start_ix = int(signal_ix[i])
        opp_ix = int(next_opp[i])
        if opp_ix < 0:
            mfe_vals.append(float("nan"))
            qualifies.append(False)
            valid.append(False)
            continue

        end_ix = int(signal_ix[opp_ix])
        if end_ix <= start_ix:
            mfe_vals.append(float("nan"))
            qualifies.append(False)
            valid.append(False)
            continue

        entry = closes[start_ix]
        if dirs[i] == "long":
            mfe = float(np.max(highs[start_ix + 1 : end_ix + 1]) - entry)
        else:
            mfe = float(entry - np.min(lows[start_ix + 1 : end_ix + 1]))

        mfe_vals.append(mfe)
        qualifies.append(bool(mfe >= min_gain))
        valid.append(True)

    out = signals.copy()
    out["next_opp_signal_row"] = next_opp
    out["mfe_to_next_opp_signal"] = mfe_vals
    out["valid_window"] = valid
    out["qualifies"] = qualifies

    qualified = out[out["qualifies"]].copy()
    quantile_rows: list[dict[str, object]] = []
    if not qualified.empty:
        qualified["body_q"] = pd.qcut(
            qualified["prior_comp_body_size"],
            q=min(10, qualified["prior_comp_body_size"].nunique()),
            duplicates="drop",
        )
        grouped = qualified.groupby("body_q", dropna=False, observed=False)["prior_comp_body_size"]
        for bucket, vals in grouped:
            quantile_rows.append(
                {
                    "body_q": str(bucket),
                    "count": int(len(vals)),
                    "q_lo": float(vals.min()),
                    "q_hi": float(vals.max()),
                }
            )

    return out, pd.DataFrame(quantile_rows)


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True)
    parser.add_argument("--output-signals", default=r"C:\Temp-C\MQL5\Files\scr_composite_signals.csv")
    parser.add_argument("--output-quantiles", default=r"C:\Temp-C\MQL5\Files\scr_composite_quantiles.csv")
    parser.add_argument("--base-cancel-pct", type=float, default=0.50)
    parser.add_argument("--waffle-penalty-scale", type=float, default=0.25)
    parser.add_argument("--min-composite-body", type=float, default=0.01)
    parser.add_argument("--min-gain", type=float, default=15.0)
    parser.add_argument("--dedupe-same-direction", action="store_true", default=False)
    args = parser.parse_args()

    df = load(args.input)
    signals = detect_scrs(
        df,
        base_cancel_pct=args.base_cancel_pct,
        waffle_penalty_scale=args.waffle_penalty_scale,
        min_composite_body=args.min_composite_body,
    )
    if args.dedupe_same_direction:
        signals = dedupe_same_direction(signals)

    analyzed, quantiles = analyze_to_next_opposite(df, signals, args.min_gain)
    analyzed.to_csv(args.output_signals, index=False)
    quantiles.to_csv(args.output_quantiles, index=False)

    print(f"Signals: {len(analyzed)}")
    print(f"Valid windows: {int(analyzed['valid_window'].sum()) if len(analyzed) else 0}")
    print(f"Qualified >= ${args.min_gain:.2f}: {int(analyzed['qualifies'].sum()) if len(analyzed) else 0}")
    if len(analyzed):
        print(f"Long: {int((analyzed['direction'] == 'long').sum())}  Short: {int((analyzed['direction'] == 'short').sum())}")
    if not quantiles.empty:
        for row in quantiles.itertuples(index=False):
            print(f"{row.body_q},{row.count},{row.q_lo:.2f},{row.q_hi:.2f}")


if __name__ == "__main__":
    main()

Step-by-step logic:

  1. Load data: Reads CSV with OHLC bars plus TEMA9, EMA4_TEMA9, EMA2_HLCC4 columns (though indicators aren't used in detection).

  2. Build 2-bar composite (for each bar i starting at index 2):

    • Composite spans bars i-2 and i-1
    • Composite direction = direction from open[i-2] to close[i-1]
    • Composite body: top = max(open[i-2], close[i-1]), bottom = min(open[i-2], close[i-1])
    • Composite range = max(high[i-2], high[i-1]) to min(low[i-2], low[i-1])
    • Calculates body_percent and wick_percent of total range

  3. Detect SCR signal when:

    • Current bar i moves opposite direction from composite
    • Current bar's body "cancels" enough of the composite body
    • Cancel amount = overlap between current body and composite body (directional)
    • Required cancel percent = base_cancel_pct + (composite.wick_percent * waffle_penalty_scale)
    • Signal fires if (cancel_amount / composite.body_size) >= required_cancel_pct

  4. Optional deduplication: Keeps only first signal in each consecutive same-direction sequence.

  5. Analyze to next opposite signal:

    • For each signal, finds next signal in opposite direction
    • Measures MFE (max favorable excursion) from entry close to that boundary
    • Long MFE = highest high in window - entry
    • Short MFE = entry - lowest low in window
    • Signal "qualifies" if MFE >= min_gain

  6. Quantile bucketing: Groups qualifying signals into up to 10 buckets by composite body size.

Parameter/threshold/limit values:

Parameter Default Value Purpose
base_cancel_pct 0.50 Base percent of composite body that must be canceled
waffle_penalty_scale 0.25 Multiplier for wick_percent added to required cancel
min_composite_body 0.01 Minimum composite body size to consider
min_gain 15.0 Minimum MFE (dollars) for signal to qualify
dedupe_same_direction False Whether to keep only first of consecutive same-dir signals
Signal detection start index 2 First bar that can generate signal (needs 2 prior bars)
Max quantile buckets 10 Maximum qcut buckets for body size grouping

Derived threshold:

  • Actual required cancel percent = 0.50 + (wick_percent * 0.25)
  • For 100% wick composite (pure doji-like): requires 75% cancel
  • For 0% wick composite (pure body): requires 50% cancel
