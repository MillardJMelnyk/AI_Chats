{
   if(InpReadoutUnits == READOUT_UNITS_PIPS)

   return priceDelta;

string ReadoutUnitText()
{

      return "pips";
   return "price";

int ReadoutDigits()
{

void UpdateOpenTradeLiveState(const int bar, const double liveExitPrice)
{
   if(g_currentTrade.direction == 0 || g_currentTrade.entryBar < 0)
      return;

   if(bar >= ArraySize(BufferTradePL))
      TS_GrowStatsArrays(bar + 1, BufferTradePL, BufferRunningPL, BufferTradeCount, BufferWinPct,
                         BufferShortCount, BufferShortWinPct, BufferLongCount, BufferLongWinPct);

   //--- Compute in raw price
   double livePL = (liveExitPrice - g_currentTrade.entryPrice) * g_currentTrade.direction;
   double liveRunning = g_tradeStats.runningPL + livePL;

   //--- Store display values in Data Window buffers
   BufferTradePL[bar]    = PriceDeltaToReadout(livePL);
   BufferRunningPL[bar]  = PriceDeltaToReadout(liveRunning);
   BufferTradeCount[bar] = g_tradeStats.TotalTrades();
   BufferWinPct[bar]     = g_tradeStats.WinPct();
   BufferShortCount[bar] = g_tradeStats.shortCount;
   BufferShortWinPct[bar]= g_tradeStats.ShortWinPct();
   BufferLongCount[bar]  = g_tradeStats.longCount;
   BufferLongWinPct[bar] = g_tradeStats.LongWinPct();

   if(g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0)
   {
      int displayDigits   = ReadoutDigits();
      string readoutUnit  = ReadoutUnitText();
      string tip = StringFormat("Entry: %s | Exit: %s | P/L: %s %s | Running: %s %s",
                                DoubleToString(g_currentTrade.entryPrice, _Digits),
                                DoubleToString(liveExitPrice, _Digits),
                                DoubleToString(BufferTradePL[bar], displayDigits), readoutUnit,
                                DoubleToString(BufferRunningPL[bar], displayDigits), readoutUnit);
      ObjectSetString(0, g_currentTrade.arrowName, OBJPROP_TOOLTIP, tip);
   }
}

                        const double lo,

                        const double tema, 
                        const double emaTema,

   FileWrite(h,
             TimeToString(dt, TIME_DATE|TIME_MINUTES),
             bar,
             opn, hi, lo, cls,
             tema, emaTema,

//+------------------------------------------------------------------+
void ResetCycleForDirection(int iDirection, int bar, const double &open[], const double &high[], const double &low[],
                            bool preserveTriggerState = false)
{
   iDir = iDirection;
   g_crossBar = bar;
   g_extremeBar = bar;
   UpdateExtremeOpenReference(g_extremeBar, open);
   g_pendingConfirm = true;

   if(iDirection > 0)
      g_extremePrice = high[bar];
   else
      g_extremePrice = low[bar];

   g_isCocked = false;
   g_cockedDir = 0;
   g_cockLevel = 0.0;
   if(!preserveTriggerState)
   {
      g_triggerPending = false;
      g_triggerDir = 0;
      g_triggerCrossBar = -1;
   }
}

void RevertProvisionalSignal(const int bar, const datetime &time[])
{
   if(g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0)
      ObjectDelete(0, g_currentTrade.arrowName);
   if(g_provisionalBar >= 0 && g_provisionalBar < ArraySize(BufferArrowBuy))
   {
      BufferArrowBuy[g_provisionalBar] = EMPTY_VALUE;
      BufferArrowSell[g_provisionalBar] = EMPTY_VALUE;
   }
   g_currentTrade.Reset();
   if(g_tradeStats.tradeCount > 0)
      g_tradeStats.tradeCount--;
   g_provisionalSignal = false;
   g_provisionalBar = -1;
}

   g_maxPeriodsBack = MathMax(g_maxPeriodsBack, InpKCPeriod);
   g_maxPeriodsBack = MathMax(g_maxPeriodsBack, InpKCATRPeriod);
   g_maxPeriodsBack += (3 + InpOpenRefOffset);
   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
   ObjectsDeleteAll(0, SIGNAL_PREFIX+"_");
   if(iHndlKC != INVALID_HANDLE)
      IndicatorRelease(iHndlKC);
   if(iHndlEMAofTEMA != INVALID_HANDLE)
      IndicatorRelease(iHndlEMAofTEMA);
   if(iHndlTEMA != INVALID_HANDLE)
      IndicatorRelease(iHndlTEMA);

   //--- Clean up trade simulation objects
   if(InpEnableTradeSim)
   {
      TS_Cleanup();
   }

}

int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   if(rates_total <= g_maxPeriodsBack)
      return(0);

   int start = (prev_calculated == 0) ? g_maxPeriodsBack : prev_calculated - 1;
   if(start < 1)
      start = 1;

   if(prev_calculated == 0)
   {
      ResetAllStateAndBuffers();
      g_lastExportedBar = -1;
      InitCsvDebugFile();
   }

   int copyStart = MathMax(0, start - 3);
   int copyCount = rates_total - copyStart;

   double temaTail[];
   double emaTail[];
   double kcUpperTail[];
   double kcMiddleTail[];
   double kcLowerTail[];
   ArrayResize(temaTail, copyCount);
   ArrayResize(emaTail, copyCount);
   ArrayResize(kcUpperTail, copyCount);
   ArrayResize(kcMiddleTail, copyCount);
   ArrayResize(kcLowerTail, copyCount);
   ArraySetAsSeries(temaTail, false);
   ArraySetAsSeries(emaTail, false);
   ArraySetAsSeries(kcUpperTail, false);
   ArraySetAsSeries(kcMiddleTail, false);
   ArraySetAsSeries(kcLowerTail, false);

   int temaCopied = CopyBuffer(iHndlTEMA, 0, 0, copyCount, temaTail);
   int emaCopied  = CopyBuffer(iHndlEMAofTEMA, 0, 0, copyCount, emaTail);
   int kcUpperCopied  = CopyBuffer(iHndlKC, 0, 0, copyCount, kcUpperTail);
   int kcMiddleCopied = CopyBuffer(iHndlKC, 1, 0, copyCount, kcMiddleTail);
   int kcLowerCopied  = CopyBuffer(iHndlKC, 2, 0, copyCount, kcLowerTail);
   if(temaCopied != copyCount || emaCopied != copyCount ||
      kcUpperCopied != copyCount || kcMiddleCopied != copyCount || kcLowerCopied != copyCount)
   {
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " CopyBuffer failed. temaCopied=",temaCopied,
            " emaCopied=",emaCopied,
            " kcUpperCopied=",kcUpperCopied,
            " kcMiddleCopied=",kcMiddleCopied,
            " kcLowerCopied=",kcLowerCopied,
            " expected=",copyCount,
            " err=",GetLastError());
      return(prev_calculated);
   }

   ArrayResize(BufferKCUpper, rates_total);
   ArrayResize(BufferKCMiddle, rates_total);
   ArrayResize(BufferKCLower, rates_total);

   double pointSize = MathPow(10.0, -_Digits);
   double eps = pointSize * 0.1;
   double pipSize = pointSize * 10.0;
   //--- Convert user inputs from readout units to raw price (once)
   double triggerSensPrice = ReadoutToPrice(InpTriggerSensitivity, pipSize);
   double flatSlopePrice   = ReadoutToPrice(InpFlatSlopePerBar, pipSize);
   double arrowOffsetPrice = ReadoutToPrice(InpObjectArrowOffset, pipSize);
   ENUM_TS_READOUT_MODE tsReadoutMode = TSReadoutMode();

   for(int iKX = 0; iKX < copyCount; iKX++)
   {
      int iIX = copyStart + iKX;
      BufferTEMA[iIX] = temaTail[iKX];
      BufferEMAofTEMA[iIX] = emaTail[iKX];
      BufferKCUpper[iIX] = kcUpperTail[iKX];
      BufferKCMiddle[iIX] = kcMiddleTail[iKX];
      BufferKCLower[iIX] = kcLowerTail[iKX];
      BufferTemaEmaSpread[iIX] = PriceDeltaToReadout(BufferTEMA[iIX] - BufferEMAofTEMA[iIX]);
   
      int activeBlackout = InpSignalBlackout;
      if(InpUseDynamicBlackout && 
         BufferKCUpper[iIX] != EMPTY_VALUE && 
         BufferKCLower[iIX] != EMPTY_VALUE && 
         BufferKCMiddle[iIX] != EMPTY_VALUE)
      {
         double currWidth = MathAbs(BufferKCUpper[iIX] - BufferKCLower[iIX]);
         int lb = MathMin(InpKCWidthLookback, iIX + 1);
         double widthSum = 0.0;
         int widthCnt = 0;
         int from = iIX - lb + 1;
         if(from < 0) from = 0;
         for(int j = from; j <= iIX; j++)
         {
            if(BufferKCUpper[j] == EMPTY_VALUE || BufferKCLower[j] == EMPTY_VALUE)
               continue;
            widthSum += MathAbs(BufferKCUpper[j] - BufferKCLower[j]);
            widthCnt++;
         }
         double baselineWidth = (widthCnt > 0) ? (widthSum / widthCnt) : currWidth;
         double widthRatio = (baselineWidth > 0.0) ? (currWidth / baselineWidth) : 1.0;
         double slopePerBar = 0.0;
         if(iIX >= InpKCMidSlopePeriod && BufferKCMiddle[iIX - InpKCMidSlopePeriod] != EMPTY_VALUE)
            slopePerBar = (BufferKCMiddle[iIX] - BufferKCMiddle[iIX - InpKCMidSlopePeriod]) / InpKCMidSlopePeriod;
         
         bool isNarrow = (widthRatio < InpNarrowWidthRatio);
         bool isFlat = (MathAbs(slopePerBar) <= flatSlopePrice);
         if(isNarrow && isFlat)
         {
            double extra = (InpNarrowWidthRatio - widthRatio) * InpBlackoutScaleK;
            if(extra < 0.0) extra = 0.0;
            activeBlackout = InpSignalBlackout + (int)MathRound(extra);
            if(activeBlackout > InpBlackoutMax)
               activeBlackout = InpBlackoutMax;
         }
         BufferKCWidthRatio[iIX] = widthRatio;
      }
      BufferDynBlackout[iIX] = activeBlackout;
   }
   
   bool isNewBar = (rates_total != prev_calculated);
   bool isAggressive = IsAggressiveMode();
   int liveBar = rates_total - 1;

   // Finalize provisional signal on new bar
   if(isNewBar && g_provisionalSignal)
   {
      g_lastSignalBar = g_provisionalBar;
      g_provisionalSignal = false;
      g_provisionalBar = -1;
      g_isCocked = false;
      g_cockedDir = 0;
   }

   if(BufferTEMA[liveBar] != EMPTY_VALUE && BufferEMAofTEMA[liveBar] != EMPTY_VALUE)
      BufferTemaEmaSpread[liveBar] = PriceDeltaToReadout(BufferTEMA[liveBar] - BufferEMAofTEMA[liveBar]);

   if(InpEnableTradeSim)
   {
      double liveExitPrice = GetMockTradePrice(liveBar, close);
      UpdateOpenTradeLiveState(liveBar, liveExitPrice);
   }

   // Non-aggressive: skip loop if not a new bar
   if(!isAggressive && !isNewBar)
      return(rates_total);

//   double pointSize = MathPow(10.0, -_Digits);
//   double eps = pointSize * 0.1;
//   double pipSize = pointSize * 10.0;
//   //--- Convert user inputs from readout units to raw price (once)
//   double triggerSensPrice = ReadoutToPrice(InpTriggerSensitivity, pipSize);
//   double flatSlopePrice   = ReadoutToPrice(InpFlatSlopePerBar, pipSize);
//   double arrowOffsetPrice = ReadoutToPrice(InpObjectArrowOffset, pipSize);
//   ENUM_TS_READOUT_MODE tsReadoutMode = TSReadoutMode();
//

//--- Signals processing loop: processing live bar data depends on isAggressive
   int loopEnd = isAggressive ? rates_total : liveBar;
   
   //--- The signals loop is irrelevant if we haven't reached start date yet
   if(time[start] < InpTradeStartDate)
      return(rates_total);
   
   //--- Also skip if stop date is specified and already passed
   if(InpTradeStopDate > 0 && time[start] > InpTradeStopDate)
      return(rates_total);

   for(int iIX = start; iIX < loopEnd; iIX++)   
   {
      //--- Stop processing if past stop date
      if(InpTradeStopDate > 0 && time[iIX] > InpTradeStopDate)
         break;

      if(prev_calculated == 0 || iIX >= prev_calculated)
      {
         BufferArrowBuy[iIX] = EMPTY_VALUE;
         BufferArrowSell[iIX] = EMPTY_VALUE;
      }
      
      int prevRel = RelSign(BufferTEMA[iIX - 1] - BufferEMAofTEMA[iIX - 1], eps);
      int currRel = RelSign(BufferTEMA[iIX] - BufferEMAofTEMA[iIX], eps);
//      double temaEmaSpread = BufferTEMA[iIX] - BufferEMAofTEMA[iIX];
//
//      //--- Display buffer: readout units for Data Window
//      BufferTemaEmaSpread[iIX] = PriceDeltaToReadout(temaEmaSpread);
//
      if(prevRel == 0) prevRel = iDir;
      if(currRel == 0) currRel = prevRel;

      // Cocking logic: only run on new bar, not on live bar
      if(isNewBar)
      {
      //--- DIAG #2: State snapshot at loop top
      if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " DIAG#2 BAR iIX<",iIX,"> time<",TimeToString(time[iIX],TIME_DATE|TIME_MINUTES),
               "> iDir<",iDir,"> prevRel<",prevRel,"> currRel<",currRel,
               "> g_isCocked<",g_isCocked,"> g_cockedDir<",g_cockedDir,
               "> g_triggerPending<",g_triggerPending,"> g_triggerDir<",g_triggerDir,
               "> g_crossBar<",g_crossBar,"> g_lastSignalBar<",g_lastSignalBar,">");

      //--- Cocking logic: evaluate local hi/lo candidates from prior 2 bars.
      //--- Must run before cross handling so same-bar cock+cross can trigger.
         if(iDir != 0)
         {
            bool candidateFound = false;
            bool cockedThisBar = false;
            for(int d = 1; d <= 2; d++)
            {
               int cand = iIX - d;
               if(cand <= g_crossBar || cand < 1 || cand + 1 >= rates_total)
                  continue;
               candidateFound = true;
               int refBar = MathMax(0, cand - InpOpenRefOffset);
               double refOpen = open[refBar];

            //--- Keep debug/state fields aligned with the candidate under evaluation.
               g_extremeBar = cand;
               g_extremeOpenRefBar = refBar;
               g_extremeOpen = refOpen;
               g_extremePrice = (iDir > 0) ? high[cand] : low[cand];
               if((iDir > 0 && close[iIX] < refOpen) || (iDir < 0 && close[iIX] > refOpen))
               {
                  g_isCocked = true;
                  g_cockedDir = (iDir > 0) ? -1 : 1;
                  g_cockLevel = refOpen;
                  cockedThisBar = true;

               if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate)
                  Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                        " DIAG#3 COCKED iIX<",iIX,"> iDir<",iDir,"> g_cockedDir<",g_cockedDir,
                        "> g_cockLevel<",g_cockLevel,"> close[iIX]<",close[iIX],
                        "> refOpen<",refOpen,"> cand<",cand,">");

                  break;
               }
            }
            g_pendingConfirm = (candidateFound && !cockedThisBar);
         }
      }

      bool crossUp = (prevRel < 0 && currRel > 0);
      bool crossDn = (prevRel > 0 && currRel < 0);
      bool isCross = (crossUp || crossDn);

      //--- DIAG #4: Cross detected
      if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate && isCross)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " DIAG#4 CROSS iIX<",iIX,"> crossUp<",crossUp,"> crossDn<",crossDn,
               "> prevRel<",prevRel,"> currRel<",currRel,
               "> g_isCocked<",g_isCocked,"> g_cockedDir<",g_cockedDir,">");

//      //--- KC width/slope: all internal calcs in raw price
//      int activeBlackout = InpSignalBlackout;
//      double currWidth = EMPTY_VALUE;
//      double widthRatio = EMPTY_VALUE;
//      double slopePerBar = EMPTY_VALUE;
//      if(BufferKCUpper[iIX] != EMPTY_VALUE && BufferKCLower[iIX] != EMPTY_VALUE && BufferKCMiddle[iIX] != EMPTY_VALUE)
//      {
//         currWidth = MathAbs(BufferKCUpper[iIX] - BufferKCLower[iIX]);
//         int lb = MathMin(InpKCWidthLookback, iIX + 1);

//         {

//               continue;

//            widthCnt++;
//         }
//         double baselineWidth = (widthCnt > 0) ? (widthSum / widthCnt) : currWidth;
//         widthRatio = (baselineWidth > 0.0) ? (currWidth / baselineWidth) : 1.0;
//         slopePerBar = 0.0;
//         if(iIX >= InpKCMidSlopePeriod && BufferKCMiddle[iIX - InpKCMidSlopePeriod] != EMPTY_VALUE)
//            slopePerBar = (BufferKCMiddle[iIX] - BufferKCMiddle[iIX - InpKCMidSlopePeriod]) / InpKCMidSlopePeriod;
//         if(InpUseDynamicBlackout)
//         {
//            bool isNarrow = (widthRatio < InpNarrowWidthRatio);
//            bool isFlat = (MathAbs(slopePerBar) <= flatSlopePrice);
//            if(isNarrow && isFlat)
//            {
//               double extra = (InpNarrowWidthRatio - widthRatio) * InpBlackoutScaleK;
//               if(extra < 0.0) extra = 0.0;
//               activeBlackout = InpSignalBlackout + (int)MathRound(extra);
//               if(activeBlackout > InpBlackoutMax)
//                  activeBlackout = InpBlackoutMax;
//            }
//         }
//      }

//      BufferDynBlackout[iIX] = activeBlackout;
//      BufferKCWidthRatio[iIX] = widthRatio;
      BufferTriggerThreshold[iIX] = InpTriggerSensitivity;  // display in readout units

      if(iDir == 0 && currRel != 0)
         ResetCycleForDirection(currRel, iIX, open, high, low);

      ENUM_SIGNAL_TYPE signal = SIGNAL_NONE;
      ENUM_SIGNAL_TYPE signalCandidate = SIGNAL_NONE;

      // Triggering logic
      if(isAggressive)
      {
         if(g_isCocked)
         {
            double liveClose = close[iIX];
            double temaVal = BufferTEMA[iIX];
            double emaVal = BufferEMAofTEMA[iIX];
            bool conditionMet = false;

            if(InpMockTradePriceSource == MOCK_PRICE_AGG_TEMA)
            {
               if(g_cockedDir == -1)
                  conditionMet = (liveClose < temaVal - triggerSensPrice);
               else
                  conditionMet = (liveClose > temaVal + triggerSensPrice);
            }
            else if(InpMockTradePriceSource == MOCK_PRICE_AGG_EMA)
            {
               if(g_cockedDir == -1)
                  conditionMet = (liveClose < emaVal - triggerSensPrice);
               else
                  conditionMet = (liveClose > emaVal + triggerSensPrice);
            }
            else // MOCK_PRICE_AGG_BOTH
            {
               if(g_cockedDir == -1)
                  conditionMet = (liveClose < temaVal && liveClose < emaVal - triggerSensPrice);
               else
                  conditionMet = (liveClose > temaVal && liveClose > emaVal + triggerSensPrice);
            }

            if(conditionMet && !g_provisionalSignal)
            {
               signalCandidate = (g_cockedDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
            }
            else if(!conditionMet && g_provisionalSignal)
            {
               RevertProvisionalSignal(iIX, time);
            }
         }
         else if(g_provisionalSignal)
         {
            RevertProvisionalSignal(iIX, time);
         }
      }
      else
      {
         // Non-aggressive: existing TEMA/EMA cross logic
         if(isCross)
         {
            if(g_isCocked)
            {
               if(crossDn && g_cockedDir == -1)
               {
                  g_triggerPending = true;
                  g_triggerDir = -1;
                  g_triggerCrossBar = iIX;

               //--- DIAG #5a: Trigger pending SHORT
               if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate)
                  Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                        " DIAG#5 TRIGGER_PENDING iIX<",iIX,"> dir<SHORT> g_triggerCrossBar<",iIX,
                        "> g_isCocked<",g_isCocked,"> g_cockedDir<",g_cockedDir,">");
               }
               else if(crossUp && g_cockedDir == 1)
               {
                  g_triggerPending = true;
                  g_triggerDir = 1;
                  g_triggerCrossBar = iIX;

               //--- DIAG #5b: Trigger pending LONG
               if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate)
                  Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                        " DIAG#5 TRIGGER_PENDING iIX<",iIX,"> dir<LONG> g_triggerCrossBar<",iIX,
                        "> g_isCocked<",g_isCocked,"> g_cockedDir<",g_cockedDir,">");
               }
               else
               {
                  g_triggerPending = false;
                  g_triggerDir = 0;
                  g_triggerCrossBar = -1;
               }
            }
            else
            {
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
         }

         if(g_triggerPending)
         {
            bool wrongSide = ((g_triggerDir == 1 && currRel < 0) || (g_triggerDir == -1 && currRel > 0));
            if(wrongSide)
            {
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
            else
            {
            //--- Cumulative spread in raw price from source buffers (not display buffer)
               double cumulativeSpread = 0.0;
               int sumFrom = MathMax(0, g_triggerCrossBar);
               if(sumFrom <= iIX)
               {
                  for(int s = sumFrom; s <= iIX; s++)
                  {
                     if(BufferTEMA[s] == EMPTY_VALUE || BufferEMAofTEMA[s] == EMPTY_VALUE)
                        continue;
                     cumulativeSpread += MathAbs(BufferTEMA[s] - BufferEMAofTEMA[s]);
                  }
               }

            //--- DIAG #6: Cumulative spread check
            if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate)
               Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                     " DIAG#6 SPREAD_CHECK iIX<",iIX,"> cumulativeSpread<",cumulativeSpread,
                     "> triggerSensPrice<",triggerSensPrice,"> sumFrom<",sumFrom,
                     "> met<",(cumulativeSpread >= triggerSensPrice),">");

               if(cumulativeSpread >= triggerSensPrice)
               {
                  signalCandidate = (g_triggerDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
                  g_triggerPending = false;
                  g_triggerDir = 0;
                  g_triggerCrossBar = -1;
               }
            }
         }
      }

      bool canFireNow = ((iIX - g_lastSignalBar) > BufferDynBlackout[iIX] && time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate);

      //--- DIAG #7: Signal candidate + gate check
      if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate && signalCandidate != SIGNAL_NONE)
         Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
               " DIAG#7 CANDIDATE iIX<",iIX,"> signalCandidate<",SignalToText(signalCandidate),
               "> canFireNow<",canFireNow,"> g_lastSignalBar<",g_lastSignalBar,
               "> BufferDynBlackout[iIX]<",BufferDynBlackout[iIX],"> barGap<",(iIX - g_lastSignalBar),
               "> time[iIX]<",TimeToString(time[iIX],TIME_DATE|TIME_MINUTES),">");

      if(signalCandidate != SIGNAL_NONE)
      {
         if(canFireNow)
         {
            signal = signalCandidate;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else
         {
            g_blackoutSignalPending = true;
            g_blackoutSignalDir = (signalCandidate == SIGNAL_LONG) ? 1 : -1;
            g_blackoutSignalBar = iIX;
         }
      }

      if(signal == SIGNAL_NONE && g_blackoutSignalPending)
      {
         bool retraced = ((g_blackoutSignalDir == 1 && currRel < 0) || (g_blackoutSignalDir == -1 && currRel > 0));
         if(retraced)
         {
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else if(canFireNow)
         {
            signal = (g_blackoutSignalDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
      }

      if(signal != SIGNAL_NONE)
      {
         //--- DIAG #8: Signal fired
         if(time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate)
            Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
                  " DIAG#8 SIGNAL_FIRED iIX<",iIX,"> signal<",SignalToText(signal),
                  "> time<",TimeToString(time[iIX],TIME_DATE|TIME_MINUTES),">");

         if(g_signalStartDT == 0)
            g_signalStartDT = time[iIX];

         if(InpEnableTradeSim)
         {
            if(InpMockTradePriceSource == MOCK_PRICE_TEMA)
            {
            HandleSignalEvent(iIX, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferTEMA,
                              g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                              (InpSoundTradeAlarm && time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell,
                              tsReadoutMode);
            }
            else if(InpMockTradePriceSource == MOCK_PRICE_EMA_TEMA)
            {
               HandleSignalEvent(iIX, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferEMAofTEMA,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,

                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell,
                                 tsReadoutMode);
            }
            else
            {
               HandleSignalEvent(iIX, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 close,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && time[iIX] >= InpTradeStartDate && time[iIX] <= InpTradeStopDate),
                                 BufferTradePL,
