int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   if(rates_total < InpPeriod) return(0);
   
   int start;
   if(prev_calculated == 0)
   {
      ATRBuffer[0] = high[0] - low[0];
      double sum = 0;
      for(int i = 1; i < InpPeriod; i++)
      {
         double tr = MathMax(high[i] - low[i], 
                     MathMax(MathAbs(high[i] - close[i-1]), 
                             MathAbs(low[i] - close[i-1])));
         ATRBuffer[i] = tr;
         sum += tr;
      }
      ATRBuffer[InpPeriod-1] = sum / InpPeriod;
      start = InpPeriod;
   }
   else
      start = prev_calculated - 1;
   
   for(int i = start; i < rates_total; i++)
   {
      double tr = MathMax(high[i] - low[i], 
                  MathMax(MathAbs(high[i] - close[i-1]), 
                          MathAbs(low[i] - close[i-1])));
      ATRBuffer[i] = (ATRBuffer[i-1] * (InpPeriod - 1) + tr) / InpPeriod;
   }
   
   return(rates_total);
}

void CreateParameterMetadata(string prefix, int instanceID, string shortName,
                            string &paramNames[], string &paramValues[])
{
   int count = ArraySize(paramNames);
   TextGVar_Create(prefix, instanceID, shortName, "ParmCount", IntegerToString(count));
   
   for(int i = 0; i < count; i++)

      string idx = IntegerToString(i);

      TextGVar_Create(prefix, instanceID, shortName, "Parm["+idx+"]_Value", paramValues[i]);
   }
}
