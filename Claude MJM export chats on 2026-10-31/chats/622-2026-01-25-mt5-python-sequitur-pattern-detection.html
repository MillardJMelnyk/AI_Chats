"""
sequitur_MJM.py
SEQUITUR algorithm implementation for grammar induction from symbol sequences.
"""


class Symbol:
    """Represents a symbol in the grammar (terminal or rule reference)."""
    
    def __init__(self, value, is_terminal=True, is_guard=False):
        self.value = value
        self.is_terminal = is_terminal
        self.is_guard = is_guard
        self.prev = None
        self.next = None
    
    def __repr__(self):
        return f"Symbol({self.value})"


class Rule:
    """Represents a grammar rule."""
    
    def __init__(self, name):
        self.name = name
        self.guard = Symbol(name, is_terminal=False, is_guard=True)
        self.guard.prev = self.guard
        self.guard.next = self.guard
        self.ref_count = 0
    
    def first(self):
        return self.guard.next
    
    def last(self):
        return self.guard.prev
    
    def append(self, symbol):
        symbol.prev = self.guard.prev
        symbol.next = self.guard
        self.guard.prev.next = symbol
        self.guard.prev = symbol
    
    def length(self):
        count = 0
        s = self.guard.next
        while s != self.guard:
            count += 1
            s = s.next
        return count
    
    def symbols(self):
        result = []
        s = self.guard.next
        while s != self.guard:
            result.append(s)
            s = s.next
        return result
    
    def __repr__(self):
        syms = ' '.join([s.value for s in self.symbols()])
        return f"{self.name} -> {syms}"


class Sequitur:

    def __init__(self):
        self.main_rule = Rule('S')
        self.rules = {'S': self.main_rule}

def expand_rule(self, name, max_depth=10):
        """Fully expand a rule to terminal symbols (iterative to avoid recursion limit)."""
        if name not in self.rules:
            return [name]

        result = []

        iterations = 0

            if sym.startswith('R') and sym in self.rules:
                stack.extend(reversed(self.rules[sym].symbols))
            else:
                result.append(sym)
        
        return result

def compute_thresholds(df, line_columns, dead_zone_pct=10, mag_tertiles=(33, 67)):
    """
    Compute adaptive thresholds from data.
    
    Returns dict with:
      - dead_zone: minimum absolute value to count as meaningful
      - mag_thresholds: dict of {line: (small_max, medium_max)} for magnitude buckets
    """
    thresholds = {
        'dead_zone': {},
        'mag_thresholds': {}
    }
    
    for col in line_columns:
        abs_vals = df[col].abs()
        
        # Dead zone: values below this percentile are noise
        thresholds['dead_zone'][col] = np.percentile(abs_vals, dead_zone_pct)
        
        # Magnitude tertiles for pivot classification
        # Only consider values above dead zone
        meaningful = abs_vals[abs_vals > thresholds['dead_zone'][col]]
        if len(meaningful) > 0:
            t1 = np.percentile(meaningful, mag_tertiles[0])
            t2 = np.percentile(meaningful, mag_tertiles[1])
        else:
            t1, t2 = 0, 0
        
        thresholds['mag_thresholds'][col] = (t1, t2)
    
    return thresholds


def magnitude_bucket(value, thresholds):
    """
    Classify magnitude as small/medium/large.
    thresholds = (small_max, medium_max)
    """
    abs_val = abs(value)
    if abs_val <= thresholds[0]:
        return 's'
    elif abs_val <= thresholds[1]:
        return 'm'
    else:
        return 'l'

So the fix is still needed. Rolling average absolute value as denominator handles this:

    }
}
