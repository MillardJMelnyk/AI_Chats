//--- compute Keltner arrays using CopyBuffer
bool ComputeKeltner(int count, double &kcMid[], double &kcUpper[], double &kcLower[])
{
   int hMA = iMA(NULL,0,KC_EMA_Period,0,MODE_EMA,PRICE_TYPICAL);
   int hATR= iATR(NULL,0,KC_ATR_Period);
   if(hMA==INVALID_HANDLE || hATR==INVALID_HANDLE){ Print("MA/ATR handle failed"); return false; }
   ArraySetAsSeries(kcMid,true);
   ArraySetAsSeries(kcUpper,true);
   ArraySetAsSeries(kcLower,true);
   ArrayResize(kcMid, count);
   ArrayResize(kcUpper, count);
   ArrayResize(kcLower, count);
   double ma[], atr[];
   ArraySetAsSeries(ma,true);
   ArraySetAsSeries(atr,true);
   ArrayResize(ma, count);
   ArrayResize(atr, count);
   if(CopyBuffer(hMA,0,0,count,ma)!=count || CopyBuffer(hATR,0,0,count,atr)!=count)
   {
      Print("CopyBuffer failed");
      IndicatorRelease(hMA);
      IndicatorRelease(hATR);
      return false;
   }
   for(int i=0;i<count;i++)
   {
      kcMid[i]=ma[i];
      kcUpper[i]=ma[i]+KC_ATR_Mult*atr[i];
      kcLower[i]=ma[i]-KC_ATR_Mult*atr[i];
   }
   IndicatorRelease(hMA);
   IndicatorRelease(hATR);
   return true;
}

bool ComputeStoch(int count, double &main[], double &signal[])
{
   int hSt = iStochastic(NULL,0,Stoch_K,Stoch_D,Stoch_Slowing,Stoch_Method,Stoch_Price);
   if(hSt==INVALID_HANDLE){ Print("Stoch handle failed"); return false; }
   ArraySetAsSeries(main,true);
   ArraySetAsSeries(signal,true);
   ArrayResize(main, count);
   ArrayResize(signal, count);
   if(CopyBuffer(hSt,0,0,count,main)!=count || CopyBuffer(hSt,1,0,count,signal)!=count)
   {
      Print("CopyBuffer stoch failed");
      IndicatorRelease(hSt);
      return false;
   }
   IndicatorRelease(hSt);
   return true;
}

//--- detect Stoch pivots (single line)
void DetectStochPivots(const double &arr[], const double &other[], int firstBar, int lastBar, string lineName)
{
   int minBars = MathMax(3,Pivot_MinBars);
   int follow  = MathMax(1,Pivot_FollowBars);
   // process oldest->newest: firstBar is oldest index, lastBar is newest (likely 0)
   for(int b=firstBar; b>=lastBar+minBars-1; b--)
   {
      int aIdx=b-(minBars-3);
      int bIdx=b-(minBars-2);
      int cIdx=b-(minBars-1);
      double pA=arr[aIdx], pB=arr[bIdx], pC=arr[cIdx];
      int s1=Sign(pB-pA), s2=Sign(pC-pB);
      if(s1==0 || s2==0 || s1==s2) continue;
      bool ok=true;
      int last=cIdx;
      for(int f=0; f<follow; f++)
      {
         int idx=cIdx- (f+1);
         if(idx<lastBar){ ok=false; break; }
         int sf=Sign(arr[idx]-arr[idx+1]);
         if(sf!=s2){ ok=false; break; }
         last=idx;
      }
      if(!ok) continue;
      EventRow e;
      e.startBar=bIdx;
      e.endBar  =last;
      e.startTime=g_rates[bIdx].time;
      e.endTime  =g_rates[last].time;
      e.type     ="StochPivot_"+lineName+"_"+(s2>0 ? "Up":"Down");
      e.band="";
      e.priceOpen=e.priceHigh=e.priceLow=e.priceClose=0;
      e.kcUpper=e.kcMid=e.kcLower=0;
      e.stochMain   = arr[bIdx];
      e.stochSignal = other[bIdx];
      AddEvent(e,false);
   }
}

//--- detect Stoch crosses
void DetectStochCrosses(const double &main[], const double &signal[], int firstBar, int lastBar)
{
   for(int i=firstBar-1; i>=lastBar; i--)
   {
      double p0=main[i+1]-signal[i+1];
      double p1=main[i]-signal[i];
      if(p0==0 || p1==0) continue;
      if(p0*p1<0)
      {
         EventRow e;
         e.startBar=i+1;
         e.endBar  =i;
         e.startTime=g_rates[i].time;
         e.endTime  =g_rates[i].time;
         e.type     =(p1>0 ? "StochCross_Up" : "StochCross_Down");
         e.band     ="";
         e.priceOpen=e.priceHigh=e.priceLow=e.priceClose=0;
         e.kcUpper=e.kcMid=e.kcLower=0;
         e.stochMain   = main[i];
         e.stochSignal = signal[i];
         AddEvent(e,false);
      }
   }
}

double BodyPctAbove(double open, double close, double ref)
{
   double bodyHigh=MathMax(open,close);
   double bodyLow =MathMin(open,close);
   double body    =bodyHigh-bodyLow;
   if(body==0) body=Point();
   double portion = (bodyHigh - MathMax(ref,bodyLow)) / body;
   return portion*100.0;
}

// detect price events for a band (series indexing)
void DetectPriceEventsBand(const double &kcBand[], const double &kcUpper[], const double &kcMid[], const double &kcLower[],
                           int firstBar, int lastBar, string bandName)
{
   int kissStart=-1, kissCount=0;
   int hugStart=-1, hugBars=0, hugMiss=0;
   int lastTouch=-1;

   for(int i=firstBar-1; i>=lastBar; i--)
   {
      double open=g_rates[i].open, close=g_rates[i].close, high=g_rates[i].high, low=g_rates[i].low;
      double band=kcBand[i];
      bool touch = (high>=band && low<=band);
      bool crossBody = ((open-band)*(close-band)<0); // body straddles

      // Determine approach side from prior bar (older) if available
      bool approachAbove=true;
      if(i+1<=firstBar)
      {
         double po=g_rates[i+1].open, pc=g_rates[i+1].close;
         double pctAbovePrev = BodyPctAbove(po,pc,band);
         approachAbove = (pctAbovePrev >= 50.0);
      }
      double pctAbove = BodyPctAbove(open,close,band);
      double pctOnApproach = approachAbove ? pctAbove : (100.0 - pctAbove);

      // CROSS
      if(crossBody)
      {
         bool ok=true;
         for(int f=0; f<Cross_FollowBars; f++)
         {
            int idx=i-(f+1);
            if(idx<lastBar){ ok=false; break; }
            double o2=g_rates[idx].open, c2=g_rates[idx].close;
            bool wantAbove = (close>band); // which side we ended on
            double pctAbove2 = BodyPctAbove(o2,c2,band);
            double pctOtherSide = wantAbove ? pctAbove2 : (100.0 - pctAbove2);
            if(pctOtherSide < Cross_FollowBodyPct){ ok=false; break; }
         }
         if(ok)
         {
            EventRow e;
            e.startBar=i;
            e.endBar  =i-1;
            e.startTime=g_rates[i].time;
            e.endTime  =g_rates[MathMax(i-1,lastBar)].time;
            e.type="PriceCross_"+bandName;
            e.band=bandName;
            e.priceOpen=open; e.priceHigh=high; e.priceLow=low; e.priceClose=close;
            e.kcUpper=kcUpper[i]; e.kcMid=kcMid[i]; e.kcLower=kcLower[i];
            e.stochMain=e.stochSignal=0;
            AddEvent(e,true);
         }
         else
         {
            // Demote failed cross to a 1-bar kiss (per spec)
            if(touch)
            {
               EventRow e;
               e.startBar=i;
               e.endBar=i;
               e.startTime=g_rates[i].time;
               e.endTime=g_rates[i].time;
               e.type="PriceKiss_"+bandName+"_FromFailedCross";
               e.band=bandName;
               e.priceOpen=open; e.priceHigh=high; e.priceLow=low; e.priceClose=close;
               e.kcUpper=kcUpper[i]; e.kcMid=kcMid[i]; e.kcLower=kcLower[i];
               e.stochMain=e.stochSignal=0;
               AddEvent(e,true);
            }
         }
      }

      // KISS
      bool kissCond = touch && (pctOnApproach >= Kiss_BodyPctOnApproach);
      if(kissCond)
      {
         if(kissStart<0){ kissStart=i; kissCount=1; }
         else kissCount++;
      }
      else
      {
         if(kissStart>=0 && kissCount<=Kiss_MaxBars)
         {
            EventRow e;
            e.startBar=kissStart;
            e.endBar=kissStart - (kissCount-1);
            e.startTime=g_rates[e.startBar].time;
            e.endTime=g_rates[e.endBar].time;
            e.type="PriceKiss_"+bandName;
            e.band=bandName;
            e.priceOpen=open; e.priceHigh=high; e.priceLow=low; e.priceClose=close;
            e.kcUpper=kcUpper[e.startBar]; e.kcMid=kcMid[e.startBar]; e.kcLower=kcLower[e.startBar];
            e.stochMain=e.stochSignal=0;
            AddEvent(e,true);
         }
         kissStart=-1; kissCount=0;
      }

      // HUG
      if(touch)
      {
         if(hugStart<0){ hugStart=i; hugBars=1; hugMiss=0; lastTouch=i; }
         else { hugBars++; lastTouch=i; }
      }
      else
      {
         if(hugStart>=0)
         {
            hugMiss++;
            hugBars++;
            if(hugMiss>Hug_MaxExceptions)
            {
               int endBar = i + hugMiss - 1; // include allowed exception bars, exclude the one that broke the rule
               if(hugBars-1>=Hug_MinBars && endBar<=hugStart)
               {
                  EventRow e;
                  e.startBar=hugStart;
                  e.endBar=endBar;
                  e.startTime=g_rates[e.startBar].time;
                  e.endTime=g_rates[e.endBar].time;
                  e.type="PriceHug_"+bandName;
                  e.band=bandName;
                  e.priceOpen=open; e.priceHigh=high; e.priceLow=low; e.priceClose=close;
                  e.kcUpper=kcUpper[e.startBar]; e.kcMid=kcMid[e.startBar]; e.kcLower=kcLower[e.startBar];
                  e.stochMain=e.stochSignal=0;
                  AddEvent(e,true);
               }
               hugStart=-1; hugBars=0; hugMiss=0; lastTouch=-1;
            }
         }
      }
   }
}

// correlations
void BuildCorrelations(int overlapMin)
{
   int na=ArraySize(eventsPrice), nb=ArraySize(eventsStoch);
   for(int i=0;i<na;i++)
   {
      EventRow ea=eventsPrice[i];
      for(int j=0;j<nb;j++)
      {
         EventRow eb=eventsStoch[j];
         int s=MathMax(ea.startBar, eb.startBar);
         int e=MathMin(ea.endBar,   eb.endBar);
         int ov = e - s + 1;
         if(ov >= overlapMin)
         {
            CorrRow c;
            c.aIndex=i; c.bIndex=j;
            c.aType=ea.type; c.bType=eb.type;
            c.aBand=ea.band; c.bBand=eb.band;
            c.overlapBars=ov;
            c.leadBars=eb.startBar - ea.startBar;
            AddCorr(c);
         }
      }
   }
}

int OnStart()
{
   Print("KSC build stamp: ", BUILD_STAMP, " (local compile time)");
   int barsTotal = Bars(NULL,0);
   if(barsTotal<=0){ Print("No bars"); return 0; }

   ArraySetAsSeries(g_rates,true); // IMPORTANT: set BEFORE CopyRates so [0]=newest ordering is real
   int count=0;
   if(StartDate>0)
      count = CopyRates(NULL,0,StartDate,TimeCurrent(),g_rates);

   // #endregion

   int firstIdx = count-1; // oldest in our copied window
   int lastIdx  = 0;       // newest

   // Price events
   DetectPriceEventsBand(kcUpper, kcUpper,kcMid,kcLower, firstIdx, lastIdx, "Upper");
   DetectPriceEventsBand(kcMid,   kcUpper,kcMid,kcLower, firstIdx, lastIdx, "Mid");
   DetectPriceEventsBand(kcLower, kcUpper,kcMid,kcLower, firstIdx, lastIdx, "Lower");

   // Stoch events
   DetectStochCrosses(stMain, stSignal, firstIdx, lastIdx);
   DetectStochPivots(stMain, stSignal, firstIdx, lastIdx, "Main");
   DetectStochPivots(stSignal, stMain, firstIdx, lastIdx, "Signal");

   // Correlations
   BuildCorrelations(Overlap_MinBars);

   // #region agent log
   Dbg("H3","KeltnerStochCorrelation.mq5:OnStart","counts",
       StringFormat("{\"priceEvents\":%d,\"stochEvents\":%d,\"correlations\":%d}",ArraySize(eventsPrice),ArraySize(eventsStoch),ArraySize(correlations)));
   // #endregion

   // Write CSVs
   WriteEventsCsv(Output_CSV);
   WriteCorrCsv(Output_Corr_CSV);

   Print("Price events: ", ArraySize(eventsPrice),
         " Stoch events: ", ArraySize(eventsStoch),
         " Correlations: ", ArraySize(correlations));
   return 0;
}
