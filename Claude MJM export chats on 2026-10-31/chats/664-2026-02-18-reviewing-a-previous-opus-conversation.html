class ProgressTracker:
    def __init__(self, label: str, total: int, interval: float = 2.0):
        self.label    = label
        self.total    = total
        self.interval = interval
        self.completed = 0
        self.start    = time.monotonic()
        self.last_print = 0.0

    @staticmethod
    def _fmt_seconds(s: float) -> str:
        s = int(s)
        h, rem = divmod(s, 3600)
        m, sec = divmod(rem, 60)
        return f"{h:02d}:{m:02d}:{sec:02d}"

    def tick(self, n: int = 1) -> None:
        self.completed += n
        now = time.monotonic()
        if now - self.last_print < self.interval:
            return
        self.last_print = now
        elapsed = now - self.start
        pct = self.completed / self.total * 100 if self.total > 0 else 0.0
        elapsed_str = self._fmt_seconds(elapsed)
        if elapsed > 0 and self.completed > 0:
            rate = self.completed / elapsed
            remaining = (self.total - self.completed) / rate
            eta_str = self._fmt_seconds(remaining)
        else:
            eta_str = "--:--:--"
        line = (f"\r{self.label}: {self.completed:,} / {self.total:,} "
                f"({pct:.1f}%) | {elapsed_str} elapsed | ETA: {eta_str}   ")
        print(line, end='', flush=True)

    def done(self) -> None:
        elapsed = time.monotonic() - self.start
        elapsed_str = self._fmt_seconds(elapsed)
        line = (f"\r{self.label}: {self.total:,} / {self.total:,} "
                f"(100.0%) | {elapsed_str} elapsed | done                ")
        print(line, flush=True)

def round_to_point(value: float, point: float = 0.001) -> float:
    return round(value / point) * point


def calculate_spreads(tema: float, kc_upper: float, kc_middle: float, kc_lower: float, point: float) -> dict:
    return {
        'spread_upper':  round_to_point(tema - kc_upper,  point),
        'spread_middle': round_to_point(tema - kc_middle, point),
        'spread_lower':  round_to_point(tema - kc_lower,  point)
    }


def determine_active_line(spreads: dict) -> Tuple[str, str]:
    abs_spreads = {
        'U': abs(spreads['spread_upper']),
        'M': abs(spreads['spread_middle']),
        'L': abs(spreads['spread_lower'])
    }
    active = min(abs_spreads, key=abs_spreads.get)
    return active, active


def calculate_spread_delta_state(spread_curr: float, spread_prev: float) -> str:
    if spread_curr == 0 and spread_prev == 0:
        return 'SD0'
    if spread_prev != 0 and spread_curr != 0:
        if (spread_curr > 0) != (spread_prev > 0):
            return 'SDX'
    elif spread_prev == 0 or spread_curr == 0:
        if spread_prev != 0 or spread_curr != 0:
            return 'SDX'
    delta = abs(spread_curr) - abs(spread_prev)
    if delta > 0:
        return 'SD+'
    elif delta < 0:
        return 'SD-'
    else:
        return 'SD0'


def calculate_l0_state(slope_delta: float) -> int:
    if slope_delta > 0:
        return 1
    elif slope_delta < 0:
        return -1
    else:
        return 0


def l1_to_numeric(l1_state: str) -> int:
    """Convert L1 state string to numeric for DTW."""
    mapping = {'SD+': 1, 'SD-': -1, 'SD0': 0, 'SDX': 0}
    return mapping.get(l1_state, 0)

            'kc_lower':      row['KC_Lower'],
            'spread_upper':  spreads['spread_upper'],
            'spread_middle': spreads['spread_middle'],
            'spread_lower':  spreads['spread_lower'],
            'active_line':   active_line,
            'zone':          zone
        })
        prog.tick()
    prog.done()
    return pd.DataFrame(results)


def pass2_l1_calculation(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df['L1_state']         = None
    df['zone_switch']      = 'N'
    df['active_spread']    = None
    df['active_spread_prev'] = None

    n = len(df)
    prog = ProgressTracker("Pass 2 (L1 states)  ", n - 1)
    for i in range(1, n):
        curr = df.iloc[i]
        prev = df.iloc[i - 1]
        al = curr['active_line']
        spread_key = {'U': 'spread_upper', 'M': 'spread_middle', 'L': 'spread_lower'}[al]
        spread_curr = curr[spread_key]
        spread_prev = prev[spread_key]
        l1_state    = calculate_spread_delta_state(spread_curr, spread_prev)
        zone_switch = 'Y' if curr['active_line'] != prev['active_line'] else 'N'
        df.at[i, 'L1_state']          = l1_state
        df.at[i, 'zone_switch']       = zone_switch
        df.at[i, 'active_spread']     = spread_curr
        df.at[i, 'active_spread_prev'] = spread_prev
        prog.tick()
    prog.done()
    return df


def pass3_l0_calculation(df: pd.DataFrame, point: float = 0.001) -> pd.DataFrame:
    df = df.copy()
    df['slope_1']    = None
    df['slope_2']    = None
    df['slope_delta'] = None
    df['L0_state']   = None

    n = len(df)
    prog = ProgressTracker("Pass 3 (L0 states)  ", n - 2)
    for i in range(2, n):
        tema_n  = df.iloc[i]['tema']
        tema_n1 = df.iloc[i - 1]['tema']
        tema_n2 = df.iloc[i - 2]['tema']
        slope_1     = round_to_point(tema_n1 - tema_n2, point)
        slope_2     = round_to_point(tema_n  - tema_n1, point)
        slope_delta = round_to_point(slope_2 - slope_1, point)
        df.at[i, 'slope_1']    = slope_1
        df.at[i, 'slope_2']    = slope_2
        df.at[i, 'slope_delta'] = slope_delta
        df.at[i, 'L0_state']   = calculate_l0_state(slope_delta)
        prog.tick()
    prog.done()
    return df

# ZigZag with pivot extraction
# ---------------------------------------------------------------------------

def compute_zigzag_pivots(df: pd.DataFrame, depth: int = 12, deviation: int = 5, 
                          backstep: int = 3, point: float = 0.001) -> List[Dict]:
    """
    Compute ZigZag pivots. Returns list of pivot dicts:
    {'bar': int, 'price': float, 'direction': int}
    direction: 1 = peak (high), -1 = trough (low)
    """
    highs = df['High'].values
    lows = df['Low'].values
    n = len(highs)
    dev = deviation * point

    pivots = []
    last_pivot_bar = 0
    last_pivot_price = 0.0
    last_direction = 0

    for i in range(depth, n):
        window_start = max(0, i - depth + 1)
        win_high_idx = window_start + int(np.argmax(highs[window_start:i + 1]))
        win_low_idx = window_start + int(np.argmin(lows[window_start:i + 1]))
        win_high = highs[win_high_idx]
        win_low = lows[win_low_idx]

        new_pivot = None

        # Candidate swing high
        if win_high_idx == i:
            if last_direction != 1:
                if last_direction == -1 and (win_high - last_pivot_price) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = {'bar': i, 'price': win_high, 'direction': 1}
                elif last_direction == 0:
                    new_pivot = {'bar': i, 'price': win_high, 'direction': 1}
            elif last_direction == 1 and win_high > last_pivot_price:
                if pivots:
                    pivots[-1] = {'bar': i, 'price': win_high, 'direction': 1}
                last_pivot_bar = i
                last_pivot_price = win_high
                continue

        # Candidate swing low
        if win_low_idx == i:
            if last_direction != -1:
                if last_direction == 1 and (last_pivot_price - win_low) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = {'bar': i, 'price': win_low, 'direction': -1}
                elif last_direction == 0:
                    new_pivot = {'bar': i, 'price': win_low, 'direction': -1}
            elif last_direction == -1 and win_low < last_pivot_price:
                if pivots:
                    pivots[-1] = {'bar': i, 'price': win_low, 'direction': -1}
                last_pivot_bar = i
                last_pivot_price = win_low
                continue

        if new_pivot:
            pivots.append(new_pivot)
            last_pivot_bar = new_pivot['bar']
            last_pivot_price = new_pivot['price']
            last_direction = new_pivot['direction']

    print(f"ZigZag: found {len(pivots)} pivots")
    return pivots


# ---------------------------------------------------------------------------
# Pivot unit extraction
# ---------------------------------------------------------------------------

def build_pivot_units(df: pd.DataFrame, pivots: List[Dict]) -> List[Dict]:
    """
    Build pattern units: each pivot with incoming leg + outgoing leg.
    Returns list of unit dicts with bar ranges and encoded sequences.
    """
    if len(pivots) < 3:
        print("Not enough pivots for pattern units")
        return []

    units = []
    
    for i in range(1, len(pivots) - 1):
        prev_pivot = pivots[i - 1]
        curr_pivot = pivots[i]
        next_pivot = pivots[i + 1]
        
        start_bar = prev_pivot['bar']
        pivot_bar = curr_pivot['bar']
        end_bar = next_pivot['bar']
        
        # Direction of incoming leg (determines normalization)
        incoming_direction = 1 if curr_pivot['price'] > prev_pivot['price'] else -1
        
        # Extract L0/L1 sequence for the full unit
        unit_df = df.loc[start_bar:end_bar].copy()
        
        # Skip if missing data
        if unit_df['L0_numeric'].isna().any() or unit_df['L1_numeric'].isna().any():
            continue
        
        l0_seq = unit_df['L0_numeric'].values.astype(float)
        l1_seq = unit_df['L1_numeric'].values.astype(float)
        
        # Direction normalize: flip signs if incoming leg was down
        if incoming_direction == -1:
            l0_seq = -l0_seq
            l1_seq = -l1_seq
        
        # Stack into 2D array for DTW
        sequence = np.column_stack([l0_seq, l1_seq])
        
        # Price magnitude of the pivot move
        incoming_magnitude = abs(curr_pivot['price'] - prev_pivot['price'])
        outgoing_magnitude = abs(next_pivot['price'] - curr_pivot['price'])
        
        units.append({
            'unit_id': i,
            'start_bar': start_bar,
            'pivot_bar': pivot_bar,
            'end_bar': end_bar,
            'length': end_bar - start_bar + 1,
            'incoming_direction': incoming_direction,
            'incoming_magnitude': incoming_magnitude,
            'outgoing_magnitude': outgoing_magnitude,
            'sequence': sequence,
            'start_time': df.loc[start_bar, 'time'],
            'pivot_time': df.loc[pivot_bar, 'time'],
            'end_time': df.loc[end_bar, 'time']
        })
    
    print(f"Built {len(units)} pivot units")
    return units

# DTW functions
# ---------------------------------------------------------------------------

def compute_dtw_distance(seq1: np.ndarray, seq2: np.ndarray, band_ratio: float = 0.5) -> float:
    """
    Compute DTW distance between two sequences (full alignment).
    band_ratio: Sakoe-Chiba band as fraction of average length.
    """
    if not HAS_TSLEARN:
        raise RuntimeError("tslearn required for DTW")
    
    avg_len = (len(seq1) + len(seq2)) / 2
    band_size = max(1, int(avg_len * band_ratio))
    
    distance = tslearn_dtw(seq1, seq2, global_constraint="sakoe_chiba", 
                           sakoe_chiba_radius=band_size)
    return distance


def compute_subsequence_dtw(long_seq: np.ndarray, short_seq: np.ndarray, 
                             band_ratio: float = 0.5) -> Tuple[float, int, int]:
    """
    Find best matching subsequence of long_seq that aligns with short_seq.
    Returns (distance, start_idx, end_idx) of the match within long_seq.
    
    Uses sliding window approach with DTW at each position.
    """
    if not HAS_TSLEARN:
        raise RuntimeError("tslearn required for DTW")
    
    short_len = len(short_seq)
    long_len = len(long_seq)
    
    if short_len >= long_len:
        # Not a subsequence situation, use regular DTW
        dist = compute_dtw_distance(long_seq, short_seq, band_ratio)
        return dist, 0, long_len - 1
    
    best_distance = float('inf')
    best_start = 0
    best_end = short_len - 1
    
    # Allow window sizes from short_len to short_len * 1.5 (within band tolerance)
    min_window = short_len
    max_window = min(long_len, int(short_len * 1.5))
    
    for window_size in range(min_window, max_window + 1):
        band_size = max(1, int(window_size * band_ratio))
        
        for start in range(long_len - window_size + 1):
            end = start + window_size
            subseq = long_seq[start:end]
            
            try:
                dist = tslearn_dtw(subseq, short_seq, global_constraint="sakoe_chiba",
                                   sakoe_chiba_radius=band_size)
                
                if dist < best_distance:
                    best_distance = dist
                    best_start = start
                    best_end = end - 1
            except:
                continue
    
    return best_distance, best_start, best_end


# ---------------------------------------------------------------------------
# Union-Find for clustering
# ---------------------------------------------------------------------------

class UnionFind:
    """Simple union-find data structure for clustering."""
    
    def __init__(self):
        self.parent = {}
        self.rank = {}
    
    def find(self, x):
        if x not in self.parent:
            self.parent[x] = x
            self.rank[x] = 0
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # Path compression
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return
        # Union by rank
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
    
    def get_clusters(self) -> Dict[int, List[int]]:
        """Return dict mapping cluster_id -> list of member unit_ids."""
        clusters = defaultdict(list)
        for x in self.parent:
            clusters[self.find(x)].append(x)
        return dict(clusters)

def find_pattern_matches(units: List[Dict], dtw_threshold: float, 
                         band_ratio: float = 0.5, min_length: int = 5) -> Tuple[List[Dict], Dict[int, int]]:
    """
    Find matching patterns using DTW. Compares ALL pairs (no consumption).
    Uses subsequence DTW when lengths differ significantly.
    Returns (list of match dicts, dict mapping unit_id -> cluster_id).
    """
    if not units:
        return [], {}
    
    # Filter by minimum length
    valid_units = [u for u in units if u['length'] >= min_length]
    
    if not valid_units:
        return [], {}
    
    matches = []
    uf = UnionFind()
    
    n_units = len(valid_units)
    total_comparisons = n_units * (n_units - 1) // 2
    
    print(f"DTW matching: {n_units} units, {total_comparisons:,} comparisons")
    prog = ProgressTracker("DTW matching        ", total_comparisons)
    
    comparison_count = 0
    
    # Compare all pairs
    for i in range(n_units):
        unit1 = valid_units[i]
        
        for j in range(i + 1, n_units):
            unit2 = valid_units[j]
            
            # Determine match type based on length ratio
            len_ratio = min(unit1['length'], unit2['length']) / max(unit1['length'], unit2['length'])
            
            try:
                if len_ratio >= 0.5:
                    # Similar lengths: full DTW
                    distance = compute_dtw_distance(unit1['sequence'], unit2['sequence'], band_ratio)
                    match_type = 'full'
                    match_start_1 = 0
                    match_end_1 = unit1['length'] - 1
                    match_start_2 = 0
                    match_end_2 = unit2['length'] - 1
                else:
                    # Different lengths: subsequence DTW
                    if unit1['length'] > unit2['length']:
                        distance, match_start_1, match_end_1 = compute_subsequence_dtw(
                            unit1['sequence'], unit2['sequence'], band_ratio)
                        match_start_2 = 0
                        match_end_2 = unit2['length'] - 1
                    else:
                        distance, match_start_2, match_end_2 = compute_subsequence_dtw(
                            unit2['sequence'], unit1['sequence'], band_ratio)
                        match_start_1 = 0
                        match_end_1 = unit1['length'] - 1
                    match_type = 'partial'
                
                # Normalize by the shorter sequence length for comparability
                shorter_len = min(unit1['length'], unit2['length'])
                normalized_distance = distance / shorter_len
                
                if normalized_distance <= dtw_threshold:
                    # Record match
                    matches.append({
                        'unit1_id': unit1['unit_id'],
                        'unit2_id': unit2['unit_id'],
                        'unit1_start': unit1['start_bar'],
                        'unit1_end': unit1['end_bar'],
                        'unit2_start': unit2['start_bar'],
                        'unit2_end': unit2['end_bar'],
                        'unit1_length': unit1['length'],
                        'unit2_length': unit2['length'],
                        'match_type': match_type,
                        'match_start_1': match_start_1,
                        'match_end_1': match_end_1,
                        'match_start_2': match_start_2,
                        'match_end_2': match_end_2,
                        'dtw_distance': distance,
                        'normalized_distance': normalized_distance,
                        'unit1_incoming_mag': unit1['incoming_magnitude'],
                        'unit1_outgoing_mag': unit1['outgoing_magnitude'],
                        'unit2_incoming_mag': unit2['incoming_magnitude'],
                        'unit2_outgoing_mag': unit2['outgoing_magnitude'],
                        'unit1_start_time': unit1['start_time'],
                        'unit1_end_time': unit1['end_time'],
                        'unit2_start_time': unit2['start_time'],
                        'unit2_end_time': unit2['end_time']
                    })
                    
                    # Union for clustering
                    uf.union(unit1['unit_id'], unit2['unit_id'])
                    
            except Exception as e:
                print(f"\nDTW error comparing units {unit1['unit_id']} and {unit2['unit_id']}: {e}")
            
            comparison_count += 1
            prog.tick()
    
    prog.done()
    
    # Build cluster assignments
    clusters = uf.get_clusters()
    unit_to_cluster = {}
    for cluster_id, (root, members) in enumerate(clusters.items()):
        for unit_id in members:
            unit_to_cluster[unit_id] = cluster_id
    
    print(f"Found {len(matches)} pattern matches in {len(clusters)} clusters")
    return matches, unit_to_cluster

def export_visualization_csv(df: pd.DataFrame, units: List[Dict], matches: List[Dict],
                              unit_to_cluster: Dict[int, int], output_path: str) -> None:
    """
    Export pattern visualization for MT5.
    Same cluster = same color. Brighter = lower DTW distance (tighter match).
    """
    # Base colors for clusters (will be brightness-adjusted in MQL5)
    colors = ['BLUE', 'RED', 'GREEN', 'MAGENTA', 'CYAN',
              'ORANGE', 'PURPLE', 'YELLOW', 'LIME', 'MAROON']
    
    if not matches:
        print("No matches to visualize")
        pd.DataFrame(columns=['pattern_type', 'start_time', 'end_time',
                               'high_price', 'low_price', 'color', 'cluster_id',
                               'normalized_distance', 'description']
                     ).to_csv(output_path, index=False, encoding='ANSI')
        return
    
    # Build unit lookup
    unit_lookup = {u['unit_id']: u for u in units}
    
    # Track best (lowest) distance for each unit (for brightness)
    unit_best_distance = {}
    for m in matches:
        d = m['normalized_distance']
        u1, u2 = m['unit1_id'], m['unit2_id']
        if u1 not in unit_best_distance or d < unit_best_distance[u1]:
            unit_best_distance[u1] = d
        if u2 not in unit_best_distance or d < unit_best_distance[u2]:
            unit_best_distance[u2] = d
    
    # Collect all units that are in matches
    units_in_matches = set()
    for m in matches:
        units_in_matches.add(m['unit1_id'])
        units_in_matches.add(m['unit2_id'])
    
    rows = []
    for unit_id in units_in_matches:
        if unit_id not in unit_lookup:
            continue
        
        unit = unit_lookup[unit_id]
        cluster_id = unit_to_cluster.get(unit_id, -1)
        color = colors[cluster_id % len(colors)] if cluster_id >= 0 else 'GRAY'
        best_dist = unit_best_distance.get(unit_id, 1.0)
        
        try:
            slice_df = df.loc[unit['start_bar']:unit['end_bar']]
            high = slice_df['close'].max()
            low = slice_df['close'].min()
            price_range = max(high - low, 0.001)
            
            # Pattern type: C{cluster}_U{unit}_L{length}
            pattern_type = f"C{cluster_id}_U{unit_id}_L{unit['length']}"
            
            # Description with parseable metadata
            description = (f"D={best_dist:.3f} "
                          f"InMag={unit['incoming_magnitude']:.5f} "
                          f"OutMag={unit['outgoing_magnitude']:.5f}")
            
            rows.append({
                'pattern_type': pattern_type,
                'start_time': unit['start_time'],
                'end_time': unit['end_time'],
                'high_price': round(high + price_range * 0.1, 6),
                'low_price': round(low - price_range * 0.1, 6),
                'color': color,
                'cluster_id': cluster_id,
                'normalized_distance': best_dist,
                'description': description
            })
        except Exception as e:
            print(f"Error processing unit {unit_id}: {e}")
    
    pd.DataFrame(rows).to_csv(output_path, index=False, encoding='ANSI')
    print(f"Exported {len(rows)} visualization blocks to {output_path}")

def run_analysis(input_path: str, point: float = 0.001,
                 dtw_threshold: float = 1.0, band_ratio: float = 0.5,
                 min_length: int = 5, zz_depth: int = 12, 
                 zz_deviation: int = 5, zz_backstep: int = 3,
                 max_units: int = 0) -> Tuple[pd.DataFrame, List[Dict]]:
    
    print(f"Loading {input_path}...")
    try:
        df_raw = pd.read_csv(input_path, encoding='utf-16')
    except Exception:
        df_raw = pd.read_csv(input_path, encoding='utf-8')
    print(f"Loaded {len(df_raw):,} bars")

    # Compute ZigZag pivots
    print("Computing ZigZag pivots...")
    pivots = compute_zigzag_pivots(df_raw, depth=zz_depth, deviation=zz_deviation,
                                    backstep=zz_backstep, point=point)

    # L0/L1 calculation
    df = pass1_raw_calculations(df_raw, point)
    df = pass2_l1_calculation(df)
    df = pass3_l0_calculation(df, point)
    df = pass4_symbol_encoding(df)

    valid_symbols = df['symbol'].notna().sum()
    print(f"Generated {valid_symbols:,} symbols")

    # Build pivot units
    print("Building pivot units...")
    units = build_pivot_units(df, pivots)
    
    # Limit to last N units if specified (for testing)
    if max_units > 0 and len(units) > max_units:
        print(f"Limiting to last {max_units} units (from {len(units)})")
        units = units[-max_units:]

    # DTW matching
    matches = []
    unit_to_cluster = {}
    if HAS_TSLEARN and units:
        matches, unit_to_cluster = find_pattern_matches(
            units, dtw_threshold=dtw_threshold,
            band_ratio=band_ratio, min_length=min_length)
    elif not HAS_TSLEARN:
        print("Skipping DTW matching - tslearn not installed")

    # Export
    base_name = input_path.replace('.csv', '')
    export_l1_output(df, f"{base_name}_L1_output.csv")
    export_pivots(pivots, f"{base_name}_pivots.csv")
    export_units(units, unit_to_cluster, f"{base_name}_units.csv")
    export_matches(matches, f"{base_name}_matches.csv")
    export_visualization_csv(df, units, matches, unit_to_cluster, f"{base_name}_patterns.csv")

    return df, matches

//| Adjust color brightness                                            |
//| factor > 1.0 = brighter, factor < 1.0 = darker                    |
//| Standalone utility function for general use                        |
//+------------------------------------------------------------------+
color AdjustColorBrightness(color baseColor, double factor)
{
   // Extract RGB components
   int r = (int)((baseColor & 0x0000FF));
   int g = (int)((baseColor & 0x00FF00) >> 8);
   int b = (int)((baseColor & 0xFF0000) >> 16);
   
   // Apply factor
   if(factor > 1.0)
   {
      // Brighten: move toward 255
      r = r + (int)((255 - r) * (factor - 1.0));
      g = g + (int)((255 - g) * (factor - 1.0));
      b = b + (int)((255 - b) * (factor - 1.0));
   }
   else
   {
      // Darken: multiply directly
      r = (int)(r * factor);
      g = (int)(g * factor);
      b = (int)(b * factor);
   }
   
   // Clamp to 0-255
   r = MathMax(0, MathMin(255, r));
   g = MathMax(0, MathMin(255, g));
   b = MathMax(0, MathMin(255, b));
   
   // Reconstruct color
   return (color)((b << 16) | (g << 8) | r);
}

//| Parse DTW distance from description string                         |
//| Format: "D=0.123 InMag=... OutMag=..."                            |
//+------------------------------------------------------------------+
double ParseDTWDistance(string description)
{
   int dPos = StringFind(description, "D=");
   if(dPos < 0) return -1.0;
   
   int spacePos = StringFind(description, " ", dPos);
   if(spacePos < 0) spacePos = StringLen(description);
   
   string distStr = StringSubstr(description, dPos + 2, spacePos - dPos - 2);
   return StringToDouble(distStr);
}

{
   int lPos = StringFind(patternType, "_L");
   if(lPos < 0) return 0;
   
   string lenStr = StringSubstr(patternType, lPos + 2);
   return (int)StringToInteger(lenStr);

int ParseClusterId(string patternType)
{
   if(StringFind(patternType, "C") != 0) return -1;
   
   int uPos = StringFind(patternType, "_U");
   if(uPos < 0) return -1;
   
   string clusterStr = StringSubstr(patternType, 1, uPos - 1);

//+------------------------------------------------------------------+
//| Calculate brightness factor based on DTW distance                  |
//| Lower distance = brighter (better match)                           |
//+------------------------------------------------------------------+
double CalculateBrightnessFactor(double distance, double maxDistance)
{
   if(maxDistance <= 0) return 1.0;
   
   // Clamp distance to range
   double d = MathMax(0, MathMin(distance, maxDistance));
   
   // Invert: 0 distance = max brightness, maxDistance = min brightness
   double ratio = 1.0 - (d / maxDistance);
   
   // Map to brightness range
   return InpBrightnessMin + ratio * (InpBrightnessMax - InpBrightnessMin);
}
