class ProgressTracker:
    """
    Single-line overwriting progress display.
    Call tick() each iteration (or tick(n) for batches).
    Call done() when finished.
    Prints at most once every `interval` seconds to reduce I/O overhead.
    """

    def __init__(self, label: str, total: int, interval: float = 2.0):
        self.label    = label
        self.total    = total
        self.interval = interval
        self.completed = 0
        self.start    = time.monotonic()
        self.last_print = 0.0  # force first print after first tick

    @staticmethod
    def _fmt_seconds(s: float) -> str:
        s = int(s)
        h, rem = divmod(s, 3600)
        m, sec = divmod(rem, 60)
        return f"{h:02d}:{m:02d}:{sec:02d}"

    def tick(self, n: int = 1) -> None:
        self.completed += n
        now = time.monotonic()
        if now - self.last_print < self.interval:
            return
        self.last_print = now
        elapsed = now - self.start

        pct = self.completed / self.total * 100 if self.total > 0 else 0.0
        elapsed_str = self._fmt_seconds(elapsed)

        if elapsed > 0 and self.completed > 0:
            rate = self.completed / elapsed
            remaining = (self.total - self.completed) / rate
            eta_str = self._fmt_seconds(remaining)
        else:
            eta_str = "--:--:--"

        line = (f"\r{self.label}: {self.completed:,} / {self.total:,} "
                f"({pct:.1f}%) | {elapsed_str} elapsed | ETA: {eta_str} remaining   ")
        print(line, end='', flush=True)

    def done(self) -> None:
        elapsed = time.monotonic() - self.start
        elapsed_str = self._fmt_seconds(elapsed)
        line = (f"\r{self.label}: {self.total:,} / {self.total:,} "
                f"(100.0%) | {elapsed_str} elapsed | done                          ")
        print(line, flush=True)

def round_to_point(value: float, point: float = 0.001) -> float:
    """Round to nearest point (0.001 for USDJPY)."""
    return round(value / point) * point


def calculate_spreads(tema: float, kc_upper: float, kc_middle: float, kc_lower: float, point: float) -> dict:
    """Calculate spreads from TEMA to each KC line, rounded to point."""
    return {
        'spread_upper':  round_to_point(tema - kc_upper,  point),
        'spread_middle': round_to_point(tema - kc_middle, point),
        'spread_lower':  round_to_point(tema - kc_lower,  point)
    }


def determine_active_line(spreads: dict) -> Tuple[str, str]:
    abs_spreads = {
        'U': abs(spreads['spread_upper']),
        'M': abs(spreads['spread_middle']),
        'L': abs(spreads['spread_lower'])
    }
    active = min(abs_spreads, key=abs_spreads.get)
    return active, active


def calculate_spread_delta_state(spread_curr: float, spread_prev: float) -> str:
    if spread_curr == 0 and spread_prev == 0:
        return 'SD0'
    if spread_prev != 0 and spread_curr != 0:
        if (spread_curr > 0) != (spread_prev > 0):
            return 'SDX'
    elif spread_prev == 0 or spread_curr == 0:
        if spread_prev != 0 or spread_curr != 0:
            return 'SDX'
    delta = abs(spread_curr) - abs(spread_prev)
    if delta > 0:
        return 'SD+'
    elif delta < 0:
        return 'SD-'
    else:
        return 'SD0'


def calculate_l0_state(slope_delta: float) -> int:
    if slope_delta > 0:
        return 1
    elif slope_delta < 0:
        return -1
    else:
        return 0

def zigzag_max_length(df: pd.DataFrame, depth: int = 12, deviation: int = 5, backstep: int = 3, point: float = 0.001) -> int:
    """
    Compute ZigZag(depth, deviation, backstep) on High/Low.
    Track longest leg in bars as it builds. Return 2x that value as max_length.
    """
    highs  = df['High'].values
    lows   = df['Low'].values
    n      = len(highs)
    dev    = deviation * point

    pivots          = []
    last_pivot_bar  = 0
    last_pivot_price = 0.0
    last_direction  = 0  # 0=none, 1=high, -1=low
    max_leg_bars    = 0

    for i in range(depth, n):
        window_start = max(0, i - depth + 1)
        win_high_idx = window_start + int(np.argmax(highs[window_start:i + 1]))
        win_low_idx  = window_start + int(np.argmin(lows[window_start:i + 1]))
        win_high     = highs[win_high_idx]
        win_low      = lows[win_low_idx]

        new_pivot = None

        # Candidate swing high
        if win_high_idx == i:
            if last_direction != 1:
                if last_direction == -1 and (win_high - last_pivot_price) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = (i, win_high, 1)
                elif last_direction == 0:
                    new_pivot = (i, win_high, 1)
            elif last_direction == 1 and win_high > last_pivot_price:
                if pivots:
                    pivots[-1] = (i, win_high, 1)
                last_pivot_bar   = i
                last_pivot_price = win_high
                continue

        # Candidate swing low
        if win_low_idx == i:
            if last_direction != -1:
                if last_direction == 1 and (last_pivot_price - win_low) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = (i, win_low, -1)
                elif last_direction == 0:
                    new_pivot = (i, win_low, -1)
            elif last_direction == -1 and win_low < last_pivot_price:
                if pivots:
                    pivots[-1] = (i, win_low, -1)
                last_pivot_bar   = i
                last_pivot_price = win_low
                continue

        if new_pivot:
            if pivots:
                leg_bars = new_pivot[0] - pivots[-1][0]
                if leg_bars > max_leg_bars:
                    max_leg_bars = leg_bars
            pivots.append(new_pivot)
            last_pivot_bar   = new_pivot[0]
            last_pivot_price = new_pivot[1]
            last_direction   = new_pivot[2]

    max_length = max_leg_bars * 3
    print(f"ZigZag: longest leg = {max_leg_bars} bars → max_length = {max_length}")
    return max_length


def pass5_pattern_mining(df: pd.DataFrame, min_length: int = 2, min_frequency: int = 2, max_length: int = 0) -> dict:

        return {}

    if max_length <= 0 or max_length > n // 2:
        max_length = n // 2
    print(f"Pass 5: {n:,} symbols, max_length={max_length:,}")

    total_iters = sum(n - l + 1 for l in range(min_length, max_length + 1))
    print(f"  → {total_iters:,} total iterations per sub-pass")

    # ------------------------------------------------------------------

    # Sub-pass B: record instances only for qualifying patterns

    pattern_data = {p: {'count': 0, 'instances': []} for p in qualifying}

Weighed fuzzy matching algorithms for trading pattern recognition







Two separate issues to address, and I'd recommend tackling them in order:

Step 1 — Reduce symbol dimensionality first (not fuzzy, just less strict)

Drop Z and ZS from the matching key, keep only L0+L1. That reduces from ~72 possible symbols per bar to 12. This alone may dramatically extend the longest repeating exact matches before you even introduce fuzziness. Fast change, no algorithm redesign. Worth doing regardless.

Step 2 — Fuzzy matching

Three candidates:

Levenshtein — your intuition was right. Allows substitution, insertion, deletion. Insertion/deletion means "this pattern had an extra noise bar in the middle" which is valid for trading. Straightforward to implement and interpret. Weakness: treats all substitutions equally — swapping L0:+1 for L0:-1 costs the same as swapping SD+ for SD-, even though those are very different kinds of differences.

DTW (Dynamic Time Warping) — designed specifically for time series that are structurally similar but stretched or compressed in time. A 47-bar thrust and a 38-bar thrust of the same shape would score as similar. This is almost certainly what you're actually seeing with your patterns. The gold standard for this class of problem.

Hamming — substitutions only, requires equal-length sequences. Too restrictive for your use case.

    print("WARNING: tslearn not installed. Run: pip install tslearn --break-system-packages")


# ---------------------------------------------------------------------------
# Progress tracking
# ---------------------------------------------------------------------------

class ProgressTracker:
    def __init__(self, label: str, total: int, interval: float = 2.0):
        self.label    = label
        self.total    = total
        self.interval = interval
        self.completed = 0
        self.start    = time.monotonic()
        self.last_print = 0.0

    @staticmethod
    def _fmt_seconds(s: float) -> str:
        s = int(s)
        h, rem = divmod(s, 3600)
        m, sec = divmod(rem, 60)
        return f"{h:02d}:{m:02d}:{sec:02d}"

    def tick(self, n: int = 1) -> None:
        self.completed += n
        now = time.monotonic()
        if now - self.last_print < self.interval:
            return
        self.last_print = now
        elapsed = now - self.start
        pct = self.completed / self.total * 100 if self.total > 0 else 0.0
        elapsed_str = self._fmt_seconds(elapsed)
        if elapsed > 0 and self.completed > 0:
            rate = self.completed / elapsed
            remaining = (self.total - self.completed) / rate
            eta_str = self._fmt_seconds(remaining)
        else:
            eta_str = "--:--:--"
        line = (f"\r{self.label}: {self.completed:,} / {self.total:,} "
                f"({pct:.1f}%) | {elapsed_str} elapsed | ETA: {eta_str}   ")
        print(line, end='', flush=True)

    def done(self) -> None:
        elapsed = time.monotonic() - self.start
        elapsed_str = self._fmt_seconds(elapsed)
        line = (f"\r{self.label}: {self.total:,} / {self.total:,} "
                f"(100.0%) | {elapsed_str} elapsed | done                ")
        print(line, flush=True)

# Helpers

def round_to_point(value: float, point: float = 0.001) -> float:
    return round(value / point) * point


def calculate_spreads(tema: float, kc_upper: float, kc_middle: float, kc_lower: float, point: float) -> dict:
    return {
        'spread_upper':  round_to_point(tema - kc_upper,  point),
        'spread_middle': round_to_point(tema - kc_middle, point),
        'spread_lower':  round_to_point(tema - kc_lower,  point)
    }


def determine_active_line(spreads: dict) -> Tuple[str, str]:
    abs_spreads = {
        'U': abs(spreads['spread_upper']),
        'M': abs(spreads['spread_middle']),
        'L': abs(spreads['spread_lower'])
    }
    active = min(abs_spreads, key=abs_spreads.get)
    return active, active


def calculate_spread_delta_state(spread_curr: float, spread_prev: float) -> str:
    if spread_curr == 0 and spread_prev == 0:
        return 'SD0'
    if spread_prev != 0 and spread_curr != 0:
        if (spread_curr > 0) != (spread_prev > 0):
            return 'SDX'
    elif spread_prev == 0 or spread_curr == 0:
        if spread_prev != 0 or spread_curr != 0:
            return 'SDX'
    delta = abs(spread_curr) - abs(spread_prev)
    if delta > 0:
        return 'SD+'
    elif delta < 0:
        return 'SD-'
    else:
        return 'SD0'


def calculate_l0_state(slope_delta: float) -> int:
    if slope_delta > 0:
        return 1
    elif slope_delta < 0:
        return -1
    else:
        return 0


def l1_to_numeric(l1_state: str) -> int:
    """Convert L1 state string to numeric for DTW."""
    mapping = {'SD+': 1, 'SD-': -1, 'SD0': 0, 'SDX': 0}
    return mapping.get(l1_state, 0)

def pass1_raw_calculations(df: pd.DataFrame, point: float = 0.001) -> pd.DataFrame:
    n = len(df)
    prog = ProgressTracker("Pass 1 (spreads/zones)", n)
    results = []
    for i, row in df.iterrows():
        spreads = calculate_spreads(
            row['TEMA'], row['KC_Upper'], row['KC_Middle'], row['KC_Lower'], point
        )
        active_line, zone = determine_active_line(spreads)
        results.append({
            'bar_index':     i,
            'time':          row['Time'],
            'close':         row['Close'],
            'tema':          row['TEMA'],
            'kc_upper':      row['KC_Upper'],
            'kc_middle':     row['KC_Middle'],
            'kc_lower':      row['KC_Lower'],
            'spread_upper':  spreads['spread_upper'],
            'spread_middle': spreads['spread_middle'],
            'spread_lower':  spreads['spread_lower'],
            'active_line':   active_line,
            'zone':          zone
        })
        prog.tick()
    prog.done()
    return pd.DataFrame(results)


def pass2_l1_calculation(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df['L1_state']         = None
    df['zone_switch']      = 'N'
    df['active_spread']    = None
    df['active_spread_prev'] = None

    n = len(df)
    prog = ProgressTracker("Pass 2 (L1 states)  ", n - 1)
    for i in range(1, n):
        curr = df.iloc[i]
        prev = df.iloc[i - 1]
        al = curr['active_line']
        spread_key = {'U': 'spread_upper', 'M': 'spread_middle', 'L': 'spread_lower'}[al]
        spread_curr = curr[spread_key]
        spread_prev = prev[spread_key]
        l1_state    = calculate_spread_delta_state(spread_curr, spread_prev)
        zone_switch = 'Y' if curr['active_line'] != prev['active_line'] else 'N'
        df.at[i, 'L1_state']          = l1_state
        df.at[i, 'zone_switch']       = zone_switch
        df.at[i, 'active_spread']     = spread_curr
        df.at[i, 'active_spread_prev'] = spread_prev
        prog.tick()
    prog.done()
    return df


def pass3_l0_calculation(df: pd.DataFrame, point: float = 0.001) -> pd.DataFrame:
    df = df.copy()
    df['slope_1']    = None
    df['slope_2']    = None
    df['slope_delta'] = None
    df['L0_state']   = None

    n = len(df)
    prog = ProgressTracker("Pass 3 (L0 states)  ", n - 2)
    for i in range(2, n):
        tema_n  = df.iloc[i]['tema']
        tema_n1 = df.iloc[i - 1]['tema']
        tema_n2 = df.iloc[i - 2]['tema']
        slope_1     = round_to_point(tema_n1 - tema_n2, point)
        slope_2     = round_to_point(tema_n  - tema_n1, point)
        slope_delta = round_to_point(slope_2 - slope_1, point)
        df.at[i, 'slope_1']    = slope_1
        df.at[i, 'slope_2']    = slope_2
        df.at[i, 'slope_delta'] = slope_delta
        df.at[i, 'L0_state']   = calculate_l0_state(slope_delta)
        prog.tick()
    prog.done()
    return df


def pass4_symbol_encoding(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df['symbol'] = None
    df['L0_numeric'] = None
    df['L1_numeric'] = None

    n = len(df)
    prog = ProgressTracker("Pass 4 (symbols)    ", n - 2)
    for i in range(2, n):
        row = df.iloc[i]
        if pd.isna(row['L0_state']) or pd.isna(row['L1_state']):
            prog.tick()
            continue
        l0_num = int(row['L0_state'])
        l1_num = l1_to_numeric(row['L1_state'])
        symbol = f"L0:{l0_num:+d}|L1:{row['L1_state']}"
        df.at[i, 'symbol'] = symbol
        df.at[i, 'L0_numeric'] = l0_num
        df.at[i, 'L1_numeric'] = l1_num
        prog.tick()
    prog.done()
    return df

# ZigZag with pivot extraction
# ---------------------------------------------------------------------------

def compute_zigzag_pivots(df: pd.DataFrame, depth: int = 12, deviation: int = 5, 
                          backstep: int = 3, point: float = 0.001) -> List[Dict]:
    """
    Compute ZigZag pivots. Returns list of pivot dicts:
    {'bar': int, 'price': float, 'direction': int}
    direction: 1 = peak (high), -1 = trough (low)
    """
    highs = df['High'].values
    lows = df['Low'].values
    n = len(highs)
    dev = deviation * point

    pivots = []
    last_pivot_bar = 0
    last_pivot_price = 0.0
    last_direction = 0

    for i in range(depth, n):
        window_start = max(0, i - depth + 1)
        win_high_idx = window_start + int(np.argmax(highs[window_start:i + 1]))
        win_low_idx = window_start + int(np.argmin(lows[window_start:i + 1]))
        win_high = highs[win_high_idx]
        win_low = lows[win_low_idx]

        new_pivot = None

        # Candidate swing high
        if win_high_idx == i:
            if last_direction != 1:
                if last_direction == -1 and (win_high - last_pivot_price) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = {'bar': i, 'price': win_high, 'direction': 1}
                elif last_direction == 0:
                    new_pivot = {'bar': i, 'price': win_high, 'direction': 1}
            elif last_direction == 1 and win_high > last_pivot_price:
                if pivots:
                    pivots[-1] = {'bar': i, 'price': win_high, 'direction': 1}
                last_pivot_bar = i
                last_pivot_price = win_high
                continue

        # Candidate swing low
        if win_low_idx == i:
            if last_direction != -1:
                if last_direction == 1 and (last_pivot_price - win_low) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = {'bar': i, 'price': win_low, 'direction': -1}
                elif last_direction == 0:
                    new_pivot = {'bar': i, 'price': win_low, 'direction': -1}
            elif last_direction == -1 and win_low < last_pivot_price:
                if pivots:
                    pivots[-1] = {'bar': i, 'price': win_low, 'direction': -1}
                last_pivot_bar = i
                last_pivot_price = win_low
                continue

        if new_pivot:
            pivots.append(new_pivot)
            last_pivot_bar = new_pivot['bar']
            last_pivot_price = new_pivot['price']
            last_direction = new_pivot['direction']

    print(f"ZigZag: found {len(pivots)} pivots")
    return pivots

# Pivot unit extraction
# ---------------------------------------------------------------------------

def build_pivot_units(df: pd.DataFrame, pivots: List[Dict]) -> List[Dict]:
    """
    Build pattern units: each pivot with incoming leg + outgoing leg.
    Returns list of unit dicts with bar ranges and encoded sequences.
    """
    if len(pivots) < 3:
        print("Not enough pivots for pattern units")
        return []

    units = []
    
    for i in range(1, len(pivots) - 1):
        prev_pivot = pivots[i - 1]
        curr_pivot = pivots[i]
        next_pivot = pivots[i + 1]
        
        start_bar = prev_pivot['bar']
        pivot_bar = curr_pivot['bar']
        end_bar = next_pivot['bar']
        
        # Direction of incoming leg (determines normalization)
        incoming_direction = 1 if curr_pivot['price'] > prev_pivot['price'] else -1
        
        # Extract L0/L1 sequence for the full unit
        unit_df = df.loc[start_bar:end_bar].copy()
        
        # Skip if missing data
        if unit_df['L0_numeric'].isna().any() or unit_df['L1_numeric'].isna().any():
            continue
        
        l0_seq = unit_df['L0_numeric'].values.astype(float)
        l1_seq = unit_df['L1_numeric'].values.astype(float)
        
        # Direction normalize: flip signs if incoming leg was down
        if incoming_direction == -1:
            l0_seq = -l0_seq
            l1_seq = -l1_seq
        
        # Stack into 2D array for DTW
        sequence = np.column_stack([l0_seq, l1_seq])
        
        # Price magnitude of the pivot move
        incoming_magnitude = abs(curr_pivot['price'] - prev_pivot['price'])
        outgoing_magnitude = abs(next_pivot['price'] - curr_pivot['price'])
        
        units.append({
            'unit_id': i,
            'start_bar': start_bar,
            'pivot_bar': pivot_bar,
            'end_bar': end_bar,
            'length': end_bar - start_bar + 1,
            'incoming_direction': incoming_direction,
            'incoming_magnitude': incoming_magnitude,
            'outgoing_magnitude': outgoing_magnitude,
            'sequence': sequence,
            'start_time': df.loc[start_bar, 'time'],
            'pivot_time': df.loc[pivot_bar, 'time'],
            'end_time': df.loc[end_bar, 'time']
        })
    
    print(f"Built {len(units)} pivot units")
    return units

# ---------------------------------------------------------------------------

def compute_dtw_distance(seq1: np.ndarray, seq2: np.ndarray, band_ratio: float = 0.5) -> float:
    """
    Compute DTW distance between two sequences.
    band_ratio: Sakoe-Chiba band as fraction of average length.
    """
    if not HAS_TSLEARN:
        raise RuntimeError("tslearn required for DTW")
    
    avg_len = (len(seq1) + len(seq2)) / 2
    band_size = max(1, int(avg_len * band_ratio))
    
    # tslearn expects 3D arrays: (n_timestamps, n_features)
    distance = tslearn_dtw(seq1, seq2, global_constraint="sakoe_chiba", 
                           sakoe_chiba_radius=band_size)
    return distance


def find_pattern_matches(units: List[Dict], dtw_threshold: float, 
                         band_ratio: float = 0.5, min_length: int = 5) -> List[Dict]:
    """
    Find matching patterns using DTW, processing longest-first with range consumption.
    Returns list of match dicts.
    """
    if not units:
        return []
    
    # Sort by length, longest first
    sorted_units = sorted(units, key=lambda u: -u['length'])
    
    # Track consumed bar ranges
    consumed = set()
    
    def is_consumed(unit: Dict) -> bool:
        unit_range = set(range(unit['start_bar'], unit['end_bar'] + 1))
        overlap = unit_range & consumed
        # Consider consumed if more than 50% overlap
        return len(overlap) > len(unit_range) * 0.5
    
    def consume(unit: Dict):
        for bar in range(unit['start_bar'], unit['end_bar'] + 1):
            consumed.add(bar)
    
    matches = []
    n_units = len(sorted_units)
    total_comparisons = n_units * (n_units - 1) // 2
    
    print(f"DTW matching: {n_units} units, {total_comparisons:,} comparisons max")
    prog = ProgressTracker("DTW matching        ", total_comparisons)
    
    comparison_count = 0
    
    # Process length tiers
    current_length_tier = sorted_units[0]['length'] if sorted_units else 0
    tier_units = []
    
    for unit in sorted_units:
        # Skip if already consumed
        if is_consumed(unit):
            continue
        
        # Skip if below minimum length
        if unit['length'] < min_length:
            continue
        
        tier_units.append(unit)
    
    # Compare all pairs
    for i, unit1 in enumerate(tier_units):
        if is_consumed(unit1):
            continue
            
        for j, unit2 in enumerate(tier_units[i + 1:], start=i + 1):
            if is_consumed(unit2):
                comparison_count += 1
                prog.tick()
                continue
            
            # Length compatibility check (within 50% band)
            len_ratio = min(unit1['length'], unit2['length']) / max(unit1['length'], unit2['length'])
            if len_ratio < 0.5:
                comparison_count += 1
                prog.tick()
                continue
            
            # Compute DTW
            try:
                distance = compute_dtw_distance(unit1['sequence'], unit2['sequence'], band_ratio)
                
                # Normalize by average length for comparability
                avg_len = (unit1['length'] + unit2['length']) / 2
                normalized_distance = distance / avg_len
                
                if normalized_distance <= dtw_threshold:
                    matches.append({
                        'unit1_id': unit1['unit_id'],
                        'unit2_id': unit2['unit_id'],
                        'unit1_start': unit1['start_bar'],
                        'unit1_end': unit1['end_bar'],
                        'unit2_start': unit2['start_bar'],
                        'unit2_end': unit2['end_bar'],
                        'unit1_length': unit1['length'],
                        'unit2_length': unit2['length'],
                        'dtw_distance': distance,
                        'normalized_distance': normalized_distance,
                        'unit1_incoming_mag': unit1['incoming_magnitude'],
                        'unit1_outgoing_mag': unit1['outgoing_magnitude'],
                        'unit2_incoming_mag': unit2['incoming_magnitude'],
                        'unit2_outgoing_mag': unit2['outgoing_magnitude'],
                        'unit1_start_time': unit1['start_time'],
                        'unit1_end_time': unit1['end_time'],
                        'unit2_start_time': unit2['start_time'],
                        'unit2_end_time': unit2['end_time']
                    })
                    
                    # Consume both ranges
                    consume(unit1)
                    consume(unit2)
                    
            except Exception as e:
                print(f"\nDTW error comparing units {unit1['unit_id']} and {unit2['unit_id']}: {e}")
            
            comparison_count += 1
            prog.tick()
    
    prog.done()
    print(f"Found {len(matches)} pattern matches")
    return matches

def export_l1_output(df: pd.DataFrame, output_path: str) -> None:
    output_cols = [
        'bar_index', 'time', 'close', 'tema',
        'spread_upper', 'spread_middle', 'spread_lower',
        'active_line', 'zone', 'zone_switch',
        'L1_state', 'active_spread',
        'slope_1', 'slope_2', 'slope_delta', 'L0_state',

    existing_cols = [c for c in output_cols if c in df.columns]
    df[existing_cols].to_csv(output_path, index=False, encoding='ANSI')
    print(f"Exported L1 data: {len(df)} rows to {output_path}")

def export_visualization_csv(df: pd.DataFrame, matches: List[Dict], output_path: str) -> None:
    """Export match pairs for MT5 visualization."""
    colors = ['BLUE', 'RED', 'GREEN', 'MAGENTA', 'CYAN',
              'ORANGE', 'PURPLE', 'YELLOW', 'LIME', 'MAROON']
    
    if not matches:
        print("No matches to visualize")
        pd.DataFrame(columns=['pattern_type', 'start_time', 'end_time',
                               'high_price', 'low_price', 'color', 'description']
                     ).to_csv(output_path, index=False, encoding='ANSI')
        return
    
    rows = []
    for match_idx, match in enumerate(matches):
        color = colors[match_idx % len(colors)]
        
        # Unit 1
        try:
            slice1 = df.loc[match['unit1_start']:match['unit1_end']]
            high1 = slice1['close'].max()
            low1 = slice1['close'].min()
            price_range1 = max(high1 - low1, 0.001)
            rows.append({
                'pattern_type': f"M{match_idx+1}_A",
                'start_time': match['unit1_start_time'],
                'end_time': match['unit1_end_time'],
                'high_price': round(high1 + price_range1 * 0.1, 6),
                'low_price': round(low1 - price_range1 * 0.1, 6),
                'color': color,
                'description': f"Match {match_idx+1}A L={match['unit1_length']} D={match['normalized_distance']:.3f}"
            })
        except Exception as e:
            print(f"Error processing unit1 of match {match_idx}: {e}")
        
        # Unit 2
        try:
            slice2 = df.loc[match['unit2_start']:match['unit2_end']]
            high2 = slice2['close'].max()
            low2 = slice2['close'].min()
            price_range2 = max(high2 - low2, 0.001)
            rows.append({
                'pattern_type': f"M{match_idx+1}_B",
                'start_time': match['unit2_start_time'],
                'end_time': match['unit2_end_time'],
                'high_price': round(high2 + price_range2 * 0.1, 6),
                'low_price': round(low2 - price_range2 * 0.1, 6),
                'color': color,
                'description': f"Match {match_idx+1}B L={match['unit2_length']} D={match['normalized_distance']:.3f}"
            })
        except Exception as e:
            print(f"Error processing unit2 of match {match_idx}: {e}")
    
    pd.DataFrame(rows).to_csv(output_path, index=False, encoding='ANSI')
    print(f"Exported {len(rows)} visualization blocks to {output_path}")

def run_analysis(input_path: str, point: float = 0.001,
                 dtw_threshold: float = 1.0, band_ratio: float = 0.5,
                 min_length: int = 5, zz_depth: int = 12, 
                 zz_deviation: int = 5, zz_backstep: int = 3) -> Tuple[pd.DataFrame, List[Dict]]:
    
    print(f"Loading {input_path}...")
    try:
        df_raw = pd.read_csv(input_path, encoding='utf-16')
    except Exception:
        df_raw = pd.read_csv(input_path, encoding='utf-8')
    print(f"Loaded {len(df_raw):,} bars")

    # Compute ZigZag pivots
    print("Computing ZigZag pivots...")
    pivots = compute_zigzag_pivots(df_raw, depth=zz_depth, deviation=zz_deviation,
                                    backstep=zz_backstep, point=point)

    # L0/L1 calculation
    df = pass1_raw_calculations(df_raw, point)
    df = pass2_l1_calculation(df)
    df = pass3_l0_calculation(df, point)
    df = pass4_symbol_encoding(df)

    valid_symbols = df['symbol'].notna().sum()
    print(f"Generated {valid_symbols:,} symbols")

    # Build pivot units
    print("Building pivot units...")
    units = build_pivot_units(df, pivots)

    # DTW matching
    matches = []
    if HAS_TSLEARN and units:
        matches = find_pattern_matches(units, dtw_threshold=dtw_threshold,
                                        band_ratio=band_ratio, min_length=min_length)
    elif not HAS_TSLEARN:
        print("Skipping DTW matching - tslearn not installed")

    # Export
    base_name = input_path.replace('.csv', '')
    export_l1_output(df, f"{base_name}_L1_output.csv")
    export_pivots(pivots, f"{base_name}_pivots.csv")
    export_units(units, f"{base_name}_units.csv")
    export_matches(matches, f"{base_name}_matches.csv")
    export_visualization_csv(df, matches, f"{base_name}_patterns.csv")

    return df, matches
