void CalculateKC(string symbol, ENUM_TIMEFRAMES period, int shift, int emaPeriod, int atrPeriod,
                 double multiplier, ENUM_APPLIED_PRICE appliedPrice,
                 double &upper, double &middle, double &lower)
{
   //--- Get price for middle line calculation
   double price = 0;
   switch(appliedPrice)
   {
      case PRICE_CLOSE: price = iClose(symbol, period, shift); break;
      case PRICE_OPEN: price = iOpen(symbol, period, shift); break;
      case PRICE_HIGH: price = iHigh(symbol, period, shift); break;
      case PRICE_LOW: price = iLow(symbol, period, shift); break;
      case PRICE_MEDIAN: price = (iHigh(symbol, period, shift) + iLow(symbol, period, shift)) / 2.0; break;
      case PRICE_TYPICAL: price = (iHigh(symbol, period, shift) + iLow(symbol, period, shift) + iClose(symbol, period, shift)) / 3.0; break;
      case PRICE_WEIGHTED: price = (iHigh(symbol, period, shift) + iLow(symbol, period, shift) + iClose(symbol, period, shift) * 2.0) / 4.0; break;
   }
   
   //--- Calculate middle line (EMA) manually for this specific bar
   static double emaBuffer[];
   static int lastCalcShift = -1;
   static int lastEmaPeriod = -1;
   
   if(lastEmaPeriod != emaPeriod || shift > lastCalcShift)
   {
      ArraySetAsSeries(emaBuffer, true);
      ArrayResize(emaBuffer, shift + emaPeriod + 100);
      lastEmaPeriod = emaPeriod;
   }
   lastCalcShift = shift;
   
   double emaAlpha = 2.0 / (emaPeriod + 1.0);
   
   if(shift >= ArraySize(emaBuffer) - 1)
   {
      middle = price;
   }
   else if(emaBuffer[shift + 1] == 0)
   {
      middle = price;
   }
   else
   {
      middle = price * emaAlpha + emaBuffer[shift + 1] * (1.0 - emaAlpha);
   }
   emaBuffer[shift] = middle;
   
   //--- Calculate ATR manually for this specific bar
   double tr = MathMax(iHigh(symbol, period, shift) - iLow(symbol, period, shift),
                MathMax(MathAbs(iHigh(symbol, period, shift) - iClose(symbol, period, shift + 1)),
                        MathAbs(iLow(symbol, period, shift) - iClose(symbol, period, shift + 1))));
   
   static double atrBuffer[];
   static int lastAtrPeriod = -1;
   
   if(lastAtrPeriod != atrPeriod || shift > lastCalcShift)
   {
      ArraySetAsSeries(atrBuffer, true);
      ArrayResize(atrBuffer, shift + atrPeriod + 100);
      lastAtrPeriod = atrPeriod;
   }
   
   double atrAlpha = 1.0 / atrPeriod;
   double atr;
   
   if(shift >= ArraySize(atrBuffer) - 1)
   {
      atr = tr;
   }
   else if(atrBuffer[shift + 1] == 0)
   {
      atr = tr;
   }
   else
   {
      atr = tr * atrAlpha + atrBuffer[shift + 1] * (1.0 - atrAlpha);
   }
   atrBuffer[shift] = atr;
   
   //--- Calculate bands
   upper = middle + multiplier * atr;
   lower = middle - multiplier * atr;
}
