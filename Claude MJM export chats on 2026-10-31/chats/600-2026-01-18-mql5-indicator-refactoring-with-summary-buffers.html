   //--- Hide all plots initially (they'll be shown as we configure them)
   for(int p = 0; p < MAX_BUFFERS; p++)
   {
      PlotIndexSetInteger(p, PLOT_DRAW_TYPE, DRAW_NONE);
      PlotIndexSetString(p, PLOT_LABEL, "");
   }
   
   //--- Check sub-windows
   int totalWindows = (int)ChartGetInteger(0, CHART_WINDOWS_TOTAL);
   for(int win = 0; win < totalWindows; win++)

      for(int i = 0; i < subIndicators; i++)

         if(shortName == "")
            continue;

         //--- Skip processing ourselves
         if(shortName == sMeShortName)
            continue;
      
         //--- Check indicator limit
         if(indicatorCount >= MAX_INDICATORS)
         {
            Print("ERROR: Maximum indicator limit (", MAX_INDICATORS, ") reached. Skipping indicator: ", shortName);
            continue;

         Print("Processing indicator<", shortName, ">");
         
         int handle = ChartIndicatorGet(0, win, shortName);
         
         if(handle == INVALID_HANDLE)

            Print("Bad handle <", handle, "> for indicator<", shortName, ">, skipping...");
            continue;
         }
         
         MqlParam params[];
         ENUM_INDICATOR indicator_type;
         int paramsCount = IndicatorParameters(handle, indicator_type, params);
         
         if(paramsCount < 0)
         {
            Print("No parameters to process for indicator<", shortName, ">, skipping...");
            continue;
         }

         int plotCount = GetIndicatorPlotCount(shortName);
         
         //--- Skip indicators with no plots
         if(plotCount < 1)
         {
            Print("Indicator ", shortName, " has no plots - skipping");
            continue;

         if(g_totalBuffers + plotCount > MAX_BUFFERS)

            Print("ERROR: Adding indicator ", shortName, " would exceed buffer limit (", MAX_BUFFERS, "). Skipping...");
            continue;
         }

         int maxPeriod = 0;
         for(int p = 0; p < MathMin(paramsCount, 3); p++)
         {
            if(params[p].type == TYPE_INT || params[p].type == TYPE_LONG)
            {
               int val = (int)params[p].integer_value;
               if(val > maxPeriod)
                  maxPeriod = val;
            }
         }
         if(maxPeriod < 1)
            maxPeriod = 1;
         
         Print("Checking parameters for indicator<", shortName, ">, parms count: <", paramsCount, ">");
         for(int lineIdx = 0; lineIdx < plotCount; lineIdx++)
         {
            if(g_totalBuffers >= MAX_BUFFERS)
            {
               Print("ERROR: Buffer limit (", MAX_BUFFERS, ") reached while processing ", shortName);
               break;
            }
            
            string lineName = GetPlotDisplayName(shortName, lineIdx);
            string lineIdxName = "Line " + (string)lineIdx;
            
            //--- Try to read color/width from TextGVar, fall back to defaults
            color defaultColor = GetAutoColor(g_totalPlots);
            int defaultWidth = 2;
            
            string colorStr;
            color lineColor = ReadLineColor(shortName, lineIdxName, defaultColor, colorStr);
//            color lineColor = ReadLineColor(shortName, lineIdxName, defaultColor);
            string widthStr;
            int lineWidth = ReadLineWidth(shortName, lineIdxName, defaultWidth, widthStr);
            int lineStyle = STYLE_SOLID;
            
            //--- Get current TextGVar strings for caching
//            string varPrefix = LineNameToVarPrefix(lineIdxName);
//            string colorStr = TextGVar_Read(TEXTGVAR_PREFIX, shortName, lineIdxName + "_Color", "");
//            string widthStr = TextGVar_Read(TEXTGVAR_PREFIX, shortName, lineIdxName + "_Width", "");
            
            int idx = ArraySize(g_lines);
            ArrayResize(g_lines, idx + 1);
            
            g_lines[idx].indicatorHandle = handle;
            g_lines[idx].indicatorShortName = shortName;
            g_lines[idx].lineIndex = lineIdx;
            g_lines[idx].bufferIndex = g_totalBuffers;
            g_lines[idx].period = maxPeriod;
            g_lines[idx].lineColor = lineColor;
            g_lines[idx].lineStyle = lineStyle;
            g_lines[idx].lineWidth = lineWidth;
            g_lines[idx].lineName = lineName;

            g_lines[idx].isValid = true;
            g_lines[idx].cachedColorStr = colorStr;
            g_lines[idx].cachedWidthStr = widthStr;

                  "> LineIdx<", g_lines[idx].lineIndex,

                  "> Period<", g_lines[idx].period,
                  "> Color<", g_lines[idx].lineColor,
                  "> Style<", g_lines[idx].lineStyle,
                  "> Width<", g_lines[idx].lineWidth,
                  "> LineName<", g_lines[idx].lineName,
                  "> CachedColorStr<", g_lines[idx].cachedColorStr,
                  "> CachedWidthStr<", g_lines[idx].cachedWidthStr,
                  ">");
            
            SetBufferByIndex(g_totalBuffers);
            PlotIndexSetInteger(g_totalPlots, PLOT_DRAW_TYPE, DRAW_LINE);
            PlotIndexSetInteger(g_totalPlots, PLOT_LINE_STYLE, lineStyle);
            PlotIndexSetInteger(g_totalPlots, PLOT_LINE_WIDTH, lineWidth);
            PlotIndexSetInteger(g_totalPlots, PLOT_LINE_COLOR, lineColor);
            PlotIndexSetString(g_totalPlots, PLOT_LABEL, shortName + " " + lineName + " Slope");

            g_totalBuffers++;
            g_totalPlots++;
         }
         
         indicatorCount++;
      }
   }
   
   Print("Total slope lines configured<", g_totalBuffers, "> from ", indicatorCount, " indicators");
}

//+------------------------------------------------------------------+
//| Check for REINIT signals from TextGVar objects                   |
//+------------------------------------------------------------------+
void CheckReinitSignals()
{
   //--- Track which shortNames we've already checked (avoid redundant checks for multi-buffer indicators)
   string checkedNames[];
   int checkedCount = 0;
   
   for(int i = 0; i < ArraySize(g_lines); i++)
   {
      if(!g_lines[i].isValid)
         continue;
      
      string shortName = g_lines[i].indicatorShortName;
      
      //--- Check if we've already processed this shortName
      bool alreadyChecked = false;
      for(int j = 0; j < checkedCount; j++)
      {
         if(checkedNames[j] == shortName)
         {
            alreadyChecked = true;
            break;
         }
      }
      
      if(alreadyChecked)
         continue;
      
      //--- Add to checked list
      ArrayResize(checkedNames, checkedCount + 1);
      checkedNames[checkedCount] = shortName;
      checkedCount++;
      
      //--- Check for REINIT signal
      string reinitSignal = TextGVar_Read(TEXTGVAR_PREFIX, shortName, "_REINIT", "");
      
Print("IndicatorSlopes[CheckReinitSignals] - BEFORE reinitSignal check, reinitSignal<", reinitSignal, ">");
      if(reinitSignal == "")
         continue;
      
      
      //--- Get fresh parameters from indicator
      int handle = g_lines[i].indicatorHandle;
      MqlParam params[];
      ENUM_INDICATOR indicator_type;
      int paramsCount = IndicatorParameters(handle, indicator_type, params);
Print("IndicatorSlopes[CheckReinitSignals] - AFTER getting params, paramsCount<", paramsCount, ">");
      
      //--- Calculate max period from params[0], params[1], params[2]
      int maxPeriod = 0;
      for(int p = 0; p < MathMin(paramsCount, 3); p++)
      {
Print("IndicatorSlopes[CheckReinitSignals] - BEFORE checking, params[",p,"].type<", params[p].type, ">");
         if(params[p].type == TYPE_INT || params[p].type == TYPE_LONG)
         {
            int val = (int)params[p].integer_value;
Print("IndicatorSlopes[CheckReinitSignals] - AFTER getting val<", val, "> maxPeriod<", maxPeriod, ">");
            if(val > maxPeriod)
               maxPeriod = val;
         }
      }
      if(maxPeriod < 1)
         maxPeriod = 1;
      
      //--- Update all lines belonging to this indicator
      for(int k = 0; k < ArraySize(g_lines); k++)
      {
         if(g_lines[k].indicatorShortName == shortName)
         {
            g_lines[k].period = maxPeriod;
            g_lines[k].lastRatesTotal = 0;  // Force full recalculation
            Print("Updated line ", k, " (", shortName, " ", g_lines[k].lineName, ") period to ", maxPeriod);
         }
      }
      
      //--- Delete the REINIT text object
      TextGVar_Delete(TEXTGVAR_PREFIX, shortName, "_REINIT");
      
      ChartRedraw(0);
   }
}

//| Custom indicator initialization function                         |
//+------------------------------------------------------------------+
int OnInit()
{
//Print(" "); Print(" "); 
//Print("IndicatorSlopes in OnInit - OnDeinit reason<", g_Uninit_Reason, ">");

   //--- Validate InpSlopeBarCnt
   if(InpSlopeBarCnt < 2)
   {
      Print("InpSlopeBarCnt must be >= 2");
      return(INIT_PARAMETERS_INCORRECT);
   }

   IndicatorSetString(INDICATOR_SHORTNAME, sMeShortName);
   
   //--- Initialize known indicators
   InitKnownIndicators();
   
   //--- Reset line array
   ArrayResize(g_lines, 0);
   
   //--- Scan chart for indicators
   ScanChartIndicators();
   
   if(g_totalBuffers == 0)
   {
      Print("No indicator plots found on chart!");
      
      //--- Remove this indicator from the chart
      int chartWindow = ChartWindowFind();
      if(chartWindow >= 0)
      {
         Print("Removing indicator - ChartWindow<", chartWindow, "> ShortName<", sMeShortName, ">");
         ChartIndicatorDelete(0, chartWindow, sMeShortName);
      }
      
      return(INIT_FAILED);
   }
   
   EventSetTimer(1);  // Check every 1 second
   g_lastIndicatorCount = GetTotalIndicatorCount();

   return(INIT_SUCCEEDED);
}

   //--- Kill the timer
   EventKillTimer();
   
   //--- Release all indicator handles
   for(int i = 0; i < ArraySize(g_lines); i++)

      if(g_lines[i].indicatorHandle != INVALID_HANDLE)
         IndicatorRelease(g_lines[i].indicatorHandle);
   }

}

//+------------------------------------------------------------------+
//| Timer function - detect indicator changes                        |
//+------------------------------------------------------------------+
void OnTimer()
{
   //--- Check for line property changes from TextGVar
   CheckLinePropertyChanges();
   
   //--- Check for REINIT signals
   CheckReinitSignals();
   
   //--- Count indicators across all subwindows
   int currentCount = GetTotalIndicatorCount();
   
   if(currentCount != g_lastIndicatorCount)
   {
      Print("Indicator count changed: ", g_lastIndicatorCount, " -> ", currentCount, ". Reinitializing...");
      g_lastIndicatorCount = currentCount;
      
      //--- Release existing handles
      for(int i = 0; i < ArraySize(g_lines); i++)
      {
         if(g_lines[i].indicatorHandle != INVALID_HANDLE)
            IndicatorRelease(g_lines[i].indicatorHandle);
      }
      ArrayResize(g_lines, 0);
      
      //--- Rescan and rebuild
      ScanChartIndicators();
      
      //--- Force full recalculation on next OnCalculate
      ChartSetSymbolPeriod(0, _Symbol, _Period);
   }
}

//+------------------------------------------------------------------+
//| Custom indicator iteration function                              |
//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,

   if(rates_total < InpSlopeBarCnt)
      return 0;

   for(int lineIdx = 0; lineIdx < ArraySize(g_lines); lineIdx++)
   {
      //--- Capture before cache update overwrites it
      bool forceFullRecalc = (g_lines[lineIdx].lastRatesTotal == 0);
      
      //--- Guard: need enough bars for indicator period + slope calc
      int minBarsRequired = g_lines[lineIdx].period + InpSlopeBarCnt;
      
      if(rates_total < minBarsRequired)
         continue;
      if(!g_lines[lineIdx].isValid)
         continue;
      
      int lastRates = g_lines[lineIdx].lastRatesTotal;
      int newBars = rates_total - lastRates;
      
      //--- First call or history changed: full copy
      if(lastRates == 0 || newBars < 0 || newBars > rates_total)
      {
         ArrayResize(g_lines[lineIdx].cache, rates_total);
         ArraySetAsSeries(g_lines[lineIdx].cache, true);
         
         int copyResult = CopyBuffer(g_lines[lineIdx].indicatorHandle, 
                                     g_lines[lineIdx].lineIndex, 
                                     0,
                                     rates_total,
                                     g_lines[lineIdx].cache);
         
         if(copyResult <= 0)
         {
            Print("CopyBuffer full copy FAILED for line ", lineIdx, ": got ", copyResult);
            g_lines[lineIdx].isValid = false;
            continue;
         }

         //--- Accept what we got (indicator may have fewer bars due to warmup period)
         g_lines[lineIdx].lastRatesTotal = rates_total;
      }
      else
      {
         //--- Incremental update: copy new bars + bar 0 update
         int barsToCopy = newBars + 1;  // +1 to recalc bar 0
         if(barsToCopy > rates_total)
            barsToCopy = rates_total;
         
         //--- Resize cache if new bars arrived
         if(rates_total > ArraySize(g_lines[lineIdx].cache))
         {
            ArrayResize(g_lines[lineIdx].cache, rates_total);
            ArraySetAsSeries(g_lines[lineIdx].cache, true);
         }
         
         double tempBuffer[];
         ArraySetAsSeries(tempBuffer, true);
         
         int copyResult = CopyBuffer(g_lines[lineIdx].indicatorHandle, 
                                     g_lines[lineIdx].lineIndex, 
                                     0,
                                     barsToCopy,
                                     tempBuffer);
         
         if(copyResult <= 0)
         {
            Print("CopyBuffer incremental FAILED for line ", lineIdx, ": got ", copyResult);
            g_lines[lineIdx].isValid = false;
            continue;
         }
         
         //--- Update cache with new values
         for(int i = 0; i < copyResult; i++)
         {
            g_lines[lineIdx].cache[i] = tempBuffer[i];
         }
         g_lines[lineIdx].lastRatesTotal = rates_total;       
      }

      //--- Calculate Z-score normalized slopes
      //--- Oldest bar we can calculate slope for (need room for slope bars + z-score lookback)
      int oldestSlopeBar = rates_total - 1 - g_lines[lineIdx].period - (InpSlopeBarCnt - 1) - InpZScorePeriod;
      if(oldestSlopeBar < 0)
         oldestSlopeBar = 0;
      
      //--- Determine which bars to recalculate
      int startBar;
      if(prev_calculated == 0 || forceFullRecalc)
      {
         startBar = oldestSlopeBar;
      }
      else
      {
         startBar = newBars;  // recalc new bars + bar 0
         if(startBar > oldestSlopeBar)
            startBar = oldestSlopeBar;
      }
      
      //+------------------------------------------------------------------+
      // NOTE: loop MUST run from older bars to newer for EMA correctness
      //+------------------------------------------------------------------+
      for(int CalcBarIX = startBar; CalcBarIX >= 0; CalcBarIX--)      
      {
         int PrevBarIX = CalcBarIX + InpSlopeBarCnt - 1;
      
         //--- Guard against out of range
         if(PrevBarIX >= ArraySize(g_lines[lineIdx].cache))
            continue;
      
         //--- Calculate raw slope for current bar
         double currentValue = g_lines[lineIdx].cache[CalcBarIX];
         double prevValue    = g_lines[lineIdx].cache[PrevBarIX];
      
         //--- Guard against EMPTY_VALUE
         if(currentValue == EMPTY_VALUE || prevValue == EMPTY_VALUE)
            continue;
      
         double slope = (currentValue - prevValue) / (InpSlopeBarCnt - 1);
         
         double outValue = 0.0;
         
         //================ CALCULATE outValue BY MODE =================
         if(InpRawSlope)
         {
            outValue = slope;
         }
         else if(InpSlopeMode == SLOPE_ZSCORE)
         {
            //--- Calculate mean and standard deviation of slopes over lookback period
            double sumSlope = 0.0;
            double sumSlopeSq = 0.0;
            int count = 0;
      
            for(int j = CalcBarIX;
                j < CalcBarIX + InpZScorePeriod &&
                j + InpSlopeBarCnt - 1 < ArraySize(g_lines[lineIdx].cache);
                j++)
            {
               double val     = g_lines[lineIdx].cache[j];
               double valPrev = g_lines[lineIdx].cache[j + InpSlopeBarCnt - 1];
      
               if(val == EMPTY_VALUE || valPrev == EMPTY_VALUE)
                  continue;
      
               double s = (val - valPrev) / (InpSlopeBarCnt - 1);
               sumSlope   += s;
               sumSlopeSq += s * s;
               count++;
            }
      
            if(count > 1)
            {
               double mean     = sumSlope / count;
               double variance = (sumSlopeSq / count) - (mean * mean);
               double stdDev   = (variance > 0) ? MathSqrt(variance) : 0;
      
               if(stdDev > 0)
                  outValue = (slope - mean) / stdDev;
               else
                  outValue = 0.0;
            }
            else
               outValue = 0.0;
         }
         else if(InpSlopeMode == SLOPE_EMA)
         {
            double alpha = 2.0 / (InpEmaPeriod + 1.0);
            double prev;
            
            if(CalcBarIX + 1 >= rates_total)
            {
               prev = slope;
            }
            else
            {
               switch(g_lines[lineIdx].bufferIndex)
               {
                  case 0:  prev = Buffer0[CalcBarIX + 1]; break;
                  case 1:  prev = Buffer1[CalcBarIX + 1]; break;
                  case 2:  prev = Buffer2[CalcBarIX + 1]; break;
                  case 3:  prev = Buffer3[CalcBarIX + 1]; break;
                  case 4:  prev = Buffer4[CalcBarIX + 1]; break;
                  case 5:  prev = Buffer5[CalcBarIX + 1]; break;
                  case 6:  prev = Buffer6[CalcBarIX + 1]; break;
                  case 7:  prev = Buffer7[CalcBarIX + 1]; break;
                  case 8:  prev = Buffer8[CalcBarIX + 1]; break;
                  case 9:  prev = Buffer9[CalcBarIX + 1]; break;
                  case 10: prev = Buffer10[CalcBarIX + 1]; break;
                  case 11: prev = Buffer11[CalcBarIX + 1]; break;
                  case 12: prev = Buffer12[CalcBarIX + 1]; break;
                  case 13: prev = Buffer13[CalcBarIX + 1]; break;
                  case 14: prev = Buffer14[CalcBarIX + 1]; break;
                  case 15: prev = Buffer15[CalcBarIX + 1]; break;
                  case 16: prev = Buffer16[CalcBarIX + 1]; break;

                  case 18: prev = Buffer18[CalcBarIX + 1]; break;
                  case 19: prev = Buffer19[CalcBarIX + 1]; break;
                  case 20: prev = Buffer20[CalcBarIX + 1]; break;
                  case 21: prev = Buffer21[CalcBarIX + 1]; break;
                  case 22: prev = Buffer22[CalcBarIX + 1]; break;

                  case 26: prev = Buffer26[CalcBarIX + 1]; break;
                  case 27: prev = Buffer27[CalcBarIX + 1]; break;
                  case 28: prev = Buffer28[CalcBarIX + 1]; break;
                  case 29: prev = Buffer29[CalcBarIX + 1]; break;
                  default: prev = slope; break;
               }
            }
      
            if(prev == EMPTY_VALUE)
               prev = slope;
      
            outValue = alpha * slope + (1.0 - alpha) * prev;
         }
         else if(InpSlopeMode == SLOPE_MEDIAN)
         {
            double tmp[];
            ArrayResize(tmp, InpMedianPeriod);
            int k = 0;
      
            for(int j = CalcBarIX;
                j < CalcBarIX + InpMedianPeriod &&
                j + InpSlopeBarCnt - 1 < ArraySize(g_lines[lineIdx].cache);
                j++)
            {
               double val     = g_lines[lineIdx].cache[j];
               double valPrev = g_lines[lineIdx].cache[j + InpSlopeBarCnt - 1];
      
               if(val == EMPTY_VALUE || valPrev == EMPTY_VALUE)
                  continue;
      
               tmp[k++] = (val - valPrev) / (InpSlopeBarCnt - 1);
            }
      
            if(k > 0)
            {
               ArrayResize(tmp, k);
               ArraySort(tmp);
               outValue = tmp[k / 2];
            }

               outValue = 0.0;
         }

         double minVal = currentValue;
         double maxVal = currentValue;
         
         for(int j = CalcBarIX; 
             j < CalcBarIX + InpRawNormPeriod && 
             j < ArraySize(g_lines[lineIdx].cache);

            outValue = 0.0;
         
         //================ WRITE TO BUFFER =================
         switch(g_lines[lineIdx].bufferIndex)
         {
            case 0:  Buffer0[CalcBarIX]  = outValue; break;

            case 2:  Buffer2[CalcBarIX]  = outValue; break;
            case 3:  Buffer3[CalcBarIX]  = outValue; break;
            case 4:  Buffer4[CalcBarIX]  = outValue; break;

string TextGVar_BuildName(const string prefix, int window, const string shortName, const string varName)
{
   return prefix + "_" + IntegerToString(window) + "_" + shortName + "_" + varName;

//+------------------------------------------------------------------+
//| Create text object variable                                      |
//+------------------------------------------------------------------+
string TextGVar_Create(const string prefix,
                     int window,
                     const string shortName,
                     const string varName,
                     const string value,
                     const int xBars = 0)
{
   string name = TextGVar_BuildName(prefix, window, shortName, varName);
   
   if(ObjectFind(0, name) >= 0)
      ObjectDelete(0, name);
   
   int totalBars = Bars(Symbol(), Period());
   int oldestBarIndex = totalBars - 1;
   int barIndex = oldestBarIndex - xBars;
   if(barIndex < 0) barIndex = 0;
   
   datetime timeCoord = iTime(Symbol(), Period(), barIndex);
   double pipSize = (Digits() == 5 || Digits() == 3) ? Point() * 10 : Point();
   double yPrice = iHigh(Symbol(), Period(), oldestBarIndex) + (20 * pipSize);
   
   if(!ObjectCreate(0, name, OBJ_TEXT, 0, timeCoord, yPrice))
      return "";
   
   ObjectSetString(0, name, OBJPROP_TEXT, value);
   ObjectSetInteger(0, name, OBJPROP_ANCHOR, ANCHOR_LEFT_UPPER);
   ObjectSetInteger(0, name, OBJPROP_FONTSIZE, 8);
   ObjectSetInteger(0, name, OBJPROP_COLOR, clrDarkGray);
   ObjectSetString(0, name, OBJPROP_FONT, "Consolas");
   ObjectSetInteger(0, name, OBJPROP_SELECTABLE, true);
   
   return name + "=" + value;
}

//+------------------------------------------------------------------+
//| Read text object value, return default if not found              |
//+------------------------------------------------------------------+
string TextGVar_Read(const string prefix,
                     int window,
                     const string shortName,
                     const string varName,
                     const string defaultValue = "")
{
   string name = TextGVar_BuildName(prefix, window, shortName, varName);
   
   if(ObjectFind(0, name) < 0)
      return defaultValue;
   
   return ObjectGetString(0, name, OBJPROP_TEXT);
}

//+------------------------------------------------------------------+
//| Update existing text object value                                |
//+------------------------------------------------------------------+
bool TextGVar_Update(const string prefix,
                     int window,
                     const string shortName,
                     const string varName,
                     const string newValue)
{
   string name = TextGVar_BuildName(prefix, window, shortName, varName);
   
   if(ObjectFind(0, name) < 0)
      return false;
   
   return ObjectSetString(0, name, OBJPROP_TEXT, newValue);
}

//+------------------------------------------------------------------+
//| Delete single text object                                        |
//+------------------------------------------------------------------+
bool TextGVar_Delete(const string prefix,
                     int window,
                     const string shortName,
                     const string varName)
{
   string name = TextGVar_BuildName(prefix, window, shortName, varName);
   
   if(ObjectFind(0, name) < 0)
      return false;
   
   return ObjectDelete(0, name);
}
