The dedup logic is necessary if you iterate g_lines[] — a multi-buffer indicator like MACD would have 3 entries with the same shortName, and you'd check/delete the same REINIT three times.

But the whole approach is overcomplicated. Simpler and more robust:

bool CheckReinitSignals()
{
   for(int obj = ObjectsTotal(0) - 1; obj >= 0; obj--)
   {
      string objName = ObjectName(0, obj);
      if(StringFind(objName, TEXTGVAR_PREFIX + "_") == 0 && 
         StringFind(objName, "_REINIT") > 0)
      {
         Print("REINIT detected: ", objName);
         ObjectDelete(0, objName);
         return true;
      }
   }
   return false;
}

Advantages:

  1. Catches any REINIT flag, not just for indicators IS is already tracking
  2. No dedup bookkeeping needed
  3. Works even if g_lines[] is stale or empty
  4. Directly matches the object naming convention

The current version has a design flaw: it only looks for REINITs belonging to indicators in g_lines[]. If IS hasn't scanned yet, or if the scan failed, it won't find anything.

   if(prev_calculated == 0)
   {
      ATRBuffer[0] = high[0] - low[0];
      double sum = 0;
      for(int i = 1; i < InpPeriod; i++)
      {
         double tr = MathMax(high[i] - low[i], 
                     MathMax(MathAbs(high[i] - close[i-1]), 
                             MathAbs(low[i] - close[i-1])));
         ATRBuffer[i] = tr;
         sum += tr;
      }
      ATRBuffer[InpPeriod-1] = sum / InpPeriod;
      start = InpPeriod;
   }
   else
      start = prev_calculated - 1;
   
   for(int i = start; i < rates_total; i++)
   {
      double tr = MathMax(high[i] - low[i], 
                  MathMax(MathAbs(high[i] - close[i-1]), 
                          MathAbs(low[i] - close[i-1])));
      ATRBuffer[i] = (ATRBuffer[i-1] * (InpPeriod - 1) + tr) / InpPeriod;
   }

   if(g_totalBuffers > 0)

      //--- Buffer31: Slope % UP/DN
      SetBufferByIndex(SUMMARY_PCT_BUFFER);

      PlotIndexSetInteger(SUMMARY_PCT_BUFFER, PLOT_LINE_COLOR, InpSummaryPctColor);

   }
   
   Print("Total slope lines configured<", g_totalBuffers, "> from ", indicatorCount, " indicators");
}

//| Check for REINIT signals from TextGVar objects                   |
//| Returns true if rescan is needed                                 |
//+------------------------------------------------------------------+
bool CheckReinitSignals()
{
   string checkedNames[];
   int checkedCount = 0;
   
   for(int i = 0; i < ArraySize(g_lines); i++)
   {
      if(!g_lines[i].isValid)
         continue;
      
      string shortName = g_lines[i].indicatorShortName;
      int window = g_lines[i].indicatorWindow;
      
      //--- Check if we've already processed this shortName
      bool alreadyChecked = false;
      for(int j = 0; j < checkedCount; j++)
      {
         if(checkedNames[j] == shortName)
         {
            alreadyChecked = true;
            break;
         }
      }
      
      if(alreadyChecked)
         continue;
      
      ArrayResize(checkedNames, checkedCount + 1);
      checkedNames[checkedCount] = shortName;
      checkedCount++;
      
      //--- Check for REINIT signal
      string reinitSignal = ReadTextGVar(window, shortName, TEXTGVAR_REINIT, "");
      
      if(reinitSignal == "")
         continue;
      
      Print("REINIT detected for ", shortName, ", triggering rescan");
      
      //--- Delete the REINIT flag
      TextGVar_Delete(TEXTGVAR_PREFIX, window, shortName, TEXTGVAR_REINIT);
      
      return true;
   }
   
   return false;
}

int OnInit()
{
   //--- Validate InpSlopeBarCnt
   if(InpSlopeBarCnt < 2)
   {
      Print("InpSlopeBarCnt must be >= 2");
      return(INIT_PARAMETERS_INCORRECT);
   }

   IndicatorSetString(INDICATOR_SHORTNAME, sMeShortName);
   
   //--- Reset line array
   ArrayResize(g_lines, 0);

   //--- Clean any outstanding REINIT flags (stale from previous session or just-completed SI reinits)
   for(int obj = ObjectsTotal(0) - 1; obj >= 0; obj--)
   {
      string objName = ObjectName(0, obj);
      if(StringFind(objName, TEXTGVAR_PREFIX + "_") == 0 && 
         StringFind(objName, "_REINIT") > 0)
      {
         ObjectDelete(0, objName);
      }
   }
   
   //--- Scan chart for indicators
   ScanChartIndicators();
   
   if(g_totalBuffers == 0)
   {
      Print("No indicator plots found on chart!");
      
      //--- Remove this indicator from the chart
      int chartWindow = ChartWindowFind();
      if(chartWindow >= 0)
      {
         Print("Removing indicator - ChartWindow<", chartWindow, "> ShortName<", sMeShortName, ">");
         ChartIndicatorDelete(0, chartWindow, sMeShortName);
      }
      
      return(INIT_FAILED);
   }
   
   EventSetMillisecondTimer(250);  // Check every 250 milliseconds
//   EventSetTimer(1);  // Check every 1 second
   g_lastIndicatorCount = GetTotalIndicatorCount();

   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
void OnTimer()
{
   //--- Check for line property changes from TextGVar
   CheckLinePropertyChanges();
   
   //--- Check for REINIT signals
   bool needsRescan = CheckReinitSignals();
   
   //--- Count indicators across all subwindows
   int currentCount = GetTotalIndicatorCount();
   
   if(needsRescan || currentCount != g_lastIndicatorCount)
   {
      Print("Reinitializing...");
      g_lastIndicatorCount = currentCount;
      
      //--- Release existing handles
      for(int i = 0; i < ArraySize(g_lines); i++)
      {
         if(g_lines[i].indicatorHandle != INVALID_HANDLE)
            IndicatorRelease(g_lines[i].indicatorHandle);
      }
      ArrayResize(g_lines, 0);
      
      //--- Rescan and rebuild
      ScanChartIndicators();
      
      //--- Force full recalculation on next OnCalculate
      ChartSetSymbolPeriod(0, _Symbol, _Period);
   }
}

//+------------------------------------------------------------------+
//| Check and apply TextGVar property changes for all lines          |
//+------------------------------------------------------------------+
void CheckLinePropertyChanges()
{
   for(int i = 0; i < ArraySize(g_lines); i++)
   {
      if(!g_lines[i].isValid)
         continue;
      
      string shortName = g_lines[i].indicatorShortName;
      int window = g_lines[i].indicatorWindow;
      int lineIdx = g_lines[i].lineIndex;
      
      //--- Read current values from TextGVar
      string currentColorStr = ReadTextGVar(window, shortName, "Plot[" + IntegerToString(lineIdx) + "]_Color", "");
      string currentWidthStr = ReadTextGVar(window, shortName, "Plot[" + IntegerToString(lineIdx) + "]_Width", "");
      
      bool changed = false;
      
      //--- Check color change
      if(currentColorStr != "" && currentColorStr != g_lines[i].cachedColorStr)
      {
         color newColor = StringToColor(currentColorStr);
         g_lines[i].lineColor = newColor;
         g_lines[i].cachedColorStr = currentColorStr;
         PlotIndexSetInteger(g_lines[i].bufferIndex, PLOT_LINE_COLOR, newColor);
         changed = true;
         Print("Line ", i, " (", shortName, " ", g_lines[i].lineName, ") color changed to ", currentColorStr);
      }
      
      //--- Check width change
      if(currentWidthStr != "" && currentWidthStr != g_lines[i].cachedWidthStr)
      {
         int newWidth = (int)StringToInteger(currentWidthStr);
         g_lines[i].lineWidth = newWidth;
         g_lines[i].cachedWidthStr = currentWidthStr;
         PlotIndexSetInteger(g_lines[i].bufferIndex, PLOT_LINE_WIDTH, newWidth);
         changed = true;
         Print("Line ", i, " (", shortName, " ", g_lines[i].lineName, ") width changed to ", currentWidthStr);
      }
      
      if(changed)
         ChartRedraw(0);
   }
}

   {
      //--- Capture before cache update overwrites it
      bool forceFullRecalc = (g_lines[lineIdx].lastRatesTotal == 0);
      
      //--- Guard: need enough bars for indicator period + slope calc
      int minBarsRequired = g_lines[lineIdx].period + InpSlopeBarCnt;
      
      if(rates_total < minBarsRequired)
         continue;
      if(!g_lines[lineIdx].isValid)
         continue;
      
      int lastRates = g_lines[lineIdx].lastRatesTotal;
      int newBars = rates_total - lastRates;
      
      //--- First call or history changed: full copy
      if(lastRates == 0 || newBars < 0 || newBars > rates_total)
      {
         ArrayResize(g_lines[lineIdx].cache, rates_total);
         ArraySetAsSeries(g_lines[lineIdx].cache, true);
         
         int copyResult = CopyBuffer(g_lines[lineIdx].indicatorHandle, 
                                     g_lines[lineIdx].lineIndex, 
                                     0,
                                     rates_total,
                                     g_lines[lineIdx].cache);
         
         if(copyResult <= 0)
         {
            Print("CopyBuffer full copy FAILED for line ", lineIdx, ": got ", copyResult);
            g_lines[lineIdx].isValid = false;
            continue;
         }

         //--- Accept what we got (indicator may have fewer bars due to warmup period)
         g_lines[lineIdx].lastRatesTotal = rates_total;
      }
      else
      {
         //--- Incremental update: copy new bars + bar 0 update
         int barsToCopy = newBars + 1;  // +1 to recalc bar 0
         if(barsToCopy > rates_total)
            barsToCopy = rates_total;
         
         //--- Resize cache if new bars arrived
         if(rates_total > ArraySize(g_lines[lineIdx].cache))
         {
            ArrayResize(g_lines[lineIdx].cache, rates_total);
            ArraySetAsSeries(g_lines[lineIdx].cache, true);
         }
         
         double tempBuffer[];
         ArraySetAsSeries(tempBuffer, true);
         
         int copyResult = CopyBuffer(g_lines[lineIdx].indicatorHandle, 
                                     g_lines[lineIdx].lineIndex, 
                                     0,
                                     barsToCopy,
                                     tempBuffer);
         
         if(copyResult <= 0)
         {
            Print("CopyBuffer incremental FAILED for line ", lineIdx, ": got ", copyResult);
            g_lines[lineIdx].isValid = false;
            continue;
         }
         
         //--- Update cache with new values
         for(int i = 0; i < copyResult; i++)
         {
            g_lines[lineIdx].cache[i] = tempBuffer[i];
         }
         g_lines[lineIdx].lastRatesTotal = rates_total;       
      }

      //--- Calculate Z-score normalized slopes
      //--- Oldest bar we can calculate slope for (need room for slope bars + z-score lookback)
      int oldestSlopeBar = rates_total - 1 - g_lines[lineIdx].period - (InpSlopeBarCnt - 1) - InpZScorePeriod;
      if(oldestSlopeBar < 0)
         oldestSlopeBar = 0;
      
      //--- Determine which bars to recalculate
      int startBar;
      if(prev_calculated == 0 || forceFullRecalc)
      {
         startBar = oldestSlopeBar;
      }
      else
      {
         startBar = newBars;  // recalc new bars + bar 0
         if(startBar > oldestSlopeBar)
            startBar = oldestSlopeBar;
      }
      
      //--- Track for summary calculation
      if(startBar > summaryStartBar)
         summaryStartBar = startBar;
      
      //+------------------------------------------------------------------+
      // NOTE: loop MUST run from older bars to newer for EMA correctness
      //+------------------------------------------------------------------+
      for(int CalcBarIX = startBar; CalcBarIX >= 0; CalcBarIX--)      
      {
         int PrevBarIX = CalcBarIX + InpSlopeBarCnt - 1;
      
         //--- Guard against out of range
         if(PrevBarIX >= ArraySize(g_lines[lineIdx].cache))

         //--- Calculate raw slope for current bar
         double currentValue = g_lines[lineIdx].cache[CalcBarIX];

         if(currentValue == EMPTY_VALUE || prevValue == EMPTY_VALUE)

         if(InpRawSlope)

         else if(InpSlopeMode == SLOPE_ZSCORE)
         {
            //--- Calculate mean and standard deviation of slopes over lookback period
            double sumSlope = 0.0;
            double sumSlopeSq = 0.0;
            int count = 0;
      
            for(int j = CalcBarIX;
                j < CalcBarIX + InpZScorePeriod &&
                j + InpSlopeBarCnt - 1 < ArraySize(g_lines[lineIdx].cache);
                j++)
            {
               double val     = g_lines[lineIdx].cache[j];
               double valPrev = g_lines[lineIdx].cache[j + InpSlopeBarCnt - 1];
      
               if(val == EMPTY_VALUE || valPrev == EMPTY_VALUE)
                  continue;
      
               double s = (val - valPrev) / (InpSlopeBarCnt - 1);
               sumSlope   += s;
               sumSlopeSq += s * s;
               count++;
            }
      
            if(count > 1)
            {
               double mean     = sumSlope / count;
               double variance = (sumSlopeSq / count) - (mean * mean);
               double stdDev   = (variance > 0) ? MathSqrt(variance) : 0;
      
               if(stdDev > 0)
                  outValue = (slope - mean) / stdDev;
               else
                  outValue = 0.0;
            }
            else
               outValue = 0.0;
         }
         else if(InpSlopeMode == SLOPE_EMA)
         {

               prev = slope;
            }
            else
            {

                j + InpSlopeBarCnt - 1 < ArraySize(g_lines[lineIdx].cache);

               ArrayResize(tmp, k);
               ArraySort(tmp);
               outValue = tmp[k / 2];
            }

         }
