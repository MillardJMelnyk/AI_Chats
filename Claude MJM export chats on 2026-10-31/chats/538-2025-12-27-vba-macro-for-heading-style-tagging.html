Sub PrefixHeadingsWithLevel()
    Dim para As Paragraph
    Dim headingLevel As Integer
    Dim prefix As String
    
    For Each para In ActiveDocument.Paragraphs
        If para.Range.Words.Count > 0 Then
            If InStr(para.Style, "Heading") > 0 Then
                headingLevel = Val(Right(para.Style, 1))
                If headingLevel > 0 And headingLevel <= 9 Then
                    prefix = "<<H" & headingLevel & ">>"
                    If Left(para.Range.Text, Len(prefix)) <> prefix Then
                        para.Range.InsertBefore prefix
                    End If
                End If
            End If
        End If
    Next para
End Sub

Sub RemoveHeadingTags()
    Dim para As Paragraph
    Dim text As String
    Dim closePos As Integer
    Dim tagLength As Integer
    Dim rng As Range
    
    For Each para In ActiveDocument.Paragraphs
        text = para.Range.Text
        If Left(text, 3) = "<<H" Then
            closePos = InStr(text, ">>")
            If closePos > 0 Then
                tagLength = closePos + 1
                Set rng = para.Range
                rng.End = rng.Start + tagLength
                rng.Delete
            End If
        End If
    Next para
End Sub
