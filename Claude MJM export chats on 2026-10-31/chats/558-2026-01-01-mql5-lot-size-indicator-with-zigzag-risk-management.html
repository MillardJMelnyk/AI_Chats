void CalculateDeciles()
{
   int bars = Bars(_Symbol, PERIOD_CURRENT);
   if(bars < 100)
   {
      Print("Insufficient bars for ZigZag analysis");
      return;
   }
   
   //--- Get ZigZag spread buffer (buffer 3 = dbZZ_SPRD, already in pips)
   double zz_spread_buffer[];
   ArraySetAsSeries(zz_spread_buffer, true);
   
   int copied = CopyBuffer(zz_handle, 3, 0, bars, zz_spread_buffer);
   if(copied <= 0)
   {
      Print("Failed to copy ZigZag spread buffer. Error: ", GetLastError());
      return;
   }
   
   //--- Collect non-zero spreads (absolute values since we want magnitude)
   double spreads[];
   ArrayResize(spreads, 0);
   
   for(int i = 0; i < copied; i++)
   {
      if(zz_spread_buffer[i] != 0.0)
      {
         int size = ArraySize(spreads);
         ArrayResize(spreads, size + 1);
         spreads[size] = MathAbs(zz_spread_buffer[i]);
      }
   }
   
   int spread_count = ArraySize(spreads);
   if(spread_count < 10)
   {
      Print("Insufficient spreads for decile calculation: ", spread_count);
      return;
   }
   
   //--- Sort spreads ascending
   ArraySort(spreads);
   
   //--- Calculate decile maximums
   for(int decile = 1; decile <= 10; decile++)
   {
      int end_index = (int)MathFloor((decile / 10.0) * spread_count) - 1;
      if(end_index < 0) end_index = 0;
      if(end_index >= spread_count) end_index = spread_count - 1;
      
      Decile_Max[decile] = spreads[end_index];
   }
   
   deciles_calculated = true;
   
   Print("Deciles calculated. Spreads analyzed: ", spread_count);
   for(int i = 1; i <= 10; i++)
      Print("Decile ", i, " max: ", DoubleToString(Decile_Max[i], 1), " pips");
}

//| Calculate ZigZag spreads and deciles                              |
//+------------------------------------------------------------------+
void CalculateDeciles()
{
   int bars = Bars(_Symbol, PERIOD_CURRENT);
   if(bars < 100)
   {
      Print("Insufficient bars for ZigZag analysis");
      return;
   }
   
   //--- Get ZigZag spread buffer (buffer 3 = dbZZ_SPRD, already in pips)
   double zz_spread_buffer[];
   ArraySetAsSeries(zz_spread_buffer, true);
   
   int copied = CopyBuffer(zz_handle, 3, 0, bars, zz_spread_buffer);
   if(copied <= 0)
   {
      Print("Failed to copy ZigZag spread buffer. Error: ", GetLastError());
      return;
   }
   
   //--- Collect non-zero spreads (absolute values since we want magnitude)
   double spreads[];
   ArrayResize(spreads, 0);
   
   for(int i = 0; i < copied; i++)
   {
      if(zz_spread_buffer[i] != 0.0)
      {
         int size = ArraySize(spreads);
         ArrayResize(spreads, size + 1);
         spreads[size] = MathAbs(zz_spread_buffer[i]);
      }
   }
   
   int spread_count = ArraySize(spreads);
   if(spread_count < 10)
   {
      Print("Insufficient spreads for decile calculation: ", spread_count);
      return;
   }
   
   //--- Sort spreads ascending
   ArraySort(spreads);
   
   //--- Calculate decile maximums
   for(int decile = 1; decile <= 10; decile++)
   {
      int end_index = (int)MathFloor((decile / 10.0) * spread_count) - 1;
      if(end_index < 0) end_index = 0;
      if(end_index >= spread_count) end_index = spread_count - 1;
      
      Decile_Max[decile] = spreads[end_index];
   }
   
   deciles_calculated = true;
   
   Print("Deciles calculated. Spreads analyzed: ", spread_count);
   for(int i = 1; i <= 10; i++)
      Print("Decile ", i, " max: ", DoubleToString(Decile_Max[i], 1), " pips");
}

//| Calculate ZigZag spreads and deciles                              |
//+------------------------------------------------------------------+
void CalculateDeciles()
{
   int bars = Bars(_Symbol, PERIOD_CURRENT);
   if(bars < 100)
   {
      Print("Insufficient bars for ZigZag analysis");
      return;
   }
   
   //--- Get ZigZag spread buffer (buffer 3 = dbZZ_SPRD, already in pips)
   double zz_spread_buffer[];
   ArraySetAsSeries(zz_spread_buffer, true);
   
   int copied = CopyBuffer(zz_handle, 3, 0, bars, zz_spread_buffer);
   if(copied <= 0)
   {
      Print("Failed to copy ZigZag spread buffer. Error: ", GetLastError());
      return;
   }
   
   //--- Collect non-zero spreads (absolute values since we want magnitude)
   double spreads[];
   ArrayResize(spreads, 0);
   
   for(int i = 0; i < copied; i++)
   {
      if(zz_spread_buffer[i] != 0.0)
      {
         int size = ArraySize(spreads);
         ArrayResize(spreads, size + 1);
         spreads[size] = MathAbs(zz_spread_buffer[i]);
      }
   }
   
   int spread_count = ArraySize(spreads);
   if(spread_count < 10)
   {
      Print("Insufficient spreads for decile calculation: ", spread_count);
      return;
   }
   
   //--- Sort spreads ascending
   ArraySort(spreads);
   
   //--- Calculate decile maximums
   for(int decile = 1; decile <= 10; decile++)
   {
      int end_index = (int)MathFloor((decile / 10.0) * spread_count) - 1;
      if(end_index < 0) end_index = 0;
      if(end_index >= spread_count) end_index = spread_count - 1;
      
      Decile_Max[decile] = spreads[end_index];
   }
   
   deciles_calculated = true;
   
   Print("Deciles calculated. Spreads analyzed: ", spread_count);
   for(int i = 1; i <= 10; i++)
      Print("Decile ", i, " max: ", DoubleToString(Decile_Max[i], 1), " pips");
}

{
   double tick_size = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double tick_value = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double point = _Point;
   int digits = _Digits;

   double pspl;

   else
      pspl = ref_price + (mpss * pip_size);

   int remaining_slots = MTIP - existing_trades;
   
   double lot_size = 0;
   if(remaining_slots > 0 && max_loss_per_lot > 0)
      lot_size = cfm / max_loss_per_lot / remaining_slots;

   double lot_step = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   double lot_min = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double lot_max = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   
   lot_size = MathFloor(lot_size / lot_step) * lot_step;
   lot_size = MathMax(lot_min, MathMin(lot_max, lot_size));
   
   string lot_text = DoubleToString(lot_size, 2);
   if(remaining_slots <= 0)
      lot_text = "MAX TRADES";
   
   ObjectSetString(0, "LSC_Value_LotSize", OBJPROP_TEXT, lot_text);
}

void OnChartEvent(const int id,
                  const long &lparam,
                  const double &dparam,
                  const string &sparam)
{
   if(id == CHARTEVENT_CHART_CHANGE)
   {
      RepositionPanel();
      return;
   }
   
   if(id == CHARTEVENT_OBJECT_CLICK)
   {
      if(sparam == "LSC_Btn_Direction")
      {
         trade_dir_buy = !trade_dir_buy;
         ObjectSetString(0, "LSC_Btn_Direction", OBJPROP_TEXT, trade_dir_buy ? "BUY" : "SELL");
         ObjectSetInteger(0, "LSC_Btn_Direction", OBJPROP_BGCOLOR, trade_dir_buy ? clrGreen : clrRed);
         //--- Reset button state so it can be clicked again
         ObjectSetInteger(0, "LSC_Btn_Direction", OBJPROP_STATE, false);
         //--- Update price box colors
         UpdatePriceBoxColors();
         ChartRedraw(0);
         UpdatePanel();
      }
      else if(sparam == "LSC_Edit_MktPrice")
      {
         //--- Switch to market price mode
         use_limit_price = false;
         current_limit_price = 0;
         double mkt_price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
         ObjectSetString(0, "LSC_Edit_MktPrice", OBJPROP_TEXT, DoubleToString(mkt_price, _Digits));
         ObjectSetString(0, "LSC_Edit_LmtPrice", OBJPROP_TEXT, "");
         UpdatePriceBoxColors();
         ChartRedraw(0);
         UpdatePanel();
      }
      else if(sparam == "LSC_Edit_LmtPrice")
      {
         //--- Switch to limit price mode, default to current market
         use_limit_price = true;
         current_limit_price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
         ObjectSetString(0, "LSC_Edit_LmtPrice", OBJPROP_TEXT, DoubleToString(current_limit_price, _Digits));
         ObjectSetString(0, "LSC_Edit_MktPrice", OBJPROP_TEXT, "");
         UpdatePriceBoxColors();
         ChartRedraw(0);
         UpdatePanel();
      }
   }
   
   if(id == CHARTEVENT_OBJECT_ENDEDIT)
   {
      if(sparam == "LSC_Edit_LmtPrice")
      {
         string text = ObjectGetString(0, "LSC_Edit_LmtPrice", OBJPROP_TEXT);
         double val = StringToDouble(text);
         if(val > 0)
         {
            current_limit_price = val;
            use_limit_price = true;
         }
         UpdatePanel();
      }
   }
}

         next_pivot_bars = zz_pivot_bars[next_pivot_idx] - i;
         leg_end_pips = MathAbs(next_pivot_price - leg_start) / pip_size;

            profitable = (next_pivot_price < close[i]) ? 1 : 0;
         else

      FileWrite(handle,
         TimeToString(time[i], TIME_DATE|TIME_MINUTES),
         DoubleToString(close[i], _Digits),

//| Build all historical distributions                                |
//+------------------------------------------------------------------+
void BuildDistributions(int rates_total, const double &high[], const double &low[], 
                        const double &close[], const long &tick_volume[])
{
   int bars_to_use = MathMin(History_Bars, rates_total);
   int start_bar = rates_total - bars_to_use;
   double pip_size = GetPipSize();
   
   //--- Update pivots first
   UpdatePivots(rates_total);
   
   Print("Found ", zz_pivot_count, " ZigZag pivots");
   
   if(zz_pivot_count < 3)
   {
      Print("Insufficient ZigZag pivots");
      return;
   }
   
   //--- Build leg size distribution
   ArrayResize(Leg_Dist, 0);
   Leg_Dist_Count = 0;
   
   for(int i = 0; i < zz_pivot_count - 1; i++)
   {
      double leg_size = MathAbs(zz_pivots[i + 1] - zz_pivots[i]) / pip_size;
      if(leg_size > 0)
      {
         ArrayResize(Leg_Dist, Leg_Dist_Count + 1);
         Leg_Dist[Leg_Dist_Count] = leg_size;
         Leg_Dist_Count++;
      }
   }
   ArraySort(Leg_Dist);
   
   //--- Build retracement distribution
   ArrayResize(Retrace_Dist, 0);
   Retrace_Dist_Count = 0;
   
   for(int i = 1; i < zz_pivot_count - 1; i++)
   {
      double prior_leg = MathAbs(zz_pivots[i] - zz_pivots[i - 1]);
      double current_leg = MathAbs(zz_pivots[i + 1] - zz_pivots[i]);
      
      if(prior_leg > 0)
      {
         double retrace_pct = (current_leg / prior_leg) * 100;
         ArrayResize(Retrace_Dist, Retrace_Dist_Count + 1);
         Retrace_Dist[Retrace_Dist_Count] = retrace_pct;
         Retrace_Dist_Count++;
      }
   }
   ArraySort(Retrace_Dist);
   
   //--- Build time distribution
   ArrayResize(Time_Dist, 0);
   Time_Dist_Count = 0;
   
   for(int i = 0; i < zz_pivot_count - 1; i++)
   {
      int duration = MathAbs(zz_pivot_bars[i + 1] - zz_pivot_bars[i]);
      if(duration > 0)
      {
         ArrayResize(Time_Dist, Time_Dist_Count + 1);
         Time_Dist[Time_Dist_Count] = duration;
         Time_Dist_Count++;
      }
   }
   ArraySort(Time_Dist);
   
   //--- Build velocity distribution (pips per bar for completed legs)
   ArrayResize(Velocity_Dist, 0);
   Velocity_Dist_Count = 0;
   
   for(int i = 0; i < zz_pivot_count - 1; i++)
   {
      double leg_size = MathAbs(zz_pivots[i + 1] - zz_pivots[i]) / pip_size;
      int duration = MathAbs(zz_pivot_bars[i + 1] - zz_pivot_bars[i]);
      if(duration > 0 && leg_size > 0)
      {
         double velocity = leg_size / duration;
         ArrayResize(Velocity_Dist, Velocity_Dist_Count + 1);
         Velocity_Dist[Velocity_Dist_Count] = velocity;
         Velocity_Dist_Count++;
      }
   }
   ArraySort(Velocity_Dist);
   
   //--- Build Volume profile
   Vol_Price_Min = low[ArrayMinimum(low, start_bar, bars_to_use)];
   Vol_Price_Max = high[ArrayMaximum(high, start_bar, bars_to_use)];
   Vol_Level_Size = (Vol_Price_Max - Vol_Price_Min) / Volume_Levels_Count;
   
   if(Vol_Level_Size <= 0) Vol_Level_Size = pip_size;
   
   ArrayResize(Vol_Price_Levels, Volume_Levels_Count);
   ArrayResize(Vol_At_Level, Volume_Levels_Count);
   ArrayInitialize(Vol_At_Level, 0);
   
   for(int i = 0; i < Volume_Levels_Count; i++)
      Vol_Price_Levels[i] = Vol_Price_Min + (i + 0.5) * Vol_Level_Size;
   
   for(int i = start_bar; i < rates_total; i++)
   {
      int low_level = (int)MathFloor((low[i] - Vol_Price_Min) / Vol_Level_Size);
      int high_level = (int)MathFloor((high[i] - Vol_Price_Min) / Vol_Level_Size);
      
      low_level = MathMax(0, MathMin(Volume_Levels_Count - 1, low_level));
      high_level = MathMax(0, MathMin(Volume_Levels_Count - 1, high_level));
      
      int levels_touched = high_level - low_level + 1;
      double vol_per_level = (double)tick_volume[i] / levels_touched;
      
      for(int lv = low_level; lv <= high_level; lv++)
         Vol_At_Level[lv] += vol_per_level;
   }
   
   Vol_Max = Vol_At_Level[ArrayMaximum(Vol_At_Level)];
   
   distributions_built = true;
   
   Print("Distributions built - Legs: ", Leg_Dist_Count, 
         ", Retraces: ", Retrace_Dist_Count, ", Durations: ", Time_Dist_Count,
         ", Velocities: ", Velocity_Dist_Count);
   
   if(Leg_Dist_Count > 0)
      Print("Leg size range: ", DoubleToString(Leg_Dist[0], 1), " to ", 
            DoubleToString(Leg_Dist[Leg_Dist_Count-1], 1), " pips");
   if(Velocity_Dist_Count > 0)
      Print("Velocity range: ", DoubleToString(Velocity_Dist[0], 2), " to ", 
            DoubleToString(Velocity_Dist[Velocity_Dist_Count-1], 2), " pips/bar");
}

//+------------------------------------------------------------------+
//| Calculate percentile rank in sorted array                         |
//+------------------------------------------------------------------+
double GetPercentileRank(const double &arr[], int count, double value)
{
   if(count == 0) return 50.0;
   
   int below = 0;
   for(int i = 0; i < count; i++)
   {
      if(arr[i] < value)
         below++;
      else
         break;
   }
   
   return (below * 100.0) / count;
}

//+------------------------------------------------------------------+
//| Calculate percentile rank for int array                           |
//+------------------------------------------------------------------+
double GetPercentileRankInt(const int &arr[], int count, int value)
{
   if(count == 0) return 50.0;
   
   int below = 0;
   for(int i = 0; i < count; i++)
   {
      if(arr[i] < value)
         below++;
      else
         break;
   }
   
   return (below * 100.0) / count;
}

//+------------------------------------------------------------------+
//| Get volume support level                                          |
//+------------------------------------------------------------------+
double GetVolumeSupport(double leg_start, double current_price)
{
   if(Vol_Level_Size <= 0 || Vol_Max <= 0) return 50.0;
   
   int current_level = (int)MathFloor((current_price - Vol_Price_Min) / Vol_Level_Size);
   int start_level = (int)MathFloor((leg_start - Vol_Price_Min) / Vol_Level_Size);
   
   current_level = MathMax(0, MathMin(Volume_Levels_Count - 1, current_level));
   start_level = MathMax(0, MathMin(Volume_Levels_Count - 1, start_level));
   
   double total_vol = 0;
   int start_lv = MathMin(current_level, start_level);
   int end_lv = MathMax(current_level, start_level);
   int levels = end_lv - start_lv + 1;
   
   for(int i = start_lv; i <= end_lv; i++)
      total_vol += Vol_At_Level[i];
   
   double avg_vol = total_vol / levels;
   return (avg_vol / Vol_Max) * 100.0;
}

//+------------------------------------------------------------------+
//| Custom indicator iteration function                               |
//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   if(!distributions_built || prev_calculated == 0)
   {
      BuildDistributions(rates_total, high, low, close, tick_volume);
      if(!distributions_built)
         return(0);
   }
   
   double pip_size = GetPipSize();
   int start = (prev_calculated == 0) ? 0 : prev_calculated - 1;
   
   for(int i = start; i < rates_total; i++)
   {
      int pivot_idx = FindRecentPivotIndex(i);
      
      if(pivot_idx < 0)
      {
         Historic_Buffer[i] = 50;
         Previous_Buffer[i] = 50;
         Time_Buffer[i] = 50;
         Volume_Buffer[i] = 50;
         Velocity_Buffer[i] = 50;
         Combined_Buffer[i] = 50;
         Combined_Color_Buffer[i] = 0;
         continue;
      }
      
      double leg_start = zz_pivots[pivot_idx];
      int leg_start_bar = zz_pivot_bars[pivot_idx];
      
      //--- Determine leg direction: is price above or below leg start pivot?
      bool leg_up = close[i] > leg_start;
      
      //--- Current leg size from pivot
      double current_leg_size = MathAbs(close[i] - leg_start) / pip_size;
      
      //--- Current leg duration
      int current_duration = i - leg_start_bar;
      if(current_duration < 1) current_duration = 1;
      
      //--- Current velocity (pips per bar)
      double current_velocity = current_leg_size / current_duration;
      
      //--- Pivot soon (historic): How extended is this leg compared to all historical legs?
      double historic_pct = GetPercentileRank(Leg_Dist, Leg_Dist_Count, current_leg_size);
      Historic_Buffer[i] = historic_pct;
      
      //--- Pivot soon (previous): How extended is this leg compared to the previous leg?
      double prior_leg_size = 0;
      if(pivot_idx > 0)
         prior_leg_size = MathAbs(zz_pivots[pivot_idx] - zz_pivots[pivot_idx - 1]);
      
      double current_retrace = 0;
      if(prior_leg_size > 0)
         current_retrace = (current_leg_size * pip_size / prior_leg_size) * 100;
      
      double previous_pct = GetPercentileRank(Retrace_Dist, Retrace_Dist_Count, current_retrace);
      Previous_Buffer[i] = previous_pct;
      
      //--- Time Extension: How long has this leg lasted compared to historical?
      double time_pct = GetPercentileRankInt(Time_Dist, Time_Dist_Count, current_duration);
      Time_Buffer[i] = time_pct;
      
      //--- Volume Support: How much volume support exists in this price zone?
      Volume_Buffer[i] = GetVolumeSupport(leg_start, close[i]);
      
      //--- Velocity: How fast is this leg moving compared to historical?
      //--- HIGH value = faster than most legs = sharper move = more likely to reverse sharply
      double velocity_pct = GetPercentileRank(Velocity_Dist, Velocity_Dist_Count, current_velocity);
      Velocity_Buffer[i] = velocity_pct;
      
      //--- Reversal Probability
      //--- Base: average of two pivot signals
      double pivot_base = (Historic_Buffer[i] + Previous_Buffer[i]) / 2.0;
      
      //--- Volume multiplier: 0.8x to 1.2x based on volume support (if included)
      double volume_factor = 1.0;
      if(Volume_Include)
         volume_factor = 0.8 + (Volume_Buffer[i] / 100.0) * 0.4;
      
      //--- Velocity multiplier: 0.8x to 1.2x based on velocity (if included)
      double velocity_factor = 1.0;
      if(Velocity_Include)
         velocity_factor = 0.8 + (Velocity_Buffer[i] / 100.0) * 0.4;
      
      //--- Combined with multipliers
      double combined = pivot_base * volume_factor * velocity_factor;
      Combined_Buffer[i] = MathMin(100.0, MathMax(0.0, combined));
      
      //--- Color: leg direction
      //--- leg_up (price above pivot, upward leg) = GREEN (1)
      //--- leg_down (price below pivot, downward leg) = RED (0)
      Combined_Color_Buffer[i] = leg_up ? 1 : 0;
   }
   
   //--- Export CSV after all calculations done
   if(Export_CSV && !csv_exported && prev_calculated == 0)
   {
      ExportCSV(rates_total, time, close);
   }
   
   return(rates_total);
}
