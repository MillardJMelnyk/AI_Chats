double CalcRollingStdDev(const double &data[], int barIndex, int period)
{
   if(barIndex < period - 1)
      return EMPTY_VALUE;

   double sum = 0;
   int count = 0;
   
   for(int i = 0; i < period; i++)
   {
      int idx = barIndex - i;
      if(idx < 0) break;
      if(data[idx] == EMPTY_VALUE) continue;
      sum += data[idx];
      count++;
   }
   
   if(count < 2)
      return EMPTY_VALUE;

   double mean = sum / count;
   double sumSqDiff = 0;
   
   for(int i = 0; i < period; i++)
   {
      int idx = barIndex - i;
      if(idx < 0) break;
      if(data[idx] == EMPTY_VALUE) continue;
      double diff = data[idx] - mean;
      sumSqDiff += diff * diff;
   }

   return MathSqrt(sumSqDiff / count);
}
