def highest(arr, count, idx):
    v = arr[idx]
    j = idx - 1
    while j > idx - count and j >= 0:
        if arr[j] > v:
            v = arr[j]
        j -= 1
    return v


def lowest(arr, count, idx):
    v = arr[idx]
    j = idx - 1
    while j > idx - count and j >= 0:
        if arr[j] < v:
            v = arr[j]
        j -= 1
    return v


def zigzag_mjm(high_arr, low_arr, depth, deviation_points, backstep, point):
    n = len(high_arr)
    hi_calc = np.zeros(n, dtype=float)
    lo_calc = np.zeros(n, dtype=float)

    last_low = 0.0
    last_high = 0.0

    # Candidate extremes (mirrors Zigzag MJM.mq5)
    for i in range(start, n):
        v = lowest(low_arr, depth, i)
        if v == last_low:
            v = 0.0
        else:
            last_low = v
            if (low_arr[i] - v) > deviation_price:
                v = 0.0
            else:
                for b in range(backstep, 0, -1):
                    k = i - b
                    if k >= 0 and lo_calc[k] != 0.0 and lo_calc[k] > v:
                        lo_calc[k] = 0.0
        lo_calc[i] = v if low_arr[i] == v else 0.0

        v = highest(high_arr, depth, i)
        if v == last_high:
            v = 0.0
        else:
            last_high = v
            if (v - high_arr[i]) > deviation_price:

                    last_hi_idx = i
                    search_mode = BOTTOM
                    zz_line[i] = last_high
                    zz_hi[i] = last_high

                zz_lo[last_lo_idx] = 0.0
                last_lo_idx = i
                last_low = lo_calc[i]
                zz_line[i] = last_low

            if hi_calc[i] != 0.0 and lo_calc[i] == 0.0:
                last_high = hi_calc[i]
                last_hi_idx = i
                zz_line[i] = last_high
