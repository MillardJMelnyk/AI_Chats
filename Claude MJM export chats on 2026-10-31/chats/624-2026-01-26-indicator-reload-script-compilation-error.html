Claude2026-01-26 19:46 UTC

The filename problem:

  1. What ChartIndicatorName() returns: The indicator's short name — the display string shown in the chart's indicator list. This is set by IndicatorSetString(INDICATOR_SHORTNAME, "whatever") inside the indicator, or defaults to filename if unset.

  2. What IndicatorParameters() returns: The ENUM_INDICATOR type (e.g., IND_CUSTOM, IND_MA) and an array of the indicator's input parameters only.

  3. What IndicatorCreate() requires for IND_CUSTOM: The indicator's .ex5 filename must be in params[0].string_value. The input parameters follow.

  4. The gap: IndicatorParameters() does not return the filename. MQL5 provides no function to retrieve the source filename from a handle or chart attachment.

  5. Why short name ≠ filename:

    • Indicator author set a custom INDICATOR_SHORTNAME
    • Short name includes parameter values (common: "RSI(14)")
    • Short name is completely unrelated to filename

Result: You can identify an indicator on a chart, extract its parameters, delete it — but you cannot reliably determine which .ex5 file to pass to IndicatorCreate() to recreate it.

void OnChartEvent(const int id,

   if(id == CHARTEVENT_OBJECT_CLICK && sparam == g_btnName)
   {
      if(!g_dragging)

         ReloadIndicator();
      }
      ObjectSetInteger(ChartID(), g_btnName, OBJPROP_STATE, false);
      ChartRedraw(ChartID());
   }
   
   if(id == CHARTEVENT_MOUSE_MOVE)
   {
      int mouseX = (int)lparam;
      int mouseY = (int)dparam;
      uint mouseState = (uint)StringToInteger(sparam);
      
      bool mouseDown = (mouseState & 1) == 1;
      
      if(mouseDown)
      {
         if(!g_dragging && Contains(mouseX, mouseY))
         {
            g_dragging = true;
            g_dragOffsetX = mouseX - g_btnX;
            g_dragOffsetY = mouseY - g_btnY;
            ChartSetInteger(ChartID(), CHART_MOUSE_SCROLL, false);
         }
         
         if(g_dragging)
         {
            Move(mouseX - g_dragOffsetX, mouseY - g_dragOffsetY);
         }
      }
      else
      {
         if(g_dragging)
         {
            g_dragging = false;
            ChartSetInteger(ChartID(), CHART_MOUSE_SCROLL, true);
         }
      }
   }

//+------------------------------------------------------------------+
void ReloadIndicator()
{
   long chartID = ChartID();
   
   int total = ChartIndicatorsTotal(chartID, TargetSubwindow);
   if(total == 0)
   {
      Alert("No indicator found in subwindow ", TargetSubwindow);
      return;
   }
   
   if(IndicatorIndex >= total)
   {
      Alert("Indicator index ", IndicatorIndex, " out of range. Subwindow has ", total, " indicators.");
      return;
   }
   
   string shortName = ChartIndicatorName(chartID, TargetSubwindow, IndicatorIndex);
   
   int handle = ChartIndicatorGet(chartID, TargetSubwindow, shortName);
   if(handle == INVALID_HANDLE)
   {
      Alert("Cannot get indicator handle for: ", shortName);
      return;
   }
   
   if(!ChartIndicatorDelete(chartID, TargetSubwindow, shortName))
   {
      Alert("Failed to detach indicator from chart.");
      return;
   }
   
   if(!ChartIndicatorAdd(chartID, TargetSubwindow, handle))
   {
      Alert("Failed to reattach indicator to chart.");
      return;
   }
}
