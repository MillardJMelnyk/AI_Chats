//+------------------------------------------------------------------+
//| Function to calculate a moving average                         |
//+------------------------------------------------------------------+
double GetMovingAverage(const double &pdaAray[],int piMA_Prd,ENUM_MA_METHOD peMA_Mthd,int piMA4IX=0)
  {

   int iAraySZ = ArraySize(pdaAray);
   int iStrtIX									= ((piMA4IX - piMA_Prd) < 0 || piMA4IX >= iAraySZ) ? piMA_Prd : piMA4IX;

   double result = 0.0;

   switch(peMA_Mthd)
     {
      case MODE_SMA:
         result = MovingAverage_SMA(pdaAray,piMA_Prd,piMA4IX);
         break;
      case MODE_EMA:
         result = MovingAverage_EMA(pdaAray,piMA_Prd,piMA4IX);
         break;
      case MODE_SMMA:
         result = MovingAverage_SMMA(pdaAray,piMA_Prd,piMA4IX);
         break;
      case MODE_LWMA:
         result = MovingAverage_LWMA(pdaAray,piMA_Prd,piMA4IX);
         break;
      default:
         result = 0.0;
         break;
     }

   return result;
  }


//+------------------------------------------------------------------+
//| Simple Moving Average                                            |
//+------------------------------------------------------------------+
double MovingAverage_SMA(const double &pdaAray[],int piMA_Prd,int piMA4IX=0)
  {
   double sum = 0.0;
   for (int ix = piMA4IX - piMA_Prd; ix <= piMA4IX; ix++)
     {
      sum += pdaAray[ix];
     }
   return sum / (piMA_Prd + 1);
  }

//+------------------------------------------------------------------+
//| Exponential Moving Average                                       |
//+------------------------------------------------------------------+
double MovingAverage_EMA(const double &pdaAray[],int piMA_Prd,int piMA4IX=0)
  {
   double alpha = 2.0 / (piMA_Prd + 1);
   int iStrtIX = piMA4IX - piMA_Prd;
   double ema = pdaAray[iStrtIX];
   for (int ix = iStrtIX; ix <= piMA4IX; ix++)
     {
      ema = alpha * pdaAray[ix] + (1 - alpha) * ema;
     }
   return ema;
  }

//+------------------------------------------------------------------+
//| Smoothed Moving Average                                          |
//+------------------------------------------------------------------+
double MovingAverage_SMMA(const double &pdaAray[],int piMA_Prd,int piMA4IX=0)
  {
   double smma = 0.0;
   for (int ix = piMA4IX - piMA_Prd; ix <= piMA4IX; ix++)
     {
      smma += pdaAray[ix];
     }
   smma /= piMA_Prd;

   for (int ix = piMA4IX - piMA_Prd; ix <= piMA4IX; ix++)
     {
      smma = (smma * (piMA_Prd - 1) + pdaAray[ix]) / piMA_Prd;
     }
   return smma;
  }

double MovingAverage_LWMA(const double &array[],int piPrd,int piMA4IX=0)
{

  for (int i = piPrd - 1; i >= 0; i--) {
    sum += array[piMA4IX + i] * (i + 1); // Weights increase for older data points
    weight_sum += (i + 1);
  }
  return sum / weight_sum;
}

int  iBarsFLIP_IX(int piIX,int piBars_CNT=0,string psSymb=NULL,ENUM_TIMEFRAMES peTF=PERIOD_CURRENT) 
{

	int iBars_CNT							= piBars_CNT;

		string				sSymbl  		= (psSymb == NULL)			? _Symbol 					: psSymb;

		iBars_CNT        					= (piBars_CNT == 0) 			? Bars(sSymbl,eChrtTF) 	: piBars_CNT;
	}

	int iFlippedIX							= iBars_CNT - piIX - 1;

	return(iFlippedIX); 
}
