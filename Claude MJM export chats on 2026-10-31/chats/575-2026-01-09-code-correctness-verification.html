// FindClosestTF: Iterates the g_tfList to find the best available MT5 timeframe match.
// This handles cases where a direct 4x or 1/4 result isn't a standard period.
ENUM_TIMEFRAMES FindClosestTF(int targetMins) {
   int bestMatchIndex = 0;
   long minDiff = 2147483647; 
   for(int i = 0; i < 21; i++) {
      int tfMins = PeriodSeconds(g_tfList[i]) / 60;
      long diff = MathAbs(tfMins - targetMins);
      if(diff < minDiff) {

         bestMatchIndex = i;
      }
   }
   return g_tfList[bestMatchIndex];
}

// Uses a mathematical 'sign flip' check: (Direction In * Direction Out).
// If the product is negative, price distance from the band reversed at bar 'i'.
void DetectPrice(const double &bandArr[], const double &atr[], int count, string bName, datetime filter) {
   for(int i=count-(2+PivotFollowOnLength); i>=1; i--) {

      if(i - PivotFollowOnLength < 0) continue; // safety: prevent negative index
      double valA = g_rates[i+1].close - bandArr[i+1];
      double valB = g_rates[i].close - bandArr[i];
      double valE = g_rates[i-PivotFollowOnLength].close - bandArr[i-PivotFollowOnLength];

// EvaluateOutcome: A forward-looking check to see if the signal hit a profit target or stop.
// Profit/Loss are both set to 1.0 * ATR at the time of the event.
bool EvaluateOutcome(int startIndex, string type, double entry, double atr, int totalCount) {
   double target, stop;
   bool isBuy = (type == "Price_Lower" || type == "StochUp");
   if(isBuy) { target = entry + atr; stop = entry - atr; }
   else      { target = entry - atr; stop = entry + atr; }
   for(int i=startIndex-1; i>=0; i--) {
      if(isBuy) {
         if(g_rates[i].high >= target) return true;
         if(g_rates[i].low <= stop) return false;
      } else {
         if(g_rates[i].low <= target) return true;
         if(g_rates[i].high >= stop) return false;
      }
   }
   return false;
}

// EvaluateOutcome: A forward-looking check to see if the signal hit a profit target or stop.

bool EvaluateOutcome(int startIndex, string type, double entry, double atr, int totalCount) {
   double target, stop;
   bool isBuy = (type == "Price_Lower" || type == "StochUp");
   if(isBuy) { target = entry + atr; stop = entry - atr; }
   else      { target = entry - atr; stop = entry + atr; }
   for(int i=startIndex-1; i>=0; i--) {
      if(isBuy) {
         if(g_rates[i].high >= target) return true;
         if(g_rates[i].low <= stop) return false;
      } else {
         if(g_rates[i].low <= target) return true;

//+------------------------------------------------------------------+
//| Compute Keltner Channel bands                                    |
//+------------------------------------------------------------------+
bool KC_ComputeKeltner(int count, double &up[], double &low[], KeltnerStochParams &p) {
   int hMA = iMA(_Symbol, _Period, p.KC_EMA_Period, 0, MODE_EMA, PRICE_TYPICAL);
   int hATR = iATR(_Symbol, _Period, p.KC_ATR_Period);
   
   ArraySetAsSeries(up, true); 
   ArraySetAsSeries(low, true);
   ArrayResize(up, count); 
   ArrayResize(low, count);
   
   double ma[], atr[]; 
   ArraySetAsSeries(ma, true); 
   ArraySetAsSeries(atr, true);
   
   if(CopyBuffer(hMA, 0, 0, count, ma) < count || 
      CopyBuffer(hATR, 0, 0, count, atr) < count) {
      IndicatorRelease(hMA); 
      IndicatorRelease(hATR);
      return false;
   }
   
   for(int i = 0; i < count; i++) { 
      up[i] = ma[i] + (p.KC_ATR_Mult * atr[i]); 
      low[i] = ma[i] - (p.KC_ATR_Mult * atr[i]); 
   }
   
   IndicatorRelease(hMA); 
   IndicatorRelease(hATR); 
   return true;
}

//| Compute Stochastic Oscillator                                    |
//+------------------------------------------------------------------+
bool KC_ComputeStoch(int count, double &m[], double &s[], KeltnerStochParams &p) {
   int h = iStochastic(_Symbol, _Period, p.Stoch_K, p.Stoch_D, 
                       p.Stoch_Slowing, p.Stoch_Method, p.Stoch_Price);
   
   ArraySetAsSeries(m, true); 
   ArraySetAsSeries(s, true);
   ArrayResize(m, count); 
   ArrayResize(s, count);
   
   if(CopyBuffer(h, 0, 0, count, m) < count || 
      CopyBuffer(h, 1, 0, count, s) < count) {
      IndicatorRelease(h);
      return false;
   }
   
   IndicatorRelease(h); 
   return true;
}

//+------------------------------------------------------------------+
//| Detect price reversal events at Keltner bands                    |
//+------------------------------------------------------------------+
void KC_DetectPrice(const double &bandArr[], const double &atr[], int count, 
                    string bName, datetime filter, EventRow &events[]) {
   for(int i = count - (2 + PivotFollowOnLength); i >= 1; i--) {
      if(g_rates[i].time < filter) continue; 
      if(i - PivotFollowOnLength < 0) continue;
      
      double valA = g_rates[i+1].close - bandArr[i+1];
      double valB = g_rates[i].close - bandArr[i];
      double valE = g_rates[i-PivotFollowOnLength].close - bandArr[i-PivotFollowOnLength];

      if((valB - valA) * (valE - valB) < 0) {
         EventRow e; 
         e.barIdx = i;  
         e.startTime = g_rates[i+1].time; 
         e.endTime = g_rates[i-PivotFollowOnLength].time; 
         e.type = "Price_" + bName;
         e.triggerPrice = g_rates[i].close; 
         int sIdx; 
         e.session = GetSession(e.startTime, sIdx);
         e.atr = atr[i]; 
         GetMTFStatus(e.startTime, e.type, e.confDown, e.alignUp);
         
         int sz = ArraySize(events); 
         ArrayResize(events, sz + 1); 
         events[sz] = e;
      }
   }
}

void KC_DetectStoch(const double &m[], const double &s[], const double &atr[],

      if(i - PivotFollowOnLength < 0) continue;

      double valB = m[i] - s[i];
      double valE = m[i-PivotFollowOnLength] - s[i-PivotFollowOnLength];

         EventRow e; 
         e.barIdx = i;
         e.startTime = g_rates[i+1].time;
         e.endTime = g_rates[i-PivotFollowOnLength].time;
         e.type = (m[i] > s[i]) ? "StochUp" : "StochDn";
         e.triggerPrice = g_rates[i].close; 
         int sIdx; 
         e.session = GetSession(e.startTime, sIdx);
         e.atr = atr[i];
         GetMTFStatus(e.startTime, e.type, e.confDown, e.alignUp);
         
         int sz = ArraySize(events); 
         ArrayResize(events, sz + 1); 
         events[sz] = e;
      }
   }

Current approach is correct. Here's why:

Inputs MUST stay in the script file because:

  1. input keyword only works at script/EA/indicator level (compile-time directive)
  2. They populate the MT5 Properties dialog for user configuration
  3. Each correlation script will have different parameters anyway
  4. They're the user interface for that specific pairing

Alternative approaches are worse:

Option A: Separate input .mqh per pair

  • Still requires #include at top of script before any code
  • Doesn't reduce script file size meaningfully (~12 lines)
  • Obscures what parameters the script accepts
  • Bad UX: user can't see inputs without opening include file

Option B: Struct with defaults (no input keyword)

  • Loses MT5 Properties dialog

   double KC_ATR_Mult;

//+------------------------------------------------------------------+
//| Compute Keltner Channel bands                                    |
//+------------------------------------------------------------------+
bool KC_ComputeKeltner(int count, double &up[], double &low[], KeltnerStochParams &p) {
   int hMA = iMA(_Symbol, _Period, p.KC_EMA_Period, 0, MODE_EMA, PRICE_TYPICAL);
   int hATR = iATR(_Symbol, _Period, p.KC_ATR_Period);
   
   ArraySetAsSeries(up, true); 
   ArraySetAsSeries(low, true);
   ArrayResize(up, count); 
   ArrayResize(low, count);
   
   double ma[], atr[]; 
   ArraySetAsSeries(ma, true); 
   ArraySetAsSeries(atr, true);
   
   if(CopyBuffer(hMA, 0, 0, count, ma) < count || 
      CopyBuffer(hATR, 0, 0, count, atr) < count) {
      IndicatorRelease(hMA); 
      IndicatorRelease(hATR);
      return false;
   }
   
   for(int i = 0; i < count; i++) { 
      up[i] = ma[i] + (p.KC_ATR_Mult * atr[i]); 
      low[i] = ma[i] - (p.KC_ATR_Mult * atr[i]); 
   }
   
   IndicatorRelease(hMA); 
   IndicatorRelease(hATR); 
   return true;
}

//+------------------------------------------------------------------+
//| Compute Stochastic Oscillator                                    |
//+------------------------------------------------------------------+
bool KC_ComputeStoch(int count, double &m[], double &s[], KeltnerStochParams &p) {
   int h = iStochastic(_Symbol, _Period, p.Stoch_K, p.Stoch_D, 
                       p.Stoch_Slowing, p.Stoch_Method, p.Stoch_Price);
   
   ArraySetAsSeries(m, true); 
   ArraySetAsSeries(s, true);
   ArrayResize(m, count); 
   ArrayResize(s, count);
   
   if(CopyBuffer(h, 0, 0, count, m) < count || 
      CopyBuffer(h, 1, 0, count, s) < count) {
      IndicatorRelease(h);
      return false;
   }
   
   IndicatorRelease(h); 
   return true;
}

//+------------------------------------------------------------------+
//| Detect price reversal events at Keltner bands                    |
//+------------------------------------------------------------------+
void KC_DetectPrice(const double &bandArr[], const double &atr[], int count, 
                    string bName, datetime filter, EventRow &events[]) {
   for(int i = count - (2 + PivotFollowOnLength); i >= 1; i--) {
      if(g_rates[i].time < filter) continue; 
      if(i - PivotFollowOnLength < 0) continue;
      
      double valA = g_rates[i+1].close - bandArr[i+1];
      double valB = g_rates[i].close - bandArr[i];
      double valE = g_rates[i-PivotFollowOnLength].close - bandArr[i-PivotFollowOnLength];

      if((valB - valA) * (valE - valB) < 0) {
         EventRow e; 
         e.barIdx = i;  
         e.startTime = g_rates[i+1].time; 
         e.endTime = g_rates[i-PivotFollowOnLength].time; 
         e.type = "Price_" + bName;
         e.triggerPrice = g_rates[i].close; 
         int sIdx; 
         e.session = GetSession(e.startTime, sIdx);
         e.atr = atr[i]; 
         GetMTFStatus(e.startTime, e.type, e.confDown, e.alignUp);
         
         int sz = ArraySize(events); 
         ArrayResize(events, sz + 1); 
         events[sz] = e;
      }
   }
}

void KC_DetectStoch(const double &m[], const double &s[], const double &atr[],
