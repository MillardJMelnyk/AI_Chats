//+------------------------------------------------------------------+
//| Parse TextGVar object name to extract InstanceID and ShortName   |
//| Format: MJM_[InstanceID]_[ShortName]_[VarName]                   |
//| Returns true if successfully parsed, false otherwise             |
//+------------------------------------------------------------------+
bool ParseTextGVarObjectName(const string objName, int &instanceID, string &shortName)
{
   //--- Must start with prefix
   if(StringFind(objName, TEXTGVAR_PREFIX + "_") != 0)
      return false;
   
   //--- Find first underscore after prefix (end of InstanceID)
   int prefixLen = StringLen(TEXTGVAR_PREFIX) + 1;  // "MJM_"
   int secondUnderscore = StringFind(objName, "_", prefixLen);
   if(secondUnderscore < 0)
      return false;
   
   //--- Extract InstanceID
   string idStr = StringSubstr(objName, prefixLen, secondUnderscore - prefixLen);
   instanceID = (int)StringToInteger(idStr);
   
   //--- Find last underscore (start of VarName)
   int lastUnderscore = -1;
   for(int i = StringLen(objName) - 1; i >= secondUnderscore; i--)
   {
      if(StringGetCharacter(objName, i) == '_')
      {
         lastUnderscore = i;
         break;
      }
   }
   
   if(lastUnderscore <= secondUnderscore)
      return false;
   
   //--- Extract ShortName (between second underscore and last underscore)
   shortName = StringSubstr(objName, secondUnderscore + 1, lastUnderscore - secondUnderscore - 1);
   
   return true;
}

   for(int win = 0; win < totalWindows; win++)
   {
      int count = ChartIndicatorsTotal(0, win);
      for(int i = 0; i < count; i++)

}

//| Get MaxPeriod from TextGVar parameter metadata                   |
//+------------------------------------------------------------------+
int GetMaxPeriodFromTextGVar(int instanceID, const string shortName)
{
   string countStr = ReadTextGVar(instanceID, shortName, "ParmCount", "0");
   int parmCount = (int)StringToInteger(countStr);
   int maxPeriod = 1;
   
   for(int p = 0; p < parmCount; p++)
   {
      string parmName = ReadTextGVar(instanceID, shortName, "Parm[" + IntegerToString(p) + "]_Name", "");
      
      // Case-insensitive check for "period"
      string lowerName = parmName;
      StringToLower(lowerName);
      if(StringFind(lowerName, "period") >= 0)
      {
         string parmValue = ReadTextGVar(instanceID, shortName, "Parm[" + IntegerToString(p) + "]_Value", "1");
         int val = (int)StringToInteger(parmValue);
         if(val > maxPeriod)
            maxPeriod = val;
      }
   }
   return maxPeriod;
}

//+------------------------------------------------------------------+
//| Scan chart for indicators and configure slope lines             |
//+------------------------------------------------------------------+
void ScanChartIndicators()
{
   //--- Reset globals
   g_totalBuffers = 0;
   g_totalPlots = 0;
   int indicatorCount = 0;
   
   //--- Clear all buffers to remove old plot data
   ClearAllBuffers();
   
   //--- Hide all plots initially (they'll be shown as we configure them)
   for(int p = 0; p < 32; p++)
   {
      PlotIndexSetInteger(p, PLOT_DRAW_TYPE, DRAW_NONE);
      PlotIndexSetString(p, PLOT_LABEL, "");
   }
   
   //--- Discovery phase: Find all registered indicators via TextGVar objects
   DiscoveredIndicator discovered[];
   int discoveredCount = 0;
   
   for(int obj = ObjectsTotal(0) - 1; obj >= 0; obj--)
   {
      string objName = ObjectName(0, obj);
      
      //--- Look for ShortName objects (format: MJM_[ID]_[ShortName]_ShortName)
      if(StringFind(objName, "_ShortName") < 0)
         continue;
      
      int instanceID;
      string shortName;
      if(!ParseTextGVarObjectName(objName, instanceID, shortName))
         continue;
      
      //--- Skip ourselves
      if(shortName == sMeShortName)
         continue;
      
      //--- Check if already discovered (avoid duplicates)
      bool alreadyFound = false;
      for(int d = 0; d < discoveredCount; d++)
      {
         if(discovered[d].instanceID == instanceID && discovered[d].shortName == shortName)
         {
            alreadyFound = true;
            break;
         }
      }
      if(alreadyFound)
         continue;
      
      //--- Add to discovered list
      ArrayResize(discovered, discoveredCount + 1);
      discovered[discoveredCount].instanceID = instanceID;
      discovered[discoveredCount].shortName = shortName;
      discoveredCount++;
      
      Print("Discovered indicator: InstanceID<", instanceID, "> ShortName<", shortName, ">");
   }
   
   Print("Discovery phase complete: found ", discoveredCount, " registered indicators");
   
   //--- Process each discovered indicator
   for(int d = 0; d < discoveredCount; d++)
   {
      int instanceID = discovered[d].instanceID;
      string shortName = discovered[d].shortName;
      
      //--- Check indicator limit
      if(indicatorCount >= MAX_INDICATORS)
      {
         Print("ERROR: Maximum indicator limit (", MAX_INDICATORS, ") reached. Skipping indicator: ", shortName);
         continue;
      }
      
      //--- Find which window the indicator is in
      int win = FindIndicatorWindow(shortName);
      if(win < 0)
      {
         Print("Indicator ", shortName, " not found in any window - skipping");
         continue;
      }
      
      //--- Get handle
      int handle = ChartIndicatorGet(0, win, shortName);
      if(handle == INVALID_HANDLE)
      {
         Print("Bad handle for indicator<", shortName, "> in window ", win, " - skipping");
         continue;
      }
      
      int plotCount = GetIndicatorPlotCount(instanceID, shortName);
      
      //--- Skip indicators with no plots
      if(plotCount < 1)
      {
         Print("Indicator ", shortName, " has no plots - skipping");
         continue;
      }
      
      //--- Check if adding this indicator would exceed buffer limit
      if(g_totalBuffers + plotCount > MAX_BUFFERS)
      {
         Print("ERROR: Adding indicator ", shortName, " would exceed buffer limit (", MAX_BUFFERS, "). Skipping...");
         continue;
      }
      
      //--- Get max period from TextGVar parameter metadata
      int maxPeriod = GetMaxPeriodFromTextGVar(instanceID, shortName);
      
      Print("Processing indicator<", shortName, ">, InstanceID<", instanceID, ">, Window<", win, ">, plot count: <", plotCount, ">, maxPeriod: <", maxPeriod, ">");
      
      for(int lineIdx = 0; lineIdx < plotCount; lineIdx++)
      {
         if(g_totalBuffers >= MAX_BUFFERS)
         {
            Print("ERROR: Buffer limit (", MAX_BUFFERS, ") reached while processing ", shortName);
            break;
         }
         
         string lineName = GetPlotDisplayName(instanceID, shortName, lineIdx);
         
         //--- Read color/width from TextGVar, fall back to defaults
         color defaultColor = GetAutoColor(g_totalPlots);
         int defaultWidth = 2;
         
         string colorStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIdx) + "]_Color", "");
         color lineColor = (colorStr != "") ? StringToColor(colorStr) : defaultColor;
         
         string widthStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIdx) + "]_Width", "");
         int lineWidth = (widthStr != "") ? (int)StringToInteger(widthStr) : defaultWidth;
         int lineStyle = STYLE_SOLID;
         
         int idx = ArraySize(g_lines);
         ArrayResize(g_lines, idx + 1);
         
         g_lines[idx].indicatorHandle = handle;
         g_lines[idx].indicatorShortName = shortName;
         g_lines[idx].instanceID = instanceID;
         g_lines[idx].indicatorWindow = win;
         g_lines[idx].lineIndex = lineIdx;
         g_lines[idx].bufferIndex = g_totalBuffers;
         g_lines[idx].period = maxPeriod;
         g_lines[idx].lineColor = lineColor;
         g_lines[idx].lineStyle = lineStyle;
         g_lines[idx].lineWidth = lineWidth;
         g_lines[idx].lineName = lineName;
         g_lines[idx].lastRatesTotal = 0;
         g_lines[idx].isValid = true;
         g_lines[idx].cachedColorStr = colorStr;
         g_lines[idx].cachedWidthStr = widthStr;
         
         Print("Line ", idx, ": ShortName<", shortName, 
               "> Handle<", g_lines[idx].indicatorHandle,
               "> InstanceID<", g_lines[idx].instanceID,
               "> Window<", g_lines[idx].indicatorWindow,
               "> LineIdx<", g_lines[idx].lineIndex,
               "> BufferIdx<", g_lines[idx].bufferIndex,
               "> Period<", g_lines[idx].period,
               "> Color<", g_lines[idx].lineColor,
               "> Style<", g_lines[idx].lineStyle,
               "> Width<", g_lines[idx].lineWidth,
               "> LineName<", g_lines[idx].lineName,
               "> CachedColorStr<", g_lines[idx].cachedColorStr,
               "> CachedWidthStr<", g_lines[idx].cachedWidthStr,
               ">");
         
         SetBufferByIndex(g_totalBuffers);
         PlotIndexSetInteger(g_totalPlots, PLOT_DRAW_TYPE, DRAW_LINE);
         PlotIndexSetInteger(g_totalPlots, PLOT_LINE_STYLE, lineStyle);
         PlotIndexSetInteger(g_totalPlots, PLOT_LINE_WIDTH, lineWidth);
         PlotIndexSetInteger(g_totalPlots, PLOT_LINE_COLOR, lineColor);
         PlotIndexSetString(g_totalPlots, PLOT_LABEL, shortName + " " + lineName + " Slope");

         g_totalBuffers++;
         g_totalPlots++;
      }
      
      indicatorCount++;
   }
   
   //--- Configure summary buffers if we have active line buffers
   if(g_totalBuffers > 0)
   {
      //--- Buffer30: Slopes Avg
      SetBufferByIndex(SUMMARY_AVG_BUFFER);

//+------------------------------------------------------------------+
//| Check for REINIT signals from TextGVar objects                   |
//| Returns true if rescan is needed                                 |
//+------------------------------------------------------------------+
bool CheckReinitSignals()
{
   string checkedNames[];
   int checkedCount = 0;
   
   for(int i = 0; i < ArraySize(g_lines); i++)
   {
      if(!g_lines[i].isValid)
         continue;
      
      string shortName = g_lines[i].indicatorShortName;
      int instanceID = g_lines[i].instanceID;
      
      //--- Check if we've already processed this shortName
      bool alreadyChecked = false;
      for(int j = 0; j < checkedCount; j++)
      {
         if(checkedNames[j] == shortName)
         {
            alreadyChecked = true;
            break;
         }
      }
      
      if(alreadyChecked)
         continue;
      
      ArrayResize(checkedNames, checkedCount + 1);
      checkedNames[checkedCount] = shortName;
      checkedCount++;
      
      //--- Check for REINIT signal
      string reinitSignal = ReadTextGVar(instanceID, shortName, TEXTGVAR_REINIT, "");
      
      if(reinitSignal == "")
         continue;
      
      Print("REINIT detected for ", shortName, ", triggering rescan");
      
      //--- Delete the REINIT flag
      TextGVar_Delete(TEXTGVAR_PREFIX, instanceID, shortName, TEXTGVAR_REINIT);
      
      return true;
   }
   
   return false;
}

int OnInit()
{
   //--- Validate InpSlopeBarCnt
   if(InpSlopeBarCnt < 2)
   {
      Print("InpSlopeBarCnt must be >= 2");
      return(INIT_PARAMETERS_INCORRECT);
   }

   IndicatorSetString(INDICATOR_SHORTNAME, sMeShortName);
   
   //--- Reset line array
   ArrayResize(g_lines, 0);

   //--- Clean any outstanding REINIT flags (stale from previous session or just-completed SI reinits)
   for(int obj = ObjectsTotal(0) - 1; obj >= 0; obj--)
   {
      string objName = ObjectName(0, obj);
      if(StringFind(objName, TEXTGVAR_PREFIX + "_") == 0 && 
         StringFind(objName, "_REINIT") > 0)
      {
         ObjectDelete(0, objName);
      }
   }
   
   //--- Scan chart for indicators
   ScanChartIndicators();
   
   if(g_totalBuffers == 0)
   {
      Print("No indicator plots found on chart!");
      
      //--- Remove this indicator from the chart
      int chartWindow = ChartWindowFind();
      if(chartWindow >= 0)
      {
         Print("Removing indicator - ChartWindow<", chartWindow, "> ShortName<", sMeShortName, ">");
         ChartIndicatorDelete(0, chartWindow, sMeShortName);
      }
      
      return(INIT_FAILED);
   }
   
   EventSetMillisecondTimer(250);  // Check every 250 milliseconds
//   EventSetTimer(1);  // Check every 1 second
   g_lastIndicatorCount = GetTotalIndicatorCount();

   return(INIT_SUCCEEDED);
}

//| Timer function - detect indicator changes                        |
//+------------------------------------------------------------------+
void OnTimer()
{
   //--- Check for line property changes from TextGVar
   CheckLinePropertyChanges();
   
   //--- Check for REINIT signals
   bool needsRescan = CheckReinitSignals();
   
   //--- Count indicators across all subwindows
   int currentCount = GetTotalIndicatorCount();
   
   if(needsRescan || currentCount != g_lastIndicatorCount)
   {
      Print("Reinitializing...");
      g_lastIndicatorCount = currentCount;
      
      //--- Release existing handles
      for(int i = 0; i < ArraySize(g_lines); i++)
      {
         if(g_lines[i].indicatorHandle != INVALID_HANDLE)
            IndicatorRelease(g_lines[i].indicatorHandle);
      }
      ArrayResize(g_lines, 0);
      
      //--- Rescan and rebuild
      ScanChartIndicators();
      
      //--- Force full recalculation on next OnCalculate
      ChartSetSymbolPeriod(0, _Symbol, _Period);
   }
}

//+------------------------------------------------------------------+
//| Check and apply TextGVar property changes for all lines          |
//+------------------------------------------------------------------+
void CheckLinePropertyChanges()
{
   for(int i = 0; i < ArraySize(g_lines); i++)
   {
      if(!g_lines[i].isValid)
         continue;
      
      string shortName = g_lines[i].indicatorShortName;
      int instanceID = g_lines[i].instanceID;
      int lineIdx = g_lines[i].lineIndex;
      
      //--- Read current values from TextGVar
      string currentColorStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIdx) + "]_Color", "");
      string currentWidthStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIdx) + "]_Width", "");
      
      bool changed = false;
      
      //--- Check color change
      if(currentColorStr != "" && currentColorStr != g_lines[i].cachedColorStr)
      {
         color newColor = StringToColor(currentColorStr);
         g_lines[i].lineColor = newColor;
         g_lines[i].cachedColorStr = currentColorStr;
         PlotIndexSetInteger(g_lines[i].bufferIndex, PLOT_LINE_COLOR, newColor);
         changed = true;
         Print("Line ", i, " (", shortName, " ", g_lines[i].lineName, ") color changed to ", currentColorStr);
      }
      
      //--- Check width change
      if(currentWidthStr != "" && currentWidthStr != g_lines[i].cachedWidthStr)
      {
         int newWidth = (int)StringToInteger(currentWidthStr);
         g_lines[i].lineWidth = newWidth;
         g_lines[i].cachedWidthStr = currentWidthStr;
         PlotIndexSetInteger(g_lines[i].bufferIndex, PLOT_LINE_WIDTH, newWidth);
         changed = true;
         Print("Line ", i, " (", shortName, " ", g_lines[i].lineName, ") width changed to ", currentWidthStr);
      }
      
      if(changed)
         ChartRedraw(0);
   }
}


//+------------------------------------------------------------------+
//| Custom indicator iteration function                              |
//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{

//   //--- Dump arrays on first call only
//   if(prev_calculated == 0)
//   {
//      Print("=== OnCalculate First Run Dump ===");
//      Print("g_totalBuffers<", g_totalBuffers, "> g_totalPlots<", g_totalPlots, "> g_lines size<", ArraySize(g_lines), ">");
//

//      {

//               "> ShortName<", g_lines[i].indicatorShortName,
//               "> LineIdx<", g_lines[i].lineIndex,

//               "> Period<", g_lines[i].period,
//               "> Color<", g_lines[i].lineColor,
//               "> LineName<", g_lines[i].lineName,
//               "> CacheSize<", ArraySize(g_lines[i].cache),
//               "> lastRatesTotal<", g_lines[i].lastRatesTotal,
//               ">");
//      }
//      
//      Print("g_knownIndicators size<", ArraySize(g_knownIndicators), ">");
//      for(int i = 0; i < ArraySize(g_knownIndicators); i++)
//      {
//         Print("g_knownIndicators[", i, "]: ShortName<", g_knownIndicators[i].shortName,
//               "> PlotCount<", g_knownIndicators[i].plotCount,
//               "> PlotNames size<", ArraySize(g_knownIndicators[i].plotNames),
//               ">");
//      }
//      
//      Print("=== End Dump ===");
//   }
   
   if(rates_total < InpSlopeBarCnt)
      return 0;

   //--- Track the earliest bar we need to recalculate for summary
   int summaryStartBar = 0;

   for(int lineIdx = 0; lineIdx < ArraySize(g_lines); lineIdx++)
   {
      //--- Capture before cache update overwrites it
      bool forceFullRecalc = (g_lines[lineIdx].lastRatesTotal == 0);
      
      //--- Guard: need enough bars for indicator period + slope calc
      int minBarsRequired = g_lines[lineIdx].period + InpSlopeBarCnt;
      
      if(rates_total < minBarsRequired)
         continue;
      if(!g_lines[lineIdx].isValid)
         continue;
      
      int lastRates = g_lines[lineIdx].lastRatesTotal;
      int newBars = rates_total - lastRates;
      
      //--- First call or history changed: full copy
      if(lastRates == 0 || newBars < 0 || newBars > rates_total)
      {
         ArrayResize(g_lines[lineIdx].cache, rates_total);
         ArraySetAsSeries(g_lines[lineIdx].cache, true);
         
         int copyResult = CopyBuffer(g_lines[lineIdx].indicatorHandle, 
                                     g_lines[lineIdx].lineIndex, 
                                     0,
                                     rates_total,
                                     g_lines[lineIdx].cache);
         
         if(copyResult <= 0)
         {
            Print("CopyBuffer full copy FAILED for line ", lineIdx, ": got ", copyResult);
            g_lines[lineIdx].isValid = false;
            continue;
         }

         //--- Accept what we got (indicator may have fewer bars due to warmup period)
         g_lines[lineIdx].lastRatesTotal = rates_total;
      }
      else
      {
         //--- Incremental update: copy new bars + bar 0 update
         int barsToCopy = newBars + 1;  // +1 to recalc bar 0
         if(barsToCopy > rates_total)
            barsToCopy = rates_total;
         
         //--- Resize cache if new bars arrived
         if(rates_total > ArraySize(g_lines[lineIdx].cache))
         {
            ArrayResize(g_lines[lineIdx].cache, rates_total);
            ArraySetAsSeries(g_lines[lineIdx].cache, true);
         }
         
         double tempBuffer[];
         ArraySetAsSeries(tempBuffer, true);
         
         int copyResult = CopyBuffer(g_lines[lineIdx].indicatorHandle, 
                                     g_lines[lineIdx].lineIndex, 
                                     0,
                                     barsToCopy,
                                     tempBuffer);
         
         if(copyResult <= 0)
         {
            Print("CopyBuffer incremental FAILED for line ", lineIdx, ": got ", copyResult);
            g_lines[lineIdx].isValid = false;
            continue;
         }
         
         //--- Update cache with new values
         for(int i = 0; i < copyResult; i++)
         {
            g_lines[lineIdx].cache[i] = tempBuffer[i];
         }
         g_lines[lineIdx].lastRatesTotal = rates_total;       
      }

      //--- Calculate Z-score normalized slopes
      //--- Oldest bar we can calculate slope for (need room for slope bars + z-score lookback)
      int oldestSlopeBar = rates_total - 1 - g_lines[lineIdx].period - (InpSlopeBarCnt - 1) - InpZScorePeriod;
      if(oldestSlopeBar < 0)
         oldestSlopeBar = 0;
      
      //--- Determine which bars to recalculate
      int startBar;
      if(prev_calculated == 0 || forceFullRecalc)
      {
         startBar = oldestSlopeBar;
      }
      else
      {
         startBar = newBars;  // recalc new bars + bar 0
         if(startBar > oldestSlopeBar)
            startBar = oldestSlopeBar;
      }
      
      //--- Track for summary calculation
      if(startBar > summaryStartBar)
         summaryStartBar = startBar;
      
      //+------------------------------------------------------------------+
      // NOTE: loop MUST run from older bars to newer for EMA correctness
      //+------------------------------------------------------------------+
      for(int CalcBarIX = startBar; CalcBarIX >= 0; CalcBarIX--)      
      {
         int PrevBarIX = CalcBarIX + InpSlopeBarCnt - 1;
      
         //--- Guard against out of range
         if(PrevBarIX >= ArraySize(g_lines[lineIdx].cache))
            continue;
      
         //--- Calculate raw slope for current bar
         double currentValue = g_lines[lineIdx].cache[CalcBarIX];
         double prevValue    = g_lines[lineIdx].cache[PrevBarIX];
      
         //--- Guard against EMPTY_VALUE
         if(currentValue == EMPTY_VALUE || prevValue == EMPTY_VALUE)
            continue;
      
         double slope = (currentValue - prevValue) / (InpSlopeBarCnt - 1);
         
         double outValue = 0.0;
         
         //================ CALCULATE outValue BY MODE =================
         if(InpRawSlope)
         {
            outValue = slope;
         }
         else if(InpSlopeMode == SLOPE_ZSCORE)
         {
            //--- Calculate mean and standard deviation of slopes over lookback period
            double sumSlope = 0.0;
            double sumSlopeSq = 0.0;
            int count = 0;
      
            for(int j = CalcBarIX;
                j < CalcBarIX + InpZScorePeriod &&
                j + InpSlopeBarCnt - 1 < ArraySize(g_lines[lineIdx].cache);
                j++)
            {
               double val     = g_lines[lineIdx].cache[j];
               double valPrev = g_lines[lineIdx].cache[j + InpSlopeBarCnt - 1];
      
               if(val == EMPTY_VALUE || valPrev == EMPTY_VALUE)
                  continue;
      
               double s = (val - valPrev) / (InpSlopeBarCnt - 1);
               sumSlope   += s;
               sumSlopeSq += s * s;
               count++;
            }
      
            if(count > 1)
            {
               double mean     = sumSlope / count;
               double variance = (sumSlopeSq / count) - (mean * mean);
               double stdDev   = (variance > 0) ? MathSqrt(variance) : 0;
      
               if(stdDev > 0)
                  outValue = (slope - mean) / stdDev;
               else
                  outValue = 0.0;
            }
            else
               outValue = 0.0;
         }
         else if(InpSlopeMode == SLOPE_EMA)
         {
            double alpha = 2.0 / (InpEmaPeriod + 1.0);
            double prev;
            
            if(CalcBarIX + 1 >= rates_total)
            {
               prev = slope;
            }
            else
            {
               switch(g_lines[lineIdx].bufferIndex)
               {
                  case 0:  prev = Buffer0[CalcBarIX + 1]; break;
                  case 1:  prev = Buffer1[CalcBarIX + 1]; break;
                  case 2:  prev = Buffer2[CalcBarIX + 1]; break;
                  case 3:  prev = Buffer3[CalcBarIX + 1]; break;
                  case 4:  prev = Buffer4[CalcBarIX + 1]; break;
                  case 5:  prev = Buffer5[CalcBarIX + 1]; break;
                  case 6:  prev = Buffer6[CalcBarIX + 1]; break;
                  case 7:  prev = Buffer7[CalcBarIX + 1]; break;
                  case 8:  prev = Buffer8[CalcBarIX + 1]; break;
                  case 9:  prev = Buffer9[CalcBarIX + 1]; break;
                  case 10: prev = Buffer10[CalcBarIX + 1]; break;
                  case 11: prev = Buffer11[CalcBarIX + 1]; break;
                  case 12: prev = Buffer12[CalcBarIX + 1]; break;
                  case 13: prev = Buffer13[CalcBarIX + 1]; break;
                  case 14: prev = Buffer14[CalcBarIX + 1]; break;
                  case 15: prev = Buffer15[CalcBarIX + 1]; break;
                  case 16: prev = Buffer16[CalcBarIX + 1]; break;
                  case 17: prev = Buffer17[CalcBarIX + 1]; break;
                  case 18: prev = Buffer18[CalcBarIX + 1]; break;
                  case 19: prev = Buffer19[CalcBarIX + 1]; break;
                  case 20: prev = Buffer20[CalcBarIX + 1]; break;
                  case 21: prev = Buffer21[CalcBarIX + 1]; break;
                  case 22: prev = Buffer22[CalcBarIX + 1]; break;
                  case 23: prev = Buffer23[CalcBarIX + 1]; break;
                  case 24: prev = Buffer24[CalcBarIX + 1]; break;
                  case 25: prev = Buffer25[CalcBarIX + 1]; break;
                  case 26: prev = Buffer26[CalcBarIX + 1]; break;
                  case 27: prev = Buffer27[CalcBarIX + 1]; break;
                  case 28: prev = Buffer28[CalcBarIX + 1]; break;
                  case 29: prev = Buffer29[CalcBarIX + 1]; break;
                  default: prev = slope; break;
               }
            }
      
            if(prev == EMPTY_VALUE)
               prev = slope;
      
            outValue = alpha * slope + (1.0 - alpha) * prev;
         }
         else if(InpSlopeMode == SLOPE_MEDIAN)
         {
            double tmp[];
            ArrayResize(tmp, InpMedianPeriod);
            int k = 0;
      
            for(int j = CalcBarIX;
                j < CalcBarIX + InpMedianPeriod &&
                j + InpSlopeBarCnt - 1 < ArraySize(g_lines[lineIdx].cache);
                j++)
            {
               double val     = g_lines[lineIdx].cache[j];
               double valPrev = g_lines[lineIdx].cache[j + InpSlopeBarCnt - 1];
      
               if(val == EMPTY_VALUE || valPrev == EMPTY_VALUE)
                  continue;
      
               tmp[k++] = (val - valPrev) / (InpSlopeBarCnt - 1);
            }
      
            if(k > 0)
            {
               ArrayResize(tmp, k);
               ArraySort(tmp);
               outValue = tmp[k / 2];
            }
            else
               outValue = 0.0;
         }
         
         //================ NORMALIZE outValue =================
         double minVal = currentValue;
         double maxVal = currentValue;
         
         for(int j = CalcBarIX; 
             j < CalcBarIX + InpRawNormPeriod && 
             j < ArraySize(g_lines[lineIdx].cache); 
             j++)
         {
            double val = g_lines[lineIdx].cache[j];
            if(val == EMPTY_VALUE)
               continue;
            if(val < minVal) minVal = val;
            if(val > maxVal) maxVal = val;
         }
         
         double range = maxVal - minVal;
         if(range > 0)
            outValue = outValue * 100.0 * (InpSlopeBarCnt - 1) / range;
         else
            outValue = 0.0;
         
         //================ WRITE TO BUFFER =================
         switch(g_lines[lineIdx].bufferIndex)
         {
            case 0:  Buffer0[CalcBarIX]  = outValue; break;
            case 1:  Buffer1[CalcBarIX]  = outValue; break;
            case 2:  Buffer2[CalcBarIX]  = outValue; break;
            case 3:  Buffer3[CalcBarIX]  = outValue; break;
            case 4:  Buffer4[CalcBarIX]  = outValue; break;
            case 5:  Buffer5[CalcBarIX]  = outValue; break;
            case 6:  Buffer6[CalcBarIX]  = outValue; break;
            case 7:  Buffer7[CalcBarIX]  = outValue; break;
            case 8:  Buffer8[CalcBarIX]  = outValue; break;
            case 9:  Buffer9[CalcBarIX]  = outValue; break;
            case 10: Buffer10[CalcBarIX] = outValue; break;
            case 11: Buffer11[CalcBarIX] = outValue; break;
            case 12: Buffer12[CalcBarIX] = outValue; break;
            case 13: Buffer13[CalcBarIX] = outValue; break;
            case 14: Buffer14[CalcBarIX] = outValue; break;
            case 15: Buffer15[CalcBarIX] = outValue; break;
            case 16: Buffer16[CalcBarIX] = outValue; break;
            case 17: Buffer17[CalcBarIX] = outValue; break;
            case 18: Buffer18[CalcBarIX] = outValue; break;
            case 19: Buffer19[CalcBarIX] = outValue; break;
            case 20: Buffer20[CalcBarIX] = outValue; break;
            case 21: Buffer21[CalcBarIX] = outValue; break;
            case 22: Buffer22[CalcBarIX] = outValue; break;
            case 23: Buffer23[CalcBarIX] = outValue; break;
            case 24: Buffer24[CalcBarIX] = outValue; break;
            case 25: Buffer25[CalcBarIX] = outValue; break;
            case 26: Buffer26[CalcBarIX] = outValue; break;
            case 27: Buffer27[CalcBarIX] = outValue; break;
            case 28: Buffer28[CalcBarIX] = outValue; break;
            case 29: Buffer29[CalcBarIX] = outValue; break;
         }
      }
   }
   
   //================ CALCULATE SUMMARY BUFFERS =================
   if(g_totalBuffers > 0)
   {
      for(int bar = summaryStartBar; bar >= 0; bar--)
      {
         double sum = 0.0;
         int positiveCount = 0;
         int negativeCount = 0;
         int validCount = 0;
         
         //--- Read from all active line buffers
         for(int i = 0; i < g_totalBuffers; i++)
         {
            double val = GetBufferValue(i, bar);
            if(val == EMPTY_VALUE)
               continue;

               positiveCount++;
            else if(val < 0)
               negativeCount++;
            validCount++;
         }

         {

//               Buffer30[bar] = -MathAbs(avg);
//            else
//               Buffer30[bar] = 0.0;  // 50/50 on zero line
//            
            //--- Calculate percentage UP (positive) vs DN (negative)
            //--- Encode as: positive value = % UP, negative value = % DN
            double pctUp = (double)positiveCount / validCount * 100.0;
            double pctDn = (double)negativeCount / validCount * 100.0;
            
            //--- Store signed value: positive if majority up, negative if majority down
            if(positiveCount > negativeCount)
               Buffer31[bar] = pctUp;
            else if(negativeCount > positiveCount)
               Buffer31[bar] = -pctDn;
            else
               Buffer31[bar] = 0.0;
         }
         else
         {
            Buffer30[bar] = EMPTY_VALUE;
            Buffer31[bar] = EMPTY_VALUE;
         }
      }
   }
   
   return(rates_total);
}
