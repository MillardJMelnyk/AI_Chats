void UpdateOpenTradeLiveState(const int bar, const double liveExitPrice)
{
   if(g_currentTrade.direction == 0 || g_currentTrade.entryBar < 0)
      return;

   if(bar >= ArraySize(BufferTradePL))
      TS_GrowStatsArrays(bar + 1, BufferTradePL, BufferRunningPL, BufferTradeCount, BufferWinPct,
                         BufferShortCount, BufferShortWinPct, BufferLongCount, BufferLongWinPct);

   double livePL = TS_Price2Pips(liveExitPrice - g_currentTrade.entryPrice) * g_currentTrade.direction;
   double liveRunning = g_tradeStats.runningPL + livePL;

   BufferTradePL[bar] = livePL;
   BufferRunningPL[bar] = liveRunning;
   BufferTradeCount[bar] = g_tradeStats.TotalTrades();
   BufferWinPct[bar] = g_tradeStats.WinPct();
   BufferShortCount[bar] = g_tradeStats.shortCount;
   BufferShortWinPct[bar] = g_tradeStats.ShortWinPct();
   BufferLongCount[bar] = g_tradeStats.longCount;
   BufferLongWinPct[bar] = g_tradeStats.LongWinPct();

   if(g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0)
   {
      string tip = StringFormat("Entry: %s | Exit: %s | P/L: %.1f pips | Running: %.1f pips",
                                DoubleToString(g_currentTrade.entryPrice, _Digits),
                                DoubleToString(liveExitPrice, _Digits),
                                BufferTradePL[bar],
                                BufferRunningPL[bar]);
      ObjectSetString(0, g_currentTrade.arrowName, OBJPROP_TOOLTIP, tip);
   }
}

//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   if(rates_total <= g_maxPeriodsBack)
      return(0);

   int start = (prev_calculated == 0) ? g_maxPeriodsBack : prev_calculated - 1;
   if(start < 1)
      start = 1;

   if(prev_calculated == 0)
   {
      ResetAllStateAndBuffers();
      g_lastExportedBar = -1;
      InitCsvDebugFile();
   }

   int copyStart = MathMax(0, start - 3);
   int copyCount = rates_total - copyStart;

   double temaTail[];
   double emaTail[];
   double kcUpperTail[];
   double kcMiddleTail[];
   double kcLowerTail[];
   ArrayResize(temaTail, copyCount);
   ArrayResize(emaTail, copyCount);
   ArrayResize(kcUpperTail, copyCount);
   ArrayResize(kcMiddleTail, copyCount);
   ArrayResize(kcLowerTail, copyCount);
   ArraySetAsSeries(temaTail, false);
   ArraySetAsSeries(emaTail, false);
   ArraySetAsSeries(kcUpperTail, false);

1. Scaling formula produces near-zero output in realistic scenarios.

The formula: extra = (InpNarrowWidthRatio - widthRatio) * InpBlackoutScaleK

Two viable approaches:

//+------------------------------------------------------------------+
//| Percentile helper: extracts values from g_<source> array range   |
//| [from..to], sorts, returns the requested percentile.             |
//| Uses g_percentileWork[] as scratch space.                        |
//+------------------------------------------------------------------+
double ComputePercentile(const double &source[], int from, int to, double percentile)
{
   int count = to - from + 1;
   if(count <= 0)
      return 0.0;

   if(ArraySize(g_percentileWork) < count)
      ArrayResize(g_percentileWork, count);

   ArrayCopy(g_percentileWork, source, 0, from, count);
   ArraySort(g_percentileWork);

   //--- ArraySort sorts the entire array; we only filled 'count' elements.
   //--- Since ArrayResize may leave the array larger, sort is still valid
   //--- as long as we index within [0..count-1].
   int idx = (int)MathRound(percentile / 100.0 * (count - 1));
   idx = MathMax(0, MathMin(idx, count - 1));
   return g_percentileWork[idx];
}

//+------------------------------------------------------------------+
//| Count crosses in window [barEnd - windowSize + 1 .. barEnd]      |
//| using prefix sums. O(1).                                         |
//+------------------------------------------------------------------+
int CountCrossesInWindow(int barEnd, int windowSize)
{
   if(barEnd < 0 || barEnd >= ArraySize(g_crossPrefixSum))
      return 0;

   int barStart = barEnd - windowSize + 1;
   if(barStart < 0)
      barStart = 0;

   double sumEnd = g_crossPrefixSum[barEnd];
   double sumBefore = (barStart > 0) ? g_crossPrefixSum[barStart - 1] : 0.0;
   return (int)(sumEnd - sumBefore + 0.5);
}

//+------------------------------------------------------------------+
//| Compute percentile of cross-frequency values over the baseline   |
//| lookback. Each sample = cross count in a window of crossWindow   |
//| bars ending at successive bars within [bar-lookback .. bar-1].   |
//+------------------------------------------------------------------+
double ComputeCrossFreqPercentile(int bar, int lookback, int crossWindow, double percentile)
{
   int from = bar - lookback;
   int to   = bar - 1;
   if(from < crossWindow - 1)
      from = crossWindow - 1;
   if(to < from)
      return 0.0;

   int sampleCount = to - from + 1;
   if(ArraySize(g_percentileWork) < sampleCount)
      ArrayResize(g_percentileWork, sampleCount);

   for(int j = 0; j < sampleCount; j++)
   {
      int sampleBar = from + j;
      g_percentileWork[j] = (double)CountCrossesInWindow(sampleBar, crossWindow);
   }

   //--- Sort only the filled portion by copying to a temp if needed.
   //--- ArraySort sorts the whole array, but we only read [0..sampleCount-1].
   //--- To avoid stale data in tail slots, resize exactly.
   if(ArraySize(g_percentileWork) > sampleCount)
      ArrayResize(g_percentileWork, sampleCount);

   ArraySort(g_percentileWork);

   int idx = (int)MathRound(percentile / 100.0 * (sampleCount - 1));
   idx = MathMax(0, MathMin(idx, sampleCount - 1));
   return g_percentileWork[idx];
}

   g_labelCfg.SetDefaults();

   g_labelCfg.prefix = "T/E CockedTrigger...";

   g_maxPeriodsBack = MathMax(g_maxPeriodsBack, InpKCPeriod);
   g_maxPeriodsBack = MathMax(g_maxPeriodsBack, InpKCATRPeriod);
   g_maxPeriodsBack += (3 + InpOpenRefOffset);

   return(INIT_SUCCEEDED);

//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   if(rates_total <= g_maxPeriodsBack)
      return(0);

   int start = (prev_calculated == 0) ? g_maxPeriodsBack : prev_calculated - 1;
   if(start < 1)
      start = 1;

   if(prev_calculated == 0)
   {
      ResetAllStateAndBuffers();
      g_lastExportedBar = -1;
      InitCsvDebugFile();
   }

   //--- Ensure dynamic blackout work arrays are sized
   if(ArraySize(g_kcWidthPctArr) < rates_total)
   {
      ArrayResize(g_kcWidthPctArr, rates_total);
      ArrayResize(g_absSlopeArr, rates_total);
      ArrayResize(g_crossFlagArr, rates_total);
      ArrayResize(g_crossPrefixSum, rates_total);
   }
   if(prev_calculated == 0)
   {
      ArrayInitialize(g_kcWidthPctArr, 0.0);
      ArrayInitialize(g_absSlopeArr, 0.0);
      ArrayInitialize(g_crossFlagArr, 0.0);
      ArrayInitialize(g_crossPrefixSum, 0.0);
   }

   int copyStart = MathMax(0, start - 3);
   int copyCount = rates_total - copyStart;

   double temaTail[];
   double emaTail[];
   double kcUpperTail[];
   double kcMiddleTail[];
   double kcLowerTail[];
   ArrayResize(temaTail, copyCount);
   ArrayResize(emaTail, copyCount);
   ArrayResize(kcUpperTail, copyCount);
   ArrayResize(kcMiddleTail, copyCount);
   ArrayResize(kcLowerTail, copyCount);
   ArraySetAsSeries(temaTail, false);
   ArraySetAsSeries(emaTail, false);
   ArraySetAsSeries(kcUpperTail, false);
   ArraySetAsSeries(kcMiddleTail, false);
   ArraySetAsSeries(kcLowerTail, false);

   int temaCopied = CopyBuffer(iHndlTEMA, 0, 0, copyCount, temaTail);
   int emaCopied  = CopyBuffer(iHndlEMAofTEMA, 0, 0, copyCount, emaTail);
   int kcUpperCopied  = CopyBuffer(iHndlKC, 0, 0, copyCount, kcUpperTail);
   int kcMiddleCopied = CopyBuffer(iHndlKC, 1, 0, copyCount, kcMiddleTail);
   int kcLowerCopied  = CopyBuffer(iHndlKC, 2, 0, copyCount, kcLowerTail);
   if(temaCopied != copyCount || emaCopied != copyCount ||
      kcUpperCopied != copyCount || kcMiddleCopied != copyCount || kcLowerCopied != copyCount)
   {
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " CopyBuffer failed. temaCopied=",temaCopied,
            " emaCopied=",emaCopied,
            " kcUpperCopied=",kcUpperCopied,
            " kcMiddleCopied=",kcMiddleCopied,
            " kcLowerCopied=",kcLowerCopied,
            " expected=",copyCount,
            " err=",GetLastError());
      return(prev_calculated);
   }

   ArrayResize(BufferKCUpper, rates_total);
   ArrayResize(BufferKCMiddle, rates_total);
   ArrayResize(BufferKCLower, rates_total);

   for(int k = 0; k < copyCount; k++)
   {
      BufferTEMA[copyStart + k] = temaTail[k];
      BufferEMAofTEMA[copyStart + k] = emaTail[k];
      BufferKCUpper[copyStart + k] = kcUpperTail[k];
      BufferKCMiddle[copyStart + k] = kcMiddleTail[k];
      BufferKCLower[copyStart + k] = kcLowerTail[k];
   }

   double pointSize = MathPow(10.0, -_Digits);
   double eps = pointSize * 0.1;
   double pipSize = pointSize * 10.0;
   bool bDoHistory = (prev_calculated == 0);

   for(int i = start; i < rates_total; i++)
   {
      if(prev_calculated == 0 || i >= prev_calculated)
      {
         BufferArrowBuy[i] = EMPTY_VALUE;
         BufferArrowSell[i] = EMPTY_VALUE;
      }

      int prevRel = RelSign(BufferTEMA[i - 1] - BufferEMAofTEMA[i - 1], eps);
      int currRel = RelSign(BufferTEMA[i] - BufferEMAofTEMA[i], eps);
      double temaEmaSpread = BufferTEMA[i] - BufferEMAofTEMA[i];
      double temaEmaSpreadPips = TS_Price2Pips(temaEmaSpread);
      BufferTemaEmaSpread[i] = temaEmaSpreadPips;
      if(prevRel == 0)
         prevRel = iDir;
      if(currRel == 0)
         currRel = prevRel;

      //--- Cocking logic: evaluate local hi/lo candidates from prior 2 bars.
      //--- Must run before cross handling so same-bar cock+cross can trigger.
      if(iDir != 0)
      {
         bool candidateFound = false;
         bool cockedThisBar = false;

         for(int d = 1; d <= 2; d++)
         {
            int cand = i - d;
            if(cand <= g_crossBar || cand < 1 || cand + 1 >= rates_total)
               continue;

            candidateFound = true;

            int refBar = MathMax(0, cand - InpOpenRefOffset);
            double refOpen = open[refBar];

            //--- Keep debug/state fields aligned with the candidate under evaluation.
            g_extremeBar = cand;
            g_extremeOpenRefBar = refBar;
            g_extremeOpen = refOpen;
            g_extremePrice = (iDir > 0) ? high[cand] : low[cand];

            if((iDir > 0 && close[i] < refOpen) ||
               (iDir < 0 && close[i] > refOpen))
            {
               g_isCocked = true;
               g_cockedDir = (iDir > 0) ? -1 : 1;
               g_cockLevel = refOpen;
               cockedThisBar = true;
               break;
            }
         }

         g_pendingConfirm = (candidateFound && !cockedThisBar);
      }

      bool crossUp = (prevRel < 0 && currRel > 0);
      bool crossDn = (prevRel > 0 && currRel < 0);
      bool isCross = (crossUp || crossDn);

      //=================================================================
      //  DYNAMIC BLACKOUT COMPUTATION
      //=================================================================
      double kcWidthPct = 0.0;
      double slopePipsPerBar = 0.0;
      double absSlope = 0.0;
      double widthScore = 0.0;
      double slopeScore = 0.0;
      double chopScore = 0.0;
      double crossAmplifier = 1.0;

      //--- Compute and store KC width as % of midline
      if(BufferKCUpper[i] != EMPTY_VALUE && BufferKCLower[i] != EMPTY_VALUE &&
         BufferKCMiddle[i] != EMPTY_VALUE && BufferKCMiddle[i] > 0.0)
      {
         kcWidthPct = (BufferKCUpper[i] - BufferKCLower[i]) / BufferKCMiddle[i] * 100.0;
      }
      g_kcWidthPctArr[i] = kcWidthPct;

      //--- Compute and store slope
      if(i >= InpKCMidSlopePeriod &&
         BufferKCMiddle[i] != EMPTY_VALUE &&
         BufferKCMiddle[i - InpKCMidSlopePeriod] != EMPTY_VALUE)
      {
         double rawSlopePrice = (BufferKCMiddle[i] - BufferKCMiddle[i - InpKCMidSlopePeriod]) / InpKCMidSlopePeriod;
         slopePipsPerBar = TS_Price2Pips(rawSlopePrice);

         if(InpSlopeNormMode == SLOPE_KC_NORMALIZED && BufferKCMiddle[i] > 0.0)
            absSlope = MathAbs(rawSlopePrice / BufferKCMiddle[i] * 100.0);
         else
            absSlope = MathAbs(slopePipsPerBar);
      }
      g_absSlopeArr[i] = absSlope;

      //--- Store cross flag and update prefix sum
      g_crossFlagArr[i] = isCross ? 1.0 : 0.0;
      g_crossPrefixSum[i] = (i > 0 ? g_crossPrefixSum[i - 1] : 0.0) + g_crossFlagArr[i];

      //--- Compute active blackout
      int activeBlackout = InpSignalBlackout;

      if(InpUseDynamicBlackout && i >= InpDynBaselineLookback)
      {
         //--- Width score: how contracted is current width vs. baseline?
         double baselineWidthPct = ComputePercentile(g_kcWidthPctArr,
                                                     i - InpDynBaselineLookback,
                                                     i - 1,
                                                     InpDynBaselinePercentile);
         if(baselineWidthPct > 0.0)
            widthScore = MathMax(0.0, 1.0 - (kcWidthPct / baselineWidthPct));

         //--- Slope score: how flat is current slope vs. baseline?
         double baselineSlope = ComputePercentile(g_absSlopeArr,
                                                  i - InpDynBaselineLookback,
                                                  i - 1,
                                                  InpDynBaselinePercentile);
         if(baselineSlope > 0.0)
            slopeScore = MathMax(0.0, 1.0 - (absSlope / baselineSlope));

         //--- Weighted composite chop score
         chopScore = g_normWidthWeight * widthScore + g_normSlopeWeight * slopeScore;

         //--- Cross frequency amplifier
         int currentCrossCount = CountCrossesInWindow(i, InpCrossFreqWindow);
         double baselineCrossFreq = ComputeCrossFreqPercentile(i,
                                                               InpDynBaselineLookback,
                                                               InpCrossFreqWindow,
                                                               InpDynBaselinePercentile);
         if(baselineCrossFreq > 0.0)
            crossAmplifier = MathMax(1.0, (double)currentCrossCount / baselineCrossFreq);

         chopScore *= crossAmplifier;

         //--- Map chop score to extra blackout bars
         double extraBars = 0.0;
         if(InpScoreMapMode == MAP_LINEAR)
            extraBars = chopScore * InpLinearScaleK;
         else
            extraBars = MathPow(chopScore, InpCurveExponent) * InpLinearScaleK;

         activeBlackout = InpSignalBlackout + (int)MathRound(extraBars);
         if(activeBlackout < InpBlackoutMin)
            activeBlackout = InpBlackoutMin;
      }

      BufferDynBlackout[i] = activeBlackout;
      BufferKCWidthRatio[i] = chopScore;  // Repurposed: now shows composite chop score
      BufferTriggerThreshold[i] = InpTriggerSensitivityPips;
      //=================================================================

      if(iDir == 0 && currRel != 0)
         ResetCycleForDirection(currRel, i, open, high, low);

      ENUM_SIGNAL_TYPE signal = SIGNAL_NONE;
      ENUM_SIGNAL_TYPE signalCandidate = SIGNAL_NONE;
      if(isCross)
      {
         if(g_isCocked)
         {
            if(crossDn && g_cockedDir == -1)
            {
               g_triggerPending = true;
               g_triggerDir = -1;
               g_triggerCrossBar = i;
            }
            else if(crossUp && g_cockedDir == 1)
            {
               g_triggerPending = true;
               g_triggerDir = 1;
               g_triggerCrossBar = i;
            }
            else
            {
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
         }
         else
         {
            g_triggerPending = false;
            g_triggerDir = 0;
            g_triggerCrossBar = -1;
         }
      }

      if(g_triggerPending)
      {
         bool wrongSide = ((g_triggerDir == 1 && currRel < 0) || (g_triggerDir == -1 && currRel > 0));
         if(wrongSide)
         {
            g_triggerPending = false;
            g_triggerDir = 0;
            g_triggerCrossBar = -1;
         }
         else
         {
            double cumulativeSpreadPips = 0.0;
            int sumFrom = MathMax(0, g_triggerCrossBar);
            if(sumFrom <= i)
            {
               for(int s = sumFrom; s <= i; s++)
               {
                  if(BufferTemaEmaSpread[s] == EMPTY_VALUE)
                     continue;
                  cumulativeSpreadPips += MathAbs(BufferTemaEmaSpread[s]);
               }
            }

            if(cumulativeSpreadPips >= InpTriggerSensitivityPips)
            {
               signalCandidate = (g_triggerDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
         }
      }

      bool canFireNow = ((i - g_lastSignalBar) > activeBlackout && time[i] >= InpTradeStartDate);
      if(signalCandidate != SIGNAL_NONE)
      {
         if(canFireNow)
         {
            signal = signalCandidate;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else
         {
            g_blackoutSignalPending = true;
            g_blackoutSignalDir = (signalCandidate == SIGNAL_LONG) ? 1 : -1;
            g_blackoutSignalBar = i;
         }
      }

      if(signal == SIGNAL_NONE && g_blackoutSignalPending)
      {
         bool retraced = ((g_blackoutSignalDir == 1 && currRel < 0) || (g_blackoutSignalDir == -1 && currRel > 0));
         if(retraced)
         {
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else if(canFireNow)
         {
            signal = (g_blackoutSignalDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
      }

      if(signal != SIGNAL_NONE)
      {
         if(g_signalStartDT == 0)
            g_signalStartDT = time[i];

         double mockExecPrice = GetMockTradePrice(i, close);

         if(InpEnableTradeSim)
         {
            if(InpMockTradePriceSource == MOCK_PRICE_TEMA)
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferTEMA,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell);
            }
            else if(InpMockTradePriceSource == MOCK_PRICE_EMA_TEMA)
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferEMAofTEMA,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell);
            }
            else
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 close,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell);
            }
         }

         //--- Buffer arrows should be anchored at selected mock execution price for the signal bar.
         if(signal == SIGNAL_LONG)
            BufferArrowBuy[i] = mockExecPrice;
         else
            BufferArrowSell[i] = mockExecPrice;

         //--- Reposition object arrows 100 pips outside the farther of KC outer line or TEMA from KC midline.
         if(InpShowArrows && g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0 &&
            BufferKCUpper[i] != EMPTY_VALUE && BufferKCMiddle[i] != EMPTY_VALUE &&
            BufferKCLower[i] != EMPTY_VALUE && BufferTEMA[i] != EMPTY_VALUE)
         {
            double mid = BufferKCMiddle[i];
            double arrowPrice = 0.0;
            double offset = InpObjectArrowOffsetPips * pipSize;

            if(signal == SIGNAL_SHORT)
            {
               double baseUpper = BufferKCUpper[i];
               double distUpper = MathAbs(baseUpper - mid);
               double distTema  = MathAbs(BufferTEMA[i] - mid);
               double base = (distTema > distUpper) ? BufferTEMA[i] : baseUpper;
               arrowPrice = base + offset;
               BufferArrowSell[i] = mockExecPrice;
            }
            else
            {
               double baseLower = BufferKCLower[i];
               double distLower = MathAbs(baseLower - mid);
               double distTema  = MathAbs(BufferTEMA[i] - mid);
               double base = (distTema > distLower) ? BufferTEMA[i] : baseLower;
               arrowPrice = base - offset;
               BufferArrowBuy[i] = mockExecPrice;
            }

            ObjectMove(0, g_currentTrade.arrowName, 0, time[i], arrowPrice);
         }

         g_lastSignalBar = i;
      }

      //--- Keep the most recent open trade tooltip/buffers live on the current bar.
      if(i == rates_total - 1 && InpEnableTradeSim)
      {
         double liveExitPrice = GetMockTradePrice(i, close);
         UpdateOpenTradeLiveState(i, liveExitPrice);
      }

      //--- Uncock only after trigger decisions on this bar.
      if(g_isCocked)
      {
         if(g_cockedDir == -1 && close[i] > g_cockLevel)
         {
            g_isCocked = false;
            g_cockedDir = 0;
         }
         else if(g_cockedDir == 1 && close[i] < g_cockLevel)
         {

         }
      }

                           kcWidthPct, slopePipsPerBar,
                           activeBlackout,
                           widthScore, slopeScore, chopScore, crossAmplifier,
                           crossUp, crossDn, isCross,
                           signal);
         g_lastExportedBar = i;
      }

      if(isCross && currRel != 0)
      {
         ResetCycleForDirection(currRel, i, open, high, low, g_triggerPending);
         continue;
      }

   }

   return(rates_total);
}
