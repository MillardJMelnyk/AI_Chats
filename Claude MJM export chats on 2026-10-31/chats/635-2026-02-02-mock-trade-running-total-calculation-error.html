//+------------------------------------------------------------------+
//| Parse TextGVar object name to extract InstanceID and ShortName   |
//| Format: MJM_[InstanceID]_[ShortName]_[VarName]                   |
//| Returns true if successfully parsed, false otherwise             |
//+------------------------------------------------------------------+
bool ParseTextGVarObjectName(const string objName, int &instanceID, string &shortName)
{
   //--- Must start with prefix
   if(StringFind(objName, TEXTGVAR_PREFIX + "_") != 0)
      return false;
   
   //--- Find first underscore after prefix (end of InstanceID)
   int prefixLen = StringLen(TEXTGVAR_PREFIX) + 1;  // "MJM_"
   int secondUnderscore = StringFind(objName, "_", prefixLen);
   if(secondUnderscore < 0)
      return false;
   
   //--- Extract InstanceID
   string idStr = StringSubstr(objName, prefixLen, secondUnderscore - prefixLen);
   instanceID = (int)StringToInteger(idStr);
   
   //--- Find last underscore (start of VarName)
   int lastUnderscore = -1;
   for(int i = StringLen(objName) - 1; i >= secondUnderscore; i--)
   {
      if(StringGetCharacter(objName, i) == '_')
      {
         lastUnderscore = i;
         break;
      }
   }
   
   if(lastUnderscore <= secondUnderscore)
      return false;
   
   //--- Extract ShortName (between second underscore and last underscore)
   shortName = StringSubstr(objName, secondUnderscore + 1, lastUnderscore - secondUnderscore - 1);

   return true;
}

//+------------------------------------------------------------------+
//| Find instanceID for a shortName by searching TextGVar objects    |
//| Returns instanceID or -1 if not found                            |
//+------------------------------------------------------------------+
int FindInstanceIDFromTextGVar(const string shortName)
{
   for(int obj = ObjectsTotal(0) - 1; obj >= 0; obj--)
   {
      string objName = ObjectName(0, obj);

      //--- Look for ShortName objects
      if(StringFind(objName, "_ShortName") < 0)
         continue;

      int instanceID;
      string foundShortName;
      if(!ParseTextGVarObjectName(objName, instanceID, foundShortName))
         continue;

      if(foundShortName == shortName)
         return instanceID;
   }
   return -1;  // No TextGVar found for this indicator
}

   int parmCount = (int)StringToInteger(countStr);
   int maxPeriod = 1;

      string parmName = ReadTextGVar(instanceID, shortName, "Parm[" + IntegerToString(p) + "]_Name", "");
      
      // Case-insensitive check for "period"
      string lowerName = parmName;
      StringToLower(lowerName);
      if(StringFind(lowerName, "period") >= 0)

         string parmValue = ReadTextGVar(instanceID, shortName, "Parm[" + IntegerToString(p) + "]_Value", "1");

         if(val > maxPeriod)
            maxPeriod = val;
      }
   }
   return maxPeriod;
}

         if(shortName == sMeShortName)
            continue;

         //--- Check if already discovered (avoid duplicates)
         bool alreadyFound = false;
         for(int d = 0; d < discoveredCount; d++)
         {
            if(discovered[d].shortName == shortName)
            {
               alreadyFound = true;
               break;
            }
         }
         if(alreadyFound)
            continue;

         //--- Try to find instanceID from TextGVar, use -1 if not found
         int instanceID = FindInstanceIDFromTextGVar(shortName);

         //--- Add to discovered list
         ArrayResize(discovered, discoveredCount + 1);
         discovered[discoveredCount].instanceID = instanceID;
         discovered[discoveredCount].shortName = shortName;
         discoveredCount++;
      }
   }
   
   //--- Process each discovered indicator
   for(int d = 0; d < discoveredCount; d++)
   {
      int instanceID = discovered[d].instanceID;
      string shortName = discovered[d].shortName;
      
      //--- Check indicator limit
      if(indicatorCount >= MAX_INDICATORS)
      {
         Print("ERROR: Maximum indicator limit (", MAX_INDICATORS, ") reached. Skipping indicator: ", shortName);
         continue;
      }
      
      //--- Find which window the indicator is in
      int win = FindIndicatorWindow(shortName);
      if(win < 0)
      {
         Print("Indicator ", shortName, " not found in any window - skipping");
         continue;
      }
      
      //--- Get handle
      int handle = ChartIndicatorGet(0, win, shortName);
      if(handle == INVALID_HANDLE)
      {
         Print("Bad handle for indicator<", shortName, "> in window ", win, " - skipping");
         continue;
      }
      
      int plotCount = GetIndicatorPlotCount(instanceID, shortName);
      
      //--- Skip indicators with no plots
      if(plotCount < 1)
      {
         Print("Indicator ", shortName, " has no plots - skipping");
         continue;
      }
      
      //--- Check if adding this indicator would exceed buffer limit
      if(g_totalBuffers + plotCount > MAX_BUFFERS)
      {
         Print("ERROR: Adding indicator ", shortName, " would exceed buffer limit (", MAX_BUFFERS, "). Skipping...");
         continue;
      }
      
      //--- Get max period from TextGVar parameter metadata
      int maxPeriod = GetMaxPeriodFromTextGVar(instanceID, shortName);
      
      
      for(int lineIX = 0; lineIX < plotCount; lineIX++)
      {
         if(g_totalBuffers >= MAX_BUFFERS)
         {
            Print("ERROR: Buffer limit (", MAX_BUFFERS, ") reached while processing ", shortName);
            break;
         }
         
         string lineName = GetPlotDisplayName(instanceID, shortName, lineIX);
         
         //--- Read color/width from TextGVar, fall back to defaults
         color defaultColor = GetAutoColor(g_totalBuffers);
         int defaultWidth = 2;
         
         string colorStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIX) + "]_Color", "");
         color lineColor = (colorStr != "") ? StringToColor(colorStr) : defaultColor;
         
         string widthStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIX) + "]_Width", "");
         int lineWidth = (widthStr != "") ? (int)StringToInteger(widthStr) : defaultWidth;
         int lineStyle = STYLE_SOLID;
         
         int idx = ArraySize(g_SI_line);
         ArrayResize(g_SI_line, idx + 1);
         ArrayResize(g_SI_line_Period, idx + 1);

         g_SI_line[idx].indicatorHandle = handle;
         g_SI_line[idx].indicatorShortName = shortName;
         g_SI_line[idx].instanceID = instanceID;
         g_SI_line[idx].indicatorWindow = win;
         g_SI_line[idx].lineIndex = lineIX;
         g_SI_line[idx].bufferIndex = g_totalBuffers;
         g_SI_line_Period[idx] = maxPeriod;
         g_SI_line[idx].lineColor = lineColor;
         g_SI_line[idx].lineStyle = lineStyle;
         g_SI_line[idx].lineWidth = lineWidth;
         g_SI_line[idx].lineName = lineName;
         g_SI_line[idx].lastRatesTotal = 0;
         g_SI_line[idx].oldestCalcBarIX = 0;
         g_SI_line[idx].isValid = true;
         g_SI_line[idx].ReinitIt = false;
         g_SI_line[idx].cachedColorStr = colorStr;
         g_SI_line[idx].cachedWidthStr = widthStr;
         
         
         SetBufferByIndex(g_totalBuffers);
         PlotIndexSetInteger(g_totalBuffers, PLOT_DRAW_TYPE, DRAW_LINE);
         PlotIndexSetInteger(g_totalBuffers, PLOT_LINE_STYLE, lineStyle);
         PlotIndexSetInteger(g_totalBuffers, PLOT_LINE_WIDTH, lineWidth);
         PlotIndexSetInteger(g_totalBuffers, PLOT_LINE_COLOR, lineColor);
         PlotIndexSetString(g_totalBuffers, PLOT_LABEL, shortName + " " + lineName + " Slope");
         PlotIndexSetInteger(g_totalBuffers, PLOT_SHOW_DATA, true);

         g_totalBuffers++;
      }
      
      indicatorCount++;
   }
   
   //--- Configure summary buffers if we have active line buffers
   if(g_totalBuffers > 0)
   {
      //--- Buffer30: Slopes Avg
      SetBufferByIndex(SUMMARY_AVG_BUFFER);
      PlotIndexSetInteger(SUMMARY_AVG_BUFFER, PLOT_DRAW_TYPE, DRAW_LINE);
      PlotIndexSetInteger(SUMMARY_AVG_BUFFER, PLOT_LINE_STYLE, STYLE_SOLID);
      PlotIndexSetInteger(SUMMARY_AVG_BUFFER, PLOT_LINE_WIDTH, InpSummaryAvgWidth);
      PlotIndexSetInteger(SUMMARY_AVG_BUFFER, PLOT_LINE_COLOR, InpSummaryAvgColor);
      PlotIndexSetString(SUMMARY_AVG_BUFFER, PLOT_LABEL, "Slopes Avg");
      PlotIndexSetInteger(SUMMARY_AVG_BUFFER, PLOT_SHOW_DATA, true);
      
      //--- Buffer31: Slope % UP/DN
      SetBufferByIndex(SUMMARY_PCT_BUFFER);
      PlotIndexSetInteger(SUMMARY_PCT_BUFFER, PLOT_DRAW_TYPE, DRAW_LINE);
      PlotIndexSetInteger(SUMMARY_PCT_BUFFER, PLOT_LINE_STYLE, STYLE_SOLID);
      PlotIndexSetInteger(SUMMARY_PCT_BUFFER, PLOT_LINE_WIDTH, InpSummaryPctWidth);
      PlotIndexSetInteger(SUMMARY_PCT_BUFFER, PLOT_LINE_COLOR, InpSummaryPctColor);
      PlotIndexSetString(SUMMARY_PCT_BUFFER, PLOT_LABEL, "Slope % UP/DN");
      PlotIndexSetInteger(SUMMARY_PCT_BUFFER, PLOT_SHOW_DATA, true);
   }

   //--- Setup mock trade P/L buffers (no plot, data window only)
   SetIndexBuffer(32, BufferCurrentTradePL, INDICATOR_DATA);
   PlotIndexSetInteger(32, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(32, PLOT_LABEL, "Trade P/L");
   PlotIndexSetInteger(32, PLOT_SHOW_DATA, true);

   SetIndexBuffer(33, BufferRunningPL, INDICATOR_DATA);
   PlotIndexSetInteger(33, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(33, PLOT_LABEL, "Running P/L");
   PlotIndexSetInteger(33, PLOT_SHOW_DATA, true);

   //--- Setup trade statistics buffers (no plot, data window only)
   SetIndexBuffer(34, BufferTradeCount, INDICATOR_DATA);
   PlotIndexSetInteger(34, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(34, PLOT_LABEL, "Trades");
   PlotIndexSetInteger(34, PLOT_SHOW_DATA, true);

   SetIndexBuffer(35, BufferWinPct, INDICATOR_DATA);
   PlotIndexSetInteger(35, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(35, PLOT_LABEL, "Win %");
   PlotIndexSetInteger(35, PLOT_SHOW_DATA, true);

   SetIndexBuffer(36, BufferShortCount, INDICATOR_DATA);
   PlotIndexSetInteger(36, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(36, PLOT_LABEL, "Shorts");
   PlotIndexSetInteger(36, PLOT_SHOW_DATA, true);

   SetIndexBuffer(37, BufferShortWinPct, INDICATOR_DATA);
   PlotIndexSetInteger(37, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(37, PLOT_LABEL, "Short Win %");
   PlotIndexSetInteger(37, PLOT_SHOW_DATA, true);

   SetIndexBuffer(38, BufferLongCount, INDICATOR_DATA);
   PlotIndexSetInteger(38, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(38, PLOT_LABEL, "Longs");
   PlotIndexSetInteger(38, PLOT_SHOW_DATA, true);

   SetIndexBuffer(39, BufferLongWinPct, INDICATOR_DATA);
   PlotIndexSetInteger(39, PLOT_DRAW_TYPE, DRAW_NONE);
   PlotIndexSetString(39, PLOT_LABEL, "Long Win %");
   PlotIndexSetInteger(39, PLOT_SHOW_DATA, true);


   //--- Initialize ATR handle for ScaledSlope function
   //--- Do this here in OnInit so ATR has time to calculate before OnCalculate runs
   int atrPeriod = GetTrimmedMeanPeriod(g_SI_line_Period);
   if(atrPeriod > 0)
      InitATRHandle(atrPeriod, _Symbol, _Period);

}

//+------------------------------------------------------------------+
//| Check for REINIT signals from TextGVar objects                   |
//| Returns ReinitInfo with old instanceID and shortName if found    |
//+------------------------------------------------------------------+
ReinitInfo CheckReinitSignals()
{
   ReinitInfo result;
   result.found = false;
   result.oldInstanceID = -1;
   result.oldShortName = "";

   string checkedNames[];
   int checkedCount = 0;
   
   
   for(int i = 0; i < ArraySize(g_SI_line); i++)
   {
   
      string shortName = g_SI_line[i].indicatorShortName;
      int instanceID = g_SI_line[i].instanceID;
      
      
      //--- Check if we've already processed this shortName
      bool alreadyChecked = false;
      for(int j = 0; j < checkedCount; j++)
      {
         if(checkedNames[j] == shortName)
         {
            alreadyChecked = true;
            break;
         }
      }
      
      if(alreadyChecked)
      {
         continue;
      }
      
      ArrayResize(checkedNames, checkedCount + 1);
      checkedNames[checkedCount] = shortName;
      checkedCount++;
      
      //--- Check for REINIT signal
      string expectedObjName = TEXTGVAR_PREFIX + "_" + IntegerToString(instanceID) + "_" + shortName + "_" + TEXTGVAR_REINIT;

   int iFoundInSubwindow = ObjectFind(0, expectedObjName);
      
      string reinitSignal = ReadTextGVar(instanceID, shortName, TEXTGVAR_REINIT, "");
      
      if(reinitSignal == "")
         continue;
      
      //--- Delete the REINIT flag
      TextGVar_Delete(TEXTGVAR_PREFIX, instanceID, shortName, TEXTGVAR_REINIT);
      
      result.found = true;
      result.oldInstanceID = instanceID;
      result.oldShortName = shortName;
      return result;
   }
   
   return result;
}

//+------------------------------------------------------------------+
//| REINIT Step A: Find matching lines and release handles           |
//| Returns count of matching lines found                             |
//+------------------------------------------------------------------+
int Reinit_FindMatchingLinesAndReleaseHandles(int oldInstanceID, const string &oldShortName, int &matchingLines[])
{
   ArrayResize(matchingLines, 0);

   int handlesToRelease[];
   ArrayResize(handlesToRelease, 0);
   int size = 0;
   
   for(int i = 0; i < ArraySize(g_SI_line); i++)
   {
      if(g_SI_line[i].instanceID == oldInstanceID && g_SI_line[i].indicatorShortName == oldShortName)
      {
         size = ArraySize(matchingLines);
         ArrayResize(matchingLines, size + 1);
         matchingLines[size] = i;

         int h = g_SI_line[i].indicatorHandle;
         if(h != INVALID_HANDLE)
         {
            bool alreadyQueued = false;
            for(int j = 0; j < ArraySize(handlesToRelease); j++)
            {
               if(handlesToRelease[j] == h)
               {
                  alreadyQueued = true;
                  break;
               }
            }
            if(!alreadyQueued)
            {
               int n = ArraySize(handlesToRelease);
               ArrayResize(handlesToRelease, n + 1);
               handlesToRelease[n] = h;
            }
         }
      }
   }

   //--- Release each handle once (handles are shared across plot-lines)
   for(int j = 0; j < ArraySize(handlesToRelease); j++)
   {
      IndicatorRelease(handlesToRelease[j]);
   }

   //--- Invalidate all matching line entries so no line keeps a stale handle
   for(int m = 0; m < ArraySize(matchingLines); m++)
   {
      int lineIX = matchingLines[m];
      g_SI_line[lineIX].indicatorHandle = INVALID_HANDLE;
   }
   
   size = ArraySize(matchingLines);

   return size;
}

//+------------------------------------------------------------------+
//| REINIT Step C: Search for new SI metadata                        |
//| Returns true if new metadata found                                |
//+------------------------------------------------------------------+
bool Reinit_FindNewMetadata(int oldInstanceID, const string &indyPattern, int &newInstanceID, string &newShortName)
{
   newInstanceID = -1;
   newShortName = "";

   for(int obj = ObjectsTotal(0) - 1; obj >= 0; obj--)
   {
      string objName = ObjectName(0, obj);

      if(StringFind(objName, "_ShortName") < 0)
         continue;

      int parsedInstanceID;
      string parsedShortName;
      if(!ParseTextGVarObjectName(objName, parsedInstanceID, parsedShortName))
         continue;

      if(StringFind(parsedShortName, indyPattern) != 0)
         continue;

      if(parsedInstanceID == oldInstanceID)
         continue;

      newInstanceID = parsedInstanceID;
      newShortName = parsedShortName;
      return true;
   }

   return false;
}

//+------------------------------------------------------------------+
//| REINIT Step D: Handle SI removal (clear buffers, invalidate)     |
//+------------------------------------------------------------------+
void Reinit_HandleRemoval(const int &matchingLines[])
{
   for(int m = 0; m < ArraySize(matchingLines); m++)
   {
      int lineIX = matchingLines[m];

      // Fully invalidate this line so it will never be used again
      g_SI_line[lineIX].indicatorHandle = INVALID_HANDLE;
      g_SI_line[lineIX].isValid = false;

      int bufIdx = g_SI_line[lineIX].bufferIndex;
      ClearBufferByIndex(bufIdx);
      PlotIndexSetInteger(bufIdx, PLOT_DRAW_TYPE, DRAW_NONE);
      PlotIndexSetString(bufIdx, PLOT_LABEL, "");
      PlotIndexSetInteger(bufIdx, PLOT_SHOW_DATA, false);
   }

   // Reset signal detection and mock trades so they recalculate with remaining indicators
   ResetSignalState();
}

//+------------------------------------------------------------------+
//| REINIT Step E: Update lines with new SI metadata                 |
//+------------------------------------------------------------------+
void Reinit_ApplyUpdates(const int &matchingLines[], int newInstanceID, const string &newShortName,
                         int newHandle, int win, int newPeriod, bool structuralChange)
{

   for(int m = 0; m < ArraySize(matchingLines); m++)
   {
      int lineIX    = matchingLines[m];
      int plotIndex = g_SI_line[lineIX].lineIndex;


      string newColorStr = ReadTextGVar(newInstanceID, newShortName, "Plot[" + IntegerToString(plotIndex) + "]_Color", "");
      string newWidthStr = ReadTextGVar(newInstanceID, newShortName, "Plot[" + IntegerToString(plotIndex) + "]_Width", "");


      // Always update identity/handle/window so subsequent copies use the new SI instance
      g_SI_line[lineIX].instanceID         = newInstanceID;
      g_SI_line[lineIX].indicatorShortName = newShortName;
      g_SI_line[lineIX].indicatorHandle    = newHandle;
      g_SI_line[lineIX].indicatorWindow    = win;
      g_SI_line[lineIX].isValid            = true;

      if(structuralChange)
      {
         // Period/parameters changed -> force full recalculation for this line
         g_SI_line_Period[lineIX]          = newPeriod;
         g_SI_line[lineIX].lastRatesTotal  = 0;
         g_SI_line[lineIX].oldestCalcBarIX = 0;
         g_SI_line[lineIX].ReinitIt        = true;
      }

      int bufIdx = g_SI_line[lineIX].bufferIndex;

      // Apply new color/width immediately if provided, and update caches
      if(newColorStr != "")
      {
         color newColor = StringToColor(newColorStr);
         g_SI_line[lineIX].lineColor      = newColor;
         g_SI_line[lineIX].cachedColorStr = newColorStr;
         PlotIndexSetInteger(bufIdx, PLOT_LINE_COLOR, newColor);
      }
      else
      {
         // Keep existing color, but make sure cache matches TextGVar state
         g_SI_line[lineIX].cachedColorStr = newColorStr;
      }

      if(newWidthStr != "")
      {
         int newWidth = (int)StringToInteger(newWidthStr);
         g_SI_line[lineIX].lineWidth      = newWidth;
         g_SI_line[lineIX].cachedWidthStr = newWidthStr;
         PlotIndexSetInteger(bufIdx, PLOT_LINE_WIDTH, newWidth);
      }
      else
      {
         g_SI_line[lineIX].cachedWidthStr = newWidthStr;
      }

      string newLabel = newShortName + " " + g_SI_line[lineIX].lineName + " Slope";
      PlotIndexSetString(bufIdx, PLOT_LABEL, newLabel);

      // Only when the indicator's data parameters changed do we clear the
      // slope buffers and force a full recalculation. Pure color/width
      // changes must NOT trigger data recalculation.
      if(structuralChange)
      {
         ClearBufferByIndex(bufIdx);
      }
   }

   // If structural change occurred, reset signal detection and mock trades
   if(structuralChange)
      ResetSignalState();

}

//+------------------------------------------------------------------+
//| Reinitialize a single indicator after REINIT signal              |
//+------------------------------------------------------------------+
void ReinitSingleIndicator(int oldInstanceID, const string oldShortName)
{

   //===============================================================
   // STEP A: Find ALL g_SI_line entries using old identity
   //===============================================================
   int matchingLines[];
   int matchCount = Reinit_FindMatchingLinesAndReleaseHandles(oldInstanceID, oldShortName, matchingLines);

   if(matchCount == 0)
   {
      Print("ChartID<",ChartID(),">: ["+__FILE__+"] ["+__FUNCTION__+"] LINE "+(string)__LINE__+" MATCHCOUNT CHECK: No matching g_SI_line entry found, skipping");
      return;
   }
   
   //===============================================================
   // STEP B: Extract g_IndyName (use first match for this)
   //===============================================================
   string indyName = ExtractIndyName(oldShortName);
   string indyPattern = indyName + "(";

   //===============================================================
   // STEP C: Search for new SI metadata
   //===============================================================
   int newInstanceID;
   string newShortName;
   bool foundNewMetadata = Reinit_FindNewMetadata(oldInstanceID, indyPattern, newInstanceID, newShortName);


   //===============================================================
   // STEP D: If SI removed (no new metadata found)
   //===============================================================
   if(!foundNewMetadata)
   {
      Reinit_HandleRemoval(matchingLines);
      return;
   }

   //===============================================================
   // STEP E: SI parameters changed - update ALL matching lines
   //===============================================================
   int win = FindIndicatorWindow(newShortName);

   int newHandle = ChartIndicatorGet(0, win, newShortName);

   if(newHandle == INVALID_HANDLE)
   {
      Print("ERROR: ChartIndicatorGet returned INVALID_HANDLE for newShortName=", newShortName, " win=", win);
      return;
   }

   int newPeriod = GetMaxPeriodFromTextGVar(newInstanceID, newShortName);

   // Determine whether this REINIT reflects a true data/parameter change
   // (e.g. period or other indicator settings) versus a pure visual change
   int  refLineIX        = matchingLines[0];
   int  oldPeriod        = g_SI_line_Period[refLineIX];
   bool structuralChange = (newPeriod != oldPeriod) || (newShortName != oldShortName);

   Reinit_ApplyUpdates(matchingLines, newInstanceID, newShortName, newHandle, win, newPeriod, structuralChange);


   //--- Immediately recalculate slopes for reinited lines (don't wait for OnCalculate)
   RecalculateAfterReinit(matchingLines);

}

//+------------------------------------------------------------------+
//| Timer function - detect indicator changes                        |
//+------------------------------------------------------------------+
void OnTimer()
{
   //--- Check for line property changes from TextGVar
   CheckLinePropertyChanges();
   
   //--- Check for REINIT signals
   ReinitInfo reinitInfo = CheckReinitSignals();


   if(reinitInfo.found)
   {
      ReinitSingleIndicator(reinitInfo.oldInstanceID, reinitInfo.oldShortName);
      return;
   }

   //--- Check if indicator count changed (add or remove)
   int currentCount = GetTotalIndicatorCount();
   if(currentCount != g_lastIndicatorCount)
   {
      FullRescanAndRebuild();
      return;
   }
}


//+------------------------------------------------------------------+
//| Check and apply TextGVar property changes for all lines          |
//+------------------------------------------------------------------+
void CheckLinePropertyChanges()
{
   for(int i = 0; i < ArraySize(g_SI_line); i++)
   {
      if(!g_SI_line[i].isValid)
         continue;
      
      string shortName = g_SI_line[i].indicatorShortName;
      int instanceID = g_SI_line[i].instanceID;
      int lineIX = g_SI_line[i].lineIndex;
      
      //--- Read current values from TextGVar
      string currentColorStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIX) + "]_Color", "");
      string currentWidthStr = ReadTextGVar(instanceID, shortName, "Plot[" + IntegerToString(lineIX) + "]_Width", "");
      
      bool changed = false;
      
      //--- Check color change
      if(currentColorStr != "" && currentColorStr != g_SI_line[i].cachedColorStr)
      {
         color newColor = StringToColor(currentColorStr);
         g_SI_line[i].lineColor = newColor;
         g_SI_line[i].cachedColorStr = currentColorStr;
         PlotIndexSetInteger(g_SI_line[i].bufferIndex, PLOT_LINE_COLOR, newColor);
         changed = true;
      }
      
      //--- Check width change
      if(currentWidthStr != "" && currentWidthStr != g_SI_line[i].cachedWidthStr)
      {
         int newWidth = (int)StringToInteger(currentWidthStr);
         g_SI_line[i].lineWidth = newWidth;
         g_SI_line[i].cachedWidthStr = currentWidthStr;
         PlotIndexSetInteger(g_SI_line[i].bufferIndex, PLOT_LINE_WIDTH, newWidth);
         changed = true;
      }
      
      if(changed)
         ChartRedraw(0);
   }
}

   }

}

//+------------------------------------------------------------------+
//| Release all stored indicator handles (dedup by handle)           |
//| NOTE: Multiple g_SI_line entries can share the same handle       |
//+------------------------------------------------------------------+
void ReleaseStoredIndicatorHandles()
{
   int releasedHandles[];
   ArrayResize(releasedHandles, 0);

   for(int i = 0; i < ArraySize(g_SI_line); i++)
   {
      int handle = g_SI_line[i].indicatorHandle;
      if(handle == INVALID_HANDLE)
         continue;

      bool alreadyReleased = false;
      for(int j = 0; j < ArraySize(releasedHandles); j++)
      {
         if(releasedHandles[j] == handle)
         {
            alreadyReleased = true;
            break;
         }
      }

      if(!alreadyReleased)
      {
         IndicatorRelease(handle);
         int n = ArraySize(releasedHandles);
         ArrayResize(releasedHandles, n + 1);
         releasedHandles[n] = handle;
      }

      // Mark invalid regardless (handles are owned per-indicator-instance, not per-plot-line)
      g_SI_line[i].indicatorHandle = INVALID_HANDLE;
   }
}

//+------------------------------------------------------------------+

{

   ObjectsDeleteAll(0, "Signal_");

   g_historyProcessed = false;
   g_signalDirection = 0;

   // Reset mock trade state
   g_tradeCount = 0;
   g_entryPrice = 0;
   g_entryBar = 0;
   g_runningPL = 0;

   // Reset trade statistics
   g_shortCount = 0;
   g_longCount = 0;
   g_shortWins = 0;
   g_longWins = 0;

   // Clear mock trade P/L buffers
   ArrayInitialize(BufferCurrentTradePL, EMPTY_VALUE);
   ArrayInitialize(BufferRunningPL, EMPTY_VALUE);

   // Clear trade statistics buffers
   ArrayInitialize(BufferTradeCount, EMPTY_VALUE);
   ArrayInitialize(BufferWinPct, EMPTY_VALUE);

   ArrayInitialize(BufferShortWinPct, EMPTY_VALUE);

   ArrayInitialize(BufferLongWinPct, EMPTY_VALUE);
}

//+------------------------------------------------------------------+
//| Full rescan helper                                               |
//+------------------------------------------------------------------+
void FullRescanAndRebuild()
{
   ReleaseStoredIndicatorHandles();
   ArrayResize(g_SI_line, 0);
   ArrayResize(g_SI_line_Period, 0);

   ScanChartIndicators();
   g_lastIndicatorCount = GetTotalIndicatorCount();

   // Reset signal detection and mock trade state so they get recalculated
   ResetSignalState();

   // Force full recalculation on next OnCalculate
   ChartSetSymbolPeriod(0, _Symbol, _Period);

   // Force chart redraw to apply plot property changes
   ChartRedraw(0);
}

//| Calculate slopes for a single line                                |
//| Called from OnCalculate loop and from ReinitSingleIndicator       |
//| Returns the earliest bar calculated (for summary buffer update)   |
//+------------------------------------------------------------------+
int CalculateSlopesForLine(int lineIX, int rates_total, int prev_calculated,
                            const double &close[], bool forceFullRecalc)
{
   //--- Guard: need enough bars for indicator period + slope calc
   int minBarsRequired = g_SI_line_Period[lineIX] + InpSlopePeriod;

   if(rates_total < minBarsRequired)
      return rates_total;  // No bars calculated

   int oldestCalcBar = (prev_calculated > 0) ? prev_calculated - 1 : minBarsRequired;

   if(!g_SI_line[lineIX].isValid)
      return rates_total;  // No bars calculated

   int lastRates = g_SI_line[lineIX].lastRatesTotal;
   int newBars = rates_total - lastRates;

   //--- First call or history changed: full copy
   if(lastRates == 0 || newBars < 0 || newBars > rates_total || forceFullRecalc == true)
   {
      ArrayResize(g_SI_line[lineIX].cache, rates_total);
      ArraySetAsSeries(g_SI_line[lineIX].cache, false);

      //--- Get cached handle for this line
      int handle = g_SI_line[lineIX].indicatorHandle;
      if(handle == INVALID_HANDLE)
         return rates_total;  // No bars calculated

      int copyResult = CopyBuffer(handle,
                                  g_SI_line[lineIX].lineIndex,
                                  0,
                                  rates_total,
                                  g_SI_line[lineIX].cache);

      if(copyResult <= 0)
         return rates_total;  // No bars calculated

      //--- Accept what we got
      g_SI_line[lineIX].lastRatesTotal = rates_total;
      oldestCalcBar = minBarsRequired;
   }
   else
   {
      //--- Incremental update: copy new bars + bar 0 update
      int barsToCopy = newBars + 1;
      if(barsToCopy > rates_total)
         barsToCopy = rates_total;

      //--- Resize cache if new bars arrived
      if(rates_total > ArraySize(g_SI_line[lineIX].cache))
      {
         ArrayResize(g_SI_line[lineIX].cache, rates_total);
         ArraySetAsSeries(g_SI_line[lineIX].cache, false);
      }

      double tempBuffer[];
      ArraySetAsSeries(tempBuffer, false);

      //--- Use stored handle
      int handle = g_SI_line[lineIX].indicatorHandle;
      if(handle == INVALID_HANDLE)
         return rates_total;  // No bars calculated

      int copyResult = CopyBuffer(handle,
                                  g_SI_line[lineIX].lineIndex,
                                  0,
                                  barsToCopy,
                                  tempBuffer);

      if(copyResult <= 0)
         return rates_total;  // No bars calculated

      //--- Update cache with new values
      int cacheStartIdx = rates_total - copyResult;
      for(int i = 0; i < copyResult; i++)
      {
         g_SI_line[lineIX].cache[cacheStartIdx + i] = tempBuffer[i];
      }
      g_SI_line[lineIX].lastRatesTotal = rates_total;
   }

   //--- Calculate slopes
   g_SI_line[lineIX].oldestCalcBarIX = oldestCalcBar;

   int startBar = g_SI_line[lineIX].oldestCalcBarIX;

   //--- Loop from older bars to newer
   for(int CalcBarIX = startBar; CalcBarIX < rates_total; CalcBarIX++)
   {
      int PrevBarIX = CalcBarIX - InpSlopePeriod;

      double currentValue = g_SI_line[lineIX].cache[CalcBarIX];
      double prevValue    = g_SI_line[lineIX].cache[PrevBarIX];

      if(currentValue == EMPTY_VALUE || prevValue == EMPTY_VALUE)
         continue;

      //--- Compute ATR-adjusted scaled slope
      double outSlopeVLU = ScaledSlope(prevValue, currentValue, close, g_SI_line[lineIX].cache,
                                       g_SI_line_Period[lineIX], CalcBarIX, rates_total, g_pipSize,
                                       g_SI_line[lineIX].indicatorShortName, _Symbol, _Period);

      //--- Write to buffer
      WriteBufferValue(g_SI_line[lineIX].bufferIndex, CalcBarIX, outSlopeVLU);
   }

   g_SI_line[lineIX].ReinitIt = false;

   return startBar;  // Return earliest bar calculated
}

//+------------------------------------------------------------------+
//| Calculate summary buffers (avg slope, % up/dn)                    |
//+------------------------------------------------------------------+
void CalculateSummaryBuffers(int startBar, int rates_total)
{
   for(int bar = startBar; bar < rates_total; bar++)
   {
      double sum = 0.0;
      int positiveCount = 0;
      int negativeCount = 0;
      int validCount = 0;

      //--- Read from all active line buffers
      for(int i = 0; i < g_totalBuffers; i++)
      {
         double val = GetBufferValue(i, bar);
         if(val == EMPTY_VALUE)
            continue;
         sum += val;
         if(val > 0)
            positiveCount++;
         else if(val < 0)
            negativeCount++;
         validCount++;
      }

      if(validCount > 0)
      {
         double avg = sum / validCount;

         //--- Mean average, all slopes
         Buffer30[bar] = avg;

         //--- Calculate percentage UP (positive) vs DN (negative)
         double pctUp = (double)positiveCount / validCount * 100.0;
         double pctDn = (double)negativeCount / validCount * 100.0;

         //--- Store signed value: positive if majority up, negative if majority down
         if(positiveCount > negativeCount)
            Buffer31[bar] = pctUp;
         else if(negativeCount > positiveCount)
            Buffer31[bar] = -pctDn;
         else
            Buffer31[bar] = 0.0;

         Buffer31[bar] = 0.0; // DEBUG ONLY REMOVE FOR PRODUCTION
      }
      else
      {
         Buffer30[bar] = EMPTY_VALUE;
         Buffer31[bar] = EMPTY_VALUE;
      }
   }
}

//| Handle mock trade P/L tracking

void HandleSignalEvent(int bar, ENUM_SIGNAL_TYPE signal, int prevDirection, 
                       const datetime &time[], const double &close[])
{
   double exitPrice = close[bar];
   double tradePL = 0;
   
   // Calculate P/L for closing trade (if there was one)
   if(g_tradeCount > 0)
   {
      tradePL = Price2Pips(exitPrice - g_entryPrice) * prevDirection;

      bool wasLong = (prevDirection == 1);
      bool wasShort = (prevDirection == -1);
      bool isWin = (tradePL > 0);

      if(wasLong)
      {
         g_longCount++;
         if(isWin) g_longWins++;
      }
      else if(wasShort)
      {
         g_shortCount++;
         if(isWin) g_shortWins++;
      }

      int totalTrades = g_shortCount + g_longCount;
      int totalWins = g_shortWins + g_longWins;
      double winPct = (totalTrades > 0) ? (totalWins * 100.0) / totalTrades : 0;
      double shortWinPct = (g_shortCount > 0) ? (g_shortWins * 100.0) / g_shortCount : 0;
      double longWinPct = (g_longCount > 0) ? (g_longWins * 100.0) / g_longCount : 0;

      for(int i = g_entryBar + 1; i <= bar; i++)
      {
         BufferCurrentTradePL[i] = tradePL;
         BufferRunningPL[i] = g_runningPL;
         BufferTradeCount[i] = totalTrades;
         BufferWinPct[i] = winPct;
         BufferShortCount[i] = g_shortCount;
         BufferShortWinPct[i] = shortWinPct;
         BufferLongCount[i] = g_longCount;
         BufferLongWinPct[i] = longWinPct;
      }

      g_runningPL += tradePL;
   }

   // Create arrow with trade info in tooltip
   ENUM_ARROW_CODE arrow;
   color clr;
   ENUM_ARROW_ANCHOR anchor;
   string signalName;
   int newDirection;

   if(signal == SIGNAL_SHORT)
   {
      arrow = ARROW_TRADE_ENTRY_SHORT;
      clr = clrYellow;
      anchor = ANCHOR_BOTTOM;
      signalName = "Signal_Short_";
      newDirection = -1;
   }
   else // SIGNAL_LONG
   {
      arrow = ARROW_TRADE_ENTRY_LONG;
      clr = clrLime;
      anchor = ANCHOR_TOP;
      signalName = "Signal_Long_";
      newDirection = 1;
   }

   // Build tooltip with trade info
   string tooltip = signalName;
   if(g_tradeCount > 0)
   {
      tooltip = "Entry: " + DoubleToString(g_entryPrice, _Digits) +
                " | Exit: " + DoubleToString(exitPrice, _Digits) +
                " | P/L: " + DoubleToString(tradePL, 1) + " pips" +
                " | Running: " + DoubleToString(g_runningPL, 1) + " pips";
   }

   SArrowConfig config;
   config.arrowCode = arrow;
   config.clr = clr;
   config.width = 2;
   config.anchor = anchor;
   config.priceOffset = 90;
   config.direction = prevDirection;

   CreateArrow(signalName + IntegerToString(bar), time[bar], close[bar], config, tooltip);

   // Set up new trade
   g_entryPrice = exitPrice;
   g_entryBar = bar;
   g_tradeCount++;
   g_signalDirection = newDirection;
}

void HandleSignalEvent(int bar, ENUM_SIGNAL_TYPE signal, int prevDirection, 
                       const datetime &time[], const double &close[])
{
   double exitPrice = close[bar];

   // Calculate P/L for closing trade (if there was one)
   if(g_tradeCount > 0)
   {
      tradePL = Price2Pips(exitPrice - g_entryPrice) * prevDirection;

      bool wasLong = (prevDirection == 1);
      bool wasShort = (prevDirection == -1);
      bool isWin = (tradePL > 0);

      if(wasLong)
      {
         g_longCount++;
         if(isWin) g_longWins++;
      }
      else if(wasShort)
      {
         g_shortCount++;
         if(isWin) g_shortWins++;
      }

      int totalTrades = g_shortCount + g_longCount;
      int totalWins = g_shortWins + g_longWins;
      double winPct = (totalTrades > 0) ? (totalWins * 100.0) / totalTrades : 0;
      double shortWinPct = (g_shortCount > 0) ? (g_shortWins * 100.0) / g_shortCount : 0;
      double longWinPct = (g_longCount > 0) ? (g_longWins * 100.0) / g_longCount : 0;

      g_runningPL += tradePL;

      for(int i = g_entryBar; i <= bar - 1; i++)
      {
         BufferCurrentTradePL[i] = tradePL;
         BufferRunningPL[i] = g_runningPL;
         BufferTradeCount[i] = totalTrades;
         BufferWinPct[i] = winPct;
         BufferShortCount[i] = g_shortCount;
         BufferShortWinPct[i] = shortWinPct;
         BufferLongCount[i] = g_longCount;
         BufferLongWinPct[i] = longWinPct;
      }

      // Update PREVIOUS arrow with closed trade results
      if(g_lastArrowName != "")
      {
         string tooltip = "Entry: " + DoubleToString(g_entryPrice, _Digits) +
                          " | Exit: " + DoubleToString(exitPrice, _Digits) +
                          " | P/L: " + DoubleToString(tradePL, 1) + " pips" +
                          " | Running: " + DoubleToString(g_runningPL, 1) + " pips";
         ObjectSetString(0, g_lastArrowName, OBJPROP_TOOLTIP, tooltip);
      }
   }

   // Create new arrow for new trade entry
   ENUM_ARROW_CODE arrow;
   color clr;
   ENUM_ARROW_ANCHOR anchor;
   string signalName;
   int newDirection;

   if(signal == SIGNAL_SHORT)
   {
      arrow = ARROW_TRADE_ENTRY_SHORT;
