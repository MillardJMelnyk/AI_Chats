//| Find the fingerprint with highest occurrences in the library     |
//+------------------------------------------------------------------+
string GetTopFingerprint()
{
   string top_fp = "";
   int max_occ = -1;
   int total = g_L3Library.GetPatternCount();
   
   if(total <= 0) return "";

   for(int i = 0; i < total; i++)
   {
      string current_fp = g_L3Library.GetFingerprintByIndex(i);
      
      // Ignore empty or invalid default fingerprints
      if(current_fp == "" || current_fp == "0") continue;
      
      int current_occ = g_L3Library.GetOccurrences(current_fp);
      
      if(current_occ > max_occ)
      {
         max_occ = current_occ;
         top_fp = current_fp;
      }
   }
   return top_fp;
}

double CPatternLibrary::CalculateDTW(const string &seq1, const string &seq2)
  {
   int n = StringLen(seq1);
   int m = StringLen(seq2);
   
   if(n == 0 || m == 0)
      return DBL_MAX;
   
   double dtw[];
   ArrayResize(dtw, (n + 1) * (m + 1));
   
   for(int i = 0; i <= n; i++)
      for(int j = 0; j <= m; j++)
         dtw[i * (m + 1) + j] = DBL_MAX;
   
   dtw[0] = 0;
   
   for(int i = 1; i <= n; i++)
     {
      for(int j = 1; j <= m; j++)
        {
         double cost = SaxCharDistance((char)StringGetCharacter(seq1, i - 1),
                                       (char)StringGetCharacter(seq2, j - 1));
         
         double prev_min = MathMin(dtw[(i - 1) * (m + 1) + j],
                           MathMin(dtw[i * (m + 1) + (j - 1)],
                                   dtw[(i - 1) * (m + 1) + (j - 1)]));
         
         dtw[i * (m + 1) + j] = cost + prev_min;
        }
     }
   
   return dtw[n * (m + 1) + m];
  }

void CPatternLibrary::UpdateExistingPattern(int idx, ENUM_PATTERN_OUTCOME outcome, double pips)
  {
   if(idx < 0 || idx >= m_pattern_count) return;
   
   switch(outcome)
     {
      case OUTCOME_SUCCESS:
         m_patterns[idx].success_count++;
         m_patterns[idx].avg_pips_success = 
            ((m_patterns[idx].avg_pips_success * (m_patterns[idx].success_count - 1)) + pips) 
            / m_patterns[idx].success_count;
         break;
         
      case OUTCOME_FAILURE:
         m_patterns[idx].failure_count++;
         m_patterns[idx].avg_pips_failure = 
            ((m_patterns[idx].avg_pips_failure * (m_patterns[idx].failure_count - 1)) + MathAbs(pips)) 
            / m_patterns[idx].failure_count;
         break;
         
      case OUTCOME_NEUTRAL:
         m_patterns[idx].neutral_count++;
         break;
     }
   
   int total_resolved = m_patterns[idx].success_count + 
                        m_patterns[idx].failure_count + 
                        m_patterns[idx].neutral_count;
   if(total_resolved > 0)
      m_patterns[idx].success_rate = (double)m_patterns[idx].success_count / total_resolved * 100.0;
  }

bool CPatternLibrary::ResolvePendingOutcomes(double current_price)
  {
   double point = SymbolInfoDouble(m_symbol, SYMBOL_POINT);
   if(point == 0) point = 0.00001;
   
   for(int i = m_pending_count - 1; i >= 0; i--)
     {
      m_pending[i].bars_elapsed++;
      
      double pips_moved = (current_price - m_pending[i].entry_price) / point / 10.0;
      
      ENUM_PATTERN_OUTCOME outcome = OUTCOME_PENDING;
      
      if(pips_moved >= m_pending[i].target_pips)
         outcome = OUTCOME_SUCCESS;
      else if(pips_moved <= -m_pending[i].stop_pips)
         outcome = OUTCOME_FAILURE;
      else if(m_pending[i].bars_elapsed >= m_pending[i].bars_to_resolve)
         outcome = OUTCOME_NEUTRAL;
      
      if(outcome != OUTCOME_PENDING)
        {
         int pattern_idx = FindPattern(m_pending[i].fingerprint);
         if(pattern_idx >= 0)
           {
            UpdateExistingPattern(pattern_idx, outcome, pips_moved);
            
            PrintFormat("L3 Resolve | FP: %s | Outcome: %s | Pips: %.1f | WinRate: %.1f%%",
                        m_pending[i].fingerprint,
                        EnumToString(outcome),
                        pips_moved,
                        m_patterns[pattern_idx].success_rate);
           }
         
         for(int j = i; j < m_pending_count - 1; j++)
            m_pending[j] = m_pending[j + 1];
         m_pending_count--;
        }
     }
   
   return true;
  }

                        InpZZDepth, InpZZDeviation, InpZZBackstep);
   if(g_zzHandle == INVALID_HANDLE)
     {
      Print("ERROR: Failed to create ZigZag indicator");
      return INIT_FAILED;
     }
   
   //--- Create Stochastic indicator
   g_stochHandle = iStochastic(_Symbol, _Period, InpStochK, InpStochD, InpStochSlowing, 
                               MODE_SMA, STO_LOWHIGH);
   if(g_stochHandle == INVALID_HANDLE)
     {
      Print("ERROR: Failed to create Stochastic indicator");
      return INIT_FAILED;
     }
   
   //--- Create L1 encoder
   g_L1Encoder = new CLayer1Encoder(_Symbol, _Period, InpPAASegments, InpAlphabetSize);
   if(g_L1Encoder == NULL)
     {
      Print("ERROR: Failed to create L1 Encoder");
      return INIT_FAILED;
     }
   
   //--- Create L2 encoder
   g_L2Encoder = new CRelationalEncoder(InpPAASegments);
   if(g_L2Encoder == NULL)
     {
      Print("ERROR: Failed to create L2 Encoder");
      return INIT_FAILED;
     }
   
   //--- Create L3 library
   g_L3Library = new CPatternLibrary(_Symbol, _Period,
                                     InpBarsToResolve, InpSuccessPips, InpFailurePips,
                                     InpMatchType, InpDTWThreshold, InpSimilarityMinPct);
   g_L3Library.Clear(); // This forces a fresh scan by wiping the loaded cache
   
   if(g_L3Library == NULL)
     {
      Print("ERROR: Failed to create L3 Library");
      return INIT_FAILED;
     }
   
   Print("ZZ Params: Depth=", InpZZDepth, " Dev=", InpZZDeviation, " Back=", InpZZBackstep);
   Print("Leg Percentages: Approach=", InpApproachPct, "% | Reaction=", InpReactionPct, "%");
   Print("Match Type: ", (InpMatchType == MATCH_EXACT ? "Exact" : "DTW Fuzzy"));
   
   //--- Process History
   datetime start_dt = GetStartTimeFromInput(InpHistoryStart);
   ProcessHistoricalPivots(start_dt, g_zzHandle);
   
   Print("==============================================");
   
   return INIT_SUCCEEDED;
  }

//+------------------------------------------------------------------+
//| Check for new ZigZag pivot                                       |
//+------------------------------------------------------------------+
void CheckForNewPivot()
  {
   //--- Get ZigZag buffer - need enough bars to find 3 pivots
   double zzBuffer[];
   ArraySetAsSeries(zzBuffer, true);
   
   int bars_needed = 200;  // Look back enough to find pivots
   if(CopyBuffer(g_zzHandle, 0, 0, bars_needed, zzBuffer) < bars_needed)
      return;
   
   //--- Find the 3 most recent pivots (need 3 to have complete approach+reaction)
   int pivot1_bar = -1, pivot2_bar = -1, pivot3_bar = -1;
   double pivot1_price = 0, pivot2_price = 0, pivot3_price = 0;
   
   for(int i = 1; i < bars_needed; i++)  // Start at 1 to skip current forming bar
     {
      if(zzBuffer[i] != 0)
        {
         if(pivot1_bar < 0)
           {
            pivot1_bar = i;
            pivot1_price = zzBuffer[i];
           }
         else if(pivot2_bar < 0)
           {
            pivot2_bar = i;
            pivot2_price = zzBuffer[i];
           }
         else if(pivot3_bar < 0)
           {
            pivot3_bar = i;
            pivot3_price = zzBuffer[i];
            break;
           }
        }
     }
   
   //--- Need at least 3 pivots
   if(pivot3_bar < 0)
      return;
   
   //--- Check if pivot2 (the middle one) is new
   datetime pivot2_time = iTime(_Symbol, _Period, pivot2_bar);
   
   if(pivot2_time == g_lastPivotTime)
      return;  // Already processed
   
   //--- New pivot confirmed - pivot2 is complete with approach (from pivot3) and reaction (to pivot1)
   g_lastPivotTime = pivot2_time;
   g_lastPivotBar = pivot2_bar;
   g_prevPivotBar = pivot3_bar;
   g_lastPivotPrice = pivot2_price;
   
   //--- Determine pivot type
   bool isPivotHigh = (pivot2_price > pivot1_price && pivot2_price > pivot3_price);
   g_lastPivotWasHigh = isPivotHigh;
   
   //--- Process the pivot
   ProcessPivot(pivot3_bar, pivot2_bar, pivot1_bar, pivot2_price, pivot2_time, isPivotHigh);
  }

//+------------------------------------------------------------------+
//| Process a confirmed pivot                                        |
//+------------------------------------------------------------------+
void ProcessPivot(int approach_start_bar, int pivot_bar, int reaction_end_bar,
                  double pivot_price, datetime pivot_time, bool is_pivot_high)
{
   //--- Calculate leg lengths
   int approach_len = approach_start_bar - pivot_bar;
   int reaction_len = pivot_bar - reaction_end_bar;
   
   if(approach_len < 2 || reaction_len < 2)
      return;  // Legs too short
   
   //--- Apply percentage trimming
   int approach_bars_used = (int)MathCeil(approach_len * InpApproachPct / 100.0);
   int reaction_bars_used = (int)MathCeil(reaction_len * InpReactionPct / 100.0);
   
   if(approach_bars_used < 2) approach_bars_used = 2;
   if(reaction_bars_used < 2) reaction_bars_used = 2;
   
   //--- Extract price data for approach leg (from percentage of leg before pivot)
   double approach_prices[];
   ArrayResize(approach_prices, approach_bars_used);
   
   int approach_start = pivot_bar + approach_bars_used - 1;
   for(int i = 0; i < approach_bars_used; i++)
     {
      approach_prices[i] = iClose(_Symbol, _Period, approach_start - i);
     }
   
   //--- Extract price data for reaction leg (from pivot to percentage of leg after)
   double reaction_prices[];
   ArrayResize(reaction_prices, reaction_bars_used);
   
   for(int i = 0; i < reaction_bars_used; i++)
     {
      reaction_prices[i] = iClose(_Symbol, _Period, pivot_bar - i);
     }
   
   //--- Extract stochastic data for approach window
   double stochK[], stochD[];
   ArrayResize(stochK, approach_bars_used);
   ArrayResize(stochD, approach_bars_used);
   
   double stochK_buf[], stochD_buf[];
   ArraySetAsSeries(stochK_buf, true);
   ArraySetAsSeries(stochD_buf, true);
   
   if(CopyBuffer(g_stochHandle, 0, pivot_bar, approach_bars_used, stochK_buf) < approach_bars_used)
      return;
   if(CopyBuffer(g_stochHandle, 1, pivot_bar, approach_bars_used, stochD_buf) < approach_bars_used)
      return;
   
   //--- Reverse to chronological order
   for(int i = 0; i < approach_bars_used; i++)
     {
      stochK[i] = stochK_buf[approach_bars_used - 1 - i];
      stochD[i] = stochD_buf[approach_bars_used - 1 - i];
     }
   
   //--- Encode to SAX
   string approach_sax = g_L1Encoder.EncodeArray(approach_prices);
   string reaction_sax = g_L1Encoder.EncodeArray(reaction_prices);
   string stochK_sax = g_L1Encoder.EncodeBoundedArray(stochK, 0.0, 100.0);
   string stochD_sax = g_L1Encoder.EncodeBoundedArray(stochD, 0.0, 100.0);
   
   if(StringLen(approach_sax) == 0 || StringLen(reaction_sax) == 0)
      return;

   l1_output.pivot_time = pivot_time;
   l1_output.pivot_price = pivot_price;
   l1_output.approach_bars = approach_bars_used;
   l1_output.reaction_bars = reaction_bars_used;
   l1_output.is_pivot_high = is_pivot_high;

   l1_output.std_dev = 0;
   
   //--- Encode to L2
   PairedPattern pattern;
   if(!g_L2Encoder.Encode(l1_output, pattern))
      return;
   
   //--- HIGHLIGHT DYNAMIC TOP PATTERN
   string topFP = GetTopFingerprint();
   if(pattern.fingerprint == topFP && topFP != "")
     {
      DrawPatternBox(pattern.fingerprint, approach_start_bar, reaction_end_bar);
     }

   //--- Log
//   PrintFormat("Pivot %s @ %s | Price: %.5f | Approach: %s | Reaction: %s",
//               (is_pivot_high ? "HIGH" : "LOW"),
//               TimeToString(pivot_time),
//               pivot_price,

//--- Store occurrence for later highlighting
   string used_fp = g_L3Library.RegisterPattern(pattern);

   g_occurrences[g_occurrence_count].approach_bar = approach_start_bar;

   g_occurrences[g_occurrence_count].stochK_sax = stochK_sax;
   g_occurrences[g_occurrence_count].stochD_sax = stochD_sax;

   g_occurrences[g_occurrence_count].reaction_bars = reaction_bars_used;

//+------------------------------------------------------------------+
//| Find the fingerprint with highest occurrences in the library     |
//+------------------------------------------------------------------+
string GetTopFingerprint()
{
   string top_fp = "";
   int max_occ = -1;
   int total = g_L3Library.GetPatternCount();
   
   if(total <= 0) return "";

   for(int i = 0; i < total; i++)
   {
      string current_fp = g_L3Library.GetFingerprintByIndex(i);
      
      // Ignore empty or invalid default fingerprints
      if(current_fp == "" || current_fp == "0") continue;
      
      int current_occ = g_L3Library.GetOccurrences(current_fp);
      
      if(current_occ > max_occ)
      {
         max_occ = current_occ;
         top_fp = current_fp;
      }
   }
   return top_fp;
}

//+------------------------------------------------------------------+
//| Detect Pivot at Offset
//+------------------------------------------------------------------+
void DetectPivotAtOffset(int offset, int zz_handle)
{
   double zzBuffer[];
   ArraySetAsSeries(zzBuffer, true);
   
   int bars_needed = 200;
   if(CopyBuffer(zz_handle, 0, offset, bars_needed, zzBuffer) < bars_needed)
      return;

   int p1_idx = -1, p2_idx = -1, p3_idx = -1;
   double p2_price = 0;

   for(int i = 1; i < bars_needed; i++)
   {
      if(zzBuffer[i] != 0)
      {
         if(p1_idx < 0) p1_idx = i;
         else if(p2_idx < 0) { p2_idx = i; p2_price = zzBuffer[i]; }
         else if(p3_idx < 0) { p3_idx = i; break; }
      }
   }

   if(p3_idx < 0) return;
   
   datetime p2_time = iTime(_Symbol, _Period, offset + p2_idx);
   
   // Block if we've already handled this exact pivot time anywhere in the scan
   if(p2_time <= g_lastPivotTime) 
      return;

   // Update the global gatekeeper
   g_lastPivotTime = p2_time;


   bool isHigh = (p2_price > zzBuffer[p1_idx] && p2_price > zzBuffer[p3_idx]);
   
   ProcessPivot(offset + p3_idx, offset + p2_idx, offset + p1_idx, p2_price, p2_time, isHigh);
}

//| Process historical pivots
//+------------------------------------------------------------------+
void ProcessHistoricalPivots(datetime start_time, int zz_handle)
{
   g_lastPivotTime = 0;
   g_occurrence_count = 0;
   ArrayResize(g_occurrences, 0);
   
   int start_bar = iBarShift(_Symbol, _Period, start_time);
   int total_bars = iBars(_Symbol, _Period);
   int scan_limit = MathMin(start_bar + 200, total_bars - 1);
   
   Print("==============================================");
   PrintFormat("MDT: Starting historical scan | Start Date: %s | Start Bar: %d", 
               TimeToString(start_time), scan_limit);

   // Single pass: populate library and store occurrences
   for(int i = scan_limit; i >= 1; i--)
   {
      DetectPivotAtOffset(i, zz_handle);
   }
   
   // Find top fingerprint
   string finalTopFP = GetTopFingerprint();
   PrintFormat("MDT: Scan complete. Top Fingerprint: %s | Total occurrences stored: %d", finalTopFP, g_occurrence_count);

   // Draw boxes from stored occurrences
   if(finalTopFP != "" && finalTopFP != "0")
   {
      int boxes_drawn = 0;
      for(int i = 0; i < g_occurrence_count; i++)
      {
         if(g_occurrences[i].matched_fingerprint == finalTopFP)
         //if(g_occurrences[i].fingerprint == finalTopFP)
         {
            DrawPatternBox(finalTopFP, g_occurrences[i].approach_bar, g_occurrences[i].reaction_bar);
            PrintFormat("MDT: HIGHLIGHTED [%s] at pivot_time=%s", finalTopFP, TimeToString(g_occurrences[i].pivot_time));
            boxes_drawn++;
         }
      }
      PrintFormat("MDT: Drew %d boxes for fingerprint %s", boxes_drawn, finalTopFP);
      WriteHighlightedPatternsCSV(finalTopFP);
   }
   
   Print("==============================================");
   ChartRedraw();
}

void ProcessPivot(int approach_start_bar, int pivot_bar, int reaction_end_bar,
                  double pivot_price, datetime pivot_time, bool is_pivot_high)
{
   //--- Calculate leg lengths
   int approach_len = approach_start_bar - pivot_bar;
   int reaction_len = pivot_bar - reaction_end_bar;
   
   if(approach_len < 2 || reaction_len < 2)
      return;
   
   //--- Apply percentage trimming
   int approach_bars_used = (int)MathCeil(approach_len * InpApproachPct / 100.0);
   int reaction_bars_used = (int)MathCeil(reaction_len * InpReactionPct / 100.0);
   
   if(approach_bars_used < 2) approach_bars_used = 2;
   if(reaction_bars_used < 2) reaction_bars_used = 2;
   
   //--- Extract price data for approach leg (from percentage of leg before pivot)
   double approach_prices[];
   ArrayResize(approach_prices, approach_bars_used);
   
   int approach_start = pivot_bar + approach_bars_used - 1;
   for(int i = 0; i < approach_bars_used; i++)
   {
      approach_prices[i] = iClose(_Symbol, _Period, approach_start - i);
   }
   
   //--- Extract price data for reaction leg (from pivot to percentage of leg after)
   double reaction_prices[];
   ArrayResize(reaction_prices, reaction_bars_used);
   
   for(int i = 0; i < reaction_bars_used; i++)
   {
      reaction_prices[i] = iClose(_Symbol, _Period, pivot_bar - i);
   }
   
   //--- Extract stochastic data for approach window
   double stochK[], stochD[];
   ArrayResize(stochK, approach_bars_used);
   ArrayResize(stochD, approach_bars_used);
   
   double stochK_buf[], stochD_buf[];
   ArraySetAsSeries(stochK_buf, true);
   ArraySetAsSeries(stochD_buf, true);
   
   if(CopyBuffer(g_stochHandle, 0, pivot_bar, approach_bars_used, stochK_buf) < approach_bars_used)
      return;
   if(CopyBuffer(g_stochHandle, 1, pivot_bar, approach_bars_used, stochD_buf) < approach_bars_used)
      return;
   
   //--- Reverse to chronological order
   for(int i = 0; i < approach_bars_used; i++)
   {
      stochK[i] = stochK_buf[approach_bars_used - 1 - i];
      stochD[i] = stochD_buf[approach_bars_used - 1 - i];

   l1_output.reaction_bars = reaction_bars_used;

   l1_output.std_dev = 0;

//+------------------------------------------------------------------+
//| Calculate mean of array                                           |
//+------------------------------------------------------------------+
double CLayer1Encoder::CalculateMean(const double &data[])
  {
   int size = ArraySize(data);
   if(size == 0) return 0;
   
   double sum = 0;
   for(int i = 0; i < size; i++)
      sum += data[i];
   
   return sum / size;
  }

//+------------------------------------------------------------------+
//| Calculate standard deviation                                      |
//+------------------------------------------------------------------+
double CLayer1Encoder::CalculateStdDev(const double &data[], double mean)
  {
   int size = ArraySize(data);
   if(size < 2) return 0.0001;
   
   double sum_sq = 0;
   for(int i = 0; i < size; i++)
     {
      double diff = data[i] - mean;
      sum_sq += diff * diff;
     }
   
   double variance = sum_sq / (size - 1);
   double std = MathSqrt(variance);
   
   return (std < 0.0001) ? 0.0001 : std;
  }

//+------------------------------------------------------------------+
//| Convert z-score to SAX character                                  |
//+------------------------------------------------------------------+
char CLayer1Encoder::ValueToSAXChar(double z_score)
  {
   for(int i = 0; i < ArraySize(m_breakpoints); i++)
     {
      if(z_score < m_breakpoints[i])
         return (char)('a' + i);
     }
   return (char)('a' + ArraySize(m_breakpoints));
  }

//+------------------------------------------------------------------+
//| Convert bounded value to SAX character                            |
//+------------------------------------------------------------------+
char CLayer1Encoder::BoundedValueToSAXChar(double value, double min_val, double max_val)
  {
   if(max_val <= min_val) return 'c';
   
   double normalized = (value - min_val) / (max_val - min_val);
   
   int idx = (int)(normalized * m_alphabet_size);
   if(idx >= m_alphabet_size) idx = m_alphabet_size - 1;
   if(idx < 0) idx = 0;
   
   return (char)('a' + idx);
  }

//+------------------------------------------------------------------+
//| Encode array to SAX - one character per bar (no PAA)              |
//+------------------------------------------------------------------+
string CLayer1Encoder::EncodeArray(const double &data[])
  {
   int size = ArraySize(data);
   if(size == 0) return "";
   
   //--- Calculate statistics for z-score normalization
   double mean = CalculateMean(data);
   double std_dev = CalculateStdDev(data, mean);
   
   //--- Convert each bar to SAX character via z-score
   string sax = "";
   for(int i = 0; i < size; i++)
     {
      double z_score = (data[i] - mean) / std_dev;
      sax += CharToString(ValueToSAXChar(z_score));
     }
   
   return sax;
  }

//+------------------------------------------------------------------+
//| Encode bounded array - one character per bar (no PAA)             |
//+------------------------------------------------------------------+
string CLayer1Encoder::EncodeBoundedArray(const double &data[], double min_val, double max_val)
  {
   int size = ArraySize(data);
   if(size == 0) return "";
   
   //--- Convert each bar to SAX character via bounded mapping
   string sax = "";
   for(int i = 0; i < size; i++)
     {
      sax += CharToString(BoundedValueToSAXChar(data[i], min_val, max_val));
     }
   
   return sax;
  }

ENUM_DIVERGENCE_TYPE CRelationalEncoder::DetectDivergence(const string &price_sax, const string &stoch_sax)
  {
   int len = MathMin(StringLen(price_sax), StringLen(stoch_sax));
   if(len < 2) return DIV_NONE;
   
   double price_start = SaxCharToValue((char)StringGetCharacter(price_sax, 0));
   double price_end   = SaxCharToValue((char)StringGetCharacter(price_sax, len-1));
   double stoch_start = SaxCharToValue((char)StringGetCharacter(stoch_sax, 0));
   double stoch_end   = SaxCharToValue((char)StringGetCharacter(stoch_sax, len-1));
   
   double price_delta = price_end - price_start;
   double stoch_delta = stoch_end - stoch_start;
   double threshold = 0.3;
   
   if(price_delta < -threshold && stoch_delta > threshold)
      return DIV_BULLISH_REGULAR;
   if(price_delta > threshold && stoch_delta < -threshold)
      return DIV_BEARISH_REGULAR;
   if(price_delta > threshold && stoch_delta < -threshold && price_end < 0 && stoch_end < 0)
      return DIV_BULLISH_HIDDEN;
   if(price_delta < -threshold && stoch_delta > threshold && price_end > 0 && stoch_end > 0)
      return DIV_BEARISH_HIDDEN;
   
   return DIV_NONE;
  }

   char last_char = (char)StringGetCharacter(stochK_sax, len-1);

  }

//+------------------------------------------------------------------+
ENUM_CROSS_EVENT CRelationalEncoder::DetectCrossEvent(const string &stochK_sax, const string &stochD_sax)
  {
   int len = MathMin(StringLen(stochK_sax), StringLen(stochD_sax));
   if(len < 2) return CROSS_NONE;
   
   double k_prev = SaxCharToValue((char)StringGetCharacter(stochK_sax, len-2));
   double k_curr = SaxCharToValue((char)StringGetCharacter(stochK_sax, len-1));
   double d_prev = SaxCharToValue((char)StringGetCharacter(stochD_sax, len-2));
   double d_curr = SaxCharToValue((char)StringGetCharacter(stochD_sax, len-1));
   
   bool was_below = (k_prev < d_prev);
   bool is_above  = (k_curr > d_curr);
   bool was_above = (k_prev > d_prev);
   bool is_below  = (k_curr < d_curr);
   
   if(was_below && is_above)
      return CROSS_BULLISH;
   if(was_above && is_below)
      return CROSS_BEARISH;
   
   return CROSS_NONE;
  }

//+------------------------------------------------------------------+
ENUM_MOMENTUM_STATE CRelationalEncoder::CalculateMomentum(const string &sax_string)
  {
   int len = StringLen(sax_string);
   if(len < 3) return MOM_FLAT;
   
   double total_velocity = 0;
   double velocity_change = 0;
   double prev_velocity = 0;
   
   for(int i = 1; i < len; i++)
     {
      double current = SaxCharToValue((char)StringGetCharacter(sax_string, i));
      double prev = SaxCharToValue((char)StringGetCharacter(sax_string, i-1));
      double velocity = current - prev;
      
      total_velocity += velocity;
      if(i > 1)
         velocity_change += (velocity - prev_velocity);
      prev_velocity = velocity;
     }
   
   double avg_velocity = total_velocity / (len - 1);
   double avg_accel = (len > 2) ? velocity_change / (len - 2) : 0;
   double v_thresh = 0.1;
   double a_thresh = 0.05;
   
   if(MathAbs(avg_velocity) < v_thresh)
      return MOM_FLAT;
   
   if(avg_velocity > v_thresh)
     {
      if(avg_accel > a_thresh)
         return MOM_ACCELERATING_UP;
      else if(avg_accel < -a_thresh)
         return MOM_DECELERATING_UP;
     }
   
   if(avg_velocity < -v_thresh)
     {
      if(avg_accel < -a_thresh)
         return MOM_ACCELERATING_DOWN;
      else if(avg_accel > a_thresh)
         return MOM_DECELERATING_DOWN;
     }
   
   return MOM_FLAT;
  }

//+------------------------------------------------------------------+
double CPatternLibrary::CalculateDTW(const string &seq1, const string &seq2)
  {
   int n = StringLen(seq1);
   int m = StringLen(seq2);
   
   if(n == 0 || m == 0)
      return DBL_MAX;
   
   double dtw[];
   ArrayResize(dtw, (n + 1) * (m + 1));
   
   for(int i = 0; i <= n; i++)
      for(int j = 0; j <= m; j++)
         dtw[i * (m + 1) + j] = DBL_MAX;
   
   dtw[0] = 0;
   
   for(int i = 1; i <= n; i++)
     {
      for(int j = 1; j <= m; j++)
        {
         double cost = SaxCharDistance((char)StringGetCharacter(seq1, i - 1),
                                       (char)StringGetCharacter(seq2, j - 1));
         
         double prev_min = MathMin(dtw[(i - 1) * (m + 1) + j],
                           MathMin(dtw[i * (m + 1) + (j - 1)],
                                   dtw[(i - 1) * (m + 1) + (j - 1)]));
         
         dtw[i * (m + 1) + j] = cost + prev_min;
        }
     }
   
   return dtw[n * (m + 1) + m];
  }

      if(m_patterns[i].pivot_type != pattern.pivot_type)
         continue;
      
      double dist_approach = CalculateDTW(pattern.approach_sax, m_patterns[i].approach_sax);
      double dist_reaction = CalculateDTW(pattern.reaction_sax, m_patterns[i].reaction_sax);
      double dist_stochK = CalculateDTW(pattern.stochK_sax, m_patterns[i].stochK_sax);
      double dist_stochD = CalculateDTW(pattern.stochD_sax, m_patterns[i].stochD_sax);
      
      double total_dist = (dist_approach + dist_reaction + dist_stochK + dist_stochD) / 4.0;
      
      if(total_dist < best_distance)
        {
         best_distance = total_dist;
         best_idx = i;
        }
     }
   
   if(best_distance <= m_dtw_threshold)
      return best_idx;
   
   PrintFormat("DTW: Best distance %.2f (threshold %.2f) - no match", best_distance, m_dtw_threshold);
   
   return -1;
  }

//+------------------------------------------------------------------+
int CPatternLibrary::FindSimilarPatterns(const PairedPattern &pattern, DTWMatch &matches[], int max_matches)
  {
   DTWMatch all_matches[];
   ArrayResize(all_matches, m_pattern_count);
   
   int valid_count = 0;
   
   for(int i = 0; i < m_pattern_count; i++)
     {
      if(m_patterns[i].pivot_type != pattern.pivot_type)
         continue;
      
      double dist_approach = CalculateDTW(pattern.approach_sax, m_patterns[i].approach_sax);
      double dist_reaction = CalculateDTW(pattern.reaction_sax, m_patterns[i].reaction_sax);
      double dist_stochK = CalculateDTW(pattern.stochK_sax, m_patterns[i].stochK_sax);
      double dist_stochD = CalculateDTW(pattern.stochD_sax, m_patterns[i].stochD_sax);
      
      double total_dist = (dist_approach + dist_reaction + dist_stochK + dist_stochD) / 4.0;
      double max_dist = 32.0;
      double similarity = MathMax(0, (1.0 - total_dist / max_dist) * 100.0);
      
      if(similarity >= m_similarity_min_pct)
        {
         all_matches[valid_count].pattern_idx = i;
         all_matches[valid_count].dtw_distance = total_dist;
         all_matches[valid_count].similarity_pct = similarity;
         valid_count++;
        }
     }
   
   for(int i = 0; i < valid_count - 1; i++)
     {
      for(int j = i + 1; j < valid_count; j++)
        {
         if(all_matches[j].dtw_distance < all_matches[i].dtw_distance)
           {
            DTWMatch temp = all_matches[i];
            all_matches[i] = all_matches[j];
            all_matches[j] = temp;
           }
        }
     }
   
   int result_count = MathMin(valid_count, max_matches);
   ArrayResize(matches, result_count);
   
   for(int i = 0; i < result_count; i++)
      matches[i] = all_matches[i];
   
   return result_count;
  }

void CPatternLibrary::UpdateExistingPattern(int idx, ENUM_PATTERN_OUTCOME outcome, double pips)
  {
   if(idx < 0 || idx >= m_pattern_count) return;
   
   switch(outcome)
     {
      case OUTCOME_SUCCESS:
         m_patterns[idx].success_count++;
         m_patterns[idx].avg_pips_success = 
            ((m_patterns[idx].avg_pips_success * (m_patterns[idx].success_count - 1)) + pips) 
            / m_patterns[idx].success_count;
         break;
         
      case OUTCOME_FAILURE:
         m_patterns[idx].failure_count++;
         m_patterns[idx].avg_pips_failure = 
            ((m_patterns[idx].avg_pips_failure * (m_patterns[idx].failure_count - 1)) + MathAbs(pips)) 
            / m_patterns[idx].failure_count;
         break;
         
      case OUTCOME_NEUTRAL:
         m_patterns[idx].neutral_count++;
         break;
     }
   
   int total_resolved = m_patterns[idx].success_count + 
                        m_patterns[idx].failure_count + 
                        m_patterns[idx].neutral_count;
   if(total_resolved > 0)
      m_patterns[idx].success_rate = (double)m_patterns[idx].success_count / total_resolved * 100.0;
  }

bool CPatternLibrary::ResolvePendingOutcomes(double current_price)
  {
   double point = SymbolInfoDouble(m_symbol, SYMBOL_POINT);
   if(point == 0) point = 0.00001;
   
   for(int i = m_pending_count - 1; i >= 0; i--)
     {
      m_pending[i].bars_elapsed++;
      
      double pips_moved = (current_price - m_pending[i].entry_price) / point / 10.0;
      
      ENUM_PATTERN_OUTCOME outcome = OUTCOME_PENDING;
      
      if(pips_moved >= m_pending[i].target_pips)
         outcome = OUTCOME_SUCCESS;
      else if(pips_moved <= -m_pending[i].stop_pips)
         outcome = OUTCOME_FAILURE;
      else if(m_pending[i].bars_elapsed >= m_pending[i].bars_to_resolve)
         outcome = OUTCOME_NEUTRAL;
      
      if(outcome != OUTCOME_PENDING)
        {
         int pattern_idx = FindPattern(m_pending[i].fingerprint);
         if(pattern_idx >= 0)
           {
            UpdateExistingPattern(pattern_idx, outcome, pips_moved);
            
            PrintFormat("L3 Resolve | FP: %s | Outcome: %s | Pips: %.1f | WinRate: %.1f%%",
                        m_pending[i].fingerprint,
                        EnumToString(outcome),
                        pips_moved,
                        m_patterns[pattern_idx].success_rate);
           }
         
         for(int j = i; j < m_pending_count - 1; j++)
            m_pending[j] = m_pending[j + 1];
         m_pending_count--;
        }
     }
   
   return true;
  }
