void OnDeinit(const int reason)
{
   //--- Remove indicator from chart
   if(g_indicatorSubWindow >= 0)
      ChartIndicatorDelete(0, g_indicatorSubWindow, "TEMA/MA Spread");
   
   //--- Release indicator handle

   ObjectDelete(0, LABEL_INSTRUCTIONS);
   ObjectDelete(0, LABEL_STATUS);
   ObjectDelete(0, LABEL_AUTOTRADE);
}
