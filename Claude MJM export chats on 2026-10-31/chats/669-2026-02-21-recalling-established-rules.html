void UpdateOpenTradeLiveState(const int bar, const double liveExitPrice)
{
   if(g_currentTrade.direction == 0 || g_currentTrade.entryBar < 0)
      return;

   if(bar >= ArraySize(BufferTradePL))
      TS_GrowStatsArrays(bar + 1, BufferTradePL, BufferRunningPL, BufferTradeCount, BufferWinPct,
                         BufferShortCount, BufferShortWinPct, BufferLongCount, BufferLongWinPct);

   double livePL = TS_Price2Pips(liveExitPrice - g_currentTrade.entryPrice) * g_currentTrade.direction;
   double liveRunning = g_tradeStats.runningPL + livePL;

   BufferTradePL[bar] = livePL;
   BufferRunningPL[bar] = liveRunning;
   BufferTradeCount[bar] = g_tradeStats.TotalTrades();
   BufferWinPct[bar] = g_tradeStats.WinPct();
   BufferShortCount[bar] = g_tradeStats.shortCount;
   BufferShortWinPct[bar] = g_tradeStats.ShortWinPct();
   BufferLongCount[bar] = g_tradeStats.longCount;
   BufferLongWinPct[bar] = g_tradeStats.LongWinPct();

   if(g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0)
   {
      string tip = StringFormat("Entry: %s | Exit: %s | P/L: %.1f pips | Running: %.1f pips",
                                DoubleToString(g_currentTrade.entryPrice, _Digits),
                                DoubleToString(liveExitPrice, _Digits),
                                BufferTradePL[bar],
                                BufferRunningPL[bar]);
      ObjectSetString(0, g_currentTrade.arrowName, OBJPROP_TOOLTIP, tip);
   }
}

   ArrayResize(kcUpperTail, copyCount);
   ArrayResize(kcMiddleTail, copyCount);
   ArrayResize(kcLowerTail, copyCount);
   ArraySetAsSeries(temaTail, false);
   ArraySetAsSeries(emaTail, false);
   ArraySetAsSeries(kcUpperTail, false);
   ArraySetAsSeries(kcMiddleTail, false);
   ArraySetAsSeries(kcLowerTail, false);

   int temaCopied = CopyBuffer(iHndlTEMA, 0, 0, copyCount, temaTail);
   int emaCopied  = CopyBuffer(iHndlEMAofTEMA, 0, 0, copyCount, emaTail);
   int kcUpperCopied  = CopyBuffer(iHndlKC, 0, 0, copyCount, kcUpperTail);
   int kcMiddleCopied = CopyBuffer(iHndlKC, 1, 0, copyCount, kcMiddleTail);
   int kcLowerCopied  = CopyBuffer(iHndlKC, 2, 0, copyCount, kcLowerTail);
   if(temaCopied != copyCount || emaCopied != copyCount ||
      kcUpperCopied != copyCount || kcMiddleCopied != copyCount || kcLowerCopied != copyCount)
   {
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " CopyBuffer failed. temaCopied=",temaCopied,
            " emaCopied=",emaCopied,
            " kcUpperCopied=",kcUpperCopied,
            " kcMiddleCopied=",kcMiddleCopied,
            " kcLowerCopied=",kcLowerCopied,
            " expected=",copyCount,
            " err=",GetLastError());
      return(prev_calculated);
   }

   ArrayResize(BufferKCUpper, rates_total);
   ArrayResize(BufferKCMiddle, rates_total);
   ArrayResize(BufferKCLower, rates_total);

   for(int k = 0; k < copyCount; k++)
   {
      BufferTEMA[copyStart + k] = temaTail[k];
      BufferEMAofTEMA[copyStart + k] = emaTail[k];
      BufferKCUpper[copyStart + k] = kcUpperTail[k];
      BufferKCMiddle[copyStart + k] = kcMiddleTail[k];
      BufferKCLower[copyStart + k] = kcLowerTail[k];
   }

   double pointSize = MathPow(10.0, -_Digits);
   double eps = pointSize * 0.1;
   double pipSize = pointSize * 10.0;
   bool bDoHistory = (prev_calculated == 0);

   for(int i = start; i < rates_total; i++)
   {
      if(prev_calculated == 0 || i >= prev_calculated)
      {
         BufferArrowBuy[i] = EMPTY_VALUE;
         BufferArrowSell[i] = EMPTY_VALUE;
      }

      int prevRel = RelSign(BufferTEMA[i - 1] - BufferEMAofTEMA[i - 1], eps);
      int currRel = RelSign(BufferTEMA[i] - BufferEMAofTEMA[i], eps);
      double temaEmaSpread = BufferTEMA[i] - BufferEMAofTEMA[i];
      double temaEmaSpreadPips = TS_Price2Pips(temaEmaSpread);
      BufferTemaEmaSpread[i] = temaEmaSpreadPips;
      if(prevRel == 0)
         prevRel = iDir;
      if(currRel == 0)
         currRel = prevRel;

      //--- Cocking logic: evaluate local hi/lo candidates from prior 2 bars.
      //--- Must run before cross handling so same-bar cock+cross can trigger.
      if(iDir != 0)
      {
         bool candidateFound = false;
         bool cockedThisBar = false;

         for(int d = 1; d <= 2; d++)
         {
            int cand = i - d;
            if(cand <= g_crossBar || cand < 1 || cand + 1 >= rates_total)
               continue;

            candidateFound = true;

            int refBar = MathMax(0, cand - InpOpenRefOffset);
            double refOpen = open[refBar];

            //--- Keep debug/state fields aligned with the candidate under evaluation.
            g_extremeBar = cand;
            g_extremeOpenRefBar = refBar;
            g_extremeOpen = refOpen;
            g_extremePrice = (iDir > 0) ? high[cand] : low[cand];

            if((iDir > 0 && close[i] < refOpen) ||
               (iDir < 0 && close[i] > refOpen))
            {
               g_isCocked = true;
               g_cockedDir = (iDir > 0) ? -1 : 1;
               g_cockLevel = refOpen;
               cockedThisBar = true;
               break;
            }
         }

         g_pendingConfirm = (candidateFound && !cockedThisBar);
      }

      bool crossUp = (prevRel < 0 && currRel > 0);
      bool crossDn = (prevRel > 0 && currRel < 0);
      bool isCross = (crossUp || crossDn);

      int activeBlackout = InpSignalBlackout;
      double currWidthPips = EMPTY_VALUE;
      double widthRatio = EMPTY_VALUE;
      double slopePipsPerBar = EMPTY_VALUE;
      if(BufferKCUpper[i] != EMPTY_VALUE && BufferKCLower[i] != EMPTY_VALUE && BufferKCMiddle[i] != EMPTY_VALUE)
      {
         currWidthPips = TS_Price2Pips(MathAbs(BufferKCUpper[i] - BufferKCLower[i]));

         int lb = MathMin(InpKCWidthLookback, i + 1);
         double widthSumPips = 0.0;
         int widthCnt = 0;
         int from = i - lb + 1;
         if(from < 0) from = 0;
         for(int j = from; j <= i; j++)
         {
            if(BufferKCUpper[j] == EMPTY_VALUE || BufferKCLower[j] == EMPTY_VALUE)
               continue;
            widthSumPips += TS_Price2Pips(MathAbs(BufferKCUpper[j] - BufferKCLower[j]));
            widthCnt++;
         }

         double baselineWidthPips = (widthCnt > 0) ? (widthSumPips / widthCnt) : currWidthPips;
         widthRatio = (baselineWidthPips > 0.0) ? (currWidthPips / baselineWidthPips) : 1.0;

         slopePipsPerBar = 0.0;
         if(i >= InpKCMidSlopePeriod && BufferKCMiddle[i - InpKCMidSlopePeriod] != EMPTY_VALUE)
            slopePipsPerBar = TS_Price2Pips((BufferKCMiddle[i] - BufferKCMiddle[i - InpKCMidSlopePeriod]) / InpKCMidSlopePeriod);

         if(InpUseDynamicBlackout)
         {
            bool isNarrow = (widthRatio < InpNarrowWidthRatio);
            bool isFlat = (MathAbs(slopePipsPerBar) <= InpFlatSlopePipsPerBar);
            if(isNarrow && isFlat)
            {
               double extra = (InpNarrowWidthRatio - widthRatio) * InpBlackoutScaleK;
               if(extra < 0.0) extra = 0.0;
               activeBlackout = InpSignalBlackout + (int)MathRound(extra);
               if(activeBlackout > InpBlackoutMax)
                  activeBlackout = InpBlackoutMax;
            }
         }
      }
      BufferDynBlackout[i] = activeBlackout;
      BufferKCWidthRatio[i] = widthRatio;
      BufferTriggerThreshold[i] = InpTriggerSensitivityPips;

      if(iDir == 0 && currRel != 0)
         ResetCycleForDirection(currRel, i, open, high, low);

      ENUM_SIGNAL_TYPE signal = SIGNAL_NONE;
      ENUM_SIGNAL_TYPE signalCandidate = SIGNAL_NONE;
      if(isCross)
      {
         if(g_isCocked)
         {
            if(crossDn && g_cockedDir == -1)
            {
               g_triggerPending = true;
               g_triggerDir = -1;
               g_triggerCrossBar = i;
            }
            else if(crossUp && g_cockedDir == 1)
            {
               g_triggerPending = true;
               g_triggerDir = 1;
               g_triggerCrossBar = i;
            }
            else
            {
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
         }
         else
         {
            g_triggerPending = false;
            g_triggerDir = 0;
            g_triggerCrossBar = -1;
         }
      }

      if(g_triggerPending)
      {
         bool wrongSide = ((g_triggerDir == 1 && currRel < 0) || (g_triggerDir == -1 && currRel > 0));
         if(wrongSide)
         {
            g_triggerPending = false;
            g_triggerDir = 0;
            g_triggerCrossBar = -1;
         }
         else
         {
            double cumulativeSpreadPips = 0.0;
            int sumFrom = MathMax(0, g_triggerCrossBar);
            if(sumFrom <= i)
            {
               for(int s = sumFrom; s <= i; s++)
               {
                  if(BufferTemaEmaSpread[s] == EMPTY_VALUE)
                     continue;
                  cumulativeSpreadPips += MathAbs(BufferTemaEmaSpread[s]);
               }
            }

            if(cumulativeSpreadPips >= InpTriggerSensitivityPips)
            {
               signalCandidate = (g_triggerDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
         }
      }

      bool canFireNow = ((i - g_lastSignalBar) > activeBlackout && time[i] >= InpTradeStartDate);
      if(signalCandidate != SIGNAL_NONE)
      {
         if(canFireNow)
         {
            signal = signalCandidate;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else
         {
            g_blackoutSignalPending = true;
            g_blackoutSignalDir = (signalCandidate == SIGNAL_LONG) ? 1 : -1;
            g_blackoutSignalBar = i;
         }
      }

      if(signal == SIGNAL_NONE && g_blackoutSignalPending)
      {
         bool retraced = ((g_blackoutSignalDir == 1 && currRel < 0) || (g_blackoutSignalDir == -1 && currRel > 0));
         if(retraced)
         {
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else if(canFireNow)
         {
            signal = (g_blackoutSignalDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
      }

      if(signal != SIGNAL_NONE)
      {
         if(g_signalStartDT == 0)
            g_signalStartDT = time[i];

         double mockExecPrice = GetMockTradePrice(i, close);

         if(InpEnableTradeSim)
         {
            if(InpMockTradePriceSource == MOCK_PRICE_TEMA)
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferTEMA,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell);
            }
            else if(InpMockTradePriceSource == MOCK_PRICE_EMA_TEMA)
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferEMAofTEMA,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell);
            }
            else
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 close,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell);
            }
         }

         //--- Buffer arrows should be anchored at selected mock execution price for the signal bar.
         if(signal == SIGNAL_LONG)
            BufferArrowBuy[i] = mockExecPrice;
         else
            BufferArrowSell[i] = mockExecPrice;

         //--- Reposition object arrows 100 pips outside the farther of KC outer line or TEMA from KC midline.
         if(InpShowArrows && g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0 &&
            BufferKCUpper[i] != EMPTY_VALUE && BufferKCMiddle[i] != EMPTY_VALUE &&
            BufferKCLower[i] != EMPTY_VALUE && BufferTEMA[i] != EMPTY_VALUE)
         {
            double mid = BufferKCMiddle[i];
            double arrowPrice = 0.0;
            double offset = InpObjectArrowOffsetPips * pipSize;

            if(signal == SIGNAL_SHORT)
            {
               double baseUpper = BufferKCUpper[i];
               double distUpper = MathAbs(baseUpper - mid);
               double distTema  = MathAbs(BufferTEMA[i] - mid);
               double base = (distTema > distUpper) ? BufferTEMA[i] : baseUpper;
               arrowPrice = base + offset;
               BufferArrowSell[i] = mockExecPrice;
            }
            else
            {
               double baseLower = BufferKCLower[i];
               double distLower = MathAbs(baseLower - mid);
               double distTema  = MathAbs(BufferTEMA[i] - mid);
               double base = (distTema > distLower) ? BufferTEMA[i] : baseLower;
               arrowPrice = base - offset;
               BufferArrowBuy[i] = mockExecPrice;
            }

            ObjectMove(0, g_currentTrade.arrowName, 0, time[i], arrowPrice);
         }

         g_lastSignalBar = i;
      }

      //--- Keep the most recent open trade tooltip/buffers live on the current bar.
      if(i == rates_total - 1 && InpEnableTradeSim)
      {
         double liveExitPrice = GetMockTradePrice(i, close);
         UpdateOpenTradeLiveState(i, liveExitPrice);
      }

      //--- Uncock only after trigger decisions on this bar.
      if(g_isCocked)
      {
         if(g_cockedDir == -1 && close[i] > g_cockLevel)
         {
            g_isCocked = false;
            g_cockedDir = 0;
         }
         else if(g_cockedDir == 1 && close[i] < g_cockLevel)
         {
            g_isCocked = false;
            g_cockedDir = 0;
         }
      }

      if(i < rates_total - 1 && i > g_lastExportedBar)
      {
         AppendCsvDebugRow(time[i], i,
                           open[i], high[i], low[i], close[i],
                           BufferTEMA[i], BufferEMAofTEMA[i],
                           BufferKCUpper[i], BufferKCMiddle[i], BufferKCLower[i],
                           currWidthPips, widthRatio, slopePipsPerBar,
                           activeBlackout,
                           crossUp, crossDn, isCross,
                           signal);
         g_lastExportedBar = i;
      }

      //--- Cross after signal check: trigger consumes prior cocked state, then reset cycle.
      if(isCross && currRel != 0)
      {
         ResetCycleForDirection(currRel, i, open, high, low, g_triggerPending);
         continue;
      }

   }

   return(rates_total);
}

//+------------------------------------------------------------------+
//| TS_DetectSignalUP_DN - array-based signal detection              |
//+------------------------------------------------------------------+
ENUM_SIGNAL_TYPE TS_DetectSignalUP_DN(const double &slopeValues[],
                                      const bool &tracked[],
                                      int bufferCount,
                                      int thresholdPercent,
                                      int &direction,
                                      double signalOFS = 0.0)
{
   int bullishCount = 0;
   int bearishCount = 0;
   int activeBuffers = 0;

   for(int i = 0; i < bufferCount; i++)
   {
      if(!tracked[i])
         continue;

      double currValue = slopeValues[i];
      if(currValue == EMPTY_VALUE)
         continue;

      activeBuffers++;

      if(currValue > signalOFS)
         bullishCount++;
      else if(currValue < -signalOFS)
         bearishCount++;
   }

   if(activeBuffers == 0)
      return SIGNAL_NONE;

   int bullishPercent = (bullishCount * 100) / activeBuffers;
   int bearishPercent = (bearishCount * 100) / activeBuffers;

   if(bearishPercent >= thresholdPercent && direction >= 0)
   {
      direction = -1;
      return SIGNAL_SHORT;
   }
   else if(bullishPercent >= thresholdPercent && direction <= 0)
   {
      direction = 1;
      return SIGNAL_LONG;
   }

   return SIGNAL_NONE;
}

//| 1. CloseTrade - P/L calc + stats mutation                        |
//+------------------------------------------------------------------+
STradeResult CloseTrade(double exitPrice, const STradeEntry &entry, STradeStats &stats)
{
   STradeResult result;

   result.tradePL = TS_Price2Pips(exitPrice - entry.entryPrice) * entry.direction;
   result.isWin   = (result.tradePL > 0);
   result.wasLong = (entry.direction == 1);

   if(result.wasLong)                  { stats.longCount++;  if(result.isWin) stats.longWins++;  }
   else if(entry.direction == -1)      { stats.shortCount++; if(result.isWin) stats.shortWins++; }

   stats.runningPL += result.tradePL;

   result.runningPL   = stats.runningPL;
   result.totalTrades = stats.TotalTrades();
   result.totalWins   = stats.TotalWins();
   result.winPct      = stats.WinPct();
   result.longCount   = stats.longCount;
   result.longWins    = stats.longWins;
   result.longWinPct  = stats.LongWinPct();
   result.shortCount  = stats.shortCount;
   result.shortWins   = stats.shortWins;
   result.shortWinPct = stats.ShortWinPct();

   return result;
}

//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   if(rates_total <= g_maxPeriodsBack)
      return(0);

   int start = (prev_calculated == 0) ? g_maxPeriodsBack : prev_calculated - 1;
   if(start < 1)
      start = 1;

   if(prev_calculated == 0)
   {
      ResetAllStateAndBuffers();
      g_lastExportedBar = -1;
      InitCsvDebugFile();
   }

   int copyStart = MathMax(0, start - 3);
   int copyCount = rates_total - copyStart;

   double temaTail[];
   double emaTail[];
   double kcUpperTail[];
   double kcMiddleTail[];
   double kcLowerTail[];
   ArrayResize(temaTail, copyCount);
   ArrayResize(emaTail, copyCount);
   ArrayResize(kcUpperTail, copyCount);
   ArrayResize(kcMiddleTail, copyCount);
   ArrayResize(kcLowerTail, copyCount);
   ArraySetAsSeries(temaTail, false);
   ArraySetAsSeries(emaTail, false);
   ArraySetAsSeries(kcUpperTail, false);
   ArraySetAsSeries(kcMiddleTail, false);
   ArraySetAsSeries(kcLowerTail, false);

   int temaCopied = CopyBuffer(iHndlTEMA, 0, 0, copyCount, temaTail);
   int emaCopied  = CopyBuffer(iHndlEMAofTEMA, 0, 0, copyCount, emaTail);
   int kcUpperCopied  = CopyBuffer(iHndlKC, 0, 0, copyCount, kcUpperTail);
   int kcMiddleCopied = CopyBuffer(iHndlKC, 1, 0, copyCount, kcMiddleTail);
   int kcLowerCopied  = CopyBuffer(iHndlKC, 2, 0, copyCount, kcLowerTail);
   if(temaCopied != copyCount || emaCopied != copyCount ||
      kcUpperCopied != copyCount || kcMiddleCopied != copyCount || kcLowerCopied != copyCount)
   {
      Print("ChartID<",ChartID(),">: [",__FILE__,"] [",__FUNCTION__,"] LINE ",__LINE__,
            " CopyBuffer failed. temaCopied=",temaCopied,
            " emaCopied=",emaCopied,
            " kcUpperCopied=",kcUpperCopied,
            " kcMiddleCopied=",kcMiddleCopied,
            " kcLowerCopied=",kcLowerCopied,
            " expected=",copyCount,
            " err=",GetLastError());
      return(prev_calculated);
   }

   ArrayResize(BufferKCUpper, rates_total);
   ArrayResize(BufferKCMiddle, rates_total);
   ArrayResize(BufferKCLower, rates_total);

   for(int k = 0; k < copyCount; k++)
   {
      BufferTEMA[copyStart + k] = temaTail[k];
      BufferEMAofTEMA[copyStart + k] = emaTail[k];
      BufferKCUpper[copyStart + k] = kcUpperTail[k];
      BufferKCMiddle[copyStart + k] = kcMiddleTail[k];
      BufferKCLower[copyStart + k] = kcLowerTail[k];
   }

   double pointSize = MathPow(10.0, -_Digits);
   double eps = pointSize * 0.1;
   double pipSize = pointSize * 10.0;
   bool bDoHistory = (prev_calculated == 0);

   for(int i = start; i < rates_total; i++)
   {
      if(prev_calculated == 0 || i >= prev_calculated)
      {
         BufferArrowBuy[i] = EMPTY_VALUE;
         BufferArrowSell[i] = EMPTY_VALUE;
      }

      int prevRel = RelSign(BufferTEMA[i - 1] - BufferEMAofTEMA[i - 1], eps);
      int currRel = RelSign(BufferTEMA[i] - BufferEMAofTEMA[i], eps);
      double temaEmaSpread = BufferTEMA[i] - BufferEMAofTEMA[i];
      BufferTemaEmaSpread[i] = PriceDeltaToReadout(temaEmaSpread);
      if(prevRel == 0)
         prevRel = iDir;
      if(currRel == 0)
         currRel = prevRel;

      //--- Cocking logic: evaluate local hi/lo candidates from prior 2 bars.
      //--- Must run before cross handling so same-bar cock+cross can trigger.
      if(iDir != 0)
      {
         bool candidateFound = false;
         bool cockedThisBar = false;

         for(int d = 1; d <= 2; d++)
         {
            int cand = i - d;
            if(cand <= g_crossBar || cand < 1 || cand + 1 >= rates_total)
               continue;

            candidateFound = true;

            int refBar = MathMax(0, cand - InpOpenRefOffset);
            double refOpen = open[refBar];

            //--- Keep debug/state fields aligned with the candidate under evaluation.
            g_extremeBar = cand;
            g_extremeOpenRefBar = refBar;
            g_extremeOpen = refOpen;
            g_extremePrice = (iDir > 0) ? high[cand] : low[cand];

            if((iDir > 0 && close[i] < refOpen) ||
               (iDir < 0 && close[i] > refOpen))
            {
               g_isCocked = true;
               g_cockedDir = (iDir > 0) ? -1 : 1;
               g_cockLevel = refOpen;
               cockedThisBar = true;
               break;
            }
         }

         g_pendingConfirm = (candidateFound && !cockedThisBar);
      }

      bool crossUp = (prevRel < 0 && currRel > 0);
      bool crossDn = (prevRel > 0 && currRel < 0);
      bool isCross = (crossUp || crossDn);

      int activeBlackout = InpSignalBlackout;
      double currWidthPips = EMPTY_VALUE;
      double widthRatio = EMPTY_VALUE;
      double slopePipsPerBar = EMPTY_VALUE;
      if(BufferKCUpper[i] != EMPTY_VALUE && BufferKCLower[i] != EMPTY_VALUE && BufferKCMiddle[i] != EMPTY_VALUE)
      {
         currWidthPips = PriceDeltaToReadout(MathAbs(BufferKCUpper[i] - BufferKCLower[i]));

         int lb = MathMin(InpKCWidthLookback, i + 1);
         double widthSumPips = 0.0;
         int widthCnt = 0;
         int from = i - lb + 1;
         if(from < 0) from = 0;
         for(int j = from; j <= i; j++)
         {
            if(BufferKCUpper[j] == EMPTY_VALUE || BufferKCLower[j] == EMPTY_VALUE)
               continue;
            widthSumPips += PriceDeltaToReadout(MathAbs(BufferKCUpper[j] - BufferKCLower[j]));
            widthCnt++;
         }

         double baselineWidthPips = (widthCnt > 0) ? (widthSumPips / widthCnt) : currWidthPips;
         widthRatio = (baselineWidthPips > 0.0) ? (currWidthPips / baselineWidthPips) : 1.0;

         slopePipsPerBar = 0.0;
         if(i >= InpKCMidSlopePeriod && BufferKCMiddle[i - InpKCMidSlopePeriod] != EMPTY_VALUE)
            slopePipsPerBar = PriceDeltaToReadout((BufferKCMiddle[i] - BufferKCMiddle[i - InpKCMidSlopePeriod]) / InpKCMidSlopePeriod);

         if(InpUseDynamicBlackout)
         {
            bool isNarrow = (widthRatio < InpNarrowWidthRatio);
            bool isFlat = (MathAbs(slopePipsPerBar) <= InpFlatSlopePipsPerBar);
            if(isNarrow && isFlat)
            {
               double extra = (InpNarrowWidthRatio - widthRatio) * InpBlackoutScaleK;
               if(extra < 0.0) extra = 0.0;
               activeBlackout = InpSignalBlackout + (int)MathRound(extra);
               if(activeBlackout > InpBlackoutMax)
                  activeBlackout = InpBlackoutMax;
            }
         }
      }
      BufferDynBlackout[i] = activeBlackout;
      BufferKCWidthRatio[i] = widthRatio;
      BufferTriggerThreshold[i] = InpTriggerSensitivityPips;

      if(iDir == 0 && currRel != 0)
         ResetCycleForDirection(currRel, i, open, high, low);

      ENUM_SIGNAL_TYPE signal = SIGNAL_NONE;
      ENUM_SIGNAL_TYPE signalCandidate = SIGNAL_NONE;
      if(isCross)
      {
         if(g_isCocked)
         {
            if(crossDn && g_cockedDir == -1)
            {
               g_triggerPending = true;
               g_triggerDir = -1;
               g_triggerCrossBar = i;
            }
            else if(crossUp && g_cockedDir == 1)
            {
               g_triggerPending = true;
               g_triggerDir = 1;
               g_triggerCrossBar = i;
            }
            else
            {
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
         }
         else
         {
            g_triggerPending = false;
            g_triggerDir = 0;
            g_triggerCrossBar = -1;
         }
      }

      if(g_triggerPending)
      {
         bool wrongSide = ((g_triggerDir == 1 && currRel < 0) || (g_triggerDir == -1 && currRel > 0));
         if(wrongSide)
         {
            g_triggerPending = false;
            g_triggerDir = 0;
            g_triggerCrossBar = -1;
         }
         else
         {
            double cumulativeSpreadPips = 0.0;
            int sumFrom = MathMax(0, g_triggerCrossBar);
            if(sumFrom <= i)
            {
               for(int s = sumFrom; s <= i; s++)
               {
                  if(BufferTemaEmaSpread[s] == EMPTY_VALUE)
                     continue;
                  cumulativeSpreadPips += MathAbs(BufferTemaEmaSpread[s]);
               }
            }

            if(cumulativeSpreadPips >= InpTriggerSensitivityPips)
            {
               signalCandidate = (g_triggerDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
               g_triggerPending = false;
               g_triggerDir = 0;
               g_triggerCrossBar = -1;
            }
         }
      }

      bool canFireNow = ((i - g_lastSignalBar) > activeBlackout && time[i] >= InpTradeStartDate);
      if(signalCandidate != SIGNAL_NONE)
      {
         if(canFireNow)
         {
            signal = signalCandidate;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else
         {
            g_blackoutSignalPending = true;
            g_blackoutSignalDir = (signalCandidate == SIGNAL_LONG) ? 1 : -1;
            g_blackoutSignalBar = i;
         }
      }

      if(signal == SIGNAL_NONE && g_blackoutSignalPending)
      {
         bool retraced = ((g_blackoutSignalDir == 1 && currRel < 0) || (g_blackoutSignalDir == -1 && currRel > 0));
         if(retraced)
         {
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
         else if(canFireNow)
         {
            signal = (g_blackoutSignalDir == 1) ? SIGNAL_LONG : SIGNAL_SHORT;
            g_blackoutSignalPending = false;
            g_blackoutSignalDir = 0;
            g_blackoutSignalBar = -1;
         }
      }

      if(signal != SIGNAL_NONE)
      {
         if(g_signalStartDT == 0)
            g_signalStartDT = time[i];

         double mockExecPrice = GetMockTradePrice(i, close);
         ENUM_TS_READOUT_MODE tsReadoutMode = TSReadoutMode();

         if(InpEnableTradeSim)
         {
            if(InpMockTradePriceSource == MOCK_PRICE_TEMA)
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferTEMA,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell,
                                 tsReadoutMode);
            }
            else if(InpMockTradePriceSource == MOCK_PRICE_EMA_TEMA)
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 BufferEMAofTEMA,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell,
                                 tsReadoutMode);
            }
            else
            {
               HandleSignalEvent(i, signal,
                                 true,              // closeTrade
                                 true,              // fillBuffers
                                 true,              // openTrade
                                 true,              // updateLabel
                                 InpShowArrows,     // showArrows
                                 time,
                                 close,
                                 g_signalStartDT,
                                 g_currentTrade,
                                 g_tradeStats,
                                 g_arrowCfg,
                                 g_labelCfg,
                                 (InpSoundTradeAlarm && !bDoHistory),
                                 BufferTradePL,
                                 BufferRunningPL,
                                 BufferTradeCount,
                                 BufferWinPct,
                                 BufferShortCount,
                                 BufferShortWinPct,
                                 BufferLongCount,
                                 BufferLongWinPct,
                                 BufferArrowBuy,
                                 BufferArrowSell,
                                 tsReadoutMode);
            }
         }

         //--- Buffer arrows should be anchored at selected mock execution price for the signal bar.
         if(signal == SIGNAL_LONG)
            BufferArrowBuy[i] = mockExecPrice;
         else
            BufferArrowSell[i] = mockExecPrice;

         //--- Reposition object arrows 100 pips outside the farther of KC outer line or TEMA from KC midline.
         if(InpShowArrows && g_currentTrade.arrowName != "" && ObjectFind(0, g_currentTrade.arrowName) >= 0 &&
            BufferKCUpper[i] != EMPTY_VALUE && BufferKCMiddle[i] != EMPTY_VALUE &&
            BufferKCLower[i] != EMPTY_VALUE && BufferTEMA[i] != EMPTY_VALUE)
         {
            double mid = BufferKCMiddle[i];
            double arrowPrice = 0.0;
            double offset = InpObjectArrowOffsetPips * pipSize;

            if(signal == SIGNAL_SHORT)
            {
               double baseUpper = BufferKCUpper[i];
               double distUpper = MathAbs(baseUpper - mid);
               double distTema  = MathAbs(BufferTEMA[i] - mid);
               double base = (distTema > distUpper) ? BufferTEMA[i] : baseUpper;
               arrowPrice = base + offset;
               BufferArrowSell[i] = mockExecPrice;
            }
            else
            {
               double baseLower = BufferKCLower[i];
               double distLower = MathAbs(baseLower - mid);
               double distTema  = MathAbs(BufferTEMA[i] - mid);
               double base = (distTema > distLower) ? BufferTEMA[i] : baseLower;
               arrowPrice = base - offset;
               BufferArrowBuy[i] = mockExecPrice;
            }

            ObjectMove(0, g_currentTrade.arrowName, 0, time[i], arrowPrice);
         }

         g_lastSignalBar = i;
      }

      //--- Keep the most recent open trade tooltip/buffers live on the current bar.
      if(i == rates_total - 1 && InpEnableTradeSim)
      {
         double liveExitPrice = GetMockTradePrice(i, close);
         UpdateOpenTradeLiveState(i, liveExitPrice);
      }

      //--- Uncock only after trigger decisions on this bar.
      if(g_isCocked)
      {
         if(g_cockedDir == -1 && close[i] > g_cockLevel)
         {
            g_isCocked = false;
            g_cockedDir = 0;
         }
         else if(g_cockedDir == 1 && close[i] < g_cockLevel)
         {
            g_isCocked = false;
            g_cockedDir = 0;
         }
      }

      if(i < rates_total - 1 && i > g_lastExportedBar)
      {
         AppendCsvDebugRow(time[i], i,

//| TS_DetectSignalUP_DN - array-based signal detection              |
//+------------------------------------------------------------------+
ENUM_SIGNAL_TYPE TS_DetectSignalUP_DN(const double &slopeValues[],
                                      const bool &tracked[],
                                      int bufferCount,
                                      int thresholdPercent,
                                      int &direction,
                                      double signalOFS = 0.0)
{
   int bullishCount = 0;
   int bearishCount = 0;
   int activeBuffers = 0;

   for(int i = 0; i < bufferCount; i++)
   {
      if(!tracked[i])
         continue;

      double currValue = slopeValues[i];
      if(currValue == EMPTY_VALUE)
         continue;

      activeBuffers++;

      if(currValue > signalOFS)
         bullishCount++;
      else if(currValue < -signalOFS)
         bearishCount++;
   }

   if(activeBuffers == 0)
      return SIGNAL_NONE;

   int bullishPercent = (bullishCount * 100) / activeBuffers;
   int bearishPercent = (bearishCount * 100) / activeBuffers;

   if(bearishPercent >= thresholdPercent && direction >= 0)
   {
      direction = -1;
      return SIGNAL_SHORT;
   }
   else if(bullishPercent >= thresholdPercent && direction <= 0)
   {
      direction = 1;
      return SIGNAL_LONG;
   }

   return SIGNAL_NONE;
}

                       const double &executionPriceSeries[],

   {

      closed = true;

      UpdateClosedTradeTooltip(currentTrade.arrowName, currentTrade.entryPrice,
                               exitPrice, result.tradePL, result.runningPL, readoutMode);
   }

   //--- Fill buffers
   if(fillBuffers && closed)
   {
      if(bar >= ArraySize(bufPL))
         TS_GrowStatsArrays(bar + 1, bufPL, bufRunning, bufCount, bufWinPct,
                            bufShortCount, bufShortWinPct, bufLongCount, bufLongWinPct);

      int signalStartBar = iBarShiftREG(_Symbol, PERIOD_CURRENT, pdtSignalStartDT);
      int fillStart = MathMax(currentTrade.entryBar + 1, signalStartBar);
      FillTradeBuffers(fillStart, bar, result,
                       bufPL, bufRunning, bufCount, bufWinPct,
                       bufShortCount, bufShortWinPct, bufLongCount, bufLongWinPct);
   }

   //--- Update label
   if(updateLabel)
   {
      UpdateTradeLabel(labelCfg, stats.runningPL, pdtSignalStartDT, readoutMode);
   }

   //--- Open new trade
   if(openTrade)
   {
      currentTrade = OpenTrade(bar, signal, exitPrice, time[bar], arrowCfg, currentTrade.direction, pbSoundAlarm, showArrows);
      stats.tradeCount++;
      
      //--- Populate arrow buffers with entry price
      if(signal == SIGNAL_LONG)
         bufArrowBuy[bar] = currentTrade.entryPrice;
      else
         bufArrowSell[bar] = currentTrade.entryPrice;
   }
