struct SPatternBlock

//| Parse color string to color type                                   |
//+------------------------------------------------------------------+
color ParseColorString(string colorStr)
{
   StringToLower(colorStr);
   
   if(colorStr == "green")       return clrGreen;
   if(colorStr == "red")         return clrRed;
   if(colorStr == "lime")        return clrLime;
   if(colorStr == "maroon")      return clrMaroon;
   if(colorStr == "cyan")        return clrCyan;
   if(colorStr == "magenta")     return clrMagenta;
   if(colorStr == "yellow")      return clrYellow;
   if(colorStr == "purple")      return clrPurple;
   if(colorStr == "blue")        return clrBlue;
   if(colorStr == "orange")      return clrOrange;
   if(colorStr == "white")       return clrWhite;
   if(colorStr == "gray" || colorStr == "grey") return clrGray;
   
   return clrGray;
}

//| Read pattern data from CSV                                         |
//+------------------------------------------------------------------+
int ReadPatternBlocks(string filename, SPatternBlock &blocks[])
{
//   int fileHandle = FileOpen(filename, FILE_READ|FILE_COMMON|FILE_CSV, ',', CP_ACP);
   int fileHandle = FileOpen(filename, FILE_READ|FILE_COMMON|FILE_CSV|FILE_ANSI, ',');
   if(fileHandle == INVALID_HANDLE)
   {
      Print("Error opening file: ", filename);
      return 0;
   }
   
   Print("File opened successfully, reading header...");
   
   //--- Skip header
   if(!FileIsEnding(fileHandle))
   {
      string header1 = FileReadString(fileHandle); // pattern_type
      string header2 = FileReadString(fileHandle); // start_time
      string header3 = FileReadString(fileHandle); // end_time
      string header4 = FileReadString(fileHandle); // high_price
      string header5 = FileReadString(fileHandle); // low_price
      string header6 = FileReadString(fileHandle); // color
      string header7 = FileReadString(fileHandle); // description
      Print("Header: ", header1, ",", header2, ",", header3);
   }
   else
   {
      Print("File is empty after opening!");
   }
   
   int count = 0;
   ArrayResize(blocks, 0);
   
   Print("Starting data read loop, FileIsEnding=", FileIsEnding(fileHandle));
   
   while(!FileIsEnding(fileHandle))
   {
      string patternType = FileReadString(fileHandle);
      if(FileIsEnding(fileHandle)) break;
      
      string startTimeStr = FileReadString(fileHandle);
      string endTimeStr = FileReadString(fileHandle);
      string highPriceStr = FileReadString(fileHandle);
      string lowPriceStr = FileReadString(fileHandle);
      string colorStr = FileReadString(fileHandle);
      string description = FileReadString(fileHandle);
      
      if(count < 3)
         Print("Row ", count, ": patternType='", patternType, "', startTimeStr='", startTimeStr, "', endTimeStr='", endTimeStr, "'");
      
      //--- Parse data (MQL5 StringToTime expects "yyyy.mm.dd hh:mi", remove seconds)
      if(StringLen(startTimeStr) > 16) startTimeStr = StringSubstr(startTimeStr, 0, 16);
      if(StringLen(endTimeStr) > 16) endTimeStr = StringSubstr(endTimeStr, 0, 16);
      datetime startTime = StringToTime(startTimeStr);
      datetime endTime = StringToTime(endTimeStr);
      double highPrice = StringToDouble(highPriceStr);
      double lowPrice = StringToDouble(lowPriceStr);
      
      if(startTime == 0 || endTime == 0)
      {
         Print("Skipping row ", count, ": invalid datetime. startTimeStr='", startTimeStr, "', endTimeStr='", endTimeStr, "'");
         continue;
      }
      
      //--- Add to array
      int idx = ArraySize(blocks);
      ArrayResize(blocks, idx + 1);
      
      blocks[idx].patternType = patternType;
      blocks[idx].startTime = startTime;
      blocks[idx].endTime = endTime;
      blocks[idx].highPrice = highPrice;
      blocks[idx].lowPrice = lowPrice;
      blocks[idx].boxColor = ParseColorString(colorStr);
      blocks[idx].description = description;
      
      count++;
   }
   
   FileClose(fileHandle);
   Print("ReadPatternBlocks complete. Total rows read: ", count);
   return count;
}

//| Draw pattern block on chart                                        |
//+------------------------------------------------------------------+
void DrawPatternBlock(const SPatternBlock &block, int index)
{
   string rectName = InpObjectPrefix + "Rect_" + IntegerToString(index);
   string labelName = InpObjectPrefix + "Label_" + IntegerToString(index);
   
   //--- Add padding to price range
   double range = block.highPrice - block.lowPrice;
   double padding = range * 0.05;
   double topPrice = block.highPrice + padding;
   double bottomPrice = block.lowPrice - padding;
   
   //--- Create rectangle
   if(ObjectCreate(0, rectName, OBJ_RECTANGLE, 0, block.startTime, topPrice, block.endTime, bottomPrice))
   {
      ObjectSetInteger(0, rectName, OBJPROP_COLOR, block.boxColor);
      ObjectSetInteger(0, rectName, OBJPROP_FILL, true);
      ObjectSetInteger(0, rectName, OBJPROP_BACK, true);
      ObjectSetInteger(0, rectName, OBJPROP_SELECTABLE, true);
      ObjectSetInteger(0, rectName, OBJPROP_SELECTED, true);
      ObjectSetInteger(0, rectName, OBJPROP_HIDDEN, false);
      
      //--- Build tooltip
      string tooltip = block.description + "\n" +
                      "Type: " + block.patternType + "\n" +
                      "Start: " + TimeToString(block.startTime) + "\n" +
                      "End: " + TimeToString(block.endTime) + "\n" +
                      "High: " + DoubleToString(block.highPrice, _Digits) + "\n" +
                      "Low: " + DoubleToString(block.lowPrice, _Digits);
      
      ObjectSetString(0, rectName, OBJPROP_TOOLTIP, tooltip);
   }
   else
   {
      Print("Failed to create rectangle: ", rectName);
   }
   
   //--- Create label if enabled
   if(InpShowLabels)
   {
      datetime midTime = block.startTime + (block.endTime - block.startTime) / 2;
      double midPrice = (topPrice + bottomPrice) / 2;
      
      if(ObjectCreate(0, labelName, OBJ_TEXT, 0, midTime, midPrice))
      {
         ObjectSetString(0, labelName, OBJPROP_TEXT, block.description);
         ObjectSetInteger(0, labelName, OBJPROP_COLOR, clrWhite);
         ObjectSetInteger(0, labelName, OBJPROP_FONTSIZE, InpLabelFontSize);
         ObjectSetInteger(0, labelName, OBJPROP_ANCHOR, ANCHOR_CENTER);
         ObjectSetInteger(0, labelName, OBJPROP_BACK, true);
      }
   }
}
