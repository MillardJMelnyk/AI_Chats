void InsertSorted(double &arr[], int &count, double value)
{
   ArrayResize(arr, count + 1);
   int i = count;
   while(i > 0 && arr[i-1] > value)
   {
      arr[i] = arr[i-1];
      i--;
   }
   arr[i] = value;
   count++;
}

//+------------------------------------------------------------------+
double GetPercentileValue(const double &arr[], int count, double percentile)
{
   if(count <= 0) return 0.0;
   if(percentile <= 0.0) return arr[0];
   if(percentile >= 100.0) return arr[count - 1];
   
   double rank = (percentile / 100.0) * (count - 1);
   int lo = (int)MathFloor(rank);
   int hi = (int)MathCeil(rank);
   
   if(lo < 0) lo = 0;
   if(hi >= count) hi = count - 1;
   if(hi == lo) return arr[lo];
   
   double frac = rank - lo;
   return arr[lo] + (arr[hi] - arr[lo]) * frac;
}

//+------------------------------------------------------------------+
string FormatWithCommas(double value, int decimals = 1)
{
   string str = DoubleToString(MathAbs(value), decimals);
   string sign = (value < 0) ? "-" : "";
   
   int dot_pos = StringFind(str, ".");
   string int_part = (dot_pos >= 0) ? StringSubstr(str, 0, dot_pos) : str;
   string dec_part = (dot_pos >= 0) ? StringSubstr(str, dot_pos) : "";
   
   string result = "";
   int len = StringLen(int_part);
   for(int i = 0; i < len; i++)
   {
      if(i > 0 && (len - i) % 3 == 0)
         result += ",";
      result += StringSubstr(int_part, i, 1);
   }
   return sign + result + dec_part;
}

void BuildDistribution(int rates_total)
{
   if(g_pivot_count < 2)
   {
      g_leg_count = 0;
      g_distribution_ready = false;
      return;
   }
   
   ArrayResize(g_leg_sizes, 0);
   g_leg_count = 0;
   
   double pip_size = GetPipSize();
   int confirmed_count = g_pivot_count - 1;
   
   int target_legs = (Min_ZZ_Legs == 0) ? (confirmed_count - 1) : Min_ZZ_Legs;
   int legs_with_priming = target_legs + PRIMING_BUFFER;
   
   int start_pivot_idx = confirmed_count - 1;
   int legs_counted = 0;
   
   for(int i = start_pivot_idx; i > 0 && legs_counted < legs_with_priming; i--)
   {
      double leg_size_pips = MathAbs(g_pivots[i].price - g_pivots[i-1].price) / pip_size;
      if(leg_size_pips > 0)
      {
         InsertSorted(g_leg_sizes, g_leg_count, leg_size_pips);
         legs_counted++;
         
         if(!g_initial_build_done && (legs_counted % 10 == 0 || legs_counted >= target_legs))
         {
            int display_count = MathMin(legs_counted, target_legs);
            double pct = MathMin(100.0, (display_count * 100.0) / target_legs);
            string status = StringFormat("Building distribution... %d/%d legs (%.1f%%)", 
                                        display_count, target_legs, pct);
            UpdateStatusMeter(status);
         }
         
         if(legs_counted == legs_with_priming)
            g_distribution_start_pivot_idx = i - 1;
      }
   }
   
   if(g_leg_count >= target_legs)
   {
      g_distribution_ready = true;
      
      int legs_for_start = (Min_ZZ_Legs == 0) ? 1 : Min_ZZ_Legs;
      int legs_back = 0;
      for(int i = confirmed_count - 1; i > 0; i--)
      {
         legs_back++;
         if(legs_back >= legs_for_start)
         {
            g_indicator_start_bar = g_pivots[i].bar;
            // CRITICAL FIX: Clamp to valid range
            if(g_indicator_start_bar >= rates_total)
               g_indicator_start_bar = rates_total - 1;
            break;
         }
      }
   }
}

void UpdateInWorkLeg(int bar, const double &high[], const double &low[])
{
   if(g_pivot_count < 2) return;
   
   int last_pivot_idx = g_pivot_count - 1;
   int leg_start_bar = g_pivots[last_pivot_idx].bar;
   double leg_start_price = g_pivots[last_pivot_idx].price;
   
   // Check if new leg started
   if(!g_inwork_active || g_inwork_leg_start_bar != leg_start_bar)
   {
      // Determine direction from previous pivot
      g_inwork_direction = 0;
      if(last_pivot_idx > 0)
      {
         double prev_price = g_pivots[last_pivot_idx - 1].price;
         if(leg_start_price > prev_price)
            g_inwork_direction = -1;  // Peak -> down leg
         else if(leg_start_price < prev_price)
            g_inwork_direction = 1;   // Trough -> up leg
      }
      
      g_inwork_active = true;
      g_inwork_leg_start_bar = leg_start_bar;
      g_inwork_leg_start_price = leg_start_price;
      g_inwork_extreme = leg_start_price;
   }
   
   // Update extreme incrementally
   if(g_inwork_direction > 0)
   {
      if(high[bar] > g_inwork_extreme)
         g_inwork_extreme = high[bar];
   }
   else if(g_inwork_direction < 0)
   {
      if(low[bar] < g_inwork_extreme)
         g_inwork_extreme = low[bar];
   }
}

void CalculateIndicatorForBar(int bar, const double &high[], const double &low[], int rates_total)
{
   Historic_Buffer[bar] = EMPTY_VALUE;
   Combined_Buffer[bar] = EMPTY_VALUE;
   Combined_Color_Buffer[bar] = 0;
   Debug_PivotIdx[bar] = -1;
   Debug_LegSize[bar] = 0;
   Debug_LegStart[bar] = 0;
   Debug_LegDistCount[bar] = g_leg_count;
   Debug_PivotCount[bar] = g_pivot_count;
   CumulativePL[bar] = g_cum_pl;
   CumulativeGain[bar] = g_cum_gain;
   CumulativeLoss[bar] = g_cum_loss;
   
   if(!g_distribution_ready) return;
   if(bar < g_indicator_start_bar) return;
   if(g_pivot_count < 2) return;
   
   UpdateInWorkLeg(bar, high, low);
   
   if(!g_inwork_active || g_inwork_direction == 0) return;
   
   double pip_size = GetPipSize();
   double leg_size_pips = MathAbs(g_inwork_extreme - g_inwork_leg_start_price) / pip_size;
   
   double target = GetPercentileValue(g_leg_sizes, g_leg_count, ConfidenceTargetPercentile);
   if(target <= 0) return;
   
   double confidence = MathMin(100.0, 100.0 * (leg_size_pips / target));
   
   Historic_Buffer[bar] = confidence;
   Combined_Buffer[bar] = confidence;
   Combined_Color_Buffer[bar] = (g_inwork_direction > 0) ? 1.0 : 0.0;
   Debug_PivotIdx[bar] = g_inwork_leg_start_bar;
   Debug_LegSize[bar] = leg_size_pips;
   Debug_LegStart[bar] = g_inwork_leg_start_price;
}

   if(g_pivot_count == 0) return;
   
   int last_idx = g_pivot_count - 1;
   int current_bar = g_pivots[last_idx].bar;
   double current_price = g_pivots[last_idx].price;
   
   if(g_last_pivot_bar < 0)
   {
      g_last_pivot_bar = current_bar;
      g_last_pivot_price = current_price;
      return;
   }
   
   if(current_bar != g_last_pivot_bar || MathAbs(current_price - g_last_pivot_price) > _Point)
   {
      Print("Last pivot CHANGED: was bar ", g_last_pivot_bar, 
            " @ ", DoubleToString(g_last_pivot_price, _Digits),
            " now bar ", current_bar,
            " @ ", DoubleToString(current_price, _Digits));
      
      BuildDistribution(rates_total);
      RebuildMockTrades(rates_total, time, high, low);
      
      int recalc_from = MathMin(g_last_pivot_bar, current_bar);
      RecalculateFromLegStart(recalc_from, rates_total, high, low);
      
      g_last_pivot_bar = current_bar;
      g_last_pivot_price = current_price;
   }
}
