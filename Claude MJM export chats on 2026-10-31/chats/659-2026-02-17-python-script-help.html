"""
TEMA/KC Pattern Analyzer
L0: PL slope dynamics (SL-agnostic)
L1: Shape detection (crosses/pivots via spread delta)
L2: Pattern detection (frequency analysis of shape sequences)

Input: CSV from MT5 with Time, OHLC, TEMA, KC_Upper, KC_Middle, KC_Lower
Output: Bar-by-bar L0/L1 data, then L2 pattern catalog

def round_to_point(value: float, point: float = 0.001) -> float:
    """Round to nearest point (0.001 for USDJPY)."""
    return round(value / point) * point


def calculate_spreads(tema: float, kc_upper: float, kc_middle: float, kc_lower: float, point: float) -> dict:
    """Calculate spreads from TEMA to each KC line, rounded to point."""
    return {
        'spread_upper': round_to_point(tema - kc_upper, point),
        'spread_middle': round_to_point(tema - kc_middle, point),
        'spread_lower': round_to_point(tema - kc_lower, point)
    }


def determine_active_line(spreads: dict) -> Tuple[str, str]:
    """
    Determine active line (nearest) and zone.
    Returns (active_line, zone) where active_line and zone are U, M, or L.
    """
    abs_spreads = {
        'U': abs(spreads['spread_upper']),
        'M': abs(spreads['spread_middle']),
        'L': abs(spreads['spread_lower'])
    }
    active = min(abs_spreads, key=abs_spreads.get)
    return active, active  # zone = active line


def calculate_spread_delta_state(spread_curr: float, spread_prev: float) -> str:
    """
    Calculate L1 spread delta state.
    SD+: diverging (spread increasing in absolute terms, same sign)
    SD-: converging (spread decreasing in absolute terms, same sign)
    SD0: parallel (spread unchanged)
    SDX: cross (spread sign flipped)
    """
    if spread_curr == 0 and spread_prev == 0:
        return 'SD0'
    
    # Check for cross (sign flip)
    if spread_prev != 0 and spread_curr != 0:
        if (spread_curr > 0) != (spread_prev > 0):
            return 'SDX'
    elif spread_prev == 0 or spread_curr == 0:
        # One is zero - edge case, treat as cross if other is non-zero
        if spread_prev != 0 or spread_curr != 0:
            return 'SDX'
    
    # Same sign - check converging/diverging
    delta = abs(spread_curr) - abs(spread_prev)
    
    if delta > 0:
        return 'SD+'  # diverging
    elif delta < 0:
        return 'SD-'  # converging
    else:
        return 'SD0'  # parallel


def calculate_l0_state(slope_delta: float) -> int:
    """
    Calculate L0 state from slope delta.
    +1: pivot toward up
     0: continuation
    -1: pivot toward down
    """
    if slope_delta > 0:
        return 1
    elif slope_delta < 0:
        return -1
    else:
        return 0

def pass2_l1_calculation(df: pd.DataFrame) -> pd.DataFrame:

    Pass 2: Calculate L1 spread delta state and zone switch flag.
    Requires bar 2 onward (index 1+).

    df = df.copy()

    df['active_spread'] = None
    df['active_spread_prev'] = None

        curr = df.iloc[i]

            spread_curr = curr['spread_upper']
        elif curr['active_line'] == 'M':
            spread_curr = curr['spread_middle']
        else:
            spread_curr = curr['spread_lower']

        df.at[i, 'active_spread'] = spread_curr

    return df


def pass3_l0_calculation(df: pd.DataFrame, point: float = 0.001) -> pd.DataFrame:

def pass5_pattern_mining(df: pd.DataFrame, min_length: int = 2, min_frequency: int = 2) -> dict:
    """
    Pass 5: Mine patterns from symbol sequence using suffix-based approach.

    Instances reference df index positions.

    valid_mask = df['symbol'].notna()

    if len(symbols) < min_length:
        return {}
    
    pattern_data = defaultdict(lambda: {'count': 0, 'instances': []})
    
    # Sliding window approach for pattern counting - no max_length limit
    for length in range(min_length, len(symbols) + 1):
        if length % 1000 == 0:
            print(f"\n  Mining progress: checking length {length}", end='')
        if length % 1000 == 0:
            print(".", end='', flush=True)
        for start in range(len(symbols) - length + 1):
            if start % 1000 == 0:

            start_idx = indices[start]
            end_idx = indices[start + length - 1]
            pattern_data[pattern]['count'] += 1
            pattern_data[pattern]['instances'].append((start_idx, end_idx))
    
    # Filter by minimum frequency
    significant_patterns = {
        pattern: data

    }
    
    return significant_patterns

By the time it's halfway through, it has accumulated counts and location lists for millions of distinct patterns — most of which appear only once and will be thrown away at the end when the min_frequency filter runs. All those location lists for throwaway patterns are sitting in memory doing nothing useful. As that pile grows, Python periodically stops the program to reclaim that memory. That's the freeze.

What the two-pass fix does:

Sub-pass A runs the exact same nested loops, but only increments counts. No location lists are stored at all. This is fast and stays lean in memory throughout.

At the end of sub-pass A, we know exactly which patterns meet min_frequency. That's typically a small fraction of the total.

Sub-pass B runs the same nested loops again, but this time only records location lists for patterns already known to qualify. Everything else is ignored. Since qualifying patterns are a small minority, very little is stored.

Total work is roughly doubled (two full passes), but memory stays controlled the entire time, so no freezes.

The trade-off: it takes twice as long on the iteration count, but in practice sub-pass B does far less storage work than the current single pass, so wall-clock time will likely be similar or better — and no stalls.

L0: PL slope dynamics (SL-agnostic)
L1: Shape detection (crosses/pivots via spread delta)
L2: Pattern detection (frequency analysis of shape sequences)

class ProgressTracker:
    """
    Single-line overwriting progress display.
    Call tick() each iteration (or tick(n) for batches).
    Call done() when finished.
    Prints at most once every `interval` seconds to reduce I/O overhead.
    """

    def __init__(self, label: str, total: int, interval: float = 2.0):
        self.label    = label
        self.total    = total
        self.interval = interval
        self.completed = 0
        self.start    = time.monotonic()
        self.last_print = 0.0  # force first print after first tick

    @staticmethod
    def _fmt_seconds(s: float) -> str:
        s = int(s)
        h, rem = divmod(s, 3600)
        m, sec = divmod(rem, 60)
        return f"{h:02d}:{m:02d}:{sec:02d}"

    def tick(self, n: int = 1) -> None:
        self.completed += n
        now = time.monotonic()
        if now - self.last_print < self.interval:
            return
        self.last_print = now
        elapsed = now - self.start

        pct = self.completed / self.total * 100 if self.total > 0 else 0.0
        elapsed_str = self._fmt_seconds(elapsed)

        if elapsed > 0 and self.completed > 0:
            rate = self.completed / elapsed
            remaining = (self.total - self.completed) / rate
            eta_str = self._fmt_seconds(remaining)
        else:
            eta_str = "--:--:--"

        line = (f"\r{self.label}: {self.completed:,} / {self.total:,} "
                f"({pct:.1f}%) | {elapsed_str} elapsed | ETA: {eta_str} remaining   ")
        print(line, end='', flush=True)

    def done(self) -> None:
        elapsed = time.monotonic() - self.start
        elapsed_str = self._fmt_seconds(elapsed)
        line = (f"\r{self.label}: {self.total:,} / {self.total:,} "
                f"(100.0%) | {elapsed_str} elapsed | done                          ")
        print(line, flush=True)

def round_to_point(value: float, point: float = 0.001) -> float:
    """Round to nearest point (0.001 for USDJPY)."""
    return round(value / point) * point


def calculate_spreads(tema: float, kc_upper: float, kc_middle: float, kc_lower: float, point: float) -> dict:
    """Calculate spreads from TEMA to each KC line, rounded to point."""
    return {
        'spread_upper':  round_to_point(tema - kc_upper,  point),
        'spread_middle': round_to_point(tema - kc_middle, point),
        'spread_lower':  round_to_point(tema - kc_lower,  point)
    }


def determine_active_line(spreads: dict) -> Tuple[str, str]:
    abs_spreads = {
        'U': abs(spreads['spread_upper']),
        'M': abs(spreads['spread_middle']),
        'L': abs(spreads['spread_lower'])
    }
    active = min(abs_spreads, key=abs_spreads.get)
    return active, active


def calculate_spread_delta_state(spread_curr: float, spread_prev: float) -> str:
    if spread_curr == 0 and spread_prev == 0:
        return 'SD0'
    if spread_prev != 0 and spread_curr != 0:
        if (spread_curr > 0) != (spread_prev > 0):
            return 'SDX'
    elif spread_prev == 0 or spread_curr == 0:
        if spread_prev != 0 or spread_curr != 0:
            return 'SDX'
    delta = abs(spread_curr) - abs(spread_prev)
    if delta > 0:
        return 'SD+'
    elif delta < 0:
        return 'SD-'
    else:
        return 'SD0'


def calculate_l0_state(slope_delta: float) -> int:
    if slope_delta > 0:
        return 1
    elif slope_delta < 0:
        return -1
    else:
        return 0

def pass1_raw_calculations(df: pd.DataFrame, point: float = 0.001) -> pd.DataFrame:
    n = len(df)
    prog = ProgressTracker("Pass 1 (spreads/zones)", n)
    results = []
    for i, row in df.iterrows():
        spreads = calculate_spreads(
            row['TEMA'], row['KC_Upper'], row['KC_Middle'], row['KC_Lower'], point
        )
        active_line, zone = determine_active_line(spreads)
        results.append({
            'bar_index':     i,
            'time':          row['Time'],
            'close':         row['Close'],
            'tema':          row['TEMA'],
            'kc_upper':      row['KC_Upper'],
            'kc_middle':     row['KC_Middle'],
            'kc_lower':      row['KC_Lower'],
            'spread_upper':  spreads['spread_upper'],
            'spread_middle': spreads['spread_middle'],
            'spread_lower':  spreads['spread_lower'],
            'active_line':   active_line,
            'zone':          zone
        })
        prog.tick()
    prog.done()
    return pd.DataFrame(results)


def pass2_l1_calculation(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df['L1_state']         = None
    df['zone_switch']      = 'N'
    df['active_spread']    = None
    df['active_spread_prev'] = None

    n = len(df)
    prog = ProgressTracker("Pass 2 (L1 states)  ", n - 1)
    for i in range(1, n):
        curr = df.iloc[i]
        prev = df.iloc[i - 1]
        al = curr['active_line']
        spread_key = {'U': 'spread_upper', 'M': 'spread_middle', 'L': 'spread_lower'}[al]
        spread_curr = curr[spread_key]
        spread_prev = prev[spread_key]
        l1_state    = calculate_spread_delta_state(spread_curr, spread_prev)
        zone_switch = 'Y' if curr['active_line'] != prev['active_line'] else 'N'
        df.at[i, 'L1_state']          = l1_state
        df.at[i, 'zone_switch']       = zone_switch
        df.at[i, 'active_spread']     = spread_curr
        df.at[i, 'active_spread_prev'] = spread_prev
        prog.tick()
    prog.done()
    return df


def pass3_l0_calculation(df: pd.DataFrame, point: float = 0.001) -> pd.DataFrame:
    df = df.copy()
    df['slope_1']    = None
    df['slope_2']    = None
    df['slope_delta'] = None
    df['L0_state']   = None

    n = len(df)
    prog = ProgressTracker("Pass 3 (L0 states)  ", n - 2)
    for i in range(2, n):
        tema_n  = df.iloc[i]['tema']
        tema_n1 = df.iloc[i - 1]['tema']
        tema_n2 = df.iloc[i - 2]['tema']
        slope_1     = round_to_point(tema_n1 - tema_n2, point)
        slope_2     = round_to_point(tema_n  - tema_n1, point)
        slope_delta = round_to_point(slope_2 - slope_1, point)
        df.at[i, 'slope_1']    = slope_1
        df.at[i, 'slope_2']    = slope_2
        df.at[i, 'slope_delta'] = slope_delta
        df.at[i, 'L0_state']   = calculate_l0_state(slope_delta)
        prog.tick()
    prog.done()
    return df


def pass4_symbol_encoding(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df['symbol'] = None

    n = len(df)
    prog = ProgressTracker("Pass 4 (symbols)    ", n - 2)
    for i in range(2, n):
        row = df.iloc[i]
        if pd.isna(row['L0_state']) or pd.isna(row['L1_state']):
            prog.tick()
            continue
        symbol = (f"L0:{int(row['L0_state']):+d}"
                  f"|L1:{row['L1_state']}"
                  f"|ZS:{row['zone_switch']}"
                  f"|Z:{row['zone']}")
        df.at[i, 'symbol'] = symbol
        prog.tick()
    prog.done()
    return df


def zigzag_max_length(df: pd.DataFrame, depth: int = 12, deviation: int = 5, backstep: int = 3, point: float = 0.001) -> int:
    """
    Compute ZigZag(depth, deviation, backstep) on High/Low.
    Track longest leg in bars as it builds. Return 2x that value as max_length.
    """
    highs  = df['High'].values
    lows   = df['Low'].values
    n      = len(highs)
    dev    = deviation * point

    pivots          = []
    last_pivot_bar  = 0
    last_pivot_price = 0.0
    last_direction  = 0  # 0=none, 1=high, -1=low
    max_leg_bars    = 0

    for i in range(depth, n):
        window_start = max(0, i - depth + 1)
        win_high_idx = window_start + int(np.argmax(highs[window_start:i + 1]))
        win_low_idx  = window_start + int(np.argmin(lows[window_start:i + 1]))
        win_high     = highs[win_high_idx]
        win_low      = lows[win_low_idx]

        new_pivot = None

        # Candidate swing high
        if win_high_idx == i:
            if last_direction != 1:
                if last_direction == -1 and (win_high - last_pivot_price) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = (i, win_high, 1)
                elif last_direction == 0:
                    new_pivot = (i, win_high, 1)
            elif last_direction == 1 and win_high > last_pivot_price:
                if pivots:
                    pivots[-1] = (i, win_high, 1)
                last_pivot_bar   = i
                last_pivot_price = win_high
                continue

        # Candidate swing low
        if win_low_idx == i:
            if last_direction != -1:
                if last_direction == 1 and (last_pivot_price - win_low) >= dev:
                    if (i - last_pivot_bar) >= backstep:
                        new_pivot = (i, win_low, -1)
                elif last_direction == 0:
                    new_pivot = (i, win_low, -1)
            elif last_direction == -1 and win_low < last_pivot_price:
                if pivots:
                    pivots[-1] = (i, win_low, -1)
                last_pivot_bar   = i
                last_pivot_price = win_low
                continue

        if new_pivot:
            if pivots:
                leg_bars = new_pivot[0] - pivots[-1][0]
                if leg_bars > max_leg_bars:
                    max_leg_bars = leg_bars
            pivots.append(new_pivot)
            last_pivot_bar   = new_pivot[0]
            last_pivot_price = new_pivot[1]
            last_direction   = new_pivot[2]

    max_length = max_leg_bars * 3
    print(f"ZigZag: longest leg = {max_leg_bars} bars → max_length = {max_length}")
    return max_length


def pass5_pattern_mining(df: pd.DataFrame, min_length: int = 2, min_frequency: int = 2, max_length: int = 0) -> dict:
    valid_mask = df['symbol'].notna()
    valid_df   = df[valid_mask].copy()
    symbols    = valid_df['symbol'].tolist()
    indices    = valid_df.index.tolist()
    n          = len(symbols)

    if n < min_length:
        return {}

    if max_length <= 0 or max_length > n // 2:
        max_length = n // 2
    print(f"Pass 5: {n:,} symbols, max_length={max_length:,}")

    total_iters = sum(n - l + 1 for l in range(min_length, max_length + 1))
    print(f"  → {total_iters:,} total iterations per sub-pass")

    TICK_BATCH = 500

    # ------------------------------------------------------------------
    # Sub-pass A: count only, no instance storage
    # ------------------------------------------------------------------
    counts     = defaultdict(int)
    prog       = ProgressTracker("Pass 5a (count)     ", total_iters)
    tick_accum = 0

    for length in range(min_length, max_length + 1):
        for start in range(n - length + 1):
            counts[tuple(symbols[start:start + length])] += 1
            tick_accum += 1
            if tick_accum >= TICK_BATCH:
                prog.tick(tick_accum)
                tick_accum = 0

    if tick_accum:
        prog.tick(tick_accum)
    prog.done()

    # Keep only patterns that meet the frequency threshold
    qualifying = {p for p, c in counts.items() if c >= min_frequency}
    del counts  # release memory before sub-pass B
    print(f"  {len(qualifying):,} patterns qualify (freq >= {min_frequency})")

    # ------------------------------------------------------------------
    # Sub-pass B: record instances only for qualifying patterns
    # ------------------------------------------------------------------
    pattern_data = {p: {'count': 0, 'instances': []} for p in qualifying}
    prog         = ProgressTracker("Pass 5b (instances) ", total_iters)
    tick_accum   = 0

    for length in range(min_length, max_length + 1):
        for start in range(n - length + 1):
            pattern = tuple(symbols[start:start + length])
            if pattern in qualifying:
                pattern_data[pattern]['count'] += 1
                pattern_data[pattern]['instances'].append(
                    (indices[start], indices[start + length - 1])
                )
            tick_accum += 1
            if tick_accum >= TICK_BATCH:
                prog.tick(tick_accum)
                tick_accum = 0

    if tick_accum:
        prog.tick(tick_accum)
    prog.done()

    return pattern_data

def export_visualization_csv(df: pd.DataFrame, patterns: dict, output_path: str,
                              min_length: int = 4, min_frequency: int = 2,
                              filter_mode: str = 'AND') -> None:
    colors = ['BLUE', 'RED', 'GREEN', 'MAGENTA', 'CYAN',
              'ORANGE', 'PURPLE', 'YELLOW', 'LIME', 'MAROON']

    filtered_patterns = []
    for pattern, data in patterns.items():
        meets_length = len(pattern) >= min_length
        meets_freq   = data['count'] >= min_frequency
        include = (meets_length and meets_freq) if filter_mode == 'AND' else (meets_length or meets_freq)
        if include:
            filtered_patterns.append((pattern, data))

    if not filtered_patterns:

        pd.DataFrame(columns=['pattern_type', 'start_time', 'end_time',

        return

    print(f"Found {len(filtered_patterns)} patterns meeting criteria")
    filtered_patterns.sort(key=lambda x: -x[1]['count'])

    rows = []
    for pattern_idx, (pattern, data) in enumerate(filtered_patterns):
        color       = colors[pattern_idx % len(colors)]
        frequency   = data['count']
        pattern_len = len(pattern)
        for inst_idx, (start_bar, end_bar) in enumerate(data['instances']):
            try:
                start_row     = df.loc[start_bar]
                end_row       = df.loc[end_bar]
                pattern_slice = df.loc[start_bar:end_bar]
                if 'high' in pattern_slice.columns and 'low' in pattern_slice.columns:
                    high_price = pattern_slice['high'].max()
                    low_price  = pattern_slice['low'].min()
                else:
                    high_price = pattern_slice['close'].max()
                    low_price  = pattern_slice['close'].min()
                price_range = max(high_price - low_price, 0.001)
                high_price += price_range * 0.1
                low_price  -= price_range * 0.1
                rows.append({
                    'pattern_type': f"P{pattern_idx+1}_L{pattern_len}",
                    'start_time':   start_row['time'],
                    'end_time':     end_row['time'],
                    'high_price':   round(high_price, 6),
                    'low_price':    round(low_price,  6),
                    'color':        color,
                    'description':  f"P{pattern_idx+1} L={pattern_len} F={frequency} #{inst_idx+1}"
                })
            except Exception as e:
                print(f"Error processing instance {start_bar}-{end_bar}: {e}")

    pd.DataFrame(rows).to_csv(output_path, index=False, encoding='ANSI')
    print(f"Exported {len(rows)} shading blocks for {len(filtered_patterns)} patterns to {output_path}")

def run_analysis(input_path: str, output_dir: str = None, point: float = 0.001,
                 min_length: int = 4, min_frequency: int = 2,
                 filter_mode: str = 'AND') -> Tuple[pd.DataFrame, dict]:
    if output_dir is None:
        output_dir = '.'

    print(f"Loading {input_path}...")
    try:
        df = pd.read_csv(input_path, encoding='utf-16')
    except Exception:
        df = pd.read_csv(input_path, encoding='utf-8')
    print(f"Loaded {len(df):,} bars")

    print("Computing ZigZag max_length...")
    zz_max = zigzag_max_length(df, point=point)

    df = pass1_raw_calculations(df, point)
    df = pass2_l1_calculation(df)
    df = pass3_l0_calculation(df, point)
    df = pass4_symbol_encoding(df)

    valid_symbols = df['symbol'].notna().sum()
    print(f"Generated {valid_symbols:,} symbols")

    patterns = pass5_pattern_mining(df, min_length=2, min_frequency=2, max_length=zz_max)
    print(f"Found {len(patterns):,} significant patterns")

    base_name = input_path.replace('.csv', '')
    export_l1_output(df,       f"{base_name}_L1_output.csv")
    export_l2_patterns(patterns, f"{base_name}_L2_catalog.csv")
    export_visualization_csv(df, patterns, f"{base_name}_patterns.csv",
                             min_length=min_length, min_frequency=min_frequency,
                             filter_mode=filter_mode)
    return df, patterns

Three candidates:

Levenshtein — your intuition was right. Allows substitution, insertion, deletion. Insertion/deletion means "this pattern had an extra noise bar in the middle" which is valid for trading. Weakness: treats all substitutions equally — swapping L0:+1 for L0:-1 costs the same as swapping SD+ for SD-, even though those are very different kinds of differences.

Yes. DTW produces a distance score — 0 means identical, higher means more different, with no fixed upper bound (it scales with sequence length and symbol difference magnitude).

However there's a complication for your use case: DTW compares two sequences of known, fixed identity. Your problem is different — you have one long sequence (~5,000 bars) and need to find all pairs of subsequences within it that are similar to each other. That's called subsequence DTW or similarity search, and it's significantly more expensive than pairwise comparison.

The practical approach for your scale is:

  1. Extract all subsequences of each length tier
  2. Use DTW distance to cluster them, then report clusters as "this pattern recurs"

The leading Python library for this is tslearn, which has subsequence DTW built in. But before committing to that architecture, Step 1 (dropping Z and ZS from the symbol) is still worth doing first — it's free information and may change what you find.
