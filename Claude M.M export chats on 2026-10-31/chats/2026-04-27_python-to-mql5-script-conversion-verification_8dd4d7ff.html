void ComputeEma(const double &values[], const int period, double &output[])
{
   int count = ArraySize(values);
   ArrayResize(output, count);
   double alpha = 2.0 / ((double)period + 1.0);
   double previous = EMPTY_VALUE;   // was: 0.0
   bool hasPrevious = false;

   for(int index = 0; index < count; index++)
   {
      double value = values[index];
      if(!MathIsValidNumber(value))
      {
         output[index] = previous;   // emits EMPTY_VALUE until first valid input
         continue;
      }

      if(!hasPrevious)
      {
         previous = value;
         hasPrevious = true;
      }
      else
      {
         previous = alpha * value + (1.0 - alpha) * previous;
      }
      output[index] = previous;
   }
}

That matches Python's behavior of carrying NaN through leading invalid bars instead of emitting 0.0.
