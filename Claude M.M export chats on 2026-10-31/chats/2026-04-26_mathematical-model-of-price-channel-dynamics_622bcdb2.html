def ema(values: np.ndarray, period: int) -> np.ndarray:
alpha = 2.0 / (period + 1.0)
output = np.full(len(values), np.nan, dtype=float)
previous = np.nan

for index, value in enumerate(values):
    if np.isnan(value):
        output[index] = previous
        continue
    previous = value if np.isnan(previous) else alpha * value + (1.0 - alpha) * previous
    output[index] = previous

return output

)

def leaky_accumulator(values: np.ndarray, decay: float) -> np.ndarray:
output = np.zeros(len(values), dtype=float)
previous = 0.0
update_weight = 1.0 - decay

for index, value in enumerate(values):
    previous = decay * previous + update_weight * value
    output[index] = previous

return output

ScoreComponent = Callable[[pd.DataFrame, SignalParams], pd.Series]

@register_score_component("force")
def _score_force(state: pd.DataFrame, params: SignalParams) -> pd.Series:
return params.force_weight * state["net_force"]

@register_score_component("slope_and_accel")
def _score_slope_and_accel(state: pd.DataFrame, params: SignalParams) -> pd.Series:
return params.slope_weight * (state["relative_slope"] + state["relative_accel"])

@register_score_component("ma1_lead")
def _score_ma1_lead(state: pd.DataFrame, params: SignalParams) -> pd.Series:

@register_score_component("stall")
def _score_stall(state: pd.DataFrame, params: SignalParams) -> pd.Series:
travel_sign = np.sign(state["travel"].replace(0.0, np.nan).ffill().fillna(0.0).to_numpy(float))
return pd.Series(
params.stall_weight * (state["stall_pressure"].to_numpy(float) * -travel_sign),
index=state.index,
)

@register_score_component("cross_signal")
def _score_cross_signal(state: pd.DataFrame, params: SignalParams) -> pd.Series:
"""
Recent cross momentum across all lines (clean weighted higher than weak).
Signed: +1 means recent crosses were upward (continuation up).
"""
clean = state["clean_cross_signal_all"].to_numpy(float)
weak = state["weak_cross_signal_all"].to_numpy(float)
combined = clean + 0.5 * weak
return pd.Series(params.cross_signal_weight * combined, index=state.index)

@register_score_component("bounce_signal")
def _score_bounce_signal(state: pd.DataFrame, params: SignalParams) -> pd.Series:
"""
Recent bounce activity. bounce_dir is the kick-back direction (away from
the bounced-off line), so a +1 bounce_signal already means upward push.
"""
bounce = state["bounce_signal_all"].to_numpy(float)
return pd.Series(params.bounce_signal_weight * bounce, index=state.index)

@register_score_component("cross_history")
def _score_cross_history(state: pd.DataFrame, params: SignalParams) -> pd.Series:
"""
Imbalance in rolling cross counts (unsigned). When clean crosses dominate
weak crosses, momentum is established; sign of recent cross_signal_all
carries direction.
"""
clean_count = state["all_clean_cross_count"].to_numpy(float)
weak_count = state["all_weak_cross_count"].to_numpy(float)
imbalance = clean_count - weak_count
direction_proxy = np.sign(state["cross_signal_all"].to_numpy(float))
return pd.Series(params.cross_history_weight * imbalance * direction_proxy, index=state.index)

@register_score_component("bounce_imbalance")
def _score_bounce_imbalance(state: pd.DataFrame, params: SignalParams) -> pd.Series:
"""
bounce_minus_clean is high when the channel keeps rejecting MA2 and
rarely lets it through. Signed against the recent relative_slope: a
bounce-heavy regime opposes the prevailing direction.
"""
composite = state["bounce_minus_clean"].to_numpy(float)
slope_dir = np.sign(state["relative_slope"].to_numpy(float))
return pd.Series(params.bounce_imbalance_weight * composite * -slope_dir, index=state.index)

def compute_direction_score(state: pd.DataFrame, params: SignalParams) -> pd.Series:

for _name, component_fn in SCORE_COMPONENTS:
total = total + component_fn(state, params).reindex(state.index).fillna(0.0)
return total

def generate_signals(state: pd.DataFrame, params: SignalParams, config: RunConfig) -> pd.DataFrame:
output = pd.DataFrame(index=state.index)
output["direction_score"] = compute_direction_score(state, params)
output["raw_direction"] = np.where(
output["direction_score"] >= params.threshold,
1,
np.where(output["direction_score"] <= -params.threshold, -1, 0),
)

raw_direction_array = output["raw_direction"].to_numpy(int)
bar_count = len(raw_direction_array)

eligibility_mask = np.ones(bar_count, dtype=bool)
perbar_gates: list[tuple[str, SignalGate]] = []
for name, gate_fn in SIGNAL_GATES:
    vectorized = getattr(gate_fn, "gate_vectorized", None)
    if vectorized is not None:
        eligibility_mask &= vectorized(raw_direction_array, params, config)
    else:
        perbar_gates.append((name, gate_fn))

signals = np.zeros(bar_count, dtype=int)
last_signal_index = -10_000_000
current_direction = 0

eligible_indices = np.where(eligibility_mask)[0]
for index in eligible_indices:
    ctx = GateContext(
        bar_index=int(index),
        raw_direction=int(raw_direction_array[index]),
        current_direction=current_direction,
        last_signal_index=last_signal_index,
        params=params,
        config=config,
    )
    if not all(gate_fn(ctx) for _name, gate_fn in perbar_gates):
        continue
    signals[index] = ctx.raw_direction
    current_direction = ctx.raw_direction
    last_signal_index = int(index)

output["signal"] = signals
return output

def backtest_signals(data: pd.DataFrame, signals: pd.Series) -> dict[str, float | int]:
close = data["Close"].to_numpy(float)
signal_values = signals.to_numpy(int)
entry_index = None
entry_price = 0.0
direction = 0
trade_pl: list[float] = []

for index, signal in enumerate(signal_values):
    if signal == 0:
        continue
    if direction != 0 and entry_index is not None:
        trade_pl.append((close[index] - entry_price) * direction)
    direction = signal
    entry_index = index
    entry_price = close[index]

if direction != 0 and entry_index is not None:
    trade_pl.append((close[-1] - entry_price) * direction)

if not trade_pl:
    return {
        "net_pl": 0.0,
        "trade_count": 0,
        "win_rate": 0.0,
        "avg_trade": 0.0,
        "best_trade": 0.0,
        "worst_trade": 0.0,
    }

pl_array = np.array(trade_pl, dtype=float)
return {
    "net_pl": float(pl_array.sum()),
    "trade_count": int(len(pl_array)),
    "win_rate": float((pl_array > 0.0).mean()),
    "avg_trade": float(pl_array.mean()),
    "best_trade": float(pl_array.max()),
    "worst_trade": float(pl_array.min()),
}

def build_trade_shading_rows(state: pd.DataFrame, point: float) -> pd.DataFrame:
signals = state[state["signal"] != 0][["dt", "close", "signal"]].reset_index(drop=True)
rows: list[dict[str, str | float]] = []

for index in range(len(signals) - 1):
    entry = signals.iloc[index]
    exit_trade = signals.iloc[index + 1]
    direction_value = int(entry["signal"])
    direction = "long" if direction_value > 0 else "short"
    entry_price = float(entry["close"])
    exit_price = float(exit_trade["close"])
    trade_pl = (exit_price - entry_price) * direction_value
    rounded_entry = round(entry_price / point) * point
    rounded_exit = round(exit_price / point) * point
    rounded_pl = round(trade_pl / point) * point
    entry_dt = str(entry["dt"])
    exit_dt = str(exit_trade["dt"])

    rows.append(
        {
            "scr_start_dt": entry_dt,
            "scr_end_dt": exit_dt,
            "upper_price": max(rounded_entry, rounded_exit),
            "lower_price": min(rounded_entry, rounded_exit),
            "direction": direction,
            "trade_pl": rounded_pl,
            "rule": SHADING_RULE_VALUE,
            "description": (
                f"entry={entry_dt} price={rounded_entry:.2f} "
                f"exit={exit_dt} price={rounded_exit:.2f} "
                f"trade_pl={rounded_pl:.2f}"
            ),
            "color": LONG_SHADE_COLOR if direction_value > 0 else SHORT_SHADE_COLOR,
            "run_tag": SHADING_RUN_TAG,
            "trade_entry_dt": entry_dt,
            "trade_exit_dt": exit_dt,
            "trade_entry_price": rounded_entry,
            "trade_exit_price": rounded_exit,
            "object_name": SHADING_OBJECT_NAME,
        }
    )

return pd.DataFrame(rows, columns=SHADING_COLUMNS)
